7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.anEmptyMap;
9   import static org.hamcrest.Matchers.containsString;
10  import static org.hamcrest.Matchers.hasEntry;
11  import static org.hamcrest.Matchers.is;
12  import static org.hamcrest.Matchers.not;
13  
14  import java.util.Collections;
15  import java.util.Map;
16  
17  import org.junit.jupiter.api.Test;
18  
19  class KenvUtilTest {
20  
21      @Test
22      void testParseKenv() {
23          // kenv on a DragonFly BSD 6.4 OpenStack guest, which sets no serial, UUID or baseboard strings
24          Map<String, String> kenv = KenvUtil.parseKenv("""
25                  hint.smbios.0.mem="0xeb320"
26                  smbios.bios.reldate="04/01/2014"
27                  smbios.bios.vendor="SeaBIOS"
28                  smbios.bios.version="1.16.3-4.el9.alma.1"
29                  smbios.chassis.version="RHEL 7.6.0 PC (i440FX + PIIX, 1996)"
30                  smbios.system.maker="RDO"
31                  smbios.system.product="OpenStack Compute"
32                  smbios.system.version="25.3.0-1.el9s"
33                  """.lines().toList());
34          assertThat(kenv, hasEntry("smbios.bios.vendor", "SeaBIOS"));
35          assertThat(kenv, hasEntry("smbios.bios.reldate", "04/01/2014"));
36          assertThat(kenv, hasEntry("smbios.system.product", "OpenStack Compute"));
37          assertThat(kenv, hasEntry("smbios.chassis.version", "RHEL 7.6.0 PC (i440FX + PIIX, 1996)"));
38          assertThat(kenv.containsKey("smbios.system.serial"), is(false));
39      }
40  
41      @Test
42      void testParseKenvBlankValue() {
43          // Values are trimmed, so a whitespace-only string reads as empty and the callers' fallbacks apply
44          Map<String, String> kenv = KenvUtil.parseKenv("""
45                  smbios.planar.serial="   "
46                  smbios.system.maker=" RDO "
47                  """.lines().toList());
48          assertThat(kenv, hasEntry("smbios.planar.serial", ""));
49          assertThat(kenv, hasEntry("smbios.system.maker", "RDO"));
50      }
51  
52      @Test
53      void testParseKenvEmpty() {
54          assertThat(KenvUtil.parseKenv(Collections.emptyList()), is(anEmptyMap()));
55      }
56  
57      @Test
58      void testQueryKenv() {
59          // Empty where there is no kenv command; on FreeBSD and DragonFly BSD every value arrives unquoted
60          for (String value : KenvUtil.queryKenv().values()) {
61              assertThat(value, not(containsString("\"")));
62          }
63      }
64  }
