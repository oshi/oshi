22  /**
23   * Tests for {@link FstatUtil}. The open file count is covered with sample {@code fstat} output; the integration test is
24   * gated to NetBSD.
25   */
26  class FstatUtilTest {
27  
28      @Test
29      void testParseOpenFilesCountsDescriptorsOnly() {
30          // fstat -p of a shell on NetBSD 10.1, with the text, root and trace rows fstat prints when they apply
31          List<String> fstat = """
32                  USER     CMD          PID   FD  MOUNT         INUM MODE         SZ|DV R/W
33                  oshi     sh          4759 text  /         1059845 -r-xr-xr-x  171784 r
34                  oshi     sh          4759   wd  /home     16203776 drwxr-xr-x     512 r
35                  oshi     sh          4759 root  /               2 drwxr-xr-x     512 r
36                  oshi     sh          4759   tr  /home     16203790 -rw-r--r--    1024 w
37                  oshi     sh          4759    0* pipe 0xffff0004326db900 <- 0xffff00042c48eb00 r
38                  oshi     sh          4759    1* pipe 0xffff0000f9e16100 -> 0xffff0001e9d42400 w
39                  oshi     sh          4759   10  /home     16203791 -rw-r--r--      12 r
40                  """.lines().toList();
41          assertThat(FstatUtil.parseOpenFiles(fstat, 4759), is(3L));
42      }
43  
44      @Test
45      void testParseOpenFilesCommandWithSpace() {
46          // fstat prints the command name unescaped, so a space in it shifts the FD column one place to the right
47          List<String> fstat = """
48                  USER     CMD          PID   FD  MOUNT         INUM MODE         SZ|DV R/W
49                  oshi     my tool     4760   wd  /home     16203776 drwxr-xr-x     512 r
50                  oshi     my tool     4760 text  /         1059845 -r-xr-xr-x  171784 r
51                  oshi     my tool     4760    0* pipe 0xffff0004326db900 <- 0xffff00042c48eb00 r
52                  oshi     my tool     4760    3  /home     16203791 -rw-r--r--      12 r
53                  """.lines().toList();
54          assertThat(FstatUtil.parseOpenFiles(fstat, 4760), is(2L));
55      }
56  
57      @Test
58      void testParseOpenFilesHeaderOnlyOrEmpty() {
59          assertThat(FstatUtil.parseOpenFiles(List.of("USER CMD PID FD ..."), 1), is(0L));
60          assertThat(FstatUtil.parseOpenFiles(Collections.<String>emptyList(), 1), is(0L));
61      }
62  
63      @Test
64      @EnabledIfSystemProperty(named = "os.name", matches = "(?i)netbsd")
65      void testFstatLive() {
66          // RuntimeMXBean#getName() returns "<pid>@<host>" — use it to avoid pulling oshi-core's SystemInfo into a
67          // oshi-common test. Mirrors the pattern used in the FreeBSD ProcstatUtilTest.
68          int pid = ParseUtil.parseIntOrDefault(ManagementFactory.getRuntimeMXBean().getName().split("@", 2)[0], -1);
69          assertThat("Number of open files must be nonnegative", FstatUtil.getOpenFiles(pid),
70                  is(greaterThanOrEqualTo(0L)));
71          // NetBSD builds the /proc/<pid>/cwd link through getcwd_common(), which resolves the vnode through the name
72          // cache and fails when it holds no entry for the directory, so the link reads back empty. Tolerate that, but
73          // when a path is returned it must be absolute.
74          String cwd = FstatUtil.getCwd(pid);
75          assertThat("getCwd must never return null", cwd, is(notNullValue()));
76          if (!cwd.isEmpty()) {
77              assertThat("A resolved cwd should be an absolute path", cwd, startsWith("/"));
78          }
79      }
80  }
