5   package oshi.util.common.platform.mac;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.contains;
9   import static org.hamcrest.Matchers.empty;
10  import static org.hamcrest.Matchers.is;
11  import static org.hamcrest.Matchers.sameInstance;
12  
13  import java.util.Arrays;
14  import java.util.Collections;
15  import java.util.List;
16  import java.util.concurrent.atomic.AtomicInteger;
17  import java.util.function.Supplier;
18  
19  import org.jspecify.annotations.Nullable;
20  import org.junit.jupiter.api.Test;
21  
22  import oshi.util.GlobalConfig;
23  
24  /**
25   * Tests the resolve-once-and-cache behavior the SMC sensors share, without a Mac.
26   * <p>
27   * Each test uses a property name of its own rather than a real OSHI property, because the suite runs in parallel and a
28   * shared property would race.
29   */
30  class SmcKeyCacheTest {
31  
32      private static final List<String> FALLBACK = Collections.unmodifiableList(Arrays.asList("Tg05", "Tg0D"));
33  
34      /** A discovery function that records how many times it ran. */
35      private static final class CountingDiscovery implements Supplier<@Nullable List<String>> {
36          private final AtomicInteger calls = new AtomicInteger();
37          private final @Nullable List<String> result;
38  
39          private CountingDiscovery(@Nullable List<String> result) {
40              this.result = result;
41          }
42  
43          @Override
44          public @Nullable List<String> get() {
45              calls.incrementAndGet();
46              return result;
47          }
48      }
49  
50      private static SmcKeyCache cache(String property) {
51          return new SmcKeyCache(property, "test", FALLBACK);
52      }
53  
54      @Test
55      void testCompletedDiscoveryIsCached() {
56          CountingDiscovery discovery = new CountingDiscovery(Arrays.asList("Tg0f", "Tg0j"));
57          SmcKeyCache keys = cache("oshi.test.smckeycache.completed");
58          assertThat(keys.get(discovery), contains("Tg0f", "Tg0j"));
59          assertThat(keys.get(discovery), contains("Tg0f", "Tg0j"));
60          assertThat("Discovery must run only once", discovery.calls.get(), is(1));
61      }
62  
63      @Test
64      void testIncompleteDiscoveryFallsBackWithoutCaching() {
65          // Null means "could not read". Caching it would disable the sensor for the lifetime of the JVM, so the next
66          // call must try again.
67          CountingDiscovery discovery = new CountingDiscovery(null);
68          SmcKeyCache keys = cache("oshi.test.smckeycache.incomplete");
69          assertThat(keys.get(discovery), is(sameInstance(FALLBACK)));
70          assertThat(keys.get(discovery), is(sameInstance(FALLBACK)));
71          assertThat("A failed discovery must be retried, not cached", discovery.calls.get(), is(2));
72      }
73  
74      @Test
75      void testEmptyDiscoveryIsCached() {
76          // Empty is a real answer from a completed run: this machine has no such sensor. That is cacheable, unlike null.
77          CountingDiscovery discovery = new CountingDiscovery(Collections.emptyList());
78          SmcKeyCache keys = cache("oshi.test.smckeycache.empty");
79          assertThat(keys.get(discovery), is(empty()));
80          assertThat(keys.get(discovery), is(empty()));
81          assertThat("An empty completed discovery must be cached", discovery.calls.get(), is(1));
82      }
83  
84      @Test
85      void testConfiguredKeysBypassDiscovery() {
86          String property = "oshi.test.smckeycache.configured";
87          GlobalConfig.set(property, "Tg1k,Tg1l");
88          try {
89              CountingDiscovery discovery = new CountingDiscovery(Arrays.asList("Tg0f"));
90              SmcKeyCache keys = cache(property);
91              assertThat(keys.get(discovery), contains("Tg1k", "Tg1l"));
92              assertThat(keys.get(discovery), contains("Tg1k", "Tg1l"));
93              assertThat("Configured keys must bypass discovery entirely", discovery.calls.get(), is(0));
94          } finally {
95              GlobalConfig.remove(property);
96          }
97      }
98  
99      @Test
100     void testUnusableConfiguredKeysFallThroughToDiscovery() {
101         // Every token is the wrong length, so parseConfiguredKeys drops them all and the result is empty. That must be
102         // treated as "nothing configured" rather than as an empty key list.
103         String property = "oshi.test.smckeycache.malformed";
104         GlobalConfig.set(property, "TOOLONG,ab");
105         try {
106             CountingDiscovery discovery = new CountingDiscovery(Arrays.asList("Tg0f"));
107             assertThat(cache(property).get(discovery), contains("Tg0f"));
108             assertThat(discovery.calls.get(), is(1));
109         } finally {
110             GlobalConfig.remove(property);
111         }
112     }
113 
114     @Test
115     void testFallbackIsReportedByIdentity() {
116         // Callers distinguish "fell back" from "discovered" by identity, so the fallback must be handed back as-is
117         // rather than copied.
118         SmcKeyCache keys = cache("oshi.test.smckeycache.identity");
119         assertThat(keys.fallback(), is(sameInstance(FALLBACK)));
120         assertThat(keys.get(new CountingDiscovery(null)), is(sameInstance(keys.fallback())));
121     }
122 }
