25   * Tests SMC key index logic without Mac hardware. This is why the logic lives here rather than in {@code SmcUtil},
26   * whose static {@code IOKit.INSTANCE} field makes any of its members unloadable off a Mac.

30      /** A realistic slice of a sorted SMC key index, with a Tg block in the middle. */
31      private static final String[] SORTED = { "#KEY", "ALI0", "F0Ac", "TB0T", "TC0P", "TCMb", "TCMz", "Tf00", "Tf11",
32              "Tg0W", "Tg0X", "Tg0e", "Tg0f", "Tg1g", "Tg1h", "Th00", "Tp01", "Tp09", "VP0C", "zSPc" };
33  
34      private static IntFunction<@Nullable String> lookup(String[] keys) {
35          return i -> i >= 0 && i < keys.length ? keys[i] : null;
36      }
37  
38      private static @Nullable List<String> findTg(String[] keys) {
39          return SmcKeyIndex.findKeys(keys.length, lookup(keys), "Tg", SmcKeyIndex::isGpuTemperatureKey);
40      }

101     @Test
102     void testTransientFailureIsRecovered() {
103         // Fails the first read of one index, then succeeds: the retry must recover the full block.
104         Set<Integer> failedOnce = new HashSet<>();
105         IntFunction<@Nullable String> flaky = i -> i == 12 && failedOnce.add(i) ? null : SORTED[i];
106         assertThat(SmcKeyIndex.findKeys(SORTED.length, flaky, "Tg", SmcKeyIndex::isGpuTemperatureKey),
107                 contains("Tg0W", "Tg0X", "Tg0e", "Tg0f", "Tg1g", "Tg1h"));
108     }

110     @Test
111     void testPermanentFailureMidScanSkipsOnlyThatKey() {
112         // Index 12 is Tg0f. Skipping rather than stopping preserves Tg1g/Tg1h, which a break would have discarded.
113         IntFunction<@Nullable String> flaky = i -> i == 12 ? null : SORTED[i];
114         assertThat(SmcKeyIndex.findKeys(SORTED.length, flaky, "Tg", SmcKeyIndex::isGpuTemperatureKey),
115                 contains("Tg0W", "Tg0X", "Tg0e", "Tg1g", "Tg1h"));
116     }

118     @Test
119     void testFailedSoleMatchIsNotReportedAsAbsent() {
120         // The only Tg key is unreadable and the next key ends the block. Reporting empty would be cached as "this
121         // machine has no GPU sensors" and would disable the sensor for the JVM lifetime.
122         String[] keys = { "TB0T", "Tg0f", "Th00", "Tp01" };
123         IntFunction<@Nullable String> flaky = i -> i == 1 ? null : keys[i];
124         assertThat("An unreadable sole match must not be cached as a confirmed absence",
125                 SmcKeyIndex.findKeys(keys.length, flaky, "Tg", SmcKeyIndex::isGpuTemperatureKey), is(nullValue()));
126     }
127 
128     @Test
129     void testFailedSearchProbeThatSkipsTheBlockIsNotReportedAsAbsent() {
130         // Same shape, but the failure is consumed by the binary search rather than the scan: substituting a neighbour
131         // for the unreadable probe moves the landing point past Tg0f, so the scan never attempts it at all. Failing
132         // only the first read is what makes null the proof of that -- had the scan reached index 2, its retry would
133         // have recovered Tg0f and returned a non-empty list. A permanently failing index cannot tell the two apart.
134         String[] keys = { "TB0T", "TCMb", "Tg0f", "Th00" };
135         Set<Integer> failedOnce = new HashSet<>();
136         IntFunction<@Nullable String> flaky = i -> i == 2 && failedOnce.add(i) ? null : keys[i];
137         assertThat("A search that skipped the block on a failed read must not report a confirmed absence",
138                 SmcKeyIndex.findKeys(keys.length, flaky, "Tg", SmcKeyIndex::isGpuTemperatureKey), is(nullValue()));
139     }

141     @Test
142     void testConfirmedAbsenceStillReportsEmpty() {
143         // The counterpart guard: with every read succeeding, "no Tg keys" is a real answer and must stay cacheable,
144         // so the fix above cannot degrade into "never return empty".
145         String[] keys = { "TB0T", "TCMb", "Th00", "Tp01" };
146         assertThat(findTg(keys), is(empty()));

293     /** Sentinels and genuine readings captured from hardware; see MacSensorsPlausibilityTest. */
294     private static final double FLOOR = 15d;
295 
296     private static double read(String key) {
297         return switch (key) {
298             case "Tg0W" -> 6.7d; // idle-gated sentinel
299             case "Tg0X" -> 40.7d;
300             case "Tg0f" -> 63.4d;
301             case "Tg1h" -> -4.0d; // negative sentinel, seen on M4 Max
302             default -> 0d;
303         };
304     }
305 
306     @Test
307     void testMaxPlausibleIgnoresSentinels() {
308         List<String> keys = List.of("Tg0W", "Tg0X", "Tg0f", "Tg1h");
309         assertThat("Returns the hottest genuine reading, not the global max",
310                 SmcKeyIndex.maxPlausible(keys, SmcKeyIndexTest::read, v -> v >= FLOOR), is(63.4d));
311     }
312 
313     @Test
314     void testMaxPlausibleWithNothingUsable() {
315         assertThat("All-sentinel reads must report unavailable",
316                 SmcKeyIndex.maxPlausible(List.of("Tg0W", "Tg1h"), SmcKeyIndexTest::read, v -> v >= FLOOR), is(0d));
317         assertThat(SmcKeyIndex.maxPlausible(Arrays.<String>asList(), SmcKeyIndexTest::read, v -> v >= FLOOR), is(0d));
318     }
319 
320     @Test
321     void testMaxPlausibleNeverReturnsBelowTheFloor() {
322         double result = SmcKeyIndex.maxPlausible(List.of("Tg0W", "Tg0X"), SmcKeyIndexTest::read, v -> v >= FLOOR);
323         assertThat("A result is either 0 or at least the floor, never in between", result == 0d || result >= FLOOR,
324                 is(true));
325         assertThat(6.7d, is(lessThanOrEqualTo(result)));
326     }
