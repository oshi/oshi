23  /**
24   * Tests the NVML logic shared by the JNA and FFM bindings, without an NVIDIA GPU or either native library. This is why
25   * the logic lives here rather than in the bindings, whose native library loading makes them unusable in a unit test.
26   */
27  class NvmlQueryTest {
28  
29      /** A scope that records its lifecycle calls and hands out a fixed handle. */
30      private static final class FakeScope implements NvmlScope<String> {
31          private final @Nullable String handle;
32          private final boolean initSucceeds;
33          private int inits;
34          private int uninits;
35  
36          FakeScope(@Nullable String handle, boolean initSucceeds) {
37              this.handle = handle;
38              this.initSucceeds = initSucceeds;
39          }
40  
41          @Override
42          public boolean init() {
43              inits++;
44              return initSucceeds;
45          }
46  
47          @Override
48          public void uninit() {
49              uninits++;
50          }
51  
52          @Override
53          public <R> R withDevice(String deviceId, Function<String, R> body, R sentinel) {
54              return handle == null ? sentinel : body.apply(handle);
55          }
56      }
57  
58      @Test
59      void testQueryReadsThroughAndBalancesInit() {
60          FakeScope scope = new FakeScope("device", true);
61          double value = NvmlQuery.query("0000:01:00.0", scope, handle -> 42d, -1d);
62          assertThat("reader result is returned", value, is(42d));
63          assertThat("init called once", scope.inits, is(1));
64          assertThat("uninit paired with init", scope.uninits, is(1));
65      }
66  
67      @Test
68      void testQueryRejectsBlankDeviceIdWithoutInitializing() {
69          for (String blank : new String[] { null, "" }) {
70              FakeScope scope = new FakeScope("device", true);
71              assertThat("blank id yields the sentinel", NvmlQuery.query(blank, scope, handle -> 42d, -1d), is(-1d));
72              assertThat("NVML is never initialized for a blank id", scope.inits, is(0));
73              assertThat("nothing to shut down", scope.uninits, is(0));
74          }
75      }
76  
77      @Test
78      void testQueryReturnsSentinelWhenInitFails() {
79          FakeScope scope = new FakeScope("device", false);
80          assertThat("failed init yields the sentinel", NvmlQuery.query("id", scope, handle -> 42d, -1d), is(-1d));
81          // A failed nvmlInit_v2 did not increment NVML's reference count, so it must not be decremented.
82          assertThat("uninit is not called after a failed init", scope.uninits, is(0));
83      }
84  
85      @Test
86      void testQueryReturnsSentinelWhenDeviceNotFound() {
87          FakeScope scope = new FakeScope(null, true);
88          assertThat("missing device yields the sentinel", NvmlQuery.query("id", scope, handle -> 42d, -1d), is(-1d));
89          assertThat("init is still balanced", scope.uninits, is(1));
90      }
91  
92      @Test
93      void testQueryUninitsWhenTheReaderThrows() {
94          FakeScope scope = new FakeScope("device", true);
95          try {
96              NvmlQuery.query("id", scope, handle -> {
97                  throw new IllegalStateException("native read blew up");
98              }, -1d);
99          } catch (IllegalStateException expected) {
100             // The reference count must still be released on the exceptional path.
101         }
102         assertThat("uninit runs even when the reader throws", scope.uninits, is(1));
103     }

105     @Test
106     void testMatchesIsBidirectional() {
107         assertThat("qualified matches bare", NvmlQuery.matches("00000000:01:00.0", "01:00.0"), is(true));
108         assertThat("bare matches qualified", NvmlQuery.matches("01:00.0", "00000000:01:00.0"), is(true));
109         assertThat("legacy matches modern", NvmlQuery.matches("0000:01:00.0", "00000000:01:00.0"), is(true));
110         assertThat("identical matches", NvmlQuery.matches("0000:01:00.0", "0000:01:00.0"), is(true));
111         assertThat("different devices do not match", NvmlQuery.matches("0000:01:00.0", "0000:02:00.0"), is(false));
112     }
113 
114     @Test
115     void testEmptyCandidateNeverMatches() {
116         // Bidirectional containment would otherwise make a blank PCI read answer to every query, since every string
117         // contains the empty string.
118         assertThat("blank candidate does not match", NvmlQuery.matches("", "0000:01:00.0"), is(false));
119     }

121     @Test
122     void testMatchBusIdReturnsTheEnumeratedForm() {
123         Set<String> busIds = new LinkedHashSet<>(List.of("00000000:0a:00.0", "00000000:02:00.0"));
124         assertThat("returns the canonical enumerated id", NvmlQuery.matchBusId(busIds, "02:00.0"),
125                 is("00000000:02:00.0"));
126         assertThat("is case-insensitive", NvmlQuery.matchBusId(busIds, "0A:00.0"), is("00000000:0a:00.0"));
127         assertThat("no match yields null", NvmlQuery.matchBusId(busIds, "09:00.0"), is(nullValue()));
128         assertThat("empty set yields null", NvmlQuery.matchBusId(Collections.emptySet(), "01:00.0"), is(nullValue()));
129     }
130 
131     @Test
132     void testMatchBusIdRejectsABlankFragment() {
133         // Every string contains the empty string, so without an explicit guard a blank fragment would match every
134         // enumerated id and resolve to the longest of them.
135         Set<String> busIds = new LinkedHashSet<>(List.of("0000:01:00.0", "00000000:01:00.0"));
136         assertThat("an empty fragment matches nothing", NvmlQuery.matchBusId(busIds, ""), is(nullValue()));
137         assertThat("a null fragment matches nothing", NvmlQuery.matchBusId(busIds, null), is(nullValue()));
138     }
139 
140     @Test
141     void testMatchBusIdPrefersTheQualifiedForm() {
142         // Both of a device's forms are enumerated and a bare fragment matches both, so the answer must not depend on
143         // which one iteration happens to reach first.
144         Set<String> legacyFirst = new LinkedHashSet<>(List.of("0000:01:00.0", "00000000:01:00.0"));
145         Set<String> modernFirst = new LinkedHashSet<>(List.of("00000000:01:00.0", "0000:01:00.0"));
146         assertThat("the domain-qualified form wins", NvmlQuery.matchBusId(legacyFirst, "01:00.0"),
147                 is("00000000:01:00.0"));
148         assertThat("regardless of iteration order", NvmlQuery.matchBusId(modernFirst, "01:00.0"),
149                 is("00000000:01:00.0"));
150     }

152     @Test
153     void testDeviceCacheRetriesUntilEnumerationSucceeds() {
154         NvmlDeviceCache cache = new NvmlDeviceCache("test");
155         int[] calls = new int[1];
156         Set<String> ids = new HashSet<>(List.of("00000000:01:00.0"));
157 
158         calls[0] = 0;
159         assertThat("a failed enumeration is not cached", cache.get(() -> {
160             calls[0]++;
161             return null;
162         }), is(Collections.<String>emptySet()));
163 
164         assertThat("the failure is retried", cache.get(() -> {
165             calls[0]++;
166             return ids;
167         }), is(ids));
168         assertThat("both attempts ran", calls[0], is(2));
169 
170         assertThat("success is cached", cache.get(() -> {
171             calls[0]++;
172             return Collections.<String>emptySet();
173         }), is(ids));
174         assertThat("the enumerator is not called again", calls[0], is(2));
175     }
176 
177     @Test
178     void testDeviceCacheCachesAGenuinelyEmptyResult() {
179         // An empty set means "no NVIDIA hardware", which is a real answer and must not be retried forever.
180         NvmlDeviceCache cache = new NvmlDeviceCache("test");
181         int[] calls = new int[1];
182         for (int i = 0; i < 2; i++) {
183             cache.get(() -> {
184                 calls[0]++;
185                 return Collections.<String>emptySet();
186             });
187         }
188         assertThat("an empty result is cached", calls[0], is(1));
189     }
