22  /**
23   * Tests the Linux routing table parsers against {@code /proc} content captured verbatim from a Linux container, with
24   * the corresponding {@code ip route} output used to confirm every decoded value.
25   */
26  class RouteTableTest {
27  
28      private static final Map<String, Integer> NO_INDICES = Collections.emptyMap();
29  
30      // Verbatim /proc/net/route. The matching `ip route show` reported:
31      // default via 172.17.0.1 dev eth0
32      // 172.17.0.0/16 dev eth0 scope link src 172.17.0.2
33      private static final List<String> PROC_NET_ROUTE = Arrays.asList(
34              "Iface\tDestination\tGateway \tFlags\tRefCnt\tUse\tMetric\tMask\t\tMTU\tWindow\tIRTT                   ", //
35              "eth0\t00000000\t010011AC\t0003\t0\t0\t0\t00000000\t0\t0\t0                                           ", //
36              "eth0\t000011AC\t00000000\t0001\t0\t0\t0\t0000FFFF\t0\t0\t0                                           ");
37  
38      // Verbatim /proc/net/ipv6_route, with a 2001:db8:1122:3344::1/64 address added to lo so the byte order is
39      // unambiguous. The matching `ip -6 route show` reported:
40      // 2001:db8:1122:3344::/64 dev lo metric 256
41      // fe80::/64 dev eth0 metric 256
42      private static final List<String> PROC_NET_IPV6_ROUTE = Arrays.asList(
43              "20010db8112233440000000000000000 40 00000000000000000000000000000000 00 00000000000000000000000000000000 00000100 00000001 00000000 00200200       lo", //
44              "fe800000000000000000000000000000 40 00000000000000000000000000000000 00 00000000000000000000000000000000 00000100 00000001 00000000 00000001     eth0", //
45              "00000000000000000000000000000000 00 00000000000000000000000000000000 00 00000000000000000000000000000000 ffffffff 00000001 00000000 00200200       lo", //
46              "00000000000000000000000000000001 80 00000000000000000000000000000000 00 00000000000000000000000000000000 00000000 00000003 00000000 80200001       lo", //
47              "20010db8112233440000000000000001 80 00000000000000000000000000000000 00 00000000000000000000000000000000 00000000 00000002 00000000 80200001       lo", //
48              "ff000000000000000000000000000000 08 00000000000000000000000000000000 00 00000000000000000000000000000000 00000100 00000002 00000000 00000001     eth0");
49  
50      // Verbatim /proc/net/ipv6_route from a container given an IPv6 default route, which the capture above had none of.
51      // The matching `ip -6 route show` reported:
52      // default via 2001:db8::1 dev eth0 metric 1024
53      private static final List<String> PROC_NET_IPV6_ROUTE_GATEWAY = Collections.singletonList(
54              "00000000000000000000000000000000 00 00000000000000000000000000000000 00 20010db8000000000000000000000001 00000400 00000001 00000000 00000003     eth0");
55  
56      @Test
57      void testParseIpv4Routes() {
58          List<IPRoute> routes = RouteTable.parseIpv4Routes(PROC_NET_ROUTE, NO_INDICES);
59          assertThat("The header line should be skipped", routes, hasSize(2));
60  
61          IPRoute def = routes.get(0);
62          assertThat(def.getDestination(), is(new byte[] { 0, 0, 0, 0 }));
63          assertThat(def.getPrefixLength(), is(0));
64          assertThat(def.isGateway(), is(true));
65          // 010011AC little-endian decodes to 172.17.0.1, matching "default via 172.17.0.1"
66          assertThat(def.getGateway(), is(new byte[] { -84, 17, 0, 1 }));
67          assertThat(def.getInterfaceName(), is("eth0"));
68          assertThat(def.isHost(), is(false));
69          assertThat("Linux publishes a metric column", def.getMetric(), is(0L));
70  
71          IPRoute subnet = routes.get(1);
72          // 000011AC decodes to 172.17.0.0 and mask 0000FFFF to 255.255.0.0, matching "172.17.0.0/16"
73          assertThat(subnet.getDestination(), is(new byte[] { -84, 17, 0, 0 }));
74          assertThat(subnet.getPrefixLength(), is(16));
75          assertThat(subnet.isGateway(), is(false));
76          assertThat(subnet.getGateway(), is(new byte[0]));
77          assertThat(subnet.isHost(), is(false));
78      }
79  
80      @Test
81      void testParseIpv6Routes() {
82          List<IPRoute> routes = RouteTable.parseIpv6Routes(PROC_NET_IPV6_ROUTE, NO_INDICES);
83          assertThat("There is no header line to skip", routes, hasSize(6));
84  
85          // The address is printed in network order, so no per-word reversal is applied. This is what distinguishes
86          // /proc/net/ipv6_route from /proc/net/tcp6, and 2001:db8:1122:3344 is asymmetric enough to prove it.
87          IPRoute added = routes.get(0);
88          assertThat(added.getDestination(), is(ParseUtil.parseIpv6AddressToBytes("2001:db8:1122:3344::")));
89          assertThat("Prefix length is hex: 40 is 64", added.getPrefixLength(), is(64));
90          assertThat("Metric is hex: 00000100 is 256", added.getMetric(), is(256L));
91          assertThat(added.getInterfaceName(), is("lo"));
92          assertThat(added.isGateway(), is(false));
93          assertThat(added.isHost(), is(false));
94  
95          assertThat(routes.get(1).getDestination(), is(ParseUtil.parseIpv6AddressToBytes("fe80::")));
96          assertThat(routes.get(1).getPrefixLength(), is(64));
97          assertThat(routes.get(1).getInterfaceName(), is("eth0"));
98  
99          assertThat(routes.get(2).getDestination(), is(new byte[16]));
100         assertThat(routes.get(2).getPrefixLength(), is(0));
101         assertThat("ffffffff is an unreachable route's metric, not a negative", routes.get(2).getMetric(),
102                 is(4294967295L));
103 
104         // Flags 80200001 does not include RTF_HOST (0x4), so a /128 is only recognized by its prefix width
105         IPRoute loopback = routes.get(3);
106         assertThat(loopback.getDestination(), is(ParseUtil.parseIpv6AddressToBytes("::1")));
107         assertThat(loopback.getPrefixLength(), is(128));
108         assertThat("A /128 local route carries no RTF_HOST flag, so the prefix width must decide", loopback.isHost(),
109                 is(true));
110 
111         assertThat(routes.get(4).getDestination(), is(ParseUtil.parseIpv6AddressToBytes("2001:db8:1122:3344::1")));
112         assertThat(routes.get(4).isHost(), is(true));
113 
114         assertThat(routes.get(5).getDestination(), is(ParseUtil.parseIpv6AddressToBytes("ff00::")));
115         assertThat("Prefix length is hex: 08 is 8", routes.get(5).getPrefixLength(), is(8));
116     }
117 
118     @Test
119     void testParseIpv6GatewayRoute() {
120         List<IPRoute> routes = RouteTable.parseIpv6Routes(PROC_NET_IPV6_ROUTE_GATEWAY, NO_INDICES);
121         assertThat(routes, hasSize(1));
122         IPRoute def = routes.get(0);
123         assertThat(def.getDestination(), is(new byte[16]));
124         assertThat(def.getPrefixLength(), is(0));
125         assertThat("Flags 00000003 carries RTF_GATEWAY", def.isGateway(), is(true));
126         assertThat(def.getGateway(), is(ParseUtil.parseIpv6AddressToBytes("2001:db8::1")));
127         assertThat("Metric is hex: 00000400 is 1024", def.getMetric(), is(1024L));
128         assertThat(def.getInterfaceName(), is("eth0"));
129         assertThat(def.isHost(), is(false));
130     }

132     @Test
133     void testInterfaceIndexLookup() {
134         Map<String, Integer> indices = new HashMap<>();
135         indices.put("eth0", 2);
136         List<IPRoute> routes = RouteTable.parseIpv4Routes(PROC_NET_ROUTE, indices);
137         assertThat(routes.get(0).getInterfaceIndex(), is(2));
138     }

140     @Test
141     void testEmptyAndMalformedInput() {
142         assertThat(RouteTable.parseIpv4Routes(Collections.emptyList(), NO_INDICES), hasSize(0));
143         assertThat(RouteTable.parseIpv6Routes(Collections.emptyList(), NO_INDICES), hasSize(0));
144         assertThat(RouteTable.parseIpv4Routes(Collections.singletonList("truncated\tline"), NO_INDICES), hasSize(0));
145         assertThat(RouteTable.parseIpv6Routes(Collections.singletonList("not hex at all"), NO_INDICES), hasSize(0));
146         // A line with the right number of columns but a destination that is not thirty-two hex digits. This is how the
147         // IPv6 file rejects anything unexpected, since unlike the IPv4 one it has no header to skip. The token is
148         // exactly thirty-two characters so that it is rejected for not being hex rather than for its length.
149         assertThat(RouteTable.parseIpv6Routes(
150                 Collections.singletonList(
151                         "zzzzzzzzzzzzzzzzzzzzzzzzzzzzzzzz 40 0 00 0 00000100 00000001 00000000 00000001 eth0"),
152                 NO_INDICES), hasSize(0));
153     }
