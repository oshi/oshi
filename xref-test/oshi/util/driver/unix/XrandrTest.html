309     private static Map<String, Output> byName(List<Output> outputs) {
310         Map<String, Output> byName = new LinkedHashMap<>();
311         for (Output output : outputs) {
312             byName.put(output.getName(), output);
313         }
314         return byName;
315     }

394     private static DisplayMode modeOf(Map<String, Output> data, String name) {
395         Output output = data.get(name);
396         assertNotNull(output);
397         Optional<DisplayMode> mode = output.getMode();
398         assertThat(name + " should have a mode", mode.isPresent(), is(true));
399         return mode.get();
400     }
