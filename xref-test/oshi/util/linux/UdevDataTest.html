7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.aMapWithSize;
9   import static org.hamcrest.Matchers.anEmptyMap;
10  import static org.hamcrest.Matchers.hasEntry;
11  
12  import java.io.IOException;
13  import java.nio.file.Files;
14  import java.nio.file.Path;
15  import java.util.List;
16  
17  import org.junit.jupiter.api.Test;
18  import org.junit.jupiter.api.condition.DisabledOnOs;
19  import org.junit.jupiter.api.condition.OS;
20  import org.junit.jupiter.api.io.TempDir;
21  
22  class UdevDataTest {
23  
24      // Shape of /run/udev/data/b8:1 on an Ubuntu host: symlinks, tags and timestamps around the E: properties
25      private static final String PARTITION_RECORD = """
26              S:disk/by-uuid/ABCD-1234
27              S:disk/by-partuuid/0f1e2d3c-01
28              W:12
29              I:1731000000000000
30              E:ID_FS_TYPE=vfat
31              E:ID_FS_UUID=ABCD-1234
32              E:ID_FS_LABEL=EFI
33              E:ID_PART_ENTRY_NAME=EFI System Partition
34              E:ID_MODEL=Samsung SSD 980 PRO=2TB
35              G:systemd
36              Q:systemd
37              V:1
38              """;
39  
40      @Test
41      void testParseProperties() {
42          var properties = UdevData.parseProperties(PARTITION_RECORD.lines().toList());
43          assertThat(properties, aMapWithSize(5));
44          assertThat(properties, hasEntry("ID_FS_TYPE", "vfat"));
45          assertThat("A value keeps spaces", properties, hasEntry("ID_PART_ENTRY_NAME", "EFI System Partition"));
46          assertThat("Only the first = separates key and value", properties,
47                  hasEntry("ID_MODEL", "Samsung SSD 980 PRO=2TB"));
48      }
49  
50      @Test
51      void testParsePropertiesIgnoresMalformed() {
52          assertThat(UdevData.parseProperties(List.of("E:", "E:=value", "E:NOEQUALS", "S:link", "")), anEmptyMap());
53      }
54  
55      @Test
56      // A block device's record is named major:minor, and Windows does not allow a colon in a file name
57      @DisabledOnOs(OS.WINDOWS)
58      void testQueryByDeviceNumberAndIfindex(@TempDir Path dir) throws IOException {
59          Files.writeString(dir.resolve("b8:1"), PARTITION_RECORD);
60          Files.writeString(dir.resolve("n2"), "E:ID_MODEL_FROM_DATABASE=I225-V\n");
61          String data = dir + "/";
62          assertThat(UdevData.queryBlockDevice(data, 8, 1), hasEntry("ID_FS_UUID", "ABCD-1234"));
63          assertThat(UdevData.queryNetworkInterface(data, 2), hasEntry("ID_MODEL_FROM_DATABASE", "I225-V"));
64          // No record, and the numbers a failed sysfs read yields, never touch a file
65          assertThat(UdevData.queryBlockDevice(data, 8, 2), anEmptyMap());
66          assertThat(UdevData.queryBlockDevice(data, 0, 0), anEmptyMap());
67          assertThat(UdevData.queryNetworkInterface(data, 0), anEmptyMap());
68      }
69  
70      @Test
71      void testDefaultDatabaseRejectsUnsetNumbers() {
72          assertThat(UdevData.queryBlockDevice(0, 0), anEmptyMap());
73          assertThat(UdevData.queryNetworkInterface(0), anEmptyMap());
74          assertThat(UdevData.queryDevice("pci", ""), anEmptyMap());
75          assertThat(UdevData.queryDevice("", "0000:01:00.0"), anEmptyMap());
76      }
77  }
