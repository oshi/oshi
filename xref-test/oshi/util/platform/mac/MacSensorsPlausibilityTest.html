33  /**
34   * Tests the plausibility guard that keeps an idle-gated SMC sensor's sentinel from being reported as a temperature. JNA
35   * twin of {@code oshi.ffm.MacSensorsPlausibilityFFMTest}.
36   */
37  @EnabledOnOs(OS.MAC)
38  class MacSensorsPlausibilityTest {
39  
40      /**
41       * Sentinels actually captured from hardware: -4.0, 0.0, 2.5, 4.0 and 5.2 appear in the iSMC sample reports, and
42       * 4.633, 6.033, 6.7, 7.425 and 8.425 in a sweep of all 288 temperature sensors on an M2 Max. 8.425 is the highest
43       * ever observed, so it is the value that pins the floor.
44       */
45      private static final double[] OBSERVED_SENTINELS = { -4.0, 0.0, 2.5, 4.0, 4.633, 5.2, 6.033, 6.7, 7.425, 8.425 };
46  
47      /**
48       * Genuine readings actually observed: the lowest seen anywhere in the iSMC reports (21.0 on an Intel T2), a typical
49       * Apple Silicon idle, and readings under load.
50       */
51      private static final double[] GENUINE_READINGS = { 21.0, 28.0, 35.0, 52.5, 85.0, 100.0 };
52  
53      @Test
54      void testObservedSentinelsAreRejected() {
55          for (double sentinel : OBSERVED_SENTINELS) {
56              assertThat("Sentinel " + sentinel + " from an idle-gated sensor must be rejected",
57                      SmcUtil.isPlausibleTemperature(sentinel), is(false));
58          }
59      }
60  
61      @Test
62      void testGenuineReadingsAreAccepted() {
63          for (double reading : GENUINE_READINGS) {
64              assertThat("Genuine temperature " + reading + " must be accepted", SmcUtil.isPlausibleTemperature(reading),
65                      is(true));
66          }
67      }
68  
69      /**
70       * No upper bound is applied. Every bad value observed was at the low end, and a ceiling near the ~100 C throttling
71       * point would discard real readings exactly when they matter most. This is a deliberate contract, not an observed
72       * reading, so it is asserted separately from {@link #GENUINE_READINGS}.
73       */
74      @Test
75      void testNoUpperBoundIsApplied() {
76          assertThat("A reading above the throttling point must still be accepted", SmcUtil.isPlausibleTemperature(110d),
77                  is(true));
78      }
79  
80      /**
81       * The guard sits in an empty band: across M1 through M5, A18 and Intel T2 machines, nothing was observed between
82       * the highest sentinel and the lowest genuine reading. Failing this means the floor has been moved onto one of the
83       * two populations.
84       */
85      @Test
86      void testFloorSitsBetweenTheTwoObservedPopulations() {
87          // Strictly greater: the predicate accepts values equal to the floor, so a floor of exactly 8.425 would
88          // accept the highest observed sentinel.
89          assertThat("Floor must clear the highest observed sentinel", SmcUtil.MIN_PLAUSIBLE_TEMPERATURE,
90                  is(greaterThan(8.425)));
91          assertThat("Floor must stay below the lowest observed genuine reading", SmcUtil.MIN_PLAUSIBLE_TEMPERATURE,
92                  is(lessThanOrEqualTo(21.0)));
93      }
94  
95      /**
96       * End to end against the real SMC: a reading is either unavailable or plausible, never a sentinel in between.
97       */
98      @Test
99      void testCpuTemperatureIsPlausibleOrUnavailable() {
100         double temp = new SystemInfo().getHardware().getSensors().getCpuTemperature();
101         assertThat("CPU temperature must be unavailable or plausible, never a sentinel", temp,
102                 either(notANumber()).or(is(0d)).or(greaterThanOrEqualTo(SmcUtil.MIN_PLAUSIBLE_TEMPERATURE)));
103     }

105     /**
106      * The chip-independent CPU-die aggregate keys are tried before the per-core keys, and {@code TCMb} (the die
107      * average) is preferred over {@code TCMz} (the die max) so the reported value stays close to what OSHI historically
108      * returned from a single per-core sensor.
109      */
110     @Test
111     void testCpuTempAggregateKeysArePreferredAverageFirst() {
112         assertThat("TCMb (average) must be tried before TCMz (max)", SmcUtil.SMC_KEYS_CPU_TEMP_AGGREGATE_AS,
113                 is(contains("TCMb", "TCMz")));
114         assertThat("Aggregate keys must be distinct from the per-core keys",
115                 SmcUtil.SMC_KEYS_CPU_TEMP_AGGREGATE_AS.stream().noneMatch(SmcUtil.SMC_KEYS_CPU_TEMP_AS::contains),
116                 is(true));
117     }

119     /**
120      * GPU keys are discovered from the SMC rather than hardcoded, because they are chip-specific: on this hardware only
121      * two of the four keys OSHI originally shipped exist, and six sensors appear in no published key table.
122      */
123     @Test
124     void testGpuTemperatureKeysAreDiscovered() {
125         List<String> keys = SmcUtil.getGpuTemperatureKeys();
126         assertThat("Discovery must never return null", keys, is(notNullValue()));
127         for (String key : keys) {
128             assertThat(key + " must match the GPU key convention", SmcKeyIndex.isGpuTemperatureKey(key), is(true));
129         }
130         List<String> second = SmcUtil.getGpuTemperatureKeys();
131         assertThat("Repeated calls must agree", second, is(equalTo(keys)));
132         // A completed discovery is cached; a failed one deliberately is not, and returns the shared fallback constant
133         // instead. Asserting identity unconditionally would pass vacuously in that case, since both calls would return
134         // the same constant without anything having been cached.
135         if (!keys.equals(SmcUtil.SMC_KEYS_GPU_TEMP_AS)) {
136             assertThat("A completed discovery must be cached, not repeated", second, is(sameInstance(keys)));
137         }
138     }

140     /**
141      * Regression guard against a discovery bug silently narrowing the sensor set: any of the legacy hardcoded keys that
142      * this machine can actually read must also have been discovered. Passes on any Apple Silicon Mac.
143      */
144     @Test
145     void testDiscoveryIsASupersetOfReadableLegacyKeys() {
146         IOKit.IOConnect conn = SmcUtil.smcOpen();
147         assumeTrue(conn != null, "SMC unavailable");
148         try {
149             List<String> discovered = SmcUtil.getGpuTemperatureKeys();
150             for (String legacy : SmcUtil.SMC_KEYS_GPU_TEMP_AS) {
151                 if (SmcUtil.isPlausibleTemperature(SmcUtil.smcGetFloat(conn, legacy))) {
152                     assertThat(legacy + " reads a real value but was not discovered", discovered, hasItem(legacy));
153                 }
154             }
155         } finally {
156             SmcUtil.smcClose(conn);
157         }
158     }

160     /**
161      * The reported value is the hottest cluster, so it can never be below the first-match value the previous
162      * implementation would have returned from the same key set.
163      */
164     @Test
165     void testMaxIsNotLessThanFirstMatch() {
166         IOKit.IOConnect conn = SmcUtil.smcOpen();
167         assumeTrue(conn != null, "SMC unavailable");
168         try {
169             List<String> keys = SmcUtil.getGpuTemperatureKeys();
170             double max = SmcUtil.smcGetMaxTemperature(conn, keys);
171             double first = SmcUtil.smcGetFirstTemperature(conn, keys);
172             assertThat("max must be at least the first plausible reading", max, is(greaterThanOrEqualTo(first)));
173             assertThat("A GPU temperature is unavailable or plausible, never a sentinel in between",
174                     max == 0d || SmcUtil.isPlausibleTemperature(max), is(true));
175         } finally {
176             SmcUtil.smcClose(conn);
177         }
178     }

180     /**
181      * The fallback list must remain a superset of the four keys OSHI read before discovery existed, so a machine that
182      * falls back can never report less than the previous implementation did. Widening the list by frequency across
183      * published sensor dumps once dropped {@code Tg0f}, which is the only one of the four present on some M2 hardware.
184      */
185     @Test
186     void testFallbackKeysIncludeTheHistoricalFour() {
187         assertThat(SmcUtil.SMC_KEYS_GPU_TEMP_AS, hasItems("Tg05", "Tg0D", "Tg0f", "Tg0j"));
188     }

227     @Test
228     void testFanSpeedsAreWithinTheHardwareLimits() {
229         IOKit.IOConnect conn = SmcUtil.smcOpen();
230         assumeTrue(conn != null, "SMC unavailable");
231         try {
232             int[] speeds = new SystemInfo().getHardware().getSensors().getFanSpeeds();
233             List<String> keys = SmcUtil.getFanSpeedKeys();
234             assertThat("One speed per discovered key", speeds.length, is(keys.size()));
235             for (int i = 0; i < speeds.length; i++) {
236                 assertThat("Fan speed must not be negative", speeds[i], is(greaterThanOrEqualTo(0)));
237                 // F%dMx is the fan's hardware maximum. Derive it from the discovered key rather than the loop index:
238                 // discovery returns keys in index order, so a machine numbering its fans non-contiguously would
239                 // otherwise be compared against a different fan's maximum. Only assert where the key reads a value.
240                 double max = SmcUtil.smcGetFloat(conn, keys.get(i).substring(0, 2) + "Mx");
241                 if (max > 0) {
242                     assertThat("Fan speed must not exceed the hardware maximum", (double) speeds[i],
243                             is(lessThanOrEqualTo(max * 1.5)));
244                 }
245             }
246         } finally {
247             SmcUtil.smcClose(conn);
248         }
249     }

251     @Test
252     void testStoppedFanIsReportedAsZeroNotDropped() {
253         // An empty array means "no fans detected"; a zero entry means "a fan reading zero". A stopped fan must produce
254         // the latter, so the array length tracks the key count regardless of whether any fan is spinning.
255         int[] speeds = new SystemInfo().getHardware().getSensors().getFanSpeeds();
256         assertThat("One speed per discovered key, even when all are stopped", speeds.length,
257                 is(SmcUtil.getFanSpeedKeys().size()));
258     }

269     @Test
270     void testAppleSiliconVoltageKeyReadsPlausibly() {
271         IOKit.IOConnect conn = SmcUtil.smcOpen();
272         assumeTrue(conn != null, "SMC unavailable");
273         try {
274             // VP0C is chip-specific, not simply Apple Silicon versus Intel: it reads 0.75 V on an M3 Pro but is
275             // absent on an M2 Max, whose 53 V keys contain no identified core voltage at all. So skip when the key
276             // is absent, without inferring which chip that implies.
277             assumeTrue(!SmcUtil.smcGetDataType(conn, SmcUtil.SMC_KEY_CPU_VOLTAGE_AS).isEmpty(),
278                     "VP0C absent on this chip");
279             double volts = SmcUtil.smcGetFirstVoltage(conn, SmcUtil.getCpuVoltageKeys());
280             assertThat("Apple Silicon core voltage must read plausibly", volts,
281                     is(greaterThanOrEqualTo(SmcUtil.MIN_PLAUSIBLE_VOLTAGE)));
282         } finally {
283             SmcUtil.smcClose(conn);
284         }
285     }

287     @Test
288     void testVoltageKeysAreNotDiscovered() {
289         // Guards against someone "completing the pattern" and adding a V-prefix scan that could return a supply rail
290         // (e.g. VD0R at 20 V) as the CPU voltage. The default keys are the two named keys, no discovery.
291         assertThat("Voltage keys must remain the named defaults", SmcUtil.getCpuVoltageKeys(),
292                 is(equalTo(SmcUtil.SMC_KEYS_CPU_VOLTAGE)));
293         assertThat(SmcUtil.SMC_KEYS_CPU_VOLTAGE, contains(SmcUtil.SMC_KEY_CPU_VOLTAGE_AS, SmcUtil.SMC_KEY_CPU_VOLTAGE));
