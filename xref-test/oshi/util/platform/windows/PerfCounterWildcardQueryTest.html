31  /**
32   * Tests the CIM type conversions in the wildcard WMI mapping.
33   * <p>
34   * A real query returns whichever CIM type the queried performance class happens to use, so on hardware only one of the
35   * four conversions is ever reached. Stubbing the result reaches all of them, and the unsupported-type path too.
36   */
37  class PerfCounterWildcardQueryTest {
38  
39      /** First constant names the instance; the rest are values, one per CIM type the mapping converts. */
40      private enum Prop {
41          NAME, U16, U32, U64, WHEN;
42      }
43  
44      /** A WMI result stub: per-property CIM/VT types and one value per row. */
45      private static final class StubResult implements WmiResult<Prop> {
46          private final Map<Prop, Integer> cimTypes = new EnumMap<>(Prop.class);
47          private final Map<Prop, Integer> vtTypes = new EnumMap<>(Prop.class);
48          private final Map<Prop, List<Object>> values = new EnumMap<>(Prop.class);
49          private int rows;
50  
51          private StubResult put(Prop p, int cim, int vt, Object... rowValues) {
52              cimTypes.put(p, cim);
53              vtTypes.put(p, vt);
54              values.put(p, Arrays.asList(rowValues));
55              rows = Math.max(rows, rowValues.length);
56              return this;
57          }
58  
59          @Override
60          public int getResultCount() {
61              return rows;
62          }
63  
64          @Override
65          public @Nullable Object getValue(Prop property, int index) {
66              List<Object> v = values.get(property);
67              return v == null || index >= v.size() ? null : v.get(index);
68          }
69  
70          @Override
71          public int getVtType(Prop property) {
72              return vtTypes.getOrDefault(property, 0);
73          }
74  
75          @Override
76          public int getCIMType(Prop property) {
77              return cimTypes.getOrDefault(property, 0);
78          }
79      }
80  
81      // Two instances, one per supported CIM type. WMI reports UINT64 as a string and datetimes in CIM format.
82      private static StubResult twoInstances() {
83          return new StubResult().put(Prop.NAME, CIM_STRING, VT_BSTR, "cpu0", "cpu1")
84                  .put(Prop.U16, CIM_UINT16, VT_I4, 16, 17).put(Prop.U32, CIM_UINT32, VT_I4, 32, 33)
85                  .put(Prop.U64, CIM_UINT64, VT_BSTR, "64", "65")
86                  .put(Prop.WHEN, CIM_DATETIME, VT_BSTR, "20240115143000.000000+000", "20240115143001.000000+000");
87      }
88  
89      private static Pair<List<String>, Map<Prop, List<Long>>> map(StubResult result) {
90          return PerfCounterWildcardQuery.mapInstancesAndValuesFromResult(Prop.class, result);
91      }
92  
93      @Test
94      void testEachCimTypeIsConverted() {
95          Map<Prop, List<Long>> values = map(twoInstances()).getB();
96          assertThat("UINT16 widens to long", values.get(Prop.U16), contains(16L, 17L));
97          assertThat("UINT32 widens unsigned", values.get(Prop.U32), contains(32L, 33L));
98          assertThat("UINT64 is parsed from its string form", values.get(Prop.U64), contains(64L, 65L));
99          assertThat("DATETIME becomes epoch milliseconds", values.get(Prop.WHEN),
100                 contains(1705329000000L, 1705329001000L));
101     }
102 
103     @Test
104     void testFirstPropertyNamesTheInstances() {
105         Pair<List<String>, Map<Prop, List<Long>>> mapped = map(twoInstances());
106         assertThat(mapped.getA(), contains("cpu0", "cpu1"));
107         assertThat("The instance name is not also a value", mapped.getB().containsKey(Prop.NAME), is(false));
108     }
109 
110     @Test
111     void testUnsignedIntegersAreNotSignExtended() {
112         // 0xFFFFFFFF arrives as -1 in a 32-bit slot; as an unsigned CIM_UINT32 it must read as 4294967295.
113         StubResult result = new StubResult().put(Prop.NAME, CIM_STRING, VT_BSTR, "cpu0")
114                 .put(Prop.U16, CIM_UINT16, VT_I4, 0).put(Prop.U32, CIM_UINT32, VT_I4, -1)
115                 .put(Prop.U64, CIM_UINT64, VT_BSTR, "0")
116                 .put(Prop.WHEN, CIM_DATETIME, VT_BSTR, "20240115143000.000000+000");
117         assertThat(map(result).getB().get(Prop.U32), contains(4294967295L));
118     }
119 
120     @Test
121     void testAnUnsupportedCimTypeIsRejected() {
122         // CIM_SINT32 is a type the mapping does not convert; it must fail loudly rather than silently drop the value.
123         StubResult result = new StubResult().put(Prop.NAME, CIM_STRING, VT_BSTR, "cpu0").put(Prop.U16, 3, VT_I4, 1);
124         assertThrows(ClassCastException.class, () -> map(result));
125     }
126 
127     @Test
128     void testNoRowsYieldsNoInstancesAndNoValues() {
129         Pair<List<String>, Map<Prop, List<Long>>> mapped = map(new StubResult());
130         assertThat(mapped.getA(), is(empty()));
131         assertThat(mapped.getB().isEmpty(), is(true));
132     }
