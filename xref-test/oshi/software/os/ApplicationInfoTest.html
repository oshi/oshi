5   package oshi.software.os;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.containsString;
9   import static org.hamcrest.Matchers.is;
10  import static org.hamcrest.Matchers.not;
11  import static org.junit.jupiter.api.Assertions.assertThrows;
12  
13  import java.util.LinkedHashMap;
14  import java.util.Map;
15  
16  import org.junit.jupiter.api.Test;
17  
18  class ApplicationInfoTest {
19  
20      @Test
21      void testGetters() {
22          Map<String, String> extra = new LinkedHashMap<>();
23          extra.put("location", "/Applications/App.app");
24          ApplicationInfo app = new ApplicationInfo("MyApp", "2.0", "Acme", 1_000_000L, extra);
25          assertThat(app.getName(), is("MyApp"));
26          assertThat(app.getVersion(), is("2.0"));
27          assertThat(app.getVendor(), is("Acme"));
28          assertThat(app.getTimestamp(), is(1_000_000L));
29          assertThat(app.getAdditionalInfo().get("location"), is("/Applications/App.app"));
30      }
31  
32      @Test
33      void testAdditionalInfoIsUnmodifiable() {
34          Map<String, String> extra = new LinkedHashMap<>();
35          extra.put("location", "/Applications/App.app");
36          ApplicationInfo app = new ApplicationInfo("MyApp", "2.0", "Acme", 1_000_000L, extra);
37  
38          // The caller's map is copied, so later edits to it do not show up
39          extra.put("source", "added later");
40          assertThat(app.getAdditionalInfo().containsKey("source"), is(false));
41  
42          // And what the getter returns cannot be written to
43          assertThrows(UnsupportedOperationException.class, () -> app.getAdditionalInfo().put("k", "v"));
44      }
45  
46      @Test
47      void testNullAdditionalInfoBecomesEmptyMap() {
48          ApplicationInfo app = new ApplicationInfo("App", "1.0", "Vendor", 0L, null);
49          assertThat(app.getAdditionalInfo().isEmpty(), is(true));
50      }
51  
52      @Test
53      void testEqualsAndHashCode() {
54          ApplicationInfo a = new ApplicationInfo("App", "1.0", "Vendor", 0L, null);
55          ApplicationInfo b = new ApplicationInfo("App", "1.0", "Vendor", 0L, null);
56          ApplicationInfo c = new ApplicationInfo("Other", "1.0", "Vendor", 0L, null);
57          assertThat(a, is(b));
58          assertThat(a.hashCode(), is(b.hashCode()));
59          assertThat(a, is(not(c)));
60          assertThat(a, is(a));
61          assertThat(a, is(not((Object) null)));
62      }
63  
64      @Test
65      void testToString() {
66          ApplicationInfo app = new ApplicationInfo("MyApp", "2.0", "Acme", 0L, null);
67          assertThat(app.toString(), containsString("MyApp"));
68          assertThat(app.toString(), containsString("Acme"));
69      }
70  
71      /**
72       * The getters each document an empty string for a value the platform did not report, so the constructor normalizes
73       * rather than storing what it was handed. A Windows registry key with no DisplayVersion or Publisher used to reach
74       * callers as null from getVersion() or getVendor().
75       */
76      @Test
77      void testAbsentStringsAreNormalizedToEmpty() {
78          ApplicationInfo app = new ApplicationInfo(null, null, null, 0L, null);
79  
80          assertThat(app.getName(), is(""));
81          assertThat(app.getVersion(), is(""));
82          assertThat(app.getVendor(), is(""));
83      }
84  
85      @Test
86      void testPresentStringsArePassedThrough() {
87          ApplicationInfo app = new ApplicationInfo("App", "1.0", "Vendor", 0L, null);
88  
89          assertThat(app.getName(), is("App"));
90          assertThat(app.getVersion(), is("1.0"));
91          assertThat(app.getVendor(), is("Vendor"));
92      }
93  }
