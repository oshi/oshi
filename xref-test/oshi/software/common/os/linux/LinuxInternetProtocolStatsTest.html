66      // Verbatim /proc/net/tcp and tcp6 from a little-endian (aarch64) container, with 2001:db8:1122:3344::1 added to lo
67      // so the byte order is unambiguous. The matching `ss -tna` reported:
68      // CLOSE-WAIT 127.0.0.1:46143 127.0.0.1:4949
69      // LISTEN [2001:db8:1122:3344::1]:22 [::]:*
70      private static final List<String> TCP_LITTLE_ENDIAN = """

72                 0: 0100007F:B43F 0100007F:1355 08 00000000:00000000 00:00000000 00000000     0        0 11040717 2 0000000000000000 20 4 0 10 -1
73              """
74              .lines().toList();

96      private static String write(Path dir, String name, String contents) throws IOException {
97          return Files.writeString(dir.resolve(name), contents).toString();
98      }
99  
100     private static Map<String, Long> snmpStats(Path dir, String key) throws IOException {
101         Map<String, Long> stats = ProcUtil.parseNestedStatistics(write(dir, "snmp", SNMP), key).get(key);
102         assertNotNull(stats, () -> "no " + key + " statistics in the fixture");
103         return stats;
104     }

106     @Test
107     void testTcp4Stats(@TempDir Path dir) throws IOException {
108         TcpStats tcp = LinuxInternetProtocolStats.parseTcp4Stats(snmpStats(dir, "Tcp"));
109 
110         assertThat("connections established", tcp.getConnectionsEstablished(), is(2L));
111         assertThat("active opens", tcp.getConnectionsActive(), is(1892L));
112         assertThat("passive opens", tcp.getConnectionsPassive(), is(73L));
113         assertThat("attempt fails", tcp.getConnectionFailures(), is(1138L));
114         assertThat("established resets", tcp.getConnectionsReset(), is(14L));
115         assertThat("segments sent", tcp.getSegmentsSent(), is(61550L));
116         assertThat("segments received", tcp.getSegmentsReceived(), is(63131L));
117         assertThat("segments retransmitted", tcp.getSegmentsRetransmitted(), is(510L));
118         assertThat("in errors", tcp.getInErrors(), is(156L));
119         assertThat("out resets", tcp.getOutResets(), is(333L));
120     }

142     /** A kernel built without a protocol emits no counters for it, which must report as zeros rather than throw. */
143     @Test
144     void testStatsAbsent() {
145         TcpStats tcp = LinuxInternetProtocolStats.parseTcp4Stats(Map.of());
146         assertThat("connections established", tcp.getConnectionsEstablished(), is(0L));
147         assertThat("segments sent", tcp.getSegmentsSent(), is(0L));
148 
149         for (UdpStats udp : new UdpStats[] { LinuxInternetProtocolStats.parseUdp4Stats(Map.of()),
150                 LinuxInternetProtocolStats.parseUdp6Stats(Map.of()) }) {
151             assertThat("datagrams sent", udp.getDatagramsSent(), is(0L));
152             assertThat("datagrams received", udp.getDatagramsReceived(), is(0L));
153             assertThat("datagrams with no port", udp.getDatagramsNoPort(), is(0L));
154             assertThat("datagrams received with errors", udp.getDatagramsReceivedErrors(), is(0L));
155         }
156     }

158     @Test
159     void testConnectionsLittleEndian() {
160         List<IPConnection> tcp = LinuxInternetProtocolStats.parseConnections("tcp4", TCP_LITTLE_ENDIAN,
161                 Map.of(11040717L, 42), ByteOrder.LITTLE_ENDIAN);
162         assertThat("The header line should be skipped", tcp.size(), is(1));
163         IPConnection conn = tcp.get(0);
164         assertThat(conn.getType(), is("tcp4"));
165         assertThat(conn.getLocalAddress(), is(ParseUtil.parseIpv4AddressToBytes("127.0.0.1")));
166         assertThat(conn.getLocalPort(), is(46143));
167         assertThat(conn.getForeignAddress(), is(ParseUtil.parseIpv4AddressToBytes("127.0.0.1")));
168         assertThat(conn.getForeignPort(), is(4949));
169         assertThat(conn.getState(), is(TcpState.CLOSE_WAIT));
170         assertThat(conn.getowningProcessId(), is(42));
171 
172         IPConnection conn6 = LinuxInternetProtocolStats
173                 .parseConnections("tcp6", TCP6_LITTLE_ENDIAN, Map.of(), ByteOrder.LITTLE_ENDIAN).get(0);
174         assertThat(conn6.getLocalAddress(), is(ParseUtil.parseIpv6AddressToBytes("2001:db8:1122:3344::1")));
175         assertThat(conn6.getLocalPort(), is(22));
176         assertThat(conn6.getForeignAddress(), is(new byte[16]));
177         assertThat(conn6.getState(), is(TcpState.LISTEN));
178         assertThat(conn6.getowningProcessId(), is(-1));
179     }

181     @Test
182     void testConnectionsBigEndian() {
183         IPConnection conn = LinuxInternetProtocolStats
184                 .parseConnections("tcp4", TCP_BIG_ENDIAN, Map.of(), ByteOrder.BIG_ENDIAN).get(0);
185         assertThat(conn.getLocalAddress(), is(ParseUtil.parseIpv4AddressToBytes("148.100.75.242")));
186         assertThat(conn.getLocalPort(), is(22));
187         assertThat(conn.getForeignAddress(), is(ParseUtil.parseIpv4AddressToBytes("73.83.45.212")));
188         assertThat(conn.getForeignPort(), is(51011));
189         assertThat(conn.getState(), is(TcpState.ESTABLISHED));
190         assertThat("transmit queue", conn.getTransmitQueue(), is(0x48));
191 
192         IPConnection conn6 = LinuxInternetProtocolStats
193                 .parseConnections("tcp6", TCP6_BIG_ENDIAN, Map.of(), ByteOrder.BIG_ENDIAN).get(0);
194         assertThat(conn6.getLocalAddress(), is(ParseUtil.parseIpv6AddressToBytes("2620:91:0:657:f816:3eff:fe5f:2052")));
195         assertThat(conn6.getLocalPort(), is(50123));
196         assertThat(conn6.getState(), is(TcpState.LISTEN));
