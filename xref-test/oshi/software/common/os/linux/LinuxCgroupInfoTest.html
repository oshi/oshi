168     @Test
169     void interfaceDefaults() {
170         CgroupInfo defaults = new CgroupInfo() {
171         };
172         assertFalse(defaults.isContainerized());
173         assertEquals(0, defaults.getVersion());
174         assertEquals(UNLIMITED, defaults.getCpuQuota());
175         assertEquals(DEFAULT_CPU_PERIOD, defaults.getCpuPeriod());
176         assertEquals(0L, defaults.getCpuUsage());
177         assertEquals(UNLIMITED_CPUS, defaults.getEffectiveCpus());
178         assertEquals(UNLIMITED_MEMORY, defaults.getMemoryLimit());
179         assertEquals(0L, defaults.getMemoryUsage());
180         assertEquals(UNLIMITED, defaults.getPidLimit());
181         assertEquals(0L, defaults.getPidCurrent());
182     }

380     /**
381      * Test subclass that overrides version detection and path resolution to enable unit testing of dispatch logic.
382      */
383     static class TestableLinuxCgroupInfo extends LinuxCgroupInfo {
384 
385         private final int version;
386         private final String basePath;
387 
388         TestableLinuxCgroupInfo(int version, String basePath) {
389             this.version = version;
390             this.basePath = basePath;
391         }
392 
393         @Override
394         public int getVersion() {
395             return version;
396         }
397 
398         @Override
399         long readCpuUsageV2() {
400             return super.readCpuUsageV2(basePath);
401         }
402 
403         @Override
404         long readCpuUsageV1() {
405             return super.readCpuUsageV1(basePath);
406         }
407 
408         @Override
409         long readMemoryUsageV2() {
410             return super.readMemoryUsageV2(basePath);
411         }
412 
413         @Override
414         long readMemoryUsageV1() {
415             return super.readMemoryUsageV1(basePath);
416         }
417 
418         @Override
419         long readPidCurrentV2() {
420             return super.readPidCurrentV2(basePath);
421         }
422 
423         @Override
424         long readPidCurrentV1() {
425             return super.readPidCurrentV1(basePath);
426         }
