30      @Test
31      void testValidDpkgLineWithAllFields() {
32          List<String> lines = Collections
33                  .singletonList("firefox|120.0|amd64|250000|1700000000|Mozilla Team|firefox-source|https://firefox.com");
34  
35          List<ApplicationInfo> result = LinuxInstalledApps.parseLinuxAppInfo(lines);
36  
37          assertThat(result, hasSize(1));
38          ApplicationInfo app = result.get(0);
39          assertThat(app.getName(), is("firefox"));
40          assertThat(app.getVersion(), is("120.0"));
41          assertThat(app.getVendor(), is("Mozilla Team"));
42          // dpkg and rpm print epoch seconds; the API is milliseconds
43          assertThat(app.getTimestamp(), is(1700000000_000L));
44          assertThat(app.getAdditionalInfo().get("architecture"), is("amd64"));
45          assertThat(app.getAdditionalInfo().get("installedSize"), is("250000"));
46          assertThat(app.getAdditionalInfo().get("source"), is("firefox-source"));
47          assertThat(app.getAdditionalInfo().get("homepage"), is("https://firefox.com"));
48      }

137     @Test
138     void testParseApkInstalled() {
139         List<ApplicationInfo> apps = LinuxInstalledApps.parseApkInstalled(APK_DB.lines().toList());
140         assertThat(apps, hasSize(2));
141         ApplicationInfo base = apps.get(0);
142         assertThat(base.getName(), is("alpine-baselayout-data"));
143         assertThat(base.getVersion(), is("3.7.2-r1"));
144         assertThat(base.getVendor(), is("Natanael Copa <ncopa@alpinelinux.org>"));
145         assertThat("apk records no install time, and t: is the build time", base.getTimestamp(), is(0L));
146         assertThat(base.getAdditionalInfo().get("architecture"), is("aarch64"));
147         assertThat(base.getAdditionalInfo().get("installedSize"), is("19034"));
148         assertThat(base.getAdditionalInfo().get("source"), is("alpine-baselayout"));
149         // The last record has no trailing blank line
150         assertThat(apps.get(1).getName(), is("musl"));
151         assertThat(apps.get(1).getAdditionalInfo().get("homepage"), is("https://musl.libc.org/"));
152     }

195     @Test
196     void testParsePacmanDesc() {
197         ApplicationInfo bash = LinuxInstalledApps.parsePacmanDesc(PACMAN_DESC.lines().toList());
198         assertNotNull(bash);
199         assertThat(bash.getName(), is("bash"));
200         assertThat(bash.getVersion(), is("5.3.20-1"));
201         assertThat(bash.getVendor(), is("Tobias Powalowski <tpowa@archlinux.org>"));
202         assertThat("pacman prints seconds", bash.getTimestamp(), is(1793491200_000L));
203         assertThat(bash.getAdditionalInfo().get("architecture"), is("x86_64"));
204         assertThat(bash.getAdditionalInfo().get("installedSize"), is("10055840"));
205         assertThat(bash.getAdditionalInfo().get("homepage"), is("https://www.gnu.org/software/bash/bash.html"));
206         assertThat(LinuxInstalledApps.parsePacmanDesc(List.of("%VERSION%", "1.0")), is(nullValue()));
207     }

209     @Test
210     void testQueryPackageDatabasesApk(@TempDir Path tmp) throws IOException {
211         Path installed = Files.writeString(tmp.resolve("installed"), APK_DB);
212         assertThat(LinuxInstalledApps.queryPackageDatabases(installed.toString(), tmp.resolve("absent").toString()),
213                 is(LinuxInstalledApps.parseApkInstalled(APK_DB.lines().toList())));

216     @Test
217     void testQueryPackageDatabasesPacman(@TempDir Path tmp) throws IOException {
218         Path local = Files.createDirectories(tmp.resolve("local"));
219         Files.writeString(Files.createDirectories(local.resolve("bash-5.3.20-1")).resolve("desc"), PACMAN_DESC);
220         // A package directory whose desc has no name is skipped, and pacman's version file is not a package
221         Files.writeString(Files.createDirectories(local.resolve("broken-1.0-1")).resolve("desc"), "%VERSION%\n1.0\n");
222         Files.writeString(local.resolve("ALPM_DB_VERSION"), "9\n");
223         List<ApplicationInfo> apps = LinuxInstalledApps.queryPackageDatabases(tmp.resolve("absent").toString(),
224                 local.toString());
225         assertThat(apps, hasSize(1));
226         assertThat(apps.get(0).getName(), is("bash"));
227     }
