7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.empty;
9   import static org.hamcrest.Matchers.hasSize;

12  import java.util.Collections;
13  import java.util.List;
14  
15  import org.junit.jupiter.api.Test;
16  
17  import oshi.software.os.ApplicationInfo;
18  
19  class SolarisInstalledAppsTest {
20  
21      @Test
22      void testParsePkgList() {
23          // Solaris 11.4 SPARC (cfarm216), then OpenIndiana from a vmactions VM
24          List<String> lines = """
25                  pkg://solaris/archiver/gnu-tar@1.35-11.4.95.0.1.227.0:20260810T173115Z       i--
26                  pkg://solaris/compliance/benchmark/ehc-solaris-policy@11.4-11.4.96.0.1.227.1:20261001T032447Z i--
27                  pkg://openindiana.org/SUNWcs@0.5.11-2026.0.0.23349:20261101T115717Z          i--
28                  pkg://openindiana.org/consolidation/userland/userland-incorporation@0.5.11-2026.0.0.35309:20261103T012511Z i--
29                  """
30                  .lines().toList();
31          List<ApplicationInfo> apps = SolarisInstalledApps.parsePkgList(lines);
32          assertThat(apps, hasSize(4));
33          ApplicationInfo tar = apps.get(0);
34          assertThat(tar.getName(), is("archiver/gnu-tar"));
35          assertThat(tar.getVersion(), is("1.35-11.4.95.0.1.227.0"));
36          assertThat(tar.getVendor(), is("solaris"));
37          // 2026-08-10T17:31:15Z
38          assertThat(tar.getTimestamp(), is(1_786_383_075_000L));
39          assertThat(tar.getAdditionalInfo().get("state"), is("i--"));
40          assertThat(tar.getAdditionalInfo().get("fmri"),
41                  is("pkg://solaris/archiver/gnu-tar@1.35-11.4.95.0.1.227.0:20260810T173115Z"));
42          assertThat(apps.get(2).getName(), is("SUNWcs"));
43          assertThat(apps.get(2).getVendor(), is("openindiana.org"));
44          assertThat(apps.get(3).getName(), is("consolidation/userland/userland-incorporation"));
45      }
46  
47      @Test
48      void testParsePkgListMalformed() {
49          assertThat(SolarisInstalledApps.parsePkgList(List.of("pkg list: no packages installed", "")), is(empty()));
50          assertThat(SolarisInstalledApps.parsePkgList(Collections.emptyList()), is(empty()));
51      }
52  }
