5   package oshi.software.common.os.unix.bsd;
6   
7   import static org.junit.jupiter.api.Assertions.assertFalse;
8   import static org.junit.jupiter.api.Assertions.assertTrue;
9   
10  import org.junit.jupiter.api.Test;
11  
12  class BsdOperatingSystemTest {
13  
14      @Test
15      void testIsRcService() {
16          // Services, from NetBSD 10.1's /etc/rc.d, including one with a suffix and one with a digit
17          assertTrue(BsdOperatingSystem.isRcService("sshd", true));
18          assertTrue(BsdOperatingSystem.isRcService("bootconf.sh", true));
19          assertTrue(BsdOperatingSystem.isRcService("dhcpd6", true));
20          // rcorder placeholders, from NetBSD and FreeBSD
21          for (String placeholder : new String[] { "CRITLOCALMOUNTED", "DAEMON", "DISKS", "LOGIN", "NETWORKING",
22                  "SERVERS", "FILESYSTEMS" }) {
23              assertFalse(BsdOperatingSystem.isRcService(placeholder, true), placeholder);
24          }
25          // OpenBSD's rc.subr, a library the scripts source
26          assertFalse(BsdOperatingSystem.isRcService("rc.subr", false));
27      }
28  }
