24      @Test
25      void testParseServices() {
26          // lssrc -a: header row, active subsystems (with/without a Group column), and an inoperative one
27          List<OSService> services = AixOperatingSystem.parseServices(List.of(//
28                  "Subsystem         Group            PID          Status", //
29                  " syslogd          ras              4194320      active", //
30                  " inetd            5218374          active", // 3-token active form (no group column)
31                  " portmap          portmap                       inoperative"));
32          Map<String, OSService> byName = services.stream()
33                  .collect(Collectors.toMap(OSService::getName, Function.identity()));
34  
35          OSService syslogd = byName.get("syslogd");
36          OSService inetd = byName.get("inetd");
37          OSService portmap = byName.get("portmap");
38          assertNotNull(syslogd);
39          assertNotNull(inetd);
40          assertNotNull(portmap);
41          assertThat(syslogd.getProcessID(), is(4194320));
42          assertThat(syslogd.getState(), is(OSService.State.RUNNING));
43          assertThat(inetd.getProcessID(), is(5218374));
44          assertThat(inetd.getState(), is(OSService.State.RUNNING));
45          assertThat(portmap.getProcessID(), is(0));
46          assertThat(portmap.getState(), is(OSService.State.STOPPED));
47      }
48  
49      @Test
50      void testParseServicesHeaderOnlyOrEmpty() {
51          assertThat(AixOperatingSystem.parseServices(List.of("Subsystem Group PID Status")), is(empty()));
52          assertThat(AixOperatingSystem.parseServices(Collections.emptyList()), is(empty()));
53      }
54  
55      @Test
56      void testResolveBootTimeMillis() {
57          long now = 1_600_000_000_000L;
58          long whoBootTime = 1_500_000_000_000L;
59          long upTime = 10_000_000L;
60  
61          // The up time duration is unambiguous, so it wins even when who -b also parsed
62          assertThat(AixOperatingSystem.resolveBootTimeMillis(whoBootTime, upTime, now), is(now - upTime));
63          assertThat(AixOperatingSystem.resolveBootTimeMillis(0L, upTime, now), is(now - upTime));
64          // who -b is consulted only when the uptime command gave nothing
65          assertThat(AixOperatingSystem.resolveBootTimeMillis(whoBootTime, 0L, now), is(whoBootTime));
66      }
67  
68      @Test
69      void testResolveBootTimeMillisTreatsZeroUpTimeAsUnknown() {
70          long now = 1_600_000_000_000L;
71          // who -b unparseable (its AIX output carries no year) and the uptime command failed. Reporting now - 0 would
72          // claim the system booted this instant and pin uptime to zero, so both failing must yield "unknown" instead.
73          assertThat(AixOperatingSystem.resolveBootTimeMillis(0L, 0L, now), is(0L));
74          // A who -b value below the 1000 ms floor is treated as unusable rather than as a boot time in 1970
75          assertThat(AixOperatingSystem.resolveBootTimeMillis(999L, 0L, now), is(0L));
76      }
