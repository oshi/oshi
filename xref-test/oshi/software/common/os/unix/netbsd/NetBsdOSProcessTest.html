27      @Test
28      void testParseEnvironment() {
29          byte[] env = "HOME=/root\0PATH=/bin:/usr/bin\0EMPTY=\0NOEQUALS\0=novalue\0EQ=a=b\0\0".getBytes(UTF_8);
30          Map<String, String> parsed = NetBsdOSProcess.parseEnvironment(env);
31          assertThat(parsed.keySet(), contains("HOME", "PATH", "EMPTY", "EQ"));
32          assertThat(parsed, hasEntry("HOME", "/root"));
33          assertThat(parsed, hasEntry("PATH", "/bin:/usr/bin"));
34          assertThat(parsed, hasEntry("EMPTY", ""));
35          assertThat(parsed, hasEntry("EQ", "a=b"));
36      }

52      @Test
53      void testParseArgv0() {
54          // The comm column would have clipped this to "/opt/bo", the width of ps's COMMAND header
55          assertThat(NetBsdOSProcess.parseArgv0("/opt/bootstrap/bin/java -Xmx64m -cp /tmp/a.jar Main"),
56                  is("/opt/bootstrap/bin/java"));
57          assertThat(NetBsdOSProcess.parseArgv0("/usr/sbin/sshd"), is("/usr/sbin/sshd"));
58          // Started through PATH, so argv[0] is a bare name rather than a path
59          assertThat(NetBsdOSProcess.parseArgv0("java /tmp/S.java"), is("java"));
60          assertThat(NetBsdOSProcess.parseArgv0("  /bin/sh -c true  "), is("/bin/sh"));
61          assertThat(NetBsdOSProcess.parseArgv0(null), is(""));
62          assertThat(NetBsdOSProcess.parseArgv0(""), is(""));
63      }

79          assertTrue(NetBsdOSProcess.isPlaceholder("(java)"));
80          assertTrue(NetBsdOSProcess.isPlaceholder("(sshd)"));

95      @Test
96      void testThreadNameIgnoresArgsPlaceholder() {
97          // Two readings of one thread, seconds apart on NetBSD 10.1 CI: once the kernel released the arguments, once
98          // the JVM was inside posix_spawn and ps printed its placeholder instead
99          Map<BsdPsThreadKeyword, String> released = new EnumMap<>(BsdPsThreadKeyword.class);
100         released.put(BsdPsThreadKeyword.LID, "1");
101         released.put(BsdPsThreadKeyword.ARGS, "/opt/bootstrap/bin/java -enableassertions -classpath /tmp/a.jar Main");
102         Map<BsdPsThreadKeyword, String> refused = new EnumMap<>(released);
103         refused.put(BsdPsThreadKeyword.ARGS, "(java)");
104         assertThat(new NetBsdOSThread(42, released).getName(), is("java"));
105         assertThat(new NetBsdOSThread(42, refused).getName(), is("java"));
106     }

108     @Test
109     void testThreadReadsItsOwnCpuTime() {
110         // ltime is the LWP's own CPU time; NetBSD's ps has no per-LWP faults or context switches to read
111         Map<BsdPsThreadKeyword, String> row = new EnumMap<>(BsdPsThreadKeyword.class);
112         row.put(BsdPsThreadKeyword.LID, "3");
113         row.put(BsdPsThreadKeyword.STATE, "S");
114         row.put(BsdPsThreadKeyword.ETIME, "01:00");
115         row.put(BsdPsThreadKeyword.LTIME, "0:02.50");
116         row.put(BsdPsThreadKeyword.PRI, "43");
117         row.put(BsdPsThreadKeyword.ARGS, "java");
118         NetBsdOSThread thread = new NetBsdOSThread(42, row);
119         assertThat(thread.getUserTime(), is(2500L));
120         assertThat(thread.getKernelTime(), is(0L));
121         assertThat(thread.getMinorFaults(), is(0L));
122         assertThat(thread.getMajorFaults(), is(0L));
123         assertThat(thread.getContextSwitches(), is(0L));
124         assertThat(NetBsdOSThread.PS_THREAD_COLUMNS, is("lid,state,etime,ltime,pri,args"));
125     }

127     // A process whose kernel answers come from a map keyed by kern.proc_args subtype, absent meaning no answer
128     private static NetBsdOSProcess stubProcess(String args, Map<Integer, String> kernelPaths) {
129         Map<BsdPsKeyword, String> psMap = new EnumMap<>(BsdPsKeyword.class);
130         psMap.put(BsdPsKeyword.PID, "42");
131         psMap.put(BsdPsKeyword.ARGS, args);
132         return new NetBsdOSProcess(42, psMap, new NetBsdOperatingSystem()) {
133             @Override
134             protected String queryProcArgsPath(int subtype) {
135                 return kernelPaths.getOrDefault(subtype, "");
136             }
137 
138             @Override
139             protected Map<BsdPsKeyword, String> queryPsRow() {
140                 // The fixture row has no counters, so it reads as torn; never re-read it from the host's ps
141                 return Collections.emptyMap();
142             }
143         };
144     }

146     @Test
147     void testPathAndCwdFromKernel() {
148         // sshd rewrites its arguments, so argv[0] is no path; the name still comes from argv[0], as its threads' does
149         NetBsdOSProcess proc = stubProcess("sshd: oshi@pts/0", Map.of(NetBsdOSProcess.KERN_PROC_PATHNAME,
150                 "/usr/sbin/sshd", NetBsdOSProcess.KERN_PROC_CWD, "/var/chroot/sshd"));
151         assertThat(proc.getPath(), is("/usr/sbin/sshd"));
152         assertThat(proc.getName(), is("sshd:"));
153         assertThat(proc.getCurrentWorkingDirectory(), is("/var/chroot/sshd"));
154     }

163     @Test
164     void testParseKernelUserTime() {
165         // cfarm428 (NetBSD 10.1): init, with 3 ticks of user and 61 of system time
166         String init = "1 (init) S 0 1 1 0 -1 16517 197 2054930146 64 1217734 3 61 117112575 9651110 85 0 1 0 1 2";
167         assertThat(NetBsdOSProcess.parseKernelUserTime(init), is(new long[] { 610L, 30L }));
168         // A busy shell loop on cfarm428 read 403 ticks of user time, the 4.02 s that ps showed; other fields filler
169         String busy = "6130 (sh) R 1 6130 6130 0 -1 16516 0 0 0 0 403 0 0 0 85 0 1 0 1 2";
170         assertThat(NetBsdOSProcess.parseKernelUserTime(busy), is(new long[] { 0L, 4030L }));
171         // Fields are counted from the last closing parenthesis, whatever the command name holds
172         String odd = "42 (a) b (c)) S 1 42 42 0 -1 0 0 0 0 0 7 5 0 0 85 0 1 0 1 2";
173         assertThat(NetBsdOSProcess.parseKernelUserTime(odd), is(new long[] { 50L, 70L }));
174         // procfs not mounted, or the process exited
175         assertThat(NetBsdOSProcess.parseKernelUserTime("").length, is(0));
176         assertThat(NetBsdOSProcess.parseKernelUserTime("42 (short) S 1 2 3").length, is(0));
177     }

179     @Test
180     void testIsTornRow() {
181         Map<BsdPsKeyword, String> row = new EnumMap<>(BsdPsKeyword.class);
182         row.put(BsdPsKeyword.STATE, "Sl");
183         row.put(BsdPsKeyword.CPUTIME, "0:00.00");
184         row.put(BsdPsKeyword.MINFLT, "0");
185         row.put(BsdPsKeyword.NVCSW, "0");
186         row.put(BsdPsKeyword.NIVCSW, "0");
187         // A JVM read while one of its threads was inside posix_spawn
188         assertTrue(NetBsdOSProcess.isTornRow(row));
189         // Any one counter the kernel filled in marks a real row, even one that has used no measurable CPU
190         row.put(BsdPsKeyword.MINFLT, "197");
191         assertFalse(NetBsdOSProcess.isTornRow(row));
192         row.put(BsdPsKeyword.MINFLT, "0");
193         row.put(BsdPsKeyword.NVCSW, "4");
194         assertFalse(NetBsdOSProcess.isTornRow(row));
195         row.put(BsdPsKeyword.NVCSW, "0");
196         row.put(BsdPsKeyword.CPUTIME, "0:00.01");
197         assertFalse(NetBsdOSProcess.isTornRow(row));
198         // A zombie has no counters to report
199         row.put(BsdPsKeyword.CPUTIME, "0:00.00");
200         row.put(BsdPsKeyword.STATE, "Z");
201         assertFalse(NetBsdOSProcess.isTornRow(row));
202     }
