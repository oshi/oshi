22      /**
23       * Builds a structure of {@code uint64_t} counters in which each counter holds its own index plus 1000, so a value
24       * read from the wrong offset is recognizable.
25       */
26      private static byte[] counters(int size) {
27          ByteBuffer bb = ByteBuffer.allocate(size).order(ByteOrder.nativeOrder());
28          for (int i = 0; i < size / Long.BYTES; i++) {
29              bb.putLong(i * Long.BYTES, 1000L + i);
30          }
31          return bb.array();
32      }

34      // Sizes: 10.x and 11.x; 12.x and 13.x; 14.x; 15.0; 16-CURRENT
35      @Test
36      void testParseTcpstat() {
37          assertTcpstat(880, 0);
38          assertTcpstat(904, 0);
39          assertTcpstat(984, 1);
40          assertTcpstat(1000, 1);
41          assertTcpstat(1048, 1);
42      }
43  
44      // Counter indexes from sys/netinet/tcp_var.h; shift is 1 from FreeBSD 14.0's tcps_progdrops onward
45      private static void assertTcpstat(int size, int shift) {
46          TcpStats stats = FreeBsdInternetProtocolStats.parseTcpstat(counters(size), 42L);
47          assertNotNull(stats);
48          assertThat(stats.getConnectionsEstablished(), is(42L));
49          assertThat(stats.getConnectionsActive(), is(1000L)); // tcps_connattempt
50          assertThat(stats.getConnectionsPassive(), is(1001L)); // tcps_accepts
51          assertThat(stats.getConnectionsReset(), is(1003L)); // tcps_drops
52          assertThat(stats.getConnectionFailures(), is(1004L)); // tcps_conndrops
53          assertThat(stats.getSegmentsSent(), is(1017L + shift)); // tcps_sndpack
54          assertThat(stats.getSegmentsRetransmitted(), is(1019L + shift)); // tcps_sndrexmitpack
55          assertThat(stats.getSegmentsReceived(), is(1028L + shift)); // tcps_rcvpack
56          // tcps_rcvbadsum + tcps_rcvbadoff + tcps_rcvshort
57          assertThat(stats.getInErrors(), is(1030L + 1031L + 1033L + 3L * shift));
58          assertThat(stats.getOutResets(), is(0L));
59      }

61      // FreeBSD 9 and earlier (u_long fields), or a size between the known layouts
62      @Test
63      void testParseTcpstatUnknownLayout() {
64          for (int size : new int[] { 0, 440, 872, 912, 976 }) {
65              assertNull(FreeBsdInternetProtocolStats.parseTcpstat(counters(size), 0L), "size " + size);
66          }
67      }
68  
69      @Test
70      void testParseUdpstat() {
71          UdpStats stats = FreeBsdInternetProtocolStats.parseUdpstat(counters(112));
72          assertNotNull(stats);
73          assertThat(stats.getDatagramsReceived(), is(1000L)); // udps_ipackets
74          assertThat(stats.getDatagramsSent(), is(1010L)); // udps_opackets
75          // udps_noport + udps_noportbcast + udps_noportmcast
76          assertThat(stats.getDatagramsNoPort(), is(1005L + 1006L + 1012L));
77          // udps_hdrops + udps_badsum + udps_badlen
78          assertThat(stats.getDatagramsReceivedErrors(), is(1001L + 1002L + 1004L));
79      }
80  
81      @Test
82      void testParseUdpstatTooShort() {
83          assertNull(FreeBsdInternetProtocolStats.parseUdpstat(counters(56)));
84          assertNull(FreeBsdInternetProtocolStats.parseUdpstat(new byte[0]));
85      }
