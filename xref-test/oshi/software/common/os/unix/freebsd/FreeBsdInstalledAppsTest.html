7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.empty;
9   import static org.hamcrest.Matchers.hasSize;
10  import static org.hamcrest.Matchers.is;
11  
12  import java.util.Collections;
13  import java.util.List;
14  
15  import org.junit.jupiter.api.Test;
16  
17  import oshi.software.os.ApplicationInfo;
18  
19  class FreeBsdInstalledAppsTest {
20  
21      @Test
22      void testParsePkgQuery() {
23          // cfarm430 (FreeBSD 16.0-CURRENT, pkgbase), cfarm152 (DragonFly 6.4), then a comment holding the separator
24          List<String> lines = """
25                  FreeBSD-acct|16.snap20261103180419|1794221260|base/FreeBSD-acct|116347|FreeBSD:16:amd64|https://www.FreeBSD.org|re@FreeBSD.org|System resource accounting
26                  autoconf-switch|20220527|1773585777|devel/autoconf-switch|524|dragonfly:6.4:*|UNKNOWN|tijl@FreeBSD.org|Wrapper script to switch between autoconf versions
27                  foo|1.0|1700000000|misc/foo|10|FreeBSD:14:amd64|https://example.org|me@example.org|pipes | in | comments
28                  """
29                  .lines().toList();
30          List<ApplicationInfo> apps = FreeBsdInstalledApps.parsePkgQuery(lines);
31          assertThat(apps, hasSize(3));
32          ApplicationInfo acct = apps.get(0);
33          assertThat(acct.getName(), is("FreeBSD-acct"));
34          assertThat(acct.getVersion(), is("16.snap20261103180419"));
35          assertThat(acct.getVendor(), is("re@FreeBSD.org"));
36          assertThat(acct.getTimestamp(), is(1_794_221_260_000L));
37          assertThat(acct.getAdditionalInfo().get("architecture"), is("FreeBSD:16:amd64"));
38          assertThat(acct.getAdditionalInfo().get("installedSize"), is("116347"));
39          assertThat(acct.getAdditionalInfo().get("source"), is("base/FreeBSD-acct"));
40          assertThat(acct.getAdditionalInfo().get("homepage"), is("https://www.FreeBSD.org"));
41          assertThat(acct.getAdditionalInfo().get("description"), is("System resource accounting"));
42          assertThat(apps.get(1).getAdditionalInfo().get("homepage"), is("UNKNOWN"));
43          assertThat(apps.get(2).getAdditionalInfo().get("description"), is("pipes | in | comments"));
44      }
45  
46      @Test
47      void testParsePkgQueryMalformed() {
48          assertThat(FreeBsdInstalledApps.parsePkgQuery(List.of("pkg: not found", "", "a|b|c")), is(empty()));
49          assertThat(FreeBsdInstalledApps.parsePkgQuery(Collections.emptyList()), is(empty()));
50      }
51  }
