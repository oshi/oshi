21  /**
22   * Tests the Windows routing table interpretation, which lives in {@code oshi-common} so that it can be exercised off
23   * Windows. The backends supply only the raw rows, which the stub below fabricates.
24   */
25  class WindowsNetworkParamsTest {
26  
27      /** A params instance whose IP Helper API rows are supplied by the test. */
28      private static final class StubParams extends WindowsNetworkParams {
29          private final List<WindowsNetworkParams.RouteRow> rows;
30          private final List<byte[]> dnsServers;
31  
32          StubParams(List<WindowsNetworkParams.RouteRow> rows) {
33              this(rows, List.of());
34          }
35  
36          StubParams(List<WindowsNetworkParams.RouteRow> rows, List<byte[]> dnsServers) {
37              this.rows = rows;
38              this.dnsServers = dnsServers;
39          }
40  
41          @Override
42          protected List<WindowsNetworkParams.RouteRow> queryRouteRows() {
43              return this.rows;
44          }
45  
46          /** Return codes for successive calls; once used up, every call succeeds. */
47          private final Deque<Integer> returnCodes = new ArrayDeque<>();
48          private final List<Integer> bufferSizes = new ArrayList<>();
49  
50          StubParams withReturnCodes(Integer... codes) {
51              returnCodes.addAll(List.of(codes));
52              return this;
53          }
54  
55          @Override
56          protected int queryAdapterDnsServers(int bufferSize, int[] requiredSize, List<byte[]> addresses) {
57              bufferSizes.add(bufferSize);
58              int ret = returnCodes.isEmpty() ? 0 : returnCodes.poll();
59              // A too-small buffer reports a larger one; a failed call may still have written partial results
60              requiredSize[0] = bufferSize + 1000;
61              addresses.addAll(this.dnsServers);
62              return ret;
63          }
64      }
65  
66      private static WindowsNetworkParams.RouteRow row(byte[] destination, int prefixLength, byte[] nextHop,
67              int interfaceIndex, long metric) {
68          WindowsNetworkParams.RouteRow r = new WindowsNetworkParams.RouteRow();
69          r.destination = destination;
70          r.prefixLength = prefixLength;
71          r.nextHop = nextHop;
72          r.interfaceIndex = interfaceIndex;
73          r.metric = metric;
74          return r;
75      }

77      @Test
78      void testAnUnspecifiedNextHopIsNotAGateway() {
79          // Windows publishes no flags column, so an on-link route is recognized by its all-zero next hop
80          List<IPRoute> routes = new StubParams(
81                  List.of(row(new byte[] { 10, 0, 0, 0 }, 24, new byte[] { 0, 0, 0, 0 }, 5, 256L))).getRoutes();
82          assertThat(routes, hasSize(1));
83          assertThat(routes.get(0).isGateway(), is(false));
84          assertThat("A non-gateway route reports no gateway address", routes.get(0).getGateway(), is(new byte[0]));
85          assertThat(routes.get(0).getMetric(), is(256L));
86      }

88      @Test
89      void testARealNextHopIsAGateway() {
90          List<IPRoute> routes = new StubParams(
91                  List.of(row(new byte[] { 0, 0, 0, 0 }, 0, new byte[] { -64, -88, 1, 1 }, 5, 25L))).getRoutes();
92          assertThat(routes.get(0).isGateway(), is(true));
93          assertThat(routes.get(0).getGateway(), is(new byte[] { -64, -88, 1, 1 }));
94          assertThat(routes.get(0).getPrefixLength(), is(0));
95          assertThat(routes.get(0).isHost(), is(false));
96      }

98      @Test
99      void testHostRoutesAreRecognizedByPrefixWidthInBothFamilies() {
100         byte[] v6 = new byte[16];
101         v6[15] = 1;
102         List<IPRoute> routes = new StubParams(
103                 List.of(row(new byte[] { 127, 0, 0, 1 }, 32, new byte[] { 0, 0, 0, 0 }, 1, 331L),
104                         row(v6, 128, new byte[16], 1, 331L))).getRoutes();
105         assertThat(routes, hasSize(2));
106         assertThat("A /32 IPv4 route is a host route", routes.get(0).isHost(), is(true));
107         assertThat("A /128 IPv6 route is a host route", routes.get(1).isHost(), is(true));
108         assertThat(routes.get(1).getDestination().length, is(16));
109     }

111     @Test
112     void testARowWithAnUnrecognizedFamilyIsDropped() {
113         // The backends report an empty array for a SOCKADDR_INET family that is neither AF_INET nor AF_INET6.
114         // Dropping the row keeps the promise that a destination's length identifies the family.
115         List<WindowsNetworkParams.RouteRow> rows = new ArrayList<>();
116         rows.add(row(new byte[0], 0, new byte[0], 5, 0L));
117         rows.add(row(new byte[] { 10, 0, 0, 0 }, 8, new byte[0], 5, 0L));
118         List<IPRoute> routes = new StubParams(rows).getRoutes();
119         assertThat(routes, hasSize(1));
120         assertThat(routes.get(0).getDestination().length, is(4));
121     }

123     @Test
124     void testAnUnknownInterfaceIndexReportsAnEmptyName() {
125         // Index 999999 will not be present in this machine's interface list
126         List<IPRoute> routes = new StubParams(List.of(row(new byte[] { 10, 0, 0, 0 }, 8, new byte[0], 999999, 0L)))
127                 .getRoutes();
128         assertThat(routes.get(0).getInterfaceName(), is(""));

137     @Test
138     void testDnsServersBothFamiliesDeduplicatedWithoutPlaceholders() {
139         byte[] google6 = { 0x20, 0x01, 0x48, 0x60, 0x48, 0x60, 0, 0, 0, 0, 0, 0, 0, 0, (byte) 0x88, (byte) 0x88 };
140         byte[] placeholder = { (byte) 0xfe, (byte) 0xc0, 0, 0, 0, 0, (byte) 0xff, (byte) 0xff, 0, 0, 0, 0, 0, 0, 0, 1 };
141         byte[] router = { -64, -88, 1, 1 };
142         // Two adapters that are up: one with an IPv4 and an IPv6 server, one repeating the IPv4 server and
143         // reporting the placeholder Windows lists when no IPv6 server is configured
144         String[] servers = new StubParams(List.of(), List.of(router, google6, router.clone(), placeholder, new byte[3]))
145                 .getDnsServers();
146         assertThat(servers, is(new String[] { "192.168.1.1", "2001:4860:4860::8888" }));
147     }

149     @Test
150     void testUnconfiguredIpv6DnsPlaceholders() {
151         byte[] address = { (byte) 0xfe, (byte) 0xc0, 0, 0, 0, 0, (byte) 0xff, (byte) 0xff, 0, 0, 0, 0, 0, 0, 0, 3 };
152         assertThat(WindowsNetworkParams.isUnconfiguredIpv6Dns(address), is(true));
153         address[15] = 4;
154         assertThat(WindowsNetworkParams.isUnconfiguredIpv6Dns(address), is(false));
155         address[15] = 1;
156         address[7] = 0;
157         assertThat(WindowsNetworkParams.isUnconfiguredIpv6Dns(address), is(false));
158         assertThat(WindowsNetworkParams.isUnconfiguredIpv6Dns(new byte[] { -2, -64, 0, 1 }), is(false));
159     }

161     @Test
162     void testAdapterQueryGrowsTheBufferWhenTheListOutgrowsIt() {
163         byte[] router = { -64, -88, 1, 1 };
164         StubParams params = new StubParams(List.of(), List.of(router)).withReturnCodes(111, 111);
165         assertThat(params.getDnsServers(), is(new String[] { "192.168.1.1" }));
166         assertThat("Each retry passes the size the previous call reported", params.bufferSizes,
167                 is(List.of(15_000, 16_000, 17_000)));
168     }

170     @Test
171     void testAdapterQueryGivesUpAfterThreeOverflows() {
172         StubParams params = new StubParams(List.of(), List.of(new byte[] { -64, -88, 1, 1 })).withReturnCodes(111, 111,
173                 111);
174         assertThat(params.getDnsServers(), is(new String[0]));
175         assertThat(params.bufferSizes, hasSize(3));
176     }

178     @Test
179     void testAdapterQueryErrorsReturnNothing() {
180         // ERROR_NO_DATA (no adapters) and a real error: whatever a failed call wrote is discarded
181         List<byte[]> partial = List.of(new byte[] { -64, -88, 1, 1 });
182         assertThat(new StubParams(List.of(), partial).withReturnCodes(232).getDnsServers(), is(new String[0]));
183         assertThat(new StubParams(List.of(), partial).withReturnCodes(87).getDnsServers(), is(new String[0]));
