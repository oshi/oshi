5   package oshi.software.common.os.windows;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.Collections;
11  import java.util.Map;
12  
13  import org.junit.jupiter.api.Test;
14  
15  import oshi.driver.common.windows.registry.ThreadPerfCounterBlock;
16  
17  /**
18   * Tests the parsing seams of {@link WindowsOSProcess}.
19   */
20  class WindowsOSProcessTest {
21  
22      private static ThreadPerfCounterBlock thread(int tid, int pid, long contextSwitches) {
23          return new ThreadPerfCounterBlock("java/" + tid, tid, pid, 0L, 0L, 0L, 8, 5, 0, 0L, contextSwitches);
24      }
25  
26      @Test
27      void testContextSwitchesSumTheProcessThreads() {
28          // A thread map queried for all processes holds other processes' threads too
29          Map<Integer, ThreadPerfCounterBlock> threads = Map.of(1, thread(1, 100, 66L), 2, thread(2, 100, 11L), 3,
30                  thread(3, 100, 10L), 4, thread(4, 200, 1000L));
31          assertThat(WindowsOSProcess.sumContextSwitches(threads, 100), is(87L));
32          assertThat(WindowsOSProcess.sumContextSwitches(threads, 300), is(0L));
33          assertThat(WindowsOSProcess.sumContextSwitches(Collections.emptyMap(), 100), is(0L));
34      }
35  }
