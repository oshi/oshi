5   package oshi.software.common.os.windows;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import oshi.software.os.OperatingSystem.OSVersionInfo;
13  import oshi.util.tuples.Pair;
14  
15  class WindowsOperatingSystemTest {
16  
17      @Test
18      void testParseCodeNameSingleBitEnterprise() {
19          assertThat(WindowsOperatingSystem.parseCodeName(0x00000002), is("Enterprise"));
20      }
21  
22      @Test
23      void testParseCodeNameSingleBitHome() {
24          assertThat(WindowsOperatingSystem.parseCodeName(0x00000200), is("Home"));
25      }
26  
27      @Test
28      void testParseCodeNameMultipleBits() {
29          assertThat(WindowsOperatingSystem.parseCodeName(0x00000202), is("Enterprise,Home"));
30      }
31  
32      @Test
33      void testParseCodeNameZeroMask() {
34          assertThat(WindowsOperatingSystem.parseCodeName(0), is(""));
35      }
36  
37      @Test
38      void testParseCodeNameAllBitsSet() {
39          int allBits = 0x00000002 | 0x00000004 | 0x00000008 | 0x00000080 | 0x00000200 | 0x00000400 | 0x00002000
40                  | 0x00004000 | 0x00008000;
41          assertThat(WindowsOperatingSystem.parseCodeName(allBits),
42                  is("Enterprise,BackOffice,Communications Server,Datacenter,Home,Web Server,Storage Server,"
43                          + "Compute Cluster,Home Server"));
44      }
45  
46      @Test
47      void testResolveVersionAliasWindows11() {
48          assertThat("Build 22000 is the first Windows 11 build",
49                  WindowsOperatingSystem.resolveVersionAlias("10", "22000"), is("11"));
50          assertThat("Later builds are still Windows 11", WindowsOperatingSystem.resolveVersionAlias("10", "26100"),
51                  is("11"));
52          assertThat("Earlier builds remain Windows 10", WindowsOperatingSystem.resolveVersionAlias("10", "19045"),
53                  is("10"));
54      }
55  
56      @Test
57      void testResolveVersionAliasServer() {
58          assertThat("Build 17763 is Server 2019", WindowsOperatingSystem.resolveVersionAlias("Server 2016", "17763"),
59                  is("Server 2019"));
60          assertThat("Build 20348 is Server 2022", WindowsOperatingSystem.resolveVersionAlias("Server 2016", "20348"),
61                  is("Server 2022"));
62          assertThat("Build 26100 is Server 2025", WindowsOperatingSystem.resolveVersionAlias("Server 2016", "26100"),
63                  is("Server 2025"));
64          assertThat("Build 14393 remains Server 2016",
65                  WindowsOperatingSystem.resolveVersionAlias("Server 2016", "14393"), is("Server 2016"));
66      }
67  
68      @Test
69      void testResolveVersionAliasUnmapped() {
70          assertThat("A version with no alias is returned unchanged",
71                  WindowsOperatingSystem.resolveVersionAlias("8.1", "9600"), is("8.1"));
72      }
73  
74      @Test
75      void testParseVersionInfoStripsWindowsPrefix() {
76          Pair<String, OSVersionInfo> pair = WindowsOperatingSystem.parseVersionInfo("Windows 10", "", 0, "19045");
77          assertThat("Family is always Windows", pair.getA(), is("Windows"));
78          OSVersionInfo version = pair.getB();
79          assertThat("The \"Windows \" prefix is stripped", version.getVersion(), is("10"));
80          assertThat("An empty suite mask yields an empty code name", version.getCodeName(), is(""));
81          assertThat("The build number is passed through", version.getBuildNumber(), is("19045"));
82      }
83  
84      @Test
85      void testParseVersionInfoAbbreviatesServicePack() {
86          OSVersionInfo version = WindowsOperatingSystem
87                  .parseVersionInfo("Windows 7", "Service Pack 1", 0x00000200, "7601").getB();
88          assertThat("Service Pack is abbreviated to SP", version.getVersion(), is("7 SP1"));
89          assertThat("The suite mask is decoded", version.getCodeName(), is("Home"));
90      }
91  
92      @Test
93      void testParseVersionInfoIgnoresAbsentServicePack() {
94          assertThat("An empty service pack is not appended",
95                  WindowsOperatingSystem.parseVersionInfo("Windows 10", "", 0, "19045").getB().getVersion(), is("10"));
96          assertThat("An unknown service pack is not appended",
97                  WindowsOperatingSystem.parseVersionInfo("Windows 10", "unknown", 0, "19045").getB().getVersion(),
98                  is("10"));
99      }
100 
101     @Test
102     void testParseVersionInfoAppliesAlias() {
103         assertThat("A Windows 10 name with a Windows 11 build is aliased",
104                 WindowsOperatingSystem.parseVersionInfo("Windows 10", "", 0, "22631").getB().getVersion(), is("11"));
105     }
106 
107     @Test
108     void testParseVersionInfoWithoutWindowsPrefix() {
109         assertThat("A name without the prefix is used as-is",
110                 WindowsOperatingSystem.parseVersionInfo("Windows", "", 0, "").getB().getVersion(), is("Windows"));
111     }
112 }
