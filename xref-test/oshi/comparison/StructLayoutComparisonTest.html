17  import com.sun.jna.Structure;
18  
19  import oshi.ffm.platform.mac.MacSystem;
20  import oshi.jna.platform.mac.SystemB;

22  /**
23   * Asserts that the FFM struct layouts and the JNA {@link Structure} classes describing the same native struct agree on
24   * total size and on where each shared field sits.
25   *
26   * <p>
27   * The two bindings are written independently from the same headers, so a disagreement means one has drifted. This is a
28   * different check from {@code .github/ffm-audit}, which compares one binding against the real SDK headers and needs a C
29   * toolchain; this one catches drift between the pair and needs only a JVM.
30   *
31   * <p>
32   * Neither subsumes the other. Both bindings once agreed that {@code utmpx.ut_pad} was 16 bytes where the header says
33   * 64, which only the header audit could see; and the FreeBSD FFM binding orders {@code addrinfo} the BSD way while the
34   * shared JNA one uses the glibc order, which only a parity check can see.
35   */
36  class StructLayoutComparisonTest {
37  
38      /** JNA exposes field offsets only to subclasses, so each structure supplies its own accessor. */
39      private static void assertSameLayout(String what, MemoryLayout ffm, Structure jna, ToIntFunction<String> jnaOffset,
40              String... fields) {
41          jna.size(); // JNA lays the structure out lazily
42          assertThat(ffm.byteSize()).as("%s: total size, JNA vs FFM", what).isEqualTo((long) jna.size());
43          for (String field : fields) {
44              assertThat(ffm.byteOffset(groupElement(field))).as("%s.%s: offset, JNA vs FFM", what, field)
45                      .isEqualTo((long) jnaOffset.applyAsInt(field));
46          }
47      }
48  
49      /** Exposes JNA's protected offset accessor for the one structure compared below. */
50      private static final class ProbeUtmpx extends SystemB.MacUtmpx {
51          int offsetOf(String field) {
52              return fieldOffset(field);
53          }
54      }
55  
56      @Test
57      @EnabledOnOs(OS.MAC)
58      void testUtmpxAgrees() {
59          // ut_tv is a nested struct timeval in the JNA binding but two fields plus padding in the FFM one, so it is
60          // pinned by the fields either side of it rather than by name
61          ProbeUtmpx jna = new ProbeUtmpx();
62          assertSameLayout("utmpx", MacSystem.UTMPX, jna, jna::offsetOf, "ut_user", "ut_id", "ut_line", "ut_pid",
63                  "ut_type", "ut_host", "ut_pad");
64      }
65  }
