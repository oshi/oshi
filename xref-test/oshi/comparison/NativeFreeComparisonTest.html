37  /**
38   * Compares the native-free (NF) provider against the JNA provider on the platforms the NF provider supports, Linux and
39   * NetBSD, to verify that procfs/sysfs/command-line implementations produce the same results as native calls.
40   * <p>
41   * JNA is the baseline (stable, well-tested). NF values should match for deterministic fields and be within tolerance
42   * for dynamic values.
43   * <p>
44   * On NetBSD the comparison is only meaningful where pkgsrc's {@code java-jna} supplies {@code libjnidispatch}: without
45   * it the JNA provider falls back to the same command-line code the NF provider uses, and every assertion compares a
46   * class against itself.
47   */
48  @EnabledIf("isComparedPlatform")
49  class NativeFreeComparisonTest {
50  
51      private static HardwareAbstractionLayer jnaHal;
52      private static HardwareAbstractionLayer nfHal;
53      private static OperatingSystem jnaOs;
54      private static OperatingSystem nfOs;

99          } else {
100             assertThat(nfOs.getThreadId()).as("threadId").isEqualTo(jnaOs.getThreadId());

174     @Test
175     void currentProcess() {
176         int pid = jnaOs.getProcessId();
177         OSProcess jna = jnaOs.getProcess(pid);
178         OSProcess nf = nfOs.getProcess(pid);
179         for (int attempt = 0; attempt < 10 && isDegradedOnNetBsd(jna, nf); attempt++) {
180             Util.sleep(50);
181             jna = jnaOs.getProcess(pid);
182             nf = nfOs.getProcess(pid);
183         }
184         assertThat(jna).isNotNull();
185         assertThat(nf).isNotNull();
186         assertThat(nf.getProcessID()).isEqualTo(jna.getProcessID());
187         assertThat(nf.getName()).isEqualTo(jna.getName());
188         assertThat(nf.getPath()).isEqualTo(jna.getPath());
189         assertThat(nf.getUser()).isEqualTo(jna.getUser());
190         assertThat(nf.getUserID()).isEqualTo(jna.getUserID());
191         assertThat(nf.getGroup()).isEqualTo(jna.getGroup());
192         assertThat(nf.getGroupID()).isEqualTo(jna.getGroupID());
193         assertThat(nf.getParentProcessID()).isEqualTo(jna.getParentProcessID());
194         assertThat(nf.getCommandLine()).isEqualTo(jna.getCommandLine());
195         if (isNetBsd()) {
196             // NetBSD derives the start time as now minus the seconds-resolution ps elapsed time, and each provider
197             // reads its own now
198             assertThat(Math.abs(nf.getStartTime() - jna.getStartTime())).as("startTime (NetBSD tolerance)")
199                     .isLessThanOrEqualTo(2000L);
200         } else {
201             assertThat(nf.getStartTime()).isEqualTo(jna.getStartTime());
202         }
203     }

207     @Test
208     void diskStores() {
209         List<HWDiskStore> jna = jnaHal.getDiskStores();
210         List<HWDiskStore> nf = nfHal.getDiskStores();
211         assertThat(nf).hasSameSizeAs(jna);
212         Map<String, HWDiskStore> nfByName = nf.stream()
213                 .collect(Collectors.toMap(HWDiskStore::getName, Function.identity()));
214         for (HWDiskStore j : jna) {
215             HWDiskStore n = nfByName.get(j.getName());
216             assertThat(n).as("disk %s", j.getName()).isNotNull();
217             // Model may differ in space vs underscore: sysfs uses raw kernel string,
218             // udev may normalize spaces to underscores (or vice versa depending on device)
219             assertThat(n.getModel().replace('_', ' ')).isEqualTo(j.getModel().replace('_', ' '));
220             // Serial: udev ID_SERIAL_SHORT may expose serials not available in sysfs device/serial
221             if (!n.getSerial().isEmpty()) {
222                 assertThat(n.getSerial()).isEqualTo(j.getSerial());
223             }
224             assertThat(n.getSize()).isEqualTo(j.getSize());
225             assertThat(n.getReads()).as("reads(%s)", j.getName()).isGreaterThanOrEqualTo(j.getReads());
226             assertThat(n.getPartitions()).as("partitions(%s)", j.getName()).hasSameSizeAs(j.getPartitions());
227         }
228     }

287     static boolean isComparedPlatform() {
288         PlatformEnum p = PlatformEnum.getCurrentPlatform();
289         return p == PlatformEnum.LINUX || p == PlatformEnum.NETBSD;
290     }
291 
292     static boolean isNetBsd() {
293         return PlatformEnum.getCurrentPlatform() == PlatformEnum.NETBSD;
294     }
295 
296     /**
297      * Whether either snapshot was taken while the JVM was inside {@code posix_spawn}. NetBSD's kernel then reports the
298      * process as if it were a zombie and refuses to hand out its arguments, so {@code ps} leaves the memory fields zero
299      * and substitutes {@code (comm)} for the command. See the same check in {@link NativeComparisonTest}.
300      */
301     private static boolean isDegradedOnNetBsd(@Nullable OSProcess jna, @Nullable OSProcess nf) {
302         if (!isNetBsd()) {
303             return false;
304         }
305         if (jna == null || nf == null) {
306             return true;
307         }
308         return jna.getVirtualSize() == 0 || nf.getVirtualSize() == 0 || jna.getPath().isEmpty()
309                 || nf.getPath().isEmpty();
310     }
