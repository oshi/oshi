33  /**
34   * Compares the routing table read natively through a {@code NET_RT_DUMP} sysctl against the same table read by running
35   * {@code netstat}.
36   * <p>
37   * These are two readings of one kernel table, so they should agree. They are worth comparing because the native path
38   * has to reproduce conventions the command applies on its way to printing: a routing socket embeds an interface index
39   * inside link-local and scoped multicast addresses, and lists cache entries the table does not show. Each of those was
40   * caught here rather than by reading the headers.
41   */
42  @DisabledIf("isNotBsd")
43  class RouteComparisonTest {
44  
45      private static PlatformEnum platform;
46      private static String ipv4Command;
47      private static String ipv6Command;
48      private static int ifNameIndex;
49  
50      @BeforeAll
51      static void setUp() {
52          platform = PlatformEnum.getCurrentPlatform();
53          ipv4Command = "netstat -rn -f inet";
54          ipv6Command = "netstat -rn -f inet6";
55          // The column holding the interface name, which differs by platform
56          ifNameIndex = platform == PlatformEnum.OPENBSD ? 7 : 3;
57      }
58  
59      /** Only the platforms with a native routing table implementation. */
60      static boolean isNotBsd() {
61          PlatformEnum p = PlatformEnum.getCurrentPlatform();
62          return p != PlatformEnum.MACOS && p != PlatformEnum.FREEBSD && p != PlatformEnum.OPENBSD;
63      }
64  
65      /**
66       * A route's full identity. The gateway belongs in it: an address the walk reads at the wrong offset still yields a
67       * correct destination, since that is the first address in the message and no padding has been applied yet.
68       */
69      private static String key(IPRoute route) {
70          return destination(route) + " via " + ParseUtil.byteArrayToHexString(route.getGateway());
71      }
72  
73      /**
74       * A route's destination alone. The gateway comparison has to match routes on this rather than on the full key,
75       * which carries the gateway and so would only ever pair a route with one that already agrees.
76       */
77      private static String destination(IPRoute route) {
78          return ParseUtil.byteArrayToHexString(route.getDestination()) + "/" + route.getPrefixLength();
79      }
80  
81      private static Set<String> keys(List<IPRoute> routes) {
82          Set<String> keys = new TreeSet<>();
83          for (IPRoute route : routes) {
84              keys.add(key(route));
85          }
86          return keys;
87      }
88  
89      private static List<IPRoute> commandRoutes() {
90          Map<String, Integer> indexByName = new HashMap<>();
91          for (NetworkIF nif : new SystemInfo().getHardware().getNetworkIFs(true)) {
92              indexByName.put(nif.getName(), nif.getIndex());
93          }
94          return NetstatRoute.queryRoutes(ipv4Command, ipv6Command, ifNameIndex, indexByName);
95      }
96  
97      /** The kernel's dump, and the layout for reading it. */
98      private static byte[] nativeDump() {
99          switch (platform) {
100             case MACOS:
101                 return RouteDumpJNA.queryRouteDump();
102             case FREEBSD:
103                 return BsdRouteDumpJNA.queryRouteDump(FreeBsdLibc.INSTANCE);
104             default:
105                 return BsdRouteDumpJNA.queryRouteDump(OpenBsdLibc.INSTANCE);
106         }
107     }
108 
109     private static RouteTableDump.Layout layout() {
110         switch (platform) {
111             case MACOS:
112                 return RouteTableDump.Layout.MACOS;
113             case FREEBSD:
114                 return RouteTableDump.Layout.FREEBSD;
115             default:
116                 return RouteTableDump.Layout.OPENBSD;
117         }
118     }
119 
120     /**
121      * Reads the table natively. Deliberately not through {@code getRoutes()}, which falls back to running a command
122      * when the sysctl yields nothing -- that would compare the command against itself and pass without testing
123      * anything.
124      */
125     private static List<IPRoute> nativeRoutes() {
126         byte[] dump = nativeDump();
127         assertThat(dump).as("NET_RT_DUMP buffer").isNotEmpty();
128         Map<Integer, String> nameByIndex = new HashMap<>();
129         for (NetworkIF nif : new SystemInfo().getHardware().getNetworkIFs(true)) {
130             nameByIndex.put(nif.getIndex(), nif.getName());
131         }
132         List<IPRoute> routes = RouteTableDump.parse(dump, layout(), nameByIndex);
133         assertThat(routes).as("routes parsed from the dump").isNotEmpty();
134         return routes;
135     }

137     @Test
138     void testNativeAgreesWithCommand() {
139         List<IPRoute> nativeList = nativeRoutes();
140         List<IPRoute> commandList = commandRoutes();
141         assumeTrue(!commandList.isEmpty(), "netstat produced no routes to compare against");
142 
143         Set<String> nativeKeys = keys(nativeList);
144         Set<String> commandKeys = keys(commandList);
145         Set<String> shared = new TreeSet<>(nativeKeys);
146         shared.retainAll(commandKeys);
147 
148         // Routes come and go between the two readings, so require a large majority rather than equality. A parsing
149         // fault does not look like churn: it moves every address at once.
150         assertThat(shared.size())
151                 .as("destinations seen by both of %d native and %d from netstat", nativeKeys.size(), commandKeys.size())
152                 .isGreaterThanOrEqualTo((Math.min(nativeKeys.size(), commandKeys.size()) * 3) / 4);
153     }
154 
155     @Test
156     void testGatewaysAgree() {
157         List<IPRoute> nativeList = nativeRoutes();
158         List<IPRoute> commandList = commandRoutes();
159         assumeTrue(!commandList.isEmpty(), "netstat produced no routes to compare against");
160 
161         // A destination can appear more than once with different gateways, so keep every one rather than the last
162         Map<String, Set<String>> commandGateways = new HashMap<>();
163         for (IPRoute route : commandList) {
164             commandGateways.computeIfAbsent(destination(route), d -> new TreeSet<>())
165                     .add(ParseUtil.byteArrayToHexString(route.getGateway()));
166         }
167         // Only routes that carry a gateway say anything here. Most do not, and counting them dilutes a real
168         // disagreement to a few percent of the comparison
169         List<String> mismatches = new ArrayList<>();
170         int matched = 0;
171         for (IPRoute route : nativeList) {
172             Set<String> expected = commandGateways.get(destination(route));
173             String actual = ParseUtil.byteArrayToHexString(route.getGateway());
174             if (expected == null || (expected.size() == 1 && expected.contains("") && actual.isEmpty())) {
175                 continue;
176             }
177             matched++;
178             if (!expected.contains(actual)) {
179                 mismatches.add(destination(route));
180             }
181         }
182         assumeTrue(matched >= 4, "too few routes with a gateway to compare");
183         // A gateway rarely changes between two readings, and a misread address changes all of them
184         assertThat(mismatches).as("gateways disagreeing on %d routes carrying one", matched)
185                 .hasSizeLessThanOrEqualTo(matched / 4);
186     }

188     @Test
189     void testPublicApiUsesTheNativePath() {
190         // getRoutes() should return what the dump says, not what the command fallback would. These are two readings
191         // of a table that changes, so require a large majority rather than equality, as above
192         Set<String> viaApi = keys(new SystemInfo().getOperatingSystem().getNetworkParams().getRoutes());
193         assertThat(viaApi).as("routes from getRoutes()").isNotEmpty();
194         Set<String> direct = keys(nativeRoutes());
195         Set<String> shared = new TreeSet<>(viaApi);
196         shared.retainAll(direct);
197         assertThat(shared.size()).as("routes common to getRoutes() and a direct read")
198                 .isGreaterThanOrEqualTo(threeQuartersOf(Math.min(viaApi.size(), direct.size())));
199     }
200 
201     @Test
202     void testPrefixLengthsAreValid() {
203         // A prefix outside this range means the netmask was read from the wrong place. It is worth asserting
204         // separately from the comparison above, because a misread address array corrupts only the routes whose
205         // padding differs between the candidate units, which can still leave the two lists overlapping heavily.
206         for (IPRoute route : nativeRoutes()) {
207             int bits = route.getDestination().length * 8;
208             assertThat(route.getPrefixLength()).as("prefix length of %s", key(route)).isBetween(0, bits);
209         }
210     }
211 
212     /** Three quarters, rounded up, so a small set is not let off with less than the fraction asks for. */
213     private static int threeQuartersOf(int count) {
214         return (count * 3 + 3) / 4;
215     }
