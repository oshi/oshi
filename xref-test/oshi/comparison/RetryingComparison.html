5   package oshi.comparison;
6   
7   import java.lang.annotation.ElementType;
8   import java.lang.annotation.Retention;
9   import java.lang.annotation.RetentionPolicy;
10  import java.lang.annotation.Target;
11  
12  import org.junitpioneer.jupiter.RetryingTest;
13  
14  /**
15   * Runs a comparison test up to three times, 50 ms apart, retrying only on a failed assertion.
16   * <p>
17   * The comparisons take one snapshot from each implementation in sequence, so a value that moves between the two reads
18   * can push an otherwise correct pair past its bound. Retrying lets those bounds sit close to the deviation actually
19   * measured rather than far enough out to absorb a rare race: a check that fails one run in twenty fails about one in
20   * eight thousand with three attempts, as long as each attempt fails independently. A wrong-field or wrong-unit bug
21   * fails every attempt, so it is still caught. An exception other than an assertion failure is not retried.
22   */
23  @Target(ElementType.METHOD)
24  @Retention(RetentionPolicy.RUNTIME)
25  @RetryingTest(maxAttempts = 3, suspendForMs = 50, onExceptions = AssertionError.class)
26  public @interface RetryingComparison {
27  }
