31  /**
32   * Asserts the socket struct layouts against sizes and offsets taken from {@code <sys/proc_info.h>} on macOS 15
33   * (arm64/x86_64, which agree). These layouts are pure arithmetic over {@code MemoryLayout}, performing no native call,
34   * so this test runs on every platform and guards the mappings from a host that never executes them.
35   */
36  class MacSystemLayoutTest {
37  
38      @Test
39      void testProcFdInfoLayout() {
40          assertEquals(8, PROC_FD_INFO.byteSize());
41          assertEquals(0, PROC_FD_INFO.byteOffset(PROC_FD));
42          assertEquals(4, PROC_FD_INFO.byteOffset(PROC_FDTYPE));
43      }
44  
45      @Test
46      void testInSockInfoLayout() {
47          // The trailing insi_v6 is a 12-byte struct aligned to 4, not a run of bytes. Modelling it as nine bytes
48          // leaves in_sockinfo six short, which silently shifts every field of an enclosing struct.
49          assertEquals(80, IN_SOCK_INFO.byteSize());
50          assertEquals(0, IN_SOCK_INFO.byteOffset(INSI_FPORT));
51          assertEquals(4, IN_SOCK_INFO.byteOffset(INSI_LPORT));
52          assertEquals(24, IN_SOCK_INFO.byteOffset(INSI_VFLAG));
53          assertEquals(32, IN_SOCK_INFO.byteOffset(INSI_FADDR));
54          assertEquals(48, IN_SOCK_INFO.byteOffset(INSI_LADDR));
55      }
56  
57      @Test
58      void testTcpSockInfoLayout() {
59          assertEquals(120, TCP_SOCK_INFO.byteSize());
60          assertEquals(0, TCP_SOCK_INFO.byteOffset(TCPSI_INI));
61          assertEquals(80, TCP_SOCK_INFO.byteOffset(TCPSI_STATE));
62      }
63  
64      @Test
65      void testSocketInfoLayout() {
66          assertEquals(768, SOCKET_INFO.byteSize());
67          assertEquals(160, SOCKET_INFO.byteOffset(SOI_FAMILY));
68          assertEquals(170, SOCKET_INFO.byteOffset(SOI_QLEN));
69          assertEquals(172, SOCKET_INFO.byteOffset(SOI_INCQLEN));
70          assertEquals(232, SOCKET_INFO.byteOffset(SOI_KIND));
71          assertEquals(240, SOCKET_INFO.byteOffset(SOI_PROTO));
72      }
73  
74      @Test
75      void testSocketFdInfoLayout() {
76          assertEquals(792, SOCKET_FD_INFO.byteSize());
77          assertEquals(24, SOCKET_FD_INFO.byteOffset(PSI));
78      }
79  }
