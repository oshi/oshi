31                          vtnet0     - 192.168.0.0/16            192.168.4.30                     336987     -     -    23723765    260196     -    46815572     -

39          assertThat(stats, hasEntry("Oerrs", 1L));

41          assertThat(stats, hasEntry("Coll", 2L));

53          assertThat(stats, not(hasKey("Idrop")));

55          assertThat(stats, hasEntry("Obytes", 35_703_541_052L));
56          assertThat(stats, hasEntry("Ipkts", 0L));

76                  Name    Mtu   Network     Address               Ibytes     Obytes

83  
84          Map<String, Long> packets = BsdNetworkIF.parseNetstat("""

87                  """.lines().toList());

89          assertThat(packets, hasEntry("Ofail", 1L));

94                  lo0     32768 <Link>                          70038229   70038229
95                  """.lines().toList());
96          assertThat(loopback, hasEntry("Ibytes", 70_038_229L));

155     @Test
156     void testParseMediaSpeed() {
157         // FreeBSD 16.0-CURRENT, a virtio NIC and a Realtek one
158         assertThat(speed("media: Ethernet autoselect (10Gbase-T <full-duplex>)"), is(10_000_000_000L));
159         assertThat(speed("media: Ethernet autoselect (1000baseT <full-duplex>)"), is(1_000_000_000L));
160         // OpenBSD 7.9, with flow control after the duplex
161         assertThat(speed("media: Ethernet autoselect (1000baseT full-duplex,rxpause,txpause)"), is(1_000_000_000L));
162         // Other Ethernet spellings, and a wireless rate
163         assertThat(speed("media: Ethernet autoselect (2500Base-T <full-duplex>)"), is(2_500_000_000L));
164         assertThat(speed("media: Ethernet autoselect (100baseTX <full-duplex>)"), is(100_000_000L));
165         assertThat(speed("media: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)"), is(54_000_000L));
166         // Fixed media, with no parentheses
167         assertThat(speed("media: Ethernet 10Gbase-SR <full-duplex>"), is(10_000_000_000L));
168         // The active type, in parentheses, rather than the configured one before it
169         assertThat(speed("media: Ethernet 1000baseT <full-duplex> (100baseTX <full-duplex>)"), is(100_000_000L));
170         // No link: OpenBSD prints (none), FreeBSD leaves the parentheses off
171         assertThat(speed("media: Ethernet autoselect (none)"), is(0L));
172         assertThat(speed("media: Ethernet autoselect"), is(0L));
173         // Fixed media names its rate with nothing connected; the status says there is no link
174         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("em0: flags=8802<BROADCAST,SIMPLEX,MULTICAST> mtu 1500",
175                 "\tmedia: Ethernet 1000baseT <full-duplex>", "\tstatus: no carrier")), is(0L));
176         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("wlan0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500",
177                 "\tmedia: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)", "\tstatus: no network")), is(0L));
178         // A wireless interface that is up is "associated", not "active"
179         assertThat(
180                 BsdNetworkIF.parseMediaSpeed(List.of("wlan0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500",
181                         "\tmedia: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)", "\tstatus: associated")),
182                 is(54_000_000L));
183         // No media line at all, as for NetBSD's virtio interface
184         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("vioif0: flags=0x8843<UP,BROADCAST,RUNNING> mtu 1500")),
185                 is(0L));
186         assertThat(BsdNetworkIF.parseMediaSpeed(Collections.emptyList()), is(0L));
187     }

225     @Test
226     void testParseOperStatus() {
227         assertThat(BsdNetworkIF.parseOperStatus(FREEBSD_VTNET), is(IfOperStatus.UP));
228         // No media, so no status line: RUNNING decides
229         assertThat(BsdNetworkIF.parseOperStatus(FREEBSD_LO0), is(IfOperStatus.UP));
230         assertThat(BsdNetworkIF.parseOperStatus(FREEBSD_LO1_DOWN), is(IfOperStatus.DOWN));
231         assertThat(BsdNetworkIF.parseOperStatus(NETBSD_LO0), is(IfOperStatus.UP));
232         // The status line says active, but the interface is not UP
233         assertThat(BsdNetworkIF.parseOperStatus(NETBSD_LO1_DOWN), is(IfOperStatus.DOWN));
234         assertThat(BsdNetworkIF.parseOperStatus(OPENBSD_ENC0), is(IfOperStatus.DOWN));
235         assertThat(BsdNetworkIF.parseOperStatus(List.of("em0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500",
236                 "\tmedia: Ethernet autoselect (none)", "\tstatus: no carrier")), is(IfOperStatus.DOWN));
237         assertThat(
238                 BsdNetworkIF.parseOperStatus(
239                         List.of("wlan0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500", "\tstatus: associated")),
240                 is(IfOperStatus.UP));
241         assertThat(statusOf("down"), is(IfOperStatus.DOWN));
242         assertThat(statusOf("no network"), is(IfOperStatus.DOWN));
243         assertThat(statusOf("master"), is(IfOperStatus.UP));
244         assertThat(statusOf("backup"), is(IfOperStatus.DORMANT));
245         assertThat(statusOf("unknown"), is(IfOperStatus.UNKNOWN));
246         assertThat(statusOf("invalid"), is(IfOperStatus.UNKNOWN));
247         assertThat(BsdNetworkIF.parseOperStatus(Collections.emptyList()), is(IfOperStatus.UNKNOWN));
248     }
