23      @Test
24      void testParseNetstatFreeBsd() {
25          Map<String, Long> stats = BsdNetworkIF.parseNetstat(
26                  """
27                          Name     Mtu Network                   Address                           Ipkts Ierrs Idrop      Ibytes     Opkts Oerrs      Obytes  Coll
28                          vtnet0  9000 <Link#1>                  00:15:5d:c4:00:30              15029012     0     3  1610916560   1018784     1  1718869087     2
29                          vtnet0     - 192.168.0.0/16            192.168.4.30                     336987     -     -    23723765    260196     -    46815572     -
30                          """
31                          .lines().toList());
32          assertThat(stats, is(aMapWithSize(8)));
33          assertThat(stats, hasEntry("Ipkts", 15_029_012L));
34          assertThat(stats, hasEntry("Idrop", 3L));
35          assertThat(stats, hasEntry("Ibytes", 1_610_916_560L));
36          assertThat(stats, hasEntry("Opkts", 1_018_784L));
37          assertThat(stats, hasEntry("Oerrs", 1L));
38          assertThat(stats, hasEntry("Obytes", 1_718_869_087L));
39          assertThat(stats, hasEntry("Coll", 2L));
40      }

42      @Test
43      void testParseNetstatDragonFly() {
44          // No Idrop column; the virtio driver counts bytes but not packets on the link-level row
45          Map<String, Long> stats = BsdNetworkIF.parseNetstat("""
46                  Name    Mtu   Network       Address            Ipkts Ierrs     Ibytes    Opkts Oerrs     Obytes  Coll
47                  vtnet0  1500  <Link#1>    fa:16:3e:47:64:29        0     0 52967869005        0     0 35703541052     0
48                  vtnet0  1500  140.211.167.1 140-211-167-212 20457260     - 9971032506 32796892     - 35243602884     -
49                  """.lines().toList());
50          assertThat(stats, is(aMapWithSize(7)));
51          assertThat(stats, not(hasKey("Idrop")));
52          assertThat(stats, hasEntry("Ibytes", 52_967_869_005L));
53          assertThat(stats, hasEntry("Obytes", 35_703_541_052L));
54          assertThat(stats, hasEntry("Ipkts", 0L));
55      }

70      @Test
71      void testParseNetstatOpenBsd() {
72          // OpenBSD prints only the byte counts with -b, and the packet counts under other headings without it
73          Map<String, Long> bytes = BsdNetworkIF.parseNetstat("""
74                  Name    Mtu   Network     Address               Ibytes     Obytes
75                  bge2    1500  <Link>      74:86:7a:d0:13:c0  386813736  849298713
76                  bge2    1500  46.23.83.68 46.23.83.68        386813736  849298713
77                  """.lines().toList());
78          assertThat(bytes, is(aMapWithSize(2)));
79          assertThat(bytes, hasEntry("Ibytes", 386_813_736L));
80          assertThat(bytes, hasEntry("Obytes", 849_298_713L));
81  
82          Map<String, Long> packets = BsdNetworkIF.parseNetstat("""
83                  Name    Mtu   Network     Address              Ipkts Ifail    Opkts Ofail Colls
84                  bge2    1500  <Link>      74:86:7a:d0:13:c0  2025726     0  1065503     1     0
85                  """.lines().toList());
86          assertThat(packets, hasEntry("Ipkts", 2_025_726L));
87          assertThat(packets, hasEntry("Ofail", 1L));
88          assertThat(packets, hasEntry("Colls", 0L));
89  
90          Map<String, Long> loopback = BsdNetworkIF.parseNetstat("""
91                  Name    Mtu   Network     Address               Ibytes     Obytes
92                  lo0     32768 <Link>                          70038229   70038229
93                  """.lines().toList());
94          assertThat(loopback, hasEntry("Ibytes", 70_038_229L));
95      }

97      @Test
98      void testParseNetstatNetBsd() {
99          // Like OpenBSD, NetBSD prints only the byte counts with -b; without it, the packet counts and Colls
100         Map<String, Long> bytes = BsdNetworkIF.parseNetstat("""
101                 Name  Mtu   Network       Address               Ibytes     Obytes
102                 vioif 1500  <Link>        00:15:5d:c4:00:28 8419940203  684694279
103                 vioif 1500  fe80::/64     fe80::215:5dff:fe 8419940203  684694279
104                 """.lines().toList());
105         assertThat(bytes, is(aMapWithSize(2)));
106         assertThat(bytes, hasEntry("Ibytes", 8_419_940_203L));
107         assertThat(bytes, hasEntry("Obytes", 684_694_279L));
108 
109         Map<String, Long> packets = BsdNetworkIF.parseNetstat("""
110                 Name  Mtu   Network       Address              Ipkts Ierrs    Opkts Oerrs Colls
111                 vioif 1500  <Link>        00:15:5d:c4:00:28 106586638     1  2503607     0     0
112                 vioif 1500  fe80::/64     fe80::215:5dff:fe 106586638     1  2503607     0     0
113                 """.lines().toList());
114         assertThat(packets, is(aMapWithSize(5)));
115         assertThat(packets, hasEntry("Ipkts", 106_586_638L));
116         assertThat(packets, hasEntry("Ierrs", 1L));
117         assertThat(packets, hasEntry("Opkts", 2_503_607L));
118         assertThat(packets, hasEntry("Oerrs", 0L));
119         assertThat(packets, hasEntry("Colls", 0L));
120 
121         // The loopback and npflog link-level rows have no address
122         Map<String, Long> loopback = BsdNetworkIF.parseNetstat("""
123                 Name  Mtu   Network       Address               Ibytes     Obytes
124                 lo0   33624 <Link>                           178247481  178247481
125                 lo0   33624 127/8         127.0.0.1          178247481  178247481
126                 """.lines().toList());
127         assertThat(loopback, hasEntry("Ibytes", 178_247_481L));
128         assertThat(loopback, hasEntry("Obytes", 178_247_481L));
129         Map<String, Long> npflog = BsdNetworkIF.parseNetstat("""
130                 Name  Mtu   Network       Address              Ipkts Ierrs    Opkts Oerrs Colls
131                 npflo 0     <Link>                                 0     0  1988557     0     0
132                 """.lines().toList());
133         assertThat(npflog, hasEntry("Ipkts", 0L));
134         assertThat(npflog, hasEntry("Opkts", 1_988_557L));
135     }

148     private static long speed(String media) {
149         return BsdNetworkIF.parseMediaSpeed(List.of("vtnet0: flags=1008843<UP,BROADCAST,RUNNING> metric 0 mtu 9000",
150                 "\t" + media, "\tstatus: active"));
151     }

153     @Test
154     void testParseMediaSpeed() {
155         // FreeBSD 16.0-CURRENT, a virtio NIC and a Realtek one
156         assertThat(speed("media: Ethernet autoselect (10Gbase-T <full-duplex>)"), is(10_000_000_000L));
157         assertThat(speed("media: Ethernet autoselect (1000baseT <full-duplex>)"), is(1_000_000_000L));
158         // OpenBSD 7.9, with flow control after the duplex
159         assertThat(speed("media: Ethernet autoselect (1000baseT full-duplex,rxpause,txpause)"), is(1_000_000_000L));
160         // Other Ethernet spellings, and a wireless rate
161         assertThat(speed("media: Ethernet autoselect (2500Base-T <full-duplex>)"), is(2_500_000_000L));
162         assertThat(speed("media: Ethernet autoselect (100baseTX <full-duplex>)"), is(100_000_000L));
163         assertThat(speed("media: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)"), is(54_000_000L));
164         // Fixed media, with no parentheses
165         assertThat(speed("media: Ethernet 10Gbase-SR <full-duplex>"), is(10_000_000_000L));
166         // The active type, in parentheses, rather than the configured one before it
167         assertThat(speed("media: Ethernet 1000baseT <full-duplex> (100baseTX <full-duplex>)"), is(100_000_000L));
168         // No link: OpenBSD prints (none), FreeBSD leaves the parentheses off
169         assertThat(speed("media: Ethernet autoselect (none)"), is(0L));
170         assertThat(speed("media: Ethernet autoselect"), is(0L));
171         // Fixed media names its rate with nothing connected; the status says there is no link
172         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("em0: flags=8802<BROADCAST,SIMPLEX,MULTICAST> mtu 1500",
173                 "\tmedia: Ethernet 1000baseT <full-duplex>", "\tstatus: no carrier")), is(0L));
174         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("wlan0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500",
175                 "\tmedia: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)", "\tstatus: no network")), is(0L));
176         // A wireless interface that is up is "associated", not "active"
177         assertThat(
178                 BsdNetworkIF.parseMediaSpeed(List.of("wlan0: flags=8843<UP,BROADCAST,RUNNING> mtu 1500",
179                         "\tmedia: IEEE802.11 autoselect (OFDM/54Mbps mode 11g)", "\tstatus: associated")),
180                 is(54_000_000L));
181         // No media line at all, as for NetBSD's virtio interface
182         assertThat(BsdNetworkIF.parseMediaSpeed(List.of("vioif0: flags=0x8843<UP,BROADCAST,RUNNING> mtu 1500")),
183                 is(0L));
184         assertThat(BsdNetworkIF.parseMediaSpeed(Collections.emptyList()), is(0L));
185     }
