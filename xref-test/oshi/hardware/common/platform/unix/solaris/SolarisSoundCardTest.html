7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.empty;
9   import static org.hamcrest.Matchers.hasSize;
10  import static org.hamcrest.Matchers.is;
11  
12  import java.util.Collections;
13  import java.util.List;
14  
15  import org.junit.jupiter.api.Test;
16  
17  import oshi.hardware.SoundCard;
18  
19  class SolarisSoundCardTest {
20  
21      @Test
22      void testParseLshal() {
23          // Representative lshal output with a sound device
24          List<String> lshal = """
25                  udi = '/org/freedesktop/Hal/devices/pci_8086_293e'
26                    info.solaris.driver = 'audio810'  (string)
27                    info.product = 'ICH9 HD Audio Controller'  (string)
28                    info.vendor = 'Intel Corporation'  (string)
29  
30                  udi = '/org/freedesktop/Hal/devices/pci_8086_0001'
31                    info.solaris.driver = 'e1000g'  (string)
32                    info.product = 'Network Adapter'  (string)
33                    info.vendor = 'Intel Corporation'  (string)
34                  """.lines().toList();
35          List<SoundCard> cards = SolarisSoundCard.parseLshal(lshal);
36          assertThat(cards, hasSize(1));
37          SoundCard card = cards.get(0);
38          assertThat(card.getName(), is("Intel Corporation ICH9 HD Audio Controller"));
39          assertThat(card.getCodec(), is("ICH9 HD Audio Controller"));
40          assertThat(card.getDriverVersion(), is("ICH9 HD Audio Controller audio810"));
41      }
42  
43      @Test
44      void testParseLshalMultipleAudioDevices() {
45          List<String> lshal = """
46                  udi = '/org/freedesktop/Hal/devices/pci_8086_293e'
47                    info.solaris.driver = 'audio810'  (string)
48                    info.product = 'HD Audio'  (string)
49                    info.vendor = 'Intel'  (string)
50  
51                  udi = '/org/freedesktop/Hal/devices/pci_1002_aa38'
52                    info.solaris.driver = 'audio810'  (string)
53                    info.product = 'Radeon Audio'  (string)
54                    info.vendor = 'AMD'  (string)
55                  """.lines().toList();
56          List<SoundCard> cards = SolarisSoundCard.parseLshal(lshal);
57          assertThat(cards, hasSize(2));
58          assertThat(cards.get(0).getName(), is("Intel HD Audio"));
59          assertThat(cards.get(1).getName(), is("AMD Radeon Audio"));
60      }
61  
62      @Test
63      void testParseLshalEmpty() {
64          List<SoundCard> cards = SolarisSoundCard.parseLshal(Collections.emptyList());
65          assertThat(cards, is(empty()));
66      }
67  
68      @Test
69      void testParseLshalNoAudioDriver() {
70          // Devices exist but none have audio810 driver
71          List<String> lshal = """
72                  udi = '/org/freedesktop/Hal/devices/pci_8086_0001'
73                    info.solaris.driver = 'e1000g'  (string)
74                    info.product = 'Network Adapter'  (string)
75                    info.vendor = 'Intel Corporation'  (string)
76                  """.lines().toList();
77          List<SoundCard> cards = SolarisSoundCard.parseLshal(lshal);
78          assertThat(cards, is(empty()));
79      }
