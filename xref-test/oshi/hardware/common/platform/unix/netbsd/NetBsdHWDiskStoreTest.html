5   package oshi.hardware.common.platform.unix.netbsd;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.greaterThanOrEqualTo;
9   import static org.hamcrest.Matchers.is;
10  import static org.junit.jupiter.api.Assertions.assertFalse;
11  import static org.junit.jupiter.api.Assertions.assertTrue;
12  
13  import java.util.List;
14  
15  import org.junit.jupiter.api.Test;
16  
17  class NetBsdHWDiskStoreTest {
18  
19      /** Captured from {@code iostat -x -I} on NetBSD 10.1. */
20      private static final List<String> FIRST = """
21              device  read KB/t    xfr   time     MB   write KB/t    xfr   time     MB
22              ld0         26.36   2087   8.23    53.72      25.42  74725   8.23  1855.30
23              fd0          3.34    179  92.38     0.58       0.00      0  92.38     0.00
24              dk0         27.45   1995  14.32    53.47      25.42  74725  14.32  1855.30
25              cd0          0.00      0   0.00     0.00       0.00      0   0.00     0.00
26              """.lines().toList();
27  
28      /** The same drives two seconds later. The write KB/t figure has fallen while the transfer count has risen. */
29      private static final List<String> SECOND = """
30              device  read KB/t    xfr   time     MB   write KB/t    xfr   time     MB
31              ld0         26.36   2087   8.23    53.72      25.40  74833   8.23  1856.52
32              fd0          3.34    179  92.38     0.58       0.00      0  92.38     0.00
33              dk0         27.45   1995  14.32    53.47      25.40  74833  14.32  1856.52
34              cd0          0.00      0   0.00     0.00       0.00      0   0.00     0.00
35              """.lines().toList();
36  
37      private static NetBsdHWDiskStore store(String name, List<String> iostat) {
38          return new NetBsdHWDiskStore(name, "model", "serial", 1024L, () -> iostat);
39      }
40  
41      @Test
42      void testUpdateAttributes() {
43          NetBsdHWDiskStore ld0 = store("ld0", FIRST);
44          assertTrue(ld0.updateAttributes());
45          assertThat(ld0.getReads(), is(2087L));
46          assertThat(ld0.getWrites(), is(74725L));
47          // The MB columns, not KB-per-transfer times transfers
48          assertThat(ld0.getReadBytes(), is((long) (53.72 * 1024 * 1024)));
49          assertThat(ld0.getWriteBytes(), is((long) (1855.30 * 1024 * 1024)));
50          // One drive-busy total, not the two identical columns added together
51          assertThat(ld0.getTransferTime(), is(8230L));
52          assertThat(ld0.getTimeStamp(), greaterThanOrEqualTo(1L));
53      }
54  
55      @Test
56      void testBytesNeverGoBackwards() {
57          // iostat's write KB/t for ld0 rounds down from 25.42 to 25.40 between these two samples while the transfer
58          // count rises, so a KB/t times transfers reading would report fewer bytes written than a moment earlier
59          NetBsdHWDiskStore before = store("ld0", FIRST);
60          NetBsdHWDiskStore after = store("ld0", SECOND);
61          assertTrue(before.updateAttributes());
62          assertTrue(after.updateAttributes());
63          assertThat(after.getWrites(), greaterThanOrEqualTo(before.getWrites()));
64          assertThat(after.getWriteBytes(), greaterThanOrEqualTo(before.getWriteBytes()));
65          assertThat(after.getReadBytes(), greaterThanOrEqualTo(before.getReadBytes()));
66      }
67  
68      @Test
69      void testIdleAndMissingDrives() {
70          NetBsdHWDiskStore cd0 = store("cd0", FIRST);
71          assertTrue(cd0.updateAttributes());
72          assertThat(cd0.getReadBytes(), is(0L));
73          assertThat(cd0.getWriteBytes(), is(0L));
74          assertThat(cd0.getTransferTime(), is(0L));
75  
76          NetBsdHWDiskStore absent = store("sd9", FIRST);
77          assertFalse(absent.updateAttributes());
78      }
79  }
