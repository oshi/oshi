44      private static NetBsdHWDiskStore store(String name, List<String> iostat) {
45          return new NetBsdHWDiskStore(name, "model", "serial", 1024L, () -> iostat);
46      }
47  
48      @Test
49      void testUpdateAttributes() {
50          NetBsdHWDiskStore ld0 = store("ld0", FIRST);
51          assertTrue(ld0.updateAttributes());
52          assertThat(ld0.getReads(), is(2087L));
53          assertThat(ld0.getWrites(), is(74725L));
54          // The MB columns, not KB-per-transfer times transfers
55          assertThat(ld0.getReadBytes(), is((long) (53.72 * 1024 * 1024)));
56          assertThat(ld0.getWriteBytes(), is((long) (1855.30 * 1024 * 1024)));
57          // One drive-busy total, not the two identical columns added together
58          assertThat(ld0.getTransferTime(), is(8230L));
59          assertThat(ld0.getTimeStamp(), greaterThanOrEqualTo(1L));
60      }
61  
62      @Test
63      void testBytesNeverGoBackwards() {
64          // iostat's write KB/t for ld0 rounds down from 25.42 to 25.40 between these two samples while the transfer
65          // count rises, so a KB/t times transfers reading would report fewer bytes written than a moment earlier
66          NetBsdHWDiskStore before = store("ld0", FIRST);
67          NetBsdHWDiskStore after = store("ld0", SECOND);
68          assertTrue(before.updateAttributes());
69          assertTrue(after.updateAttributes());
70          assertThat(after.getWrites(), greaterThanOrEqualTo(before.getWrites()));
71          assertThat(after.getWriteBytes(), greaterThanOrEqualTo(before.getWriteBytes()));
72          assertThat(after.getReadBytes(), greaterThanOrEqualTo(before.getReadBytes()));
73      }
74  
75      @Test
76      void testIdleAndMissingDrives() {
77          NetBsdHWDiskStore cd0 = store("cd0", FIRST);
78          assertTrue(cd0.updateAttributes());
79          assertThat(cd0.getReadBytes(), is(0L));
80          assertThat(cd0.getWriteBytes(), is(0L));
81          assertThat(cd0.getTransferTime(), is(0L));
82  
83          NetBsdHWDiskStore absent = store("sd9", FIRST);
84          assertFalse(absent.updateAttributes());
85      }

121     @Test

130     @Test
131     void testParseWedges() {
132         Map<String, List<HWPartition>> wedges = NetBsdHWDiskStore.parseWedges(DMESG, DEVICES,
133                 NetBsdHWDiskStore.parseMounts(DF), (disk, part) -> new Pair<>(168, ParseUtil.getFirstIntValue(disk)));
134         assertThat(wedges.keySet(), contains("ld4", "ld6", "sd0"));
135 
136         // Each wedge once, though the live buffer repeats two of them, and not dk5, deleted since boot. Nor dk6, whose
137         // parent is gone and which is left for getDisks to list as a disk
138         List<HWPartition> ld4 = wedges.get("ld4");
139         assertNotNull(ld4);
140         assertThat(ld4.size(), is(2));
141         HWPartition dk1 = ld4.get(1);
142         assertThat(dk1.getIdentification(), is("dk1"));
143         assertThat(dk1.getName(), is("dk1"));
144         assertThat(dk1.getType(), is("ffs"));
145         assertThat(dk1.getUuid(), is("bce3cb3f-b5ff-4aac-8049-5f748b710db0"));
146         assertThat(dk1.getLabel(), is(""));
147         assertThat(dk1.getSize(), is(133807967L * 512));
148         assertThat(dk1.getMajor(), is(168));
149         assertThat(dk1.getMinor(), is(1));
150         assertThat(dk1.getMountPoint(), is("/"));
151         assertThat(ld4.get(0).getMountPoint(), is("/boot"));
152 
153         // Swap is not mounted
154         List<HWPartition> ld6 = wedges.get("ld6");
155         assertNotNull(ld6);
156         assertThat(ld6.get(0).getType(), is("swap"));
157         assertThat(ld6.get(0).getMountPoint(), is(""));
158 
159         // Sized in the parent's sectors, and named by a label that holds a comma
160         List<HWPartition> sd0 = wedges.get("sd0");
161         assertNotNull(sd0);
162         HWPartition dk4 = sd0.get(0);
163         assertThat(dk4.getSize(), is(1000L * 4096));
164         assertThat(dk4.getUuid(), is(""));
165         assertThat(dk4.getLabel(), is("data, old"));
166         assertThat(dk4.getMountPoint(), is("/mnt/old disk"));
167     }

169     @Test
170     void testParseWedgesLastAnnouncementWins() {
171         // dk2 was deleted and its number reused on another disk after boot
172         List<String> dmesg = List.of("ld5: 1024 GB, 16383 cyl, 16 head, 63 sec, 512 bytes/sect x 2147483648 sectors",
173                 "dk2 at ld5: \"c4040fb3-6246-4f8a-9718-9d613c827f82\", 2147481567 blocks at 2048, type: ffs",
174                 "[   300.000000] sd2: 7680 MB, 980 cyl, 255 head, 63 sec, 4096 bytes/sect x 1966080 sectors",
175                 "[   300.100000] dk2 at sd2: \"usb\", 1966000 blocks at 40, type: msdos");
176         Map<String, List<HWPartition>> wedges = NetBsdHWDiskStore.parseWedges(dmesg, List.of("ld5", "sd2", "dk2"),
177                 Map.of(), (disk, part) -> new Pair<>(0, 0));
178         assertThat(wedges.keySet(), contains("sd2"));
179         List<HWPartition> sd2 = wedges.get("sd2");
180         assertNotNull(sd2);
181         assertThat(sd2.get(0).getLabel(), is("usb"));
182         assertThat(sd2.get(0).getType(), is("msdos"));
183         assertThat(sd2.get(0).getSize(), is(1966000L * 4096));
184     }

186     @Test
187     void testParseWedgesWithoutParentGeometry() {
188         // A parent whose geometry line is gone from both buffers is assumed to have 512-byte sectors
189         Map<String, List<HWPartition>> wedges = NetBsdHWDiskStore.parseWedges(
190                 List.of("dk2 at ld5: \"c4040fb3-6246-4f8a-9718-9d613c827f82\", 2048 blocks at 2048, type: ffs"),
191                 List.of("ld5", "dk2"), Map.of(), (disk, part) -> new Pair<>(0, 0));
192         List<HWPartition> ld5 = wedges.get("ld5");
193         assertNotNull(ld5);
194         assertThat(ld5.get(0).getSize(), is(2048L * 512));
195         assertThat(NetBsdHWDiskStore
196                 .parseWedges(List.of("boot device: dk1"), List.of("dk1"), Map.of(), (disk, part) -> new Pair<>(0, 0))
197                 .isEmpty(), is(true));
198     }
