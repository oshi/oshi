106     private static @Nullable TestPowerSource parse(String acpiconf, boolean acline, int state, int time, int life) {
107         return FreeBsdPowerSource.parseBattery("BAT0", acpiconf.lines().toList(), acline, state, time, life,
108                 TestPowerSource::new);
109     }
110 
111     @Test
112     void testMilliwattBatteryDischarging() {
113         TestPowerSource ps = parse(MW_DISCHARGING, false, -1, -1, -1);
114         assertNotNull(ps);
115         assertThat(ps.getName(), is("BAT0"));
116         assertThat(ps.getDeviceName(), is("5B10W13930"));
117         assertThat(ps.getSerialNumber(), is("1234"));
118         assertThat(ps.getChemistry(), is("LION"));
119         assertThat(ps.getManufacturer(), is("SMP"));
120         assertThat(ps.isDischarging(), is(true));
121         assertThat(ps.isCharging(), is(false));
122         assertThat(ps.isPowerOnLine(), is(false));
123         assertThat(ps.getRemainingCapacityPercent(), closeTo(0.8, 1e-9));
124         assertThat(ps.getTimeRemainingInstant(), closeTo(8100d, 1e-9));
125         assertThat(ps.getTimeRemainingEstimated(), closeTo(8100d, 1e-9));
126         assertThat(ps.getPowerUsageRate(), closeTo(-9120d, 1e-9));
127         assertThat(ps.getVoltage(), closeTo(12.16, 1e-9));
128         // 9120 mW / 12.16 V = 750 mA
129         assertThat(ps.getAmperage(), closeTo(-750d, 1e-9));
130         assertThat(ps.getCapacityUnits(), is(CapacityUnits.MWH));
131         assertThat(ps.getDesignCapacity(), is(57000));
132         assertThat(ps.getMaxCapacity(), is(51300));
133         assertThat(ps.getCurrentCapacity(), is(41040));
134         assertThat(ps.getCycleCount(), is(142));
135     }

137     @Test
138     void testMilliampBatteryCharging() {
139         TestPowerSource ps = parse(MA_CHARGING, true, -1, -1, -1);
140         assertNotNull(ps);
141         assertThat(ps.getSerialNumber(), is(Constants.UNKNOWN));
142         assertThat(ps.isCharging(), is(true));
143         assertThat(ps.isDischarging(), is(false));
144         assertThat(ps.isPowerOnLine(), is(true));
145         assertThat(ps.getTimeRemainingInstant(), closeTo(-1d, 1e-9));
146         assertThat(ps.getPowerUsageRate(), closeTo(12000d, 1e-9));
147         assertThat(ps.getAmperage(), closeTo(1500d, 1e-9));
148         assertThat(ps.getVoltage(), closeTo(8d, 1e-9));
149         assertThat(ps.getCapacityUnits(), is(CapacityUnits.MAH));
150         assertThat(ps.getMaxCapacity(), is(3800));
151         assertThat(ps.getCurrentCapacity(), is(1900));
152         assertThat(ps.getCycleCount(), is(-1));
153     }

160     @Test
161     void testSysctlTotalsWhenAcpiconfFails() {
162         // state 1 = discharging, 95 minutes, 64%
163         TestPowerSource ps = parse("", false, 1, 95, 64);
164         assertNotNull(ps);
165         assertThat(ps.isDischarging(), is(true));
166         assertThat(ps.getRemainingCapacityPercent(), closeTo(0.64, 1e-9));
167         assertThat(ps.getTimeRemainingEstimated(), closeTo(5700d, 1e-9));
168         assertThat(ps.getTimeRemainingInstant(), closeTo(5700d, 1e-9));
169         assertThat(ps.getCapacityUnits(), is(CapacityUnits.RELATIVE));
170         assertThat(ps.getDeviceName(), is(Constants.UNKNOWN));
171         assertThat(ps.getVoltage(), closeTo(-1d, 1e-9));
172     }
