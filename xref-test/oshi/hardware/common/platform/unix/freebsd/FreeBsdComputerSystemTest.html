5   package oshi.hardware.common.platform.unix.freebsd;

7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   import static org.hamcrest.Matchers.nullValue;
10  
11  import java.util.Collections;
12  import java.util.List;
13  
14  import org.jspecify.annotations.Nullable;
15  import org.junit.jupiter.api.Test;
16  
17  import oshi.util.tuples.Quintet;
18  
19  class FreeBsdComputerSystemTest {
20  
21      @Test
22      void testParseDmiDecode() {
23          // Representative `dmidecode -t system` (DMI type 1) block.
24          List<String> dmidecode = """
25                  Handle 0x0001, DMI type 1, 27 bytes
26                  System Information
27                  \tManufacturer: Dell Inc.
28                  \tProduct Name: PowerEdge R640
29                  \tVersion: Not Specified
30                  \tSerial Number: 7XY1234
31                  \tUUID: 4c4c4544-0058-5910-8034-c4c04f313233
32                  \tWake-up Type: Power Switch
33                  \tSKU Number: SKU=NotProvided
34                  \tFamily: PowerEdge
35                  """.lines().toList();
36          Quintet<@Nullable String, @Nullable String, @Nullable String, @Nullable String, @Nullable String> dmi = FreeBsdComputerSystem
37                  .parseDmiDecode(dmidecode);
38          assertThat(dmi.getA(), is("Dell Inc."));
39          assertThat(dmi.getB(), is("PowerEdge R640"));
40          assertThat(dmi.getC(), is("7XY1234"));
41          assertThat(dmi.getD(), is("4c4c4544-0058-5910-8034-c4c04f313233"));
42          assertThat(dmi.getE(), is("Not Specified"));
43      }
44  
45      @Test
46      void testParseDmiDecodeEmpty() {
47          // No output (e.g. dmidecode not available / not root): every field is null so the caller can fall back.
48          Quintet<@Nullable String, @Nullable String, @Nullable String, @Nullable String, @Nullable String> dmi = FreeBsdComputerSystem
49                  .parseDmiDecode(Collections.emptyList());
50          assertThat(dmi.getA(), is(nullValue()));
51          assertThat(dmi.getC(), is(nullValue()));
52          assertThat(dmi.getD(), is(nullValue()));
53      }
54  }
