22  class CupsPrinterTest {
23  
24      /** Minimal concrete subclass for testing. */
25      private static final class StubPrinter extends CupsPrinter {
26          StubPrinter(String name, String driverName, String description, PrinterStatus status, String statusReason,
27                  boolean isDefault, boolean isLocal, String portName) {
28              super(name, driverName, description, status, statusReason, isDefault, isLocal, portName);
29          }
30      }
31  
32      private static final CupsPrinter.PrinterFactory STUB_FACTORY = StubPrinter::new;
33  
34      // Fixture: lpstat -p output with two printers
35      private static final List<String> LPSTAT_P = List.of(
36              "printer HP_LaserJet is idle.  enabled since Mon 01 Jan 2024 12:00:00 AM",
37              "printer PDF_Printer disabled since Tue 02 Jan 2024 - paused by admin");
38  
39      private static final Map<String, String> PORT_MAP = new HashMap<>();
40      private static final Map<String, String> DESC_MAP = new HashMap<>();
41      static {
42          PORT_MAP.put("HP_LaserJet", "usb://HP/LaserJet");
43          PORT_MAP.put("PDF_Printer", "cups-pdf:/");
44          DESC_MAP.put("HP_LaserJet", "HP LaserJet Pro MFP");
45          DESC_MAP.put("PDF_Printer", "CUPS-PDF Virtual Printer");
46      }
47  
48      @Test
49      void testGetPrintersFromLpstatTwoPrinters() {
50          List<Printer> printers = CupsPrinter.getPrintersFromLpstat(LPSTAT_P, "HP_LaserJet", PORT_MAP, DESC_MAP,
51                  name -> "TestDriver", STUB_FACTORY);
52          assertThat(printers, hasSize(2));
53  
54          Printer hp = printers.get(0);
55          assertThat(hp.getName(), is("HP_LaserJet"));
56          assertThat(hp.getStatus(), is(PrinterStatus.IDLE));
57          assertThat(hp.isDefault(), is(true));
58          assertThat(hp.isLocal(), is(true));
59          assertThat(hp.getPortName(), is("usb://HP/LaserJet"));
60          assertThat(hp.getDescription(), is("HP LaserJet Pro MFP"));
61          assertThat(hp.getDriverName(), is("TestDriver"));
62  
63          Printer pdf = printers.get(1);
64          assertThat(pdf.getName(), is("PDF_Printer"));
65          assertThat(pdf.getStatus(), is(PrinterStatus.OFFLINE));
66          assertThat(pdf.isDefault(), is(false));
67          assertThat(pdf.getStatusReason(), is("paused by admin"));
68      }
69  
70      @Test
71      void testGetPrintersFromLpstatEmpty() {
72          List<Printer> printers = CupsPrinter.getPrintersFromLpstat(Collections.emptyList(), "", Collections.emptyMap(),
73                  Collections.emptyMap(), name -> "", STUB_FACTORY);
74          assertThat(printers, is(empty()));
75      }
76  
77      @Test
78      void testGetPrintersFromLpstatSkipsNonPrinterLines() {
79          List<String> mixed = """
80                  scheduler is running
81                  printer MyPrinter now printing MyPrinter-42. enabled since Mon 01 Jan
82                  no system default destination
83                  """.lines().toList();
84          List<Printer> printers = CupsPrinter.getPrintersFromLpstat(mixed, "", Collections.emptyMap(),
85                  Collections.emptyMap(), name -> "", STUB_FACTORY);
86          assertThat(printers, hasSize(1));
87          assertThat(printers.get(0).getName(), is("MyPrinter"));
88          assertThat(printers.get(0).getStatus(), is(PrinterStatus.PRINTING));
89      }
90  
91      @Test
92      void testParseStateFromCupsIppStates() {
93          // CUPS ipp_pstate_t: 3 = idle, 4 = processing, 5 = stopped
94          assertThat(CupsPrinter.parseStateFromCups("3", "none"), is(PrinterStatus.IDLE));
95          assertThat(CupsPrinter.parseStateFromCups("4", "none"), is(PrinterStatus.PRINTING));
96          assertThat(CupsPrinter.parseStateFromCups("5", "none"), is(PrinterStatus.OFFLINE));
97      }
98  
99      @Test
100     void testParseStateFromCupsErrorReasonsTakePrecedence() {
101         // An error or fault in the reasons overrides an otherwise healthy numeric state
102         assertThat(CupsPrinter.parseStateFromCups("3", "media-empty-error"), is(PrinterStatus.ERROR));
103         assertThat(CupsPrinter.parseStateFromCups("4", "cover-open-fault"), is(PrinterStatus.ERROR));
104         // Matching is case-insensitive
105         assertThat(CupsPrinter.parseStateFromCups("3", "Toner-Empty-ERROR"), is(PrinterStatus.ERROR));
106     }
107 
108     @Test
109     void testParseStateFromCupsNonErrorReasonsIgnored() {
110         // Reasons that are neither error nor fault leave the numeric state in control
111         assertThat(CupsPrinter.parseStateFromCups("3", "toner-low"), is(PrinterStatus.IDLE));
112         assertThat(CupsPrinter.parseStateFromCups("5", "paused"), is(PrinterStatus.OFFLINE));
113     }
114 
115     @Test
116     void testParseStateFromCupsUnknown() {
117         // Empty state, unrecognized state, and unparseable state all yield UNKNOWN
118         assertThat(CupsPrinter.parseStateFromCups("", "none"), is(PrinterStatus.UNKNOWN));
119         assertThat(CupsPrinter.parseStateFromCups("", ""), is(PrinterStatus.UNKNOWN));
120         assertThat(CupsPrinter.parseStateFromCups("9", "none"), is(PrinterStatus.UNKNOWN));
121         assertThat(CupsPrinter.parseStateFromCups("bogus", "none"), is(PrinterStatus.UNKNOWN));
122         // An empty state still defers to an error reason
123         assertThat(CupsPrinter.parseStateFromCups("", "hardware-error"), is(PrinterStatus.ERROR));
124     }
