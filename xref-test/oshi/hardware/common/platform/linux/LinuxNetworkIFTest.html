27      private static String model(Path tempDir, @Nullable String ifindex, @Nullable String udevRecord)
28              throws IOException {
29          Path sys = Files.createDirectories(tempDir.resolve("sys"));
30          Path udev = Files.createDirectories(tempDir.resolve("udev"));
31          if (ifindex != null) {
32              Path eth0 = Files.createDirectories(sys.resolve("eth0"));
33              writeFile(eth0.resolve("ifindex"), ifindex);
34              // The kernel uevent file never carries the hwdb properties, so it must not be what is read
35              writeFile(eth0.resolve("uevent"),
36                      "INTERFACE=eth0\nIFINDEX=" + ifindex + "\nID_MODEL_FROM_DATABASE=Not From Uevent");
37          }
38          if (udevRecord != null) {
39              writeFile(udev.resolve("n2"), udevRecord);
40          }
41          return LinuxNetworkIF.queryIfModelFromUdevData("eth0", sys + "/", udev + "/");
42      }
43  
44      @Test
45      void testQueryIfModelVendorAndModel(@TempDir Path tempDir) throws IOException {
46          assertThat(model(tempDir, "2", "I:1700000000\nE:ID_NET_NAME_PATH=enp3s0\n"
47                  + "E:ID_VENDOR_FROM_DATABASE=Intel Corporation\nE:ID_MODEL_FROM_DATABASE=Ethernet Controller I225-V\n"
48                  + "G:systemd"), is("Intel Corporation Ethernet Controller I225-V"));
49      }
50  
51      @Test
52      void testQueryIfModelOnlyModel(@TempDir Path tempDir) throws IOException {
53          assertThat(model(tempDir, "2", "E:ID_MODEL_FROM_DATABASE=RTL8111/8168/8411"), is("RTL8111/8168/8411"));
54      }
55  
56      @Test
57      void testQueryIfModelNoUdevRecord(@TempDir Path tempDir) throws IOException {
58          // A host without udevd, such as a container: the interface name, never the uevent text
59          assertThat(model(tempDir, "2", null), is("eth0"));
60      }
61  
62      @Test
63      void testQueryIfModelNoIfindex(@TempDir Path tempDir) throws IOException {
64          assertThat(model(tempDir, null, "E:ID_MODEL_FROM_DATABASE=RTL8111/8168/8411"), is("eth0"));
65      }
66  
67      @Test
68      void testQueryIfModelRecordWithoutModel(@TempDir Path tempDir) throws IOException {
69          assertThat(model(tempDir, "2", "E:ID_NET_DRIVER=e1000e"), is("eth0"));
70      }
