78      @Test
79      void testCreateMIDRBasic() {
80          // vendor=0x41 (ARM), stepping=r1p3, model=0xd07, family=8
81          String midr = LinuxCentralProcessor.createMIDR("0x41", "r1p3", "0xd07", "8");
82          // Implementer 0x41 << 24 = 0x41000000
83          // Variant 1 << 20 = 0x00100000
84          // Architecture 8 << 16 = 0x00080000
85          // PartNum 0xd07 << 4 = 0x0000D070
86          // Revision 3 = 0x00000003
87          // Total = 0x41180073... let me compute:
88          // 0x41 << 24 = 0x41000000
89          // 1 << 20 = 0x00100000
90          // 8 << 16 = 0x00080000
91          // 0xd07 << 4 = 0x0000D070
92          // 3 = 0x00000003
93          // Sum = 0x4118D073
94          assertThat(midr, is("4118D073"));
95      }
