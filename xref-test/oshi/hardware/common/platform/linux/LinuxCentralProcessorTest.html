80      @Test
81      void testCreateMIDRBasic() {
82          // vendor=0x41 (ARM), stepping=r1p3, model=0xd07, family=8
83          String midr = LinuxCentralProcessor.createMIDR("0x41", "r1p3", "0xd07", "8");
84          // Implementer 0x41 << 24 = 0x41000000
85          // Variant 1 << 20 = 0x00100000
86          // Architecture 8 << 16 = 0x00080000
87          // PartNum 0xd07 << 4 = 0x0000D070
88          // Revision 3 = 0x00000003
89          // Total = 0x41180073... let me compute:
90          // 0x41 << 24 = 0x41000000
91          // 1 << 20 = 0x00100000
92          // 8 << 16 = 0x00080000
93          // 0xd07 << 4 = 0x0000D070
94          // 3 = 0x00000003
95          // Sum = 0x4118D073
96          assertThat(midr, is("4118D073"));
97      }

628     @Test
629     void testApplyHybridCoreTypes() {
630         // Core i7-12700: 8 two-thread P-cores (core_id 0-28 step 4, CPUs 0-15) and 4 E-cores (core_id 32-35)
631         List<LogicalProcessor> logProcs = new ArrayList<>();
632         Map<Integer, Integer> efficiency = new HashMap<>();
633         for (int cpu = 0; cpu < 20; cpu++) {
634             int core = cpu < 16 ? (cpu / 2) * 4 : 32 + cpu - 16;
635             logProcs.add(new LogicalProcessor(cpu, core, 0));
636             efficiency.put(core, 0);
637         }
638         LinuxCentralProcessor.applyHybridCoreTypes(efficiency, logProcs, "0-15");
639         assertThat(efficiency.get(0), is(1));
640         assertThat(efficiency.get(28), is(1));
641         assertThat(efficiency.get(32), is(0));
642         assertThat(efficiency.values().stream().filter(e -> e == 1).count(), is(8L));
643 
644         // No hybrid PMU, or capacities the kernel already published, leave the values alone
645         Map<Integer, Integer> plain = new HashMap<>(Map.of(0, 0, 4, 0));
646         LinuxCentralProcessor.applyHybridCoreTypes(plain, logProcs, "");
647         assertThat(plain, is(Map.of(0, 0, 4, 0)));
648         Map<Integer, Integer> arm = new HashMap<>(Map.of(0, 446, 4, 1024));
649         LinuxCentralProcessor.applyHybridCoreTypes(arm, logProcs, "0-3");
650         assertThat(arm, is(Map.of(0, 446, 4, 1024)));
651     }
