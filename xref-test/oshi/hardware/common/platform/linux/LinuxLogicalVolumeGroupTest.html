34  // parsePhysicalVolumes dereferences DevPath.DEV, whose static initializer validates that the configured /dev path
35  // exists. That holds on Linux, macOS, and the BSD/illumos CI hosts, but not on Windows, where class init throws.
36  @DisabledOnOs(OS.WINDOWS)
37  class LinuxLogicalVolumeGroupTest {
38  
39      @Test
40      void testParsePhysicalVolumes() {
41          // pvs -o vg_name,pv_name output: header row, then "vg pv" rows (leading whitespace)
42          List<String> pvs = List.of("  VG   PV", "  vg0  /dev/sda2", "  vg0  /dev/sdb1", "  vg1  /dev/sdc1");
43          Map<String, Set<String>> map = LinuxLogicalVolumeGroup.parsePhysicalVolumes(pvs);
44          assertThat(map.keySet(), containsInAnyOrder("vg0", "vg1"));
45          assertThat(map.get("vg0"), containsInAnyOrder("/dev/sda2", "/dev/sdb1"));
46          assertThat(map.get("vg1"), contains("/dev/sdc1"));
47      }
48  
49      @Test
50      void testParsePhysicalVolumesSkipsNonDeviceAndMalformed() {
51          // A single-token line (length != 2) and a second token that is not a /dev path are both ignored
52          List<String> pvs = List.of("  novg", "  vg2  notadevice", "  vg3  /dev/mapper/pv");
53          Map<String, Set<String>> map = LinuxLogicalVolumeGroup.parsePhysicalVolumes(pvs);
54          assertThat(map.keySet(), contains("vg3"));
55          assertThat(map.get("vg3"), contains("/dev/mapper/pv"));
56      }
57  
58      @Test
59      void testParsePhysicalVolumesEmpty() {
60          assertThat(LinuxLogicalVolumeGroup.parsePhysicalVolumes(Collections.emptyList()), anEmptyMap());
61      }

65      /** A concrete group so the factory has something to build. */
66      private static final class TestLvg extends LinuxLogicalVolumeGroup {
67          private TestLvg(String name, Map<String, Set<String>> lvMap, Set<String> pvSet) {
68              super(name, lvMap, pvSet);
69          }
70      }
71  
72      // Creates a sysfs-like directory with a "slaves" entry per physical volume, returning its path.
73      private static String syspathWithSlaves(Path root, String name, String... slaves) throws IOException {
74          Path dir = Files.createDirectories(root.resolve(name).resolve("slaves"));
75          for (String slave : slaves) {
76              Files.createFile(dir.resolve(slave));
77          }
78          return root.resolve(name).toString();
79      }
80  
81      private static UdevBlockDevice lvmDevice(String syspath, String vgName, String lvName) {
82          return new UdevBlockDevice(syspath, DevPath.DM + "0", "LVM-abcdef", vgName, lvName);
83      }
84  
85      private static List<LogicalVolumeGroup> build(List<UdevBlockDevice> devices) {
86          return LinuxLogicalVolumeGroup.buildLogicalVolumeGroups(devices, new HashMap<>(), TestLvg::new);
87      }
88  
89      @Test
90      void testAssemblesGroupFromSlavesDirectory(@TempDir Path tmp) throws IOException {
91          String syspath = syspathWithSlaves(tmp, "dm-0", "sda2", "sdb1");
92          List<LogicalVolumeGroup> lvgs = build(List.of(lvmDevice(syspath, "vg0", "root")));
93  
94          assertThat(lvgs, hasSize(1));
95          assertThat(lvgs.get(0).getName(), is("vg0"));
96          assertThat("Each slaves entry becomes a /dev physical volume", lvgs.get(0).getPhysicalVolumes(),
97                  containsInAnyOrder("/dev/sda2", "/dev/sdb1"));
98          assertThat(lvgs.get(0).getLogicalVolumes().get("root"), containsInAnyOrder("/dev/sda2", "/dev/sdb1"));
99      }

101     @Test
102     void testTwoLogicalVolumesShareOneGroup(@TempDir Path tmp) throws IOException {
103         List<UdevBlockDevice> devices = List.of(lvmDevice(syspathWithSlaves(tmp, "dm-0", "sda2"), "vg0", "root"),
104                 lvmDevice(syspathWithSlaves(tmp, "dm-1", "sdb1"), "vg0", "swap"));
105         List<LogicalVolumeGroup> lvgs = build(devices);
106 
107         assertThat("Both volumes belong to one group", lvgs, hasSize(1));
108         assertThat(lvgs.get(0).getLogicalVolumes().keySet(), containsInAnyOrder("root", "swap"));
109         assertThat(lvgs.get(0).getLogicalVolumes().get("root"), contains("/dev/sda2"));
110         assertThat(lvgs.get(0).getLogicalVolumes().get("swap"), contains("/dev/sdb1"));
111         assertThat("The group's physical volumes are the union", lvgs.get(0).getPhysicalVolumes(),
112                 containsInAnyOrder("/dev/sda2", "/dev/sdb1"));
113     }

115     @Test
116     void testNonLvmDevicesAreFiltered(@TempDir Path tmp) throws IOException {
117         String syspath = syspathWithSlaves(tmp, "dev", "sda1");
118         List<UdevBlockDevice> devices = List.of(
119                 // not a device-mapper node
120                 new UdevBlockDevice(syspath, "/dev/sda", "LVM-abcdef", "vg0", "root"),
121                 // device-mapper, but owned by something other than LVM, e.g. LUKS
122                 new UdevBlockDevice(syspath, DevPath.DM + "0", "CRYPT-LUKS2-abcdef", "vg0", "root"),
123                 // device-mapper LVM, but udev reported no volume group or volume name
124                 new UdevBlockDevice(syspath, DevPath.DM + "1", "LVM-abcdef", null, "root"),
125                 new UdevBlockDevice(syspath, DevPath.DM + "2", "LVM-abcdef", "vg0", ""),
126                 // udev reported no properties at all
127                 new UdevBlockDevice(syspath, null, null, null, null));
128         assertThat("Nothing here is an LVM volume", build(devices), is(empty()));
129     }

131     @Test
132     void testMissingSlavesDirectoryStillReportsTheGroup(@TempDir Path tmp) {
133         // A volume whose sysfs entry has no slaves directory is still a volume group, just with no physical volumes.
134         List<LogicalVolumeGroup> lvgs = build(List.of(lvmDevice(tmp.resolve("absent").toString(), "vg0", "root")));
135         assertThat(lvgs, hasSize(1));
136         assertThat(lvgs.get(0).getName(), is("vg0"));
137         assertThat(lvgs.get(0).getPhysicalVolumes(), is(empty()));
138         assertThat(lvgs.get(0).getLogicalVolumes(), anEmptyMap());
139     }
140 
141     @Test
142     void testPhysicalVolumesFromPvsAreMergedWithSlaves(@TempDir Path tmp) throws IOException {
143         // pvs already reported a volume for this group; the slaves scan adds to that set rather than replacing it.
144         Map<String, Set<String>> fromPvs = new HashMap<>();
145         fromPvs.put("vg0", new HashSet<>(List.of("/dev/sdc1")));
146         List<LogicalVolumeGroup> lvgs = LinuxLogicalVolumeGroup.buildLogicalVolumeGroups(
147                 List.of(lvmDevice(syspathWithSlaves(tmp, "dm-0", "sda2"), "vg0", "root")), fromPvs, TestLvg::new);
148 
149         assertThat(lvgs, hasSize(1));
150         assertThat(lvgs.get(0).getPhysicalVolumes(), containsInAnyOrder("/dev/sdc1", "/dev/sda2"));
151     }
