7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.contains;
9   import static org.hamcrest.Matchers.is;
10  
11  import java.util.ArrayList;
12  import java.util.List;
13  
14  import org.junit.jupiter.api.Test;
15  
16  import oshi.hardware.HWPartition;
17  import oshi.hardware.common.platform.windows.WindowsHWDiskStore.PartitionMaps;
18  
19  class WindowsHWDiskStorePartitionsTest {
20  
21      @Test
22      void testUnmountedPartitionsAreListed() {
23          PartitionMaps maps = new PartitionMaps();
24          String disk = "\\\\.\\PHYSICALDRIVE0";
25          WindowsHWDiskStore.mapDriveToPartition(maps,
26                  "\\\\HOST\\root\\cimv2:Win32_DiskDrive.DeviceID=\"\\\\\\\\.\\\\PHYSICALDRIVE0\"",
27                  "\\\\HOST\\root\\cimv2:Win32_DiskPartition.DeviceID=\"Disk #0, Partition #0\"");
28          WindowsHWDiskStore.mapDriveToPartition(maps,
29                  "\\\\HOST\\root\\cimv2:Win32_DiskDrive.DeviceID=\"\\\\\\\\.\\\\PHYSICALDRIVE0\"",
30                  "\\\\HOST\\root\\cimv2:Win32_DiskPartition.DeviceID=\"Disk #0, Partition #1\"");
31          // The EFI system partition has no drive letter
32          WindowsHWDiskStore.mapUnmountedPartition(maps, "Disk #0, Partition #0", "Disk #0, Partition #0", "GPT: System",
33                  "GPT: System", 104_857_600L, 0, 0);
34          maps.getPartitionMap().computeIfAbsent("Disk #0, Partition #1", x -> new ArrayList<>())
35                  .add(new HWPartition("Disk #0, Partition #1", "GPT: Basic Data", "GPT: Basic Data", "uuid", "Windows",
36                          255_000_000_000L, 0, 1, "C:\\"));
37  
38          List<HWPartition> partitions = WindowsHWDiskStore.buildPartitionList(maps, disk);
39          // Sorted by name, which Windows fills from the partition type
40          assertThat(partitions.stream().map(HWPartition::getMountPoint).toList(), contains("C:\\", ""));
41          HWPartition efi = partitions.get(1);
42          assertThat(efi.getSize(), is(104_857_600L));
43          assertThat(efi.getUuid(), is(""));
44          assertThat(efi.getMajor(), is(0));
45          assertThat(efi.getMinor(), is(0));
46      }
47  }
