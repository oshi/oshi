19  class WindowsCentralProcessorTest {
20  
21      @Test
22      void testParseIdentifier() {
23          String id = "Intel64 Family 6 Model 142 Stepping 12";
24          assertThat(WindowsCentralProcessor.parseIdentifier(id, "Family"), is("6"));
25          assertThat(WindowsCentralProcessor.parseIdentifier(id, "Model"), is("142"));
26          assertThat(WindowsCentralProcessor.parseIdentifier(id, "Stepping"), is("12"));
27      }
28  
29      @Test
30      void testParseIdentifierNotFound() {
31          assertThat(WindowsCentralProcessor.parseIdentifier("Intel64 Family 6", "Model"), is(""));
32      }
33  
34      @Test
35      void testParseIdentifierKeyAtEnd() {
36          // Key is the last token — no value follows
37          assertThat(WindowsCentralProcessor.parseIdentifier("Family 6 Model", "Model"), is(""));
38      }
39  
40      @Test
41      void testNoFeaturesPresent() {
42          assertTrue(WindowsCentralProcessor.queryFeatureFlags(f -> false).isEmpty());
43      }
44  
45      @Test
46      void testAllFeaturesPresent() {
47          List<String> all = WindowsCentralProcessor.queryFeatureFlags(f -> true);
48          assertEquals(90, all.size());
49          assertEquals(new HashSet<>(all).size(), all.size(), "feature names should be unique");
50          for (String name : all) {
51              assertTrue(name.startsWith("PF_"), name);
52          }
53      }
54  
55      @Test
56      void testOnlyRequestedFeaturesReported() {
57          // 3 = PF_MMX_INSTRUCTIONS_AVAILABLE, 39 = PF_AVX_INSTRUCTIONS_AVAILABLE
58          Set<Integer> present = new HashSet<>(List.of(3, 39));
59          IntPredicate predicate = present::contains;
60          assertEquals(List.of("PF_MMX_INSTRUCTIONS_AVAILABLE", "PF_AVX_INSTRUCTIONS_AVAILABLE"),
61                  WindowsCentralProcessor.queryFeatureFlags(predicate));
62      }
63  
64      @Test
65      void testFeatureValuesAreContiguous() {
66          // The enum mirrors winnt.h, whose PF_ defines run 0 through 89 with no gaps
67          Set<Integer> queried = new HashSet<>();
68          WindowsCentralProcessor.queryFeatureFlags(f -> {
69              queried.add(f);
70              return false;
71          });
72          Set<Integer> expected = new HashSet<>();
73          for (int i = 0; i <= 89; i++) {
74              expected.add(i);
75          }
76          assertEquals(expected, queried);
77      }
78  
79      @Test
80      void testParseSteppingArm64() {
81          // Captured on a GitHub windows-11-arm runner (Azure Cobalt 100, Neoverse N2): Identifier and CP 4000
82          String arm = "ARMv8 (64-bit) Family 8 Model D49 Revision   0";
83          long midr = 0x410fd490L;
84          assertThat(WindowsCentralProcessor.parseIdentifier(arm, "Model"), is("D49"));
85          assertThat(WindowsCentralProcessor.parseStepping(arm, midr), is("r0p0"));
86          assertThat(WindowsCentralProcessor.parseStepping(arm, 0x412fd491L), is("r2p1"));
87          // Without the MIDR, the identifier's revision
88          assertThat(WindowsCentralProcessor.parseStepping(arm, 0L), is("0"));
89          // x86 names its stepping, whatever the MIDR
90          assertThat(WindowsCentralProcessor.parseStepping("Intel64 Family 6 Model 142 Stepping 12", 0L), is("12"));
91      }
92  
93      @Test
94      void testChooseProcessorId() {
95          // WMI's ProcessorId on the same runner is all zeros; the MIDR is formatted as Linux formats it on ARM
96          assertThat(WindowsCentralProcessor.chooseProcessorId("0000000000000000", 0x410fd490L, () -> "fallback"),
97                  is("410FD49000000000"));
98          assertThat(WindowsCentralProcessor.chooseProcessorId("", 0x410fd490L, () -> "fallback"),
99                  is("410FD49000000000"));
100         // x86: the CPUID signature from WMI, or the encoded fallback when WMI has none
101         assertThat(WindowsCentralProcessor.chooseProcessorId("BFEBFBFF000806EC", 0L, () -> "fallback"),
102                 is("BFEBFBFF000806EC"));
103         assertThat(WindowsCentralProcessor.chooseProcessorId("", 0L, () -> "fallback"), is("fallback"));
104         assertThat(WindowsCentralProcessor.chooseProcessorId("0000000000000000", 0L, () -> "fallback"),
105                 is("0000000000000000"));
106     }
