34   * Tests the shared Storage Spaces correlation without Windows, by stubbing the four WMI results. Storage Spaces needs

39      private static final String SP_GUID = "{11111111-1111-1111-1111-111111111111}";
40      private static final String OTHER_SP_GUID = "{99999999-9999-9999-9999-999999999999}";
41      private static final String PD1_GUID = "{aaaaaaaa-aaaa-aaaa-aaaa-aaaaaaaaaaaa}";
42      private static final String PD2_GUID = "{bbbbbbbb-bbbb-bbbb-bbbb-bbbbbbbbbbbb}";
43      private static final String VD_GUID = "{cccccccc-cccc-cccc-cccc-cccccccccccc}";

45      // Builds the ObjectId form the production regexes expect.
46      private static String objectId(String kind, String... guids) {
47          StringBuilder sb = new StringBuilder("{1}\\\\HOST\\root/Microsoft/Windows/Storage/Providers_v2\\SPACES_");
48          sb.append(kind).append(":ObjectId=").append(kind).append(':');
49          for (String guid : guids) {
50              sb.append(guid);
51          }
52          return sb.toString();
53      }

55      // A minimal single-column-per-property WmiResult stub.
56      private static <T extends Enum<T>> WmiResult<T> result(List<Map<T, Object>> rows, int cimType) {
57          return new WmiResult<T>() {
58              @Override
59              public int getResultCount() {
60                  return rows.size();
61              }
62  
63              @Override
64              public @Nullable Object getValue(T property, int index) {
65                  return rows.get(index).get(property);
66              }
67  
68              @Override
69              public int getVtType(T property) {
70                  return VT_BSTR;
71              }
72  
73              @Override
74              public int getCIMType(T property) {
75                  return cimType;
76              }
77          };
78      }

80      // The varargs of these helpers are fixed-size groups, so each loop is bounded on the last index it reads rather
81      // than on the first: a caller that passes a partial group gets one fewer row instead of running off the end.
82      private static WmiResult<StoragePoolProperty> pools(String... namesAndIds) {
83          List<Map<StoragePoolProperty, Object>> rows = new ArrayList<>();
84          for (int i = 0; i + 1 < namesAndIds.length; i += 2) {
85              Map<StoragePoolProperty, Object> row = new EnumMap<>(StoragePoolProperty.class);
86              row.put(StoragePoolProperty.FRIENDLYNAME, namesAndIds[i]);
87              row.put(StoragePoolProperty.OBJECTID, objectId("SP", namesAndIds[i + 1]));
88              rows.add(row);
89          }
90          return result(rows, CIM_STRING);
91      }
92  
93      private static WmiResult<VirtualDiskProperty> virtualDisks(String... namesSpAndVd) {
94          List<Map<VirtualDiskProperty, Object>> rows = new ArrayList<>();
95          for (int i = 0; i + 2 < namesSpAndVd.length; i += 3) {
96              Map<VirtualDiskProperty, Object> row = new EnumMap<>(VirtualDiskProperty.class);
97              row.put(VirtualDiskProperty.FRIENDLYNAME, namesSpAndVd[i]);
98              // A VD ObjectId carries the storage pool GUID followed by the virtual disk GUID
99              row.put(VirtualDiskProperty.OBJECTID, objectId("VD", namesSpAndVd[i + 1], namesSpAndVd[i + 2]));
100             rows.add(row);
101         }
102         return result(rows, CIM_STRING);
103     }
104 
105     private static WmiResult<PhysicalDiskProperty> physicalDisks(String... nameLocationAndId) {
106         List<Map<PhysicalDiskProperty, Object>> rows = new ArrayList<>();
107         for (int i = 0; i + 2 < nameLocationAndId.length; i += 3) {
108             Map<PhysicalDiskProperty, Object> row = new EnumMap<>(PhysicalDiskProperty.class);
109             row.put(PhysicalDiskProperty.FRIENDLYNAME, nameLocationAndId[i]);
110             row.put(PhysicalDiskProperty.PHYSICALLOCATION, nameLocationAndId[i + 1]);
111             row.put(PhysicalDiskProperty.OBJECTID, objectId("PD", nameLocationAndId[i + 2]));
112             rows.add(row);
113         }
114         return result(rows, CIM_STRING);
115     }
116 
117     private static WmiResult<StoragePoolToPhysicalDiskProperty> poolToDisk(String... spAndPdGuids) {
118         List<Map<StoragePoolToPhysicalDiskProperty, Object>> rows = new ArrayList<>();
119         for (int i = 0; i + 1 < spAndPdGuids.length; i += 2) {
120             Map<StoragePoolToPhysicalDiskProperty, Object> row = new EnumMap<>(StoragePoolToPhysicalDiskProperty.class);
121             row.put(StoragePoolToPhysicalDiskProperty.STORAGEPOOL, objectId("SP", spAndPdGuids[i]));
122             row.put(StoragePoolToPhysicalDiskProperty.PHYSICALDISK, objectId("PD", spAndPdGuids[i + 1]));
123             rows.add(row);
124         }
125         // Association properties come back as CIM references, not plain strings
126         return result(rows, CIM_REFERENCE);
127     }

129     private static List<LogicalVolumeGroup> build(WmiResult<StoragePoolProperty> sp, WmiResult<VirtualDiskProperty> vds,
130             WmiResult<PhysicalDiskProperty> pds, WmiResult<StoragePoolToPhysicalDiskProperty> sppd) {
131         return WindowsLogicalVolumeGroup.buildFromWmi(sp, vds, pds, sppd, TestLvg::new);
132     }

134     private static final class TestLvg extends WindowsLogicalVolumeGroup {
135         private TestLvg(String name, Map<String, java.util.Set<String>> lvMap, java.util.Set<String> pvSet) {
136             super(name, lvMap, pvSet);
137         }
138     }

140     @Test
141     void testCorrelatesPoolWithItsDisksAndVolumes() {
142         List<LogicalVolumeGroup> lvgs = build(pools("Pool1", SP_GUID), virtualDisks("Volume1", SP_GUID, VD_GUID),
143                 physicalDisks("Disk1", "PCISlot1", PD1_GUID, "Disk2", "PCISlot2", PD2_GUID),
144                 poolToDisk(SP_GUID, PD1_GUID, SP_GUID, PD2_GUID));
145 
146         assertThat(lvgs, hasSize(1));
147         LogicalVolumeGroup lvg = lvgs.get(0);
148         assertThat(lvg.getName(), is("Pool1"));
149         assertThat("Both pooled disks are physical volumes, named and located", lvg.getPhysicalVolumes(),
150                 containsInAnyOrder("Disk1 @ PCISlot1", "Disk2 @ PCISlot2"));
151         assertThat(lvg.getLogicalVolumes().keySet(), contains("Volume1 " + VD_GUID));
152         assertThat("A logical volume maps to the pool's physical volumes",
153                 lvg.getLogicalVolumes().get("Volume1 " + VD_GUID),
154                 containsInAnyOrder("Disk1 @ PCISlot1", "Disk2 @ PCISlot2"));
155     }

157     @Test
158     void testDisksAndVolumesOfAnotherPoolAreNotMixedIn() {
159         // The regression this correlation exists to prevent: object IDs are matched by substring, so a second pool
160         // must not pick up the first pool's members.
161         List<LogicalVolumeGroup> lvgs = build(pools("Pool1", SP_GUID, "Pool2", OTHER_SP_GUID),
162                 virtualDisks("Volume1", SP_GUID, VD_GUID),
163                 physicalDisks("Disk1", "PCISlot1", PD1_GUID, "Disk2", "PCISlot2", PD2_GUID),
164                 poolToDisk(SP_GUID, PD1_GUID, OTHER_SP_GUID, PD2_GUID));
165 
166         assertThat(lvgs, hasSize(2));
167         LogicalVolumeGroup pool1 = lvgs.get(0);
168         LogicalVolumeGroup pool2 = lvgs.get(1);
169         assertThat(pool1.getName(), is("Pool1"));
170         assertThat(pool1.getPhysicalVolumes(), contains("Disk1 @ PCISlot1"));
171         assertThat(pool1.getLogicalVolumes().keySet(), contains("Volume1 " + VD_GUID));
172         assertThat(pool2.getName(), is("Pool2"));
173         assertThat(pool2.getPhysicalVolumes(), contains("Disk2 @ PCISlot2"));
174         assertThat("Pool2 owns no virtual disk", pool2.getLogicalVolumes().keySet(), is(empty()));
175     }

193     @Test
194     void testUnparseableObjectIdsDoNotThrow() {
195         // A future or unexpected ObjectId format must degrade to no correlation rather than an exception.
196         List<Map<StoragePoolProperty, Object>> rows = new ArrayList<>();
197         Map<StoragePoolProperty, Object> row = new EnumMap<>(StoragePoolProperty.class);
198         row.put(StoragePoolProperty.FRIENDLYNAME, "Pool1");
199         row.put(StoragePoolProperty.OBJECTID, "not-an-object-id");
200         rows.add(row);
201 
202         List<LogicalVolumeGroup> lvgs = build(result(rows, CIM_STRING), virtualDisks("Volume1", SP_GUID, VD_GUID),
203                 physicalDisks("Disk1", "PCISlot1", PD1_GUID), poolToDisk(SP_GUID, PD1_GUID));
204         assertThat(lvgs, hasSize(1));
205         assertThat(lvgs.get(0).getName(), is("Pool1"));
206         assertThat("An unparseable pool ID must correlate nothing, not everything", lvgs.get(0).getPhysicalVolumes(),
207                 is(empty()));
208         assertThat(lvgs.get(0).getLogicalVolumes(), is(anEmptyMap()));
209     }

211     @Test
212     void testMissingObjectIdCorrelatesNothing() {
213         // WmiUtil maps an absent property to "", and every string contains "", so an unguarded substring match would
214         // make a pool with no ObjectId absorb every physical and virtual disk on the system.
215         List<Map<StoragePoolProperty, Object>> rows = new ArrayList<>();
216         Map<StoragePoolProperty, Object> row = new EnumMap<>(StoragePoolProperty.class);
217         row.put(StoragePoolProperty.FRIENDLYNAME, "Pool1");
218         row.put(StoragePoolProperty.OBJECTID, null);
219         rows.add(row);
220 
221         List<LogicalVolumeGroup> lvgs = build(result(rows, CIM_STRING), virtualDisks("Volume1", SP_GUID, VD_GUID),
222                 physicalDisks("Disk1", "PCISlot1", PD1_GUID), poolToDisk(SP_GUID, PD1_GUID));
223         assertThat(lvgs, hasSize(1));
224         assertThat("A pool with no ObjectId must not claim every disk", lvgs.get(0).getPhysicalVolumes(), is(empty()));
225         assertThat("A pool with no ObjectId must not claim every volume", lvgs.get(0).getLogicalVolumes(),
226                 is(anEmptyMap()));
227     }

229     @Test
230     void testMultipleVolumesInOnePool() {
231         String vd2 = "{dddddddd-dddd-dddd-dddd-dddddddddddd}";
232         List<LogicalVolumeGroup> lvgs = build(pools("Pool1", SP_GUID),
233                 virtualDisks("Volume1", SP_GUID, VD_GUID, "Volume2", SP_GUID, vd2),

237         assertThat(lvgs.get(0).getLogicalVolumes().keySet(),
238                 containsInAnyOrder("Volume1 " + VD_GUID, "Volume2 " + vd2));
239     }

241     @Test
242     void testObjectIdFixtureMatchesTheProductionRegexes() {
243         // Guards the fixture itself: if these strings stopped matching, every assertion above would still pass while
244         // testing nothing, because an unmatched ObjectId is silently left as-is.
245         assertThat("SP fixture must parse",
246                 build(pools("Pool1", SP_GUID), virtualDisks(), physicalDisks("Disk1", "PCISlot1", PD1_GUID),
247                         poolToDisk(SP_GUID, PD1_GUID)).get(0).getPhysicalVolumes(),
248                 contains("Disk1 @ PCISlot1"));
249         assertThat("VD fixture must parse into 'name guid'", List.of(
250                 build(pools("Pool1", SP_GUID), virtualDisks("Volume1", SP_GUID, VD_GUID), physicalDisks(), poolToDisk())
251                         .get(0).getLogicalVolumes().keySet().iterator().next()),
252                 contains("Volume1 " + VD_GUID));
