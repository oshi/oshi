48      /** The metrics a single vendor library can answer. Every one defaults to the unavailable sentinel. */
49      private static final class VendorMetrics {
50          private double utilization = -1d;
51          private double temperature = -1d;
52          private double powerDraw = -1d;
53          private long coreClock = -1L;
54          private long memoryClock = -1L;
55          private double fanSpeed = -1d;
56  
57          VendorMetrics utilization(double value) {
58              this.utilization = value;
59              return this;
60          }
61  
62          VendorMetrics temperature(double value) {
63              this.temperature = value;
64              return this;
65          }
66  
67          VendorMetrics powerDraw(double value) {
68              this.powerDraw = value;
69              return this;
70          }
71  
72          VendorMetrics coreClock(long value) {
73              this.coreClock = value;
74              return this;
75          }
76  
77          VendorMetrics memoryClock(long value) {
78              this.memoryClock = value;
79              return this;
80          }
81  
82          VendorMetrics fanSpeed(double value) {
83              this.fanSpeed = value;
84              return this;
85          }
86  
87          /** Fills every metric with a distinct value so a test can tell which source answered. */
88          VendorMetrics all(double base) {
89              return utilization(base).temperature(base + 1).powerDraw(base + 2).coreClock((long) base + 3)
90                      .memoryClock((long) base + 4).fanSpeed(base + 5);
91          }
92      }

94      /** A {@link WmiResult} over an in-memory list of LHM sensor rows. */
95      private static final class FakeSensorResult implements WmiResult<LhmSensorProperty> {
96          private final List<String> names = new ArrayList<>();
97          private final List<Float> values = new ArrayList<>();
98  
99          FakeSensorResult add(String name, double value) {
100             names.add(name);
101             values.add((float) value);
102             return this;
103         }
104 
105         @Override
106         public int getResultCount() {
107             return names.size();
108         }
109 
110         @Override
111         public @Nullable Object getValue(LhmSensorProperty property, int index) {
112             return switch (property) {
113                 case NAME -> names.get(index);
114                 case VALUE -> values.get(index);
115                 default -> null;
116             };
117         }
118 
119         @Override
120         public int getVtType(LhmSensorProperty property) {
121             return property == LhmSensorProperty.VALUE ? WmiConstants.VT_R4 : WmiConstants.VT_BSTR;
122         }
123 
124         @Override
125         public int getCIMType(LhmSensorProperty property) {
126             return property == LhmSensorProperty.VALUE ? WmiConstants.CIM_REAL32 : WmiConstants.CIM_STRING;
127         }
128     }

295     /** Adds every LHM sensor the class knows how to read, at values distinct from the vendor stubs. */
296     private static StubWindowsGpuStats withAllLhmSensors(StubWindowsGpuStats stats, double base) {
297         return stats.withLhmSensor("Load", "GPU Core", base).withLhmSensor("Temperature", "GPU Core", base + 1)
298                 .withLhmSensor("Power", "GPU Package", base + 2).withLhmSensor("Clock", "GPU Core", base + 3)
299                 .withLhmSensor("Clock", "GPU Memory", base + 4).withLhmSensor("Control", "GPU Fan", base + 5);
300     }

302     private static void assertAllMetrics(StubWindowsGpuStats stats, double base) {
303         assertThat(stats.getGpuUtilization(), closeTo(base, EPS));
304         assertThat(stats.getTemperature(), closeTo(base + 1, EPS));
305         assertThat(stats.getPowerDraw(), closeTo(base + 2, EPS));
306         assertThat(stats.getCoreClockMhz(), is((long) base + 3));
307         assertThat(stats.getMemoryClockMhz(), is((long) base + 4));
308         assertThat(stats.getFanSpeedPercent(), closeTo(base + 5, EPS));
309     }

369     @Test
370     void testUtilizationFallsBackToEngineTicks() {
371         try (StubWindowsGpuStats stats = StubWindowsGpuStats.nvidia().withEngineTicks(1000L, 5000L)) {
372             // The tick path needs two samples, so the first call only seeds the baseline
373             assertThat(stats.getGpuUtilization(), is(-1d));
374             stats.withEngineTicks(1300L, 5700L);
375             // 300 more active ticks of a 700-tick window
376             assertThat(stats.getGpuUtilization(), closeTo(300.0 * 100 / 700, EPS));
377         }
378     }

380     @Test
381     void testUtilizationIgnoresEngineTicksWhenVendorAnswers() {
382         try (StubWindowsGpuStats stats = StubWindowsGpuStats.nvidia().withNvml().withEngineTicks(1000L, 5000L)) {
383             stats.nvml.utilization(42);
384             assertThat(stats.getGpuUtilization(), closeTo(42, EPS));
385             stats.withEngineTicks(1300L, 5700L);
386             assertThat(stats.getGpuUtilization(), closeTo(42, EPS));
387         }

390     @Test
391     void testUtilizationReturnsSentinelWhenTicksGoBackwards() {
392         try (StubWindowsGpuStats stats = StubWindowsGpuStats.nvidia().withEngineTicks(1000L, 5000L)) {
393             assertThat(stats.getGpuUtilization(), is(-1d));
394             // A counter reset, as when the adapter's PDH instance is recreated
395             stats.withEngineTicks(10L, 50L);
396             assertThat(stats.getGpuUtilization(), is(-1d));
397             // The reset sample becomes the new baseline, so the next window is measurable again
398             stats.withEngineTicks(110L, 250L);
399             assertThat(stats.getGpuUtilization(), closeTo(100.0 * 100 / 200, EPS));
400         }
401     }
