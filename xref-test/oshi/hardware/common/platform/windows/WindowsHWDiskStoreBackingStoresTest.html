47      // Stands in for GetVolumePathName on a path whose volume is its drive root
48      private static String driveRoot(String path) {
49          return path.substring(0, 3);
50      }

52      // The ObjectId form both the object IDs and the references to them carry
53      private static String objectId(String kind, String... guids) {
54          StringBuilder sb = new StringBuilder("{1}\\\\HOST\\root/Microsoft/Windows/Storage/Providers_v2\\SPACES_");
55          sb.append(kind).append(".ObjectId=\"{5b46f354-cd98-11f1-98e6-806e6f6e6963}:").append(kind).append(':');
56          for (String guid : guids) {
57              sb.append(guid);
58          }
59          return sb.append('"').toString();
60      }

62      private static <T extends Enum<T>> WmiResult<T> result(List<Map<T, Object>> rows, Map<T, Integer> cimTypes) {
63          return new WmiResult<T>() {
64              @Override
65              public int getResultCount() {
66                  return rows.size();
67              }
68  
69              @Override
70              public @Nullable Object getValue(T property, int index) {
71                  return rows.get(index).get(property);
72              }
73  
74              @Override
75              public int getVtType(T property) {
76                  int cim = getCIMType(property);
77                  return cim == CIM_UINT32 || cim == CIM_UINT16 ? VT_I4 : VT_BSTR;
78              }
79  
80              @Override
81              public int getCIMType(T property) {
82                  return cimTypes.getOrDefault(property, CIM_STRING);
83              }
84          };
85      }

87      private static WmiResult<DiskProperty> disks(Object... numberBusLocationUniqueId) {
88          List<Map<DiskProperty, Object>> rows = new ArrayList<>();
89          for (int i = 0; i + 3 < numberBusLocationUniqueId.length; i += 4) {
90              Map<DiskProperty, Object> row = new EnumMap<>(DiskProperty.class);
91              row.put(DiskProperty.NUMBER, numberBusLocationUniqueId[i]);
92              row.put(DiskProperty.BUSTYPE, numberBusLocationUniqueId[i + 1]);
93              row.put(DiskProperty.LOCATION, numberBusLocationUniqueId[i + 2]);
94              row.put(DiskProperty.UNIQUEID, numberBusLocationUniqueId[i + 3]);
95              rows.add(row);
96          }
97          Map<DiskProperty, Integer> types = new EnumMap<>(DiskProperty.class);
98          types.put(DiskProperty.NUMBER, CIM_UINT32);
99          types.put(DiskProperty.BUSTYPE, CIM_UINT16);
100         return result(rows, types);
101     }

103     private static WmiResult<VirtualDiskIdProperty> virtualDisk() {
104         Map<VirtualDiskIdProperty, Object> row = new EnumMap<>(VirtualDiskIdProperty.class);
105         row.put(VirtualDiskIdProperty.UNIQUEID, SPACE_UNIQUE_ID);
106         row.put(VirtualDiskIdProperty.OBJECTID, objectId("VD", SP_GUID, VD_GUID));
107         List<Map<VirtualDiskIdProperty, Object>> rows = new ArrayList<>();
108         rows.add(row);
109         return result(rows, new EnumMap<>(VirtualDiskIdProperty.class));
110     }
111 
112     private static WmiResult<VirtualDiskToPhysicalDiskProperty> members(String... pdGuids) {
113         String[] refs = new String[pdGuids.length];
114         for (int i = 0; i < pdGuids.length; i++) {
115             refs[i] = objectId("PD", pdGuids[i]);
116         }
117         return memberRefs(refs);
118     }
119 
120     private static WmiResult<VirtualDiskToPhysicalDiskProperty> memberRefs(String... pdRefs) {
121         List<Map<VirtualDiskToPhysicalDiskProperty, Object>> rows = new ArrayList<>();
122         for (String pd : pdRefs) {
123             Map<VirtualDiskToPhysicalDiskProperty, Object> row = new EnumMap<>(VirtualDiskToPhysicalDiskProperty.class);
124             row.put(VirtualDiskToPhysicalDiskProperty.VIRTUALDISK, objectId("VD", SP_GUID, VD_GUID));
125             row.put(VirtualDiskToPhysicalDiskProperty.PHYSICALDISK, pd);
126             rows.add(row);
127         }
128         Map<VirtualDiskToPhysicalDiskProperty, Integer> types = new EnumMap<>(VirtualDiskToPhysicalDiskProperty.class);
129         types.put(VirtualDiskToPhysicalDiskProperty.VIRTUALDISK, CIM_REFERENCE);
130         types.put(VirtualDiskToPhysicalDiskProperty.PHYSICALDISK, CIM_REFERENCE);
131         return result(rows, types);
132     }
133 
134     private static WmiResult<PhysicalDiskIdProperty> physicalDisks() {
135         List<Map<PhysicalDiskIdProperty, Object>> rows = new ArrayList<>();
136         String[][] pds = { { "2", PD1_GUID }, { "3", PD2_GUID } };
137         for (String[] pd : pds) {
138             Map<PhysicalDiskIdProperty, Object> row = new EnumMap<>(PhysicalDiskIdProperty.class);
139             row.put(PhysicalDiskIdProperty.DEVICEID, pd[0]);
140             row.put(PhysicalDiskIdProperty.OBJECTID, objectId("PD", pd[1]));
141             rows.add(row);
142         }
143         return result(rows, new EnumMap<>(PhysicalDiskIdProperty.class));
144     }

146     private static Map<String, List<String>> runnerLayout() {
147         return WindowsHWDiskStore.buildBackingStores(
148                 disks(0, 10,
149                         "Integrated : Bus 0 : Device 63667 : Function 30746 : Adapter 0 : Port 0 : Target 0 : LUN 0",
150                         "600224808EEA07B8362DCA3D18902747", 4, 15, "D:\\oshi3.vhdx", "60022480C89306D00CF0D175C5B764A1",
151                         5, 16, "", SPACE_UNIQUE_ID),
152                 virtualDisk(), members(PD2_GUID, PD1_GUID), physicalDisks(),
153                 WindowsHWDiskStoreBackingStoresTest::driveRoot);
154     }

172     @Test
173     void testUnresolvableDisksAreNotReported() {
174         Map<String, List<String>> stores = WindowsHWDiskStore.buildBackingStores(
175                 disks(4, 15, "D:\\oshi3.vhdx", "", 6, 15, "", "", 5, 16, "", "NO-SUCH-VIRTUAL-DISK"), virtualDisk(),
176                 members(PD1_GUID), physicalDisks(), path -> "");
177         // The volume could not be read, the location is missing, and the virtual disk is unknown
178         assertThat(stores, is(anEmptyMap()));
179     }

181     @Test
182     void testSpaceMembersWithoutADeviceIdAreSkipped() {
183         // One member has no DeviceId, and one is a reference in no form the patterns know, so has no physical disk
184         List<Map<PhysicalDiskIdProperty, Object>> rows = new ArrayList<>();
185         Map<PhysicalDiskIdProperty, Object> row = new EnumMap<>(PhysicalDiskIdProperty.class);
186         row.put(PhysicalDiskIdProperty.DEVICEID, "");
187         row.put(PhysicalDiskIdProperty.OBJECTID, objectId("PD", PD1_GUID));
188         rows.add(row);
189         WmiResult<PhysicalDiskIdProperty> noDeviceId = result(rows, new EnumMap<>(PhysicalDiskIdProperty.class));
190         Map<String, List<String>> stores = WindowsHWDiskStore.buildBackingStores(disks(5, 16, "", SPACE_UNIQUE_ID),
191                 virtualDisk(), memberRefs("not an object reference", objectId("PD", PD1_GUID)), noDeviceId,
192                 WindowsHWDiskStoreBackingStoresTest::driveRoot);
193         // With no member resolvable to a physical drive the Space is not reported, rather than reported unlayered
194         assertThat(stores, is(anEmptyMap()));
195     }

197     @Test
198     void testUnexpectedPropertyTypeReportsNothing() {
199         // A Number published as a string rather than a uint32 must not escape as an exception
200         List<Map<DiskProperty, Object>> rows = new ArrayList<>();
201         Map<DiskProperty, Object> row = new EnumMap<>(DiskProperty.class);
202         row.put(DiskProperty.NUMBER, "0");
203         rows.add(row);
204         WmiResult<DiskProperty> badDisks = result(rows, new EnumMap<>(DiskProperty.class));
205         assertThat(WindowsHWDiskStore.buildBackingStores(badDisks, virtualDisk(), members(), physicalDisks(),
206                 WindowsHWDiskStoreBackingStoresTest::driveRoot), is(anEmptyMap()));
207         assertThat(runnerLayout(), not(hasKey("\\\\.\\PHYSICALDRIVE1")));
208     }
