5   package oshi.hardware.common.platform.mac;

7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.List;
11  import java.util.Optional;
12  import java.util.concurrent.atomic.AtomicInteger;
13  import java.util.function.Supplier;
14  
15  import org.jspecify.annotations.Nullable;
16  import org.junit.jupiter.api.Test;
17  
18  import oshi.driver.common.mac.CoreGraphicsDisplay;
19  import oshi.hardware.DisplayInfo;
20  import oshi.hardware.DisplayInfoImpl;
21  import oshi.hardware.DisplayMode;
22  import oshi.util.EdidUtil;
23  
24  class MacDisplayTest {
25  
26      private static final DisplayMode BUILT_IN_MODE = CoreGraphicsDisplay.toMode(0, 0, 1728, 1117, 3456, 2234, 120, 0);
27      private static final DisplayMode EXTERNAL_MODE = CoreGraphicsDisplay.toMode(1728, 0, 2560, 1440, 2560, 1440, 60, 0);
28  
29      private final AtomicInteger queries = new AtomicInteger();
30      private final Supplier<List<CoreGraphicsDisplay>> cgDisplays = this::queryCoreGraphicsDisplays;
31  
32      // A built-in panel, which is the main display, and an external monitor, counting how often they are queried
33      private List<CoreGraphicsDisplay> queryCoreGraphicsDisplays() {
34          queries.incrementAndGet();
35          return List.of(new CoreGraphicsDisplay(0x610, 0xa050, 0, true, true, BUILT_IN_MODE),
36                  new CoreGraphicsDisplay(0x1e6d, 0x5b08, 0x0b0b0b0b, false, false, EXTERNAL_MODE));
37      }
38  
39      private static final class TestDisplay extends MacDisplay {
40          TestDisplay(byte[] edid, @Nullable Boolean builtIn, Supplier<List<CoreGraphicsDisplay>> cgDisplays) {
41              super(edid, "Port-HDMI@1", builtIn, cgDisplays);
42          }
43  
44          TestDisplay(DisplayInfo info, Supplier<List<CoreGraphicsDisplay>> cgDisplays) {
45              super(info, "disp0", cgDisplays);
46          }
47      }
48  
49      // An EDID carrying the identity CoreGraphics reports for the external display above
50      private static byte[] externalEdid() {
51          byte[] edid = EdidUtil.newEdidTemplate();
52          edid[8] = 0x1e;
53          edid[9] = 0x6d;
54          edid[10] = 0x08;
55          edid[11] = 0x5b;
56          edid[12] = edid[13] = edid[14] = edid[15] = 0x0b;
57          return edid;
58      }
59  
60      @Test
61      void testExternalDisplayMatchedByEdid() {
62          TestDisplay display = new TestDisplay(externalEdid(), null, cgDisplays);
63          assertThat(queries.get(), is(0));
64          assertThat(display.getCurrentMode(), is(Optional.of(EXTERNAL_MODE)));
65          assertThat(display.isBuiltIn(), is(Optional.of(Boolean.FALSE)));
66          assertThat(display.getDevicePort(), is("Port-HDMI@1"));
67          assertThat(display.isPrimary(), is(Optional.of(Boolean.FALSE)));
68      }
69  
70      @Test
71      void testExternalMainDisplayIsPrimary() {
72          Supplier<List<CoreGraphicsDisplay>> externalMain = () -> List.of(
73                  new CoreGraphicsDisplay(0x610, 0xa050, 0, true, false, BUILT_IN_MODE),
74                  new CoreGraphicsDisplay(0x1e6d, 0x5b08, 0x0b0b0b0b, false, true, EXTERNAL_MODE));
75          assertThat(new TestDisplay(externalEdid(), null, externalMain).isPrimary(), is(Optional.of(Boolean.TRUE)));
76      }
77  
78      @Test
79      void testKnownBuiltInStatusNeedsNoQuery() {
80          TestDisplay display = new TestDisplay(externalEdid(), Boolean.FALSE, cgDisplays);
81          assertThat(display.isBuiltIn(), is(Optional.of(Boolean.FALSE)));
82          assertThat(queries.get(), is(0));
83      }
84  
85      @Test
86      void testSyntheticBuiltInPanelMatchedByBuiltInFlag() {
87          DisplayInfo info = new DisplayInfoImpl("APP", "A050", null, (byte) 0, 2023, "1.4", true, 35, 22, "3456x2234",
88                  "Built-in Retina Display", "");
89          TestDisplay display = new TestDisplay(info, cgDisplays);
90          assertThat(display.getCurrentMode(), is(Optional.of(BUILT_IN_MODE)));
91          assertThat(display.isBuiltIn(), is(Optional.of(Boolean.TRUE)));
92          assertThat(display.isPrimary(), is(Optional.of(Boolean.TRUE)));
93      }
94  
95      @Test
96      void testUnmatchedDisplayHasNoModeOrBuiltInStatus() {
97          TestDisplay display = new TestDisplay(EdidUtil.newEdidTemplate(), null, cgDisplays);
98          assertThat(display.getCurrentMode().isPresent(), is(false));
99          assertThat(display.isBuiltIn().isPresent(), is(false));
100         assertThat(display.isPrimary().isPresent(), is(false));
101     }
102 }
