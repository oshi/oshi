22      private static final List<String> SP_MEMORY_TWO_BANKS = """
23              Memory:
24  
25                  Memory Slots:
26  
27                    ECC: Disabled
28                    Upgradeable Memory: Yes
29  
30                      BANK 0/DIMM0 :
31  
32                        Size: 8 GB
33                        Type: DDR4
34                        Speed: 2400 MHz
35                        Manufacturer: Samsung
36                        Part Number: M471A1K43CB1-CRC
37                        Serial Number: 12345678
38                        Status: OK
39  
40                      BANK 1/DIMM0 :
41  
42                        Size: 8 GB
43                        Type: DDR4
44                        Speed: 2400 MHz
45                        Manufacturer: Hynix
46                        Part Number: HMA81GS6AFR8N-UH
47                        Serial Number: 87654321
48                        Status: OK
49              """.lines().toList();
50  
51      @Test
52      void testParseSystemProfilerMemoryTwoBanks() {
53          List<PhysicalMemory> result = MacGlobalMemory.parseSystemProfilerMemory(SP_MEMORY_TWO_BANKS);
54          assertThat(result, hasSize(2));
55  
56          PhysicalMemory bank0 = result.get(0);
57          assertThat(bank0.getBankLabel(), is("BANK 0/DIMM0"));
58          assertThat(bank0.getCapacity(), is(greaterThan(0L)));
59          assertThat(bank0.getMemoryType(), is("DDR4"));
60          assertThat(bank0.getManufacturer(), is("Samsung"));
61          assertThat(bank0.getClockSpeed(), is(greaterThan(0L)));
62          assertThat(bank0.getPartNumber(), is("M471A1K43CB1-CRC"));
63  
64          PhysicalMemory bank1 = result.get(1);
65          assertThat(bank1.getBankLabel(), is("BANK 1/DIMM0"));
66          assertThat(bank1.getManufacturer(), is("Hynix"));
67          assertThat(bank1.getSerialNumber(), is("87654321"));
68          assertThat(bank1.getPartNumber(), is("HMA81GS6AFR8N-UH"));
69      }
70  
71      @Test
72      void testParseSystemProfilerMemoryEmpty() {
73          List<PhysicalMemory> result = MacGlobalMemory.parseSystemProfilerMemory(Collections.emptyList());
74          assertThat(result, is(empty()));
75      }

77      @Test
78      void testParseSystemProfilerMemorySingleBank() {
79          List<String> singleBank = """
80                          BANK 0/DIMM0 :
81                            Size: 16 GB
82                            Type: LPDDR4X
83                            Speed: 4267 MHz
84                            Manufacturer: Micron
85                            Part Number: MT53E1G32D4NQ-046
86                            Serial Number: ABCDEF01
87                  """.lines().toList();
88          List<PhysicalMemory> result = MacGlobalMemory.parseSystemProfilerMemory(singleBank);
89          assertThat(result, hasSize(1));
90          assertThat(result.get(0).getBankLabel(), is("BANK 0/DIMM0"));
91          assertThat(result.get(0).getMemoryType(), is("LPDDR4X"));
92          assertThat(result.get(0).getManufacturer(), is("Micron"));
93          assertThat(result.get(0).getPartNumber(), is("MT53E1G32D4NQ-046"));
94      }
95  
96      @Test
97      void testParseSystemProfilerMemoryNoColon() {
98          List<String> noColon = List.of("        BANK 0", "          Size: 4 GB", "          Type: DDR3");
99          List<PhysicalMemory> result = MacGlobalMemory.parseSystemProfilerMemory(noColon);
100         assertThat(result, hasSize(1));
101         assertThat(result.get(0).getBankLabel(), is("BANK 0"));
102     }
