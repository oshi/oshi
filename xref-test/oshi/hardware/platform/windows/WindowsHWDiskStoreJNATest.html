7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.endsWith;
9   import static org.hamcrest.Matchers.is;
10  import static org.junit.jupiter.api.Assertions.assertNotNull;
11  
12  import java.util.Locale;
13  
14  import org.junit.jupiter.api.Test;
15  import org.junit.jupiter.api.condition.EnabledOnOs;
16  import org.junit.jupiter.api.condition.OS;
17  
18  /**
19   * Exercises the native volume lookup that names the file store holding a virtual disk's backing file, which a runner
20   * with no file-backed disk would otherwise never reach.
21   */
22  @EnabledOnOs(OS.WINDOWS)
23  class WindowsHWDiskStoreJNATest {
24  
25      @Test
26      void testVolumePathNameOfSystemRoot() {
27          String systemRoot = System.getenv("SystemRoot");
28          assertNotNull(systemRoot);
29          String volume = WindowsHWDiskStoreJNA.queryVolumePathName(systemRoot);
30          assertThat(volume, endsWith("\\"));
31          assertThat(systemRoot + " should be on " + volume,
32                  systemRoot.toUpperCase(Locale.ROOT).startsWith(volume.toUpperCase(Locale.ROOT)), is(true));
33      }
34  }
