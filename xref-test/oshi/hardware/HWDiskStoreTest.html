5   package oshi.hardware;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import oshi.hardware.HWDiskStore.StoreKind;
13  
14  class HWDiskStoreTest {
15  
16      @Test
17      void testStoreKindEntry() {
18          assertThat(StoreKind.DISK.entry("/dev/sda"), is("disk:/dev/sda"));
19          assertThat(StoreKind.PARTITION.entry("disk0s2"), is("partition:disk0s2"));
20          assertThat(StoreKind.FILE.entry("/System/Volumes/Data"), is("file:/System/Volumes/Data"));
21      }
22  
23      @Test
24      void testStoreKindMatches() {
25          assertThat(StoreKind.DISK.matches("disk:/dev/sda"), is(true));
26          assertThat(StoreKind.PARTITION.matches("disk:/dev/sda"), is(false));
27          assertThat(StoreKind.FILE.matches("disk:/dev/sda"), is(false));
28          // The name of a macOS disk starts with the kind's own word, without the colon
29          assertThat(StoreKind.DISK.matches("disk0s2"), is(false));
30          assertThat(StoreKind.DISK.matches(""), is(false));
31      }
32  
33      @Test
34      void testStoreKindNameOf() {
35          assertThat(StoreKind.PARTITION.nameOf("partition:/dev/sda2"), is("/dev/sda2"));
36          assertThat(StoreKind.DISK.nameOf("partition:/dev/sda2"), is(""));
37          // Only the first colon separates the kind; Windows names keep theirs
38          assertThat(StoreKind.FILE.nameOf("file:C:\\"), is("C:\\"));
39          assertThat(StoreKind.DISK.nameOf(StoreKind.DISK.entry("\\\\.\\PHYSICALDRIVE1")), is("\\\\.\\PHYSICALDRIVE1"));
40      }
41  
42      @Test
43      void testStoreKindPrefixes() {
44          for (StoreKind kind : StoreKind.values()) {
45              assertThat(kind.getPrefix().endsWith(":"), is(true));
46              assertThat(kind.nameOf(kind.entry("x:y")), is("x:y"));
47          }
48      }
49  }
