23      /**
24       * Builds {@code envstat -x} output in the layout NetBSD 10.1 prints (captured with the {@code swsensor} test
25       * driver): keys sorted, one sensor dictionary per sensor, and a {@code device-properties} dictionary closing each
26       * device's array. Each sensor is {description, type, value, state}.
27       *
28       * @param devices each a device name followed by its sensors
29       * @return the output lines
30       */
31      public static List<String> envstatX(Object[]... devices) {
32          List<String> out = new ArrayList<>(
33                  List.of("<?xml version=\"1.0\" encoding=\"UTF-8\"?>",
34                          "<!DOCTYPE plist PUBLIC \"-//Apple Computer//DTD PLIST 1.0//EN\" "
35                                  + "\"http://www.apple.com/DTDs/PropertyList-1.0.dtd\">",
36                          "<plist version=\"1.0\">", "<dict>"));
37          for (Object[] device : devices) {
38              out.add("\t<key>" + device[0] + "</key>");
39              out.add("\t<array>");
40              for (int i = 1; i < device.length; i++) {
41                  String[] s = (String[]) device[i];
42                  out.add("\t\t<dict>");
43                  if (!s[2].isEmpty()) {
44                      out.add("\t\t\t<key>cur-value</key>");
45                      out.add("\t\t\t<integer>" + s[2] + "</integer>");
46                  }
47                  out.add("\t\t\t<key>description</key>");
48                  out.add("\t\t\t<string>" + s[0] + "</string>");
49                  out.add("\t\t\t<key>index</key>");
50                  out.add("\t\t\t<string>sensor" + (i - 1) + "</string>");
51                  out.add("\t\t\t<key>monitoring-supported</key>");
52                  out.add("\t\t\t<false/>");
53                  out.add("\t\t\t<key>state</key>");
54                  out.add("\t\t\t<string>" + s[3] + "</string>");
55                  out.add("\t\t\t<key>type</key>");
56                  out.add("\t\t\t<string>" + s[1] + "</string>");
57                  out.add("\t\t</dict>");
58              }
59              out.add("\t\t<dict>");
60              out.add("\t\t\t<key>device-properties</key>");
61              out.add("\t\t\t<dict>");
62              out.add("\t\t\t\t<key>device-class</key>");
63              out.add("\t\t\t\t<string>other</string>");
64              out.add("\t\t\t\t<key>refresh-timeout</key>");
65              out.add("\t\t\t\t<integer>0x1e</integer>");
66              out.add("\t\t\t</dict>");
67              out.add("\t\t</dict>");
68              out.add("\t</array>");
69          }
70          out.add("</dict>");
71          out.add("</plist>");
72          return out;
73      }

75      public static String[] sensor(String description, String type, String value, String state) {
76          return new String[] { description, type, value, state };
77      }

100     @Test
101     void testParseSensors() {
102         List<Sensor> sensors = Envstat.parseSensors(laptop());
103         assertThat(sensors, hasSize(18));
104         Sensor first = sensors.get(0);
105         assertThat(first.getDevice(), is("coretemp0"));
106         assertThat(first.getDescription(), is("cpu0 temperature"));
107         assertThat(first.getType(), is(Envstat.TEMPERATURE));
108         assertThat(first.getValue(), is(323_150_000L));
109         assertThat(first.isValid(), is(true));
110 
111         Sensor missingFan = sensors.get(5);
112         assertThat(missingFan.getDevice(), is("itesio0"));
113         assertThat(missingFan.getValue(), is(0L));
114         assertThat(missingFan.isValid(), is(false));
115 
116         // A warning state still carries a current value
117         Sensor charge = sensors.stream().filter(s -> s.getDescription().equals("charge")).findFirst().orElseThrow();
118         assertThat(charge.isValid(), is(true));
119     }

121     @Test
122     void testSummarize() {
123         Triplet<Double, int[], Double> summary = Envstat.summarize(Envstat.parseSensors(laptop()));
124         // Average of the two CPU sensors, ignoring the thermal zone
125         assertThat(summary.getA(), closeTo(52d, 1e-9));
126         // The invalid fan is skipped
127         assertThat(summary.getB().length, is(1));
128         assertThat(summary.getB()[0], is(2600));
129         assertThat(summary.getC(), closeTo(1.15, 1e-9));
130     }
131 
132     @Test
133     void testSummarizeNoCpuSensor() {
134         List<String> lines = envstatX(
135                 new Object[] { "acpitz0", sensor("temperature", "Temperature", "313150000", "valid") },
136                 new Object[] { "acpitz1", sensor("temperature", "Temperature", "317150000", "valid") });
137         Triplet<Double, int[], Double> summary = Envstat.summarize(Envstat.parseSensors(lines));
138         assertThat(summary.getA(), closeTo(42d, 1e-9));
139         assertThat(summary.getB().length, is(0));
140         assertThat(summary.getC(), is(0d));
141     }

143     @Test
144     void testNoDrivers() {
145         // NetBSD under QEMU prints an empty dictionary
146         List<String> lines = List.of("<?xml version=\"1.0\" encoding=\"UTF-8\"?>", "<plist version=\"1.0\">", "<dict/>",
147                 "</plist>");
148         assertThat(Envstat.parseSensors(lines), is(empty()));
149         assertThat(Envstat.parseSensors(List.of("envstat: no drivers registered")), is(empty()));
150     }
