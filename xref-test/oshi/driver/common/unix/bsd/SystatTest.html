23   * Tests for {@link Systat} parsing methods. Each parse method is exercised with sample {@code systat -ab sensors}
24   * output drawn from real OpenBSD/NetBSD systems so the tests can run on any platform.

28      @Test
29      void testParseSensorsExtractsCpuTempVoltsAndFans() {
30          // CPU rows go to cpuTemps/volts; non-cpu temp rows go to allTemps; "fan" rows to fanRPMs.
31          List<String> systat = """
32                  cpu0.temp0    55.00 degC
33                  cpu0.volt0    1.25 V DC
34                  acpitz0.temp0 40.00 degC
35                  acpitz1.temp0 42.00 degC
36                  it0.fan0    1500 RPM
37                  it0.fan1    2000 RPM
38                  """.lines().toList();
39  
40          Triplet<Double, int[], Double> r = Systat.parseSensors(systat);
41          // cpuTemps non-empty so the average of cpu temps wins over allTemps.
42          assertThat(r.getA(), is(closeTo(55.0, 1e-9)));
43          assertThat("fan count", r.getB().length, is(2));
44          assertThat(r.getB()[0], is(1500));
45          assertThat(r.getB()[1], is(2000));
46          assertThat(r.getC(), is(closeTo(1.25, 1e-9)));
47      }

49      @Test
50      void testParseSensorsFallsBackToAllTempsWhenNoCpuTemp() {
51          List<String> systat = List.of("acpitz0.temp0 30.00 degC", "acpitz1.temp0 50.00 degC");
52          Triplet<Double, int[], Double> r = Systat.parseSensors(systat);
53          // Average of acpitz0 and acpitz1.
54          assertThat(r.getA(), is(closeTo(40.0, 1e-9)));
55          assertThat("no fans", r.getB().length, is(0));
56          assertThat(r.getC(), is(closeTo(0.0, 1e-9)));
57      }

59      @Test
60      void testParseSensorsAveragesMultipleCpuTemps() {
61          List<String> systat = List.of("cpu0.temp0 50.00 degC", "cpu1.temp0 70.00 degC");
62          Triplet<Double, int[], Double> r = Systat.parseSensors(systat);
63          assertThat(r.getA(), is(closeTo(60.0, 1e-9)));
64      }

66      @Test
67      void testParseSensorsIgnoresMalformedRowsAndEmptyInput() {
68          // Single-token rows (no value) and blank lines must be skipped without NaN propagation.
69          List<String> systat = List.of("", "cpu0.temp0", "junk", "cpu0.temp0 not-a-number");
70          Triplet<Double, int[], Double> r = Systat.parseSensors(systat);
71          // "not-a-number" parses to NaN and is excluded by listAverage; result is 0.
72          assertThat(r.getA(), is(closeTo(0.0, 1e-9)));
73          assertThat("no fans", r.getB().length, is(0));
74  
75          Triplet<Double, int[], Double> empty = Systat.parseSensors(Collections.<String>emptyList());
76          assertThat(empty.getA(), is(closeTo(0.0, 1e-9)));
77          assertThat("no fans on empty", empty.getB().length, is(0));
78          assertThat(empty.getC(), is(closeTo(0.0, 1e-9)));
79      }

95      @Test
96      void testParseBatteryFieldsWatthour() {
97          List<String> systat = """
98                  acpibat0.volt0 11.10 V DC
99                  acpibat0.current0 0.500 A
100                 acpibat0.temp0 30.0 degC
101                 acpibat0.watthour0 12.00 Wh (remaining capacity)
102                 acpibat0.watthour1 24.00 Wh (last full capacity)
103                 acpibat0.watthour2 36.00 Wh (design capacity)
104                 acpibat1.volt0 99.0 V DC
105                 """.lines().toList(); // different name — ignored
106 
107         BatteryFields b = Systat.parseBatteryFields("acpibat0", systat);
108         assertThat(b.getVoltage(), is(closeTo(11.10, 1e-9)));
109         assertThat(b.getAmperage(), is(closeTo(0.500, 1e-9)));
110         assertThat(b.getTemperature(), is(closeTo(30.0, 1e-9)));
111         assertThat(b.getCapacityUnits(), is(CapacityUnits.MWH));
112         // x1000 conversion to mWh
113         assertThat(b.getCurrentCapacity(), is(12000));
114         assertThat(b.getMaxCapacity(), is(24000));
115         assertThat(b.getDesignCapacity(), is(36000));
116     }

118     @Test
119     void testParseBatteryFieldsAmphour() {
120         List<String> systat = """
121                 acpibat0.amphour0 1.50 Ah (remaining capacity)
122                 acpibat0.amphour1 3.00 Ah (last full capacity)
123                 acpibat0.amphour2 4.00 Ah (new capacity)

126         BatteryFields b = Systat.parseBatteryFields("acpibat0", systat);
127         assertThat(b.getCapacityUnits(), is(CapacityUnits.MAH));
128         assertThat(b.getCurrentCapacity(), is(1500));
129         assertThat(b.getMaxCapacity(), is(3000));
130         assertThat(b.getDesignCapacity(), is(4000));

133     @Test
134     void testParseBatteryFieldsDefaultsWhenAbsent() {
135         BatteryFields b = Systat.parseBatteryFields("acpibat0", Collections.<String>emptyList());
136         assertThat(b.getVoltage(), is(closeTo(-1.0, 1e-9)));
137         assertThat(b.getAmperage(), is(closeTo(0.0, 1e-9)));
138         assertThat(b.getTemperature(), is(closeTo(0.0, 1e-9)));
139         assertThat(b.getCapacityUnits(), is(CapacityUnits.RELATIVE));
140         assertThat(b.getCurrentCapacity(), is(0));
141         assertThat(b.getMaxCapacity(), is(1));
142         assertThat(b.getDesignCapacity(), is(1));
143     }

153     @Test
154     void testListAverageSkipsNaNs() {
155         assertThat(Systat.listAverage(List.of(10.0, Double.NaN, 20.0)), is(closeTo(15.0, 1e-9)));
156         assertThat(Systat.listAverage(List.of(Double.NaN, Double.NaN)), is(closeTo(0.0, 1e-9)));
157         assertThat(Systat.listAverage(Collections.<Double>emptyList()), is(closeTo(0.0, 1e-9)));

161     void testParseSensorsFanZeroDefaultForNonNumeric() {
162         // Fan row with a non-numeric value should default the parsed RPM to 0 rather than throw.

164         Triplet<Double, int[], Double> r = Systat.parseSensors(systat);
165         assertThat("one fan parsed", r.getB().length, is(1));
