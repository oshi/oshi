24      private static void assertEntry(MountTable.Entry e, String volume, String uuid, String path, String type,
25              String options) {
26          assertThat(e.getVolume(), is(volume));
27          assertThat(e.getUuid(), is(uuid));
28          assertThat(e.getPath(), is(path));
29          assertThat(e.getType(), is(type));
30          assertThat(e.getOptions(), is(options));
31      }

33      @Test
34      void testParseFreeBsdLibxo() {
35          // mount -p --libxo json on FreeBSD 16.0-CURRENT (cfarm430), trimmed, with an invented mount point holding
36          // spaces long enough that mount -p would separate it from the type with a single space, and an escaped quote
37          List<String> json = List.of("{\"mount\": {\"fstab\": [{\"device\":\"/dev/da0p2\",\"mntpoint\":\"/\","
38                  + "\"fstype\":\"ufs\",\"opts\":\"rw,noatime\",\"dump\":1,\"pass\":1}, "
39                  + "{\"device\":\"tmpfs\",\"mntpoint\":\"/tmp\",\"fstype\":\"tmpfs\",\"opts\":\"rw\",\"dump\":0,"
40                  + "\"pass\":0}, {\"device\":\"/dev/da1p1\",\"mntpoint\":\"/mnt/My External \\\"Backup\\\" Drive\","
41                  + "\"fstype\":\"ufs\",\"opts\":\"rw\",\"dump\":2,\"pass\":2}]}}");
42          List<MountTable.Entry> entries = MountTable.parseFreeBsdLibxo(json);
43          assertThat(entries, hasSize(3));
44          assertEntry(entries.get(0), "/dev/da0p2", "", "/", "ufs", "rw,noatime");
45          assertEntry(entries.get(1), "tmpfs", "", "/tmp", "tmpfs", "rw");
46          assertEntry(entries.get(2), "/dev/da1p1", "", "/mnt/My External \"Backup\" Drive", "ufs", "rw");
47      }

49      @Test
50      void testParseFreeBsdLibxoEscapes() {
51          // Each JSON escape a path may carry: a backslash, a tab, a slash and a unicode escape, and one ending the value
52          String path = "/mnt/a\\\\b\\tc\\/d\\u00e9\\u0041";
53          List<String> json = List.of("{\"mount\": {\"fstab\": [{\"device\":\"/dev/da1p1\",\"mntpoint\":\"" + path
54                  + "\",\"fstype\":\"ufs\",\"opts\":\"rw\",\"dump\":2,\"pass\":2}]}}");
55          List<MountTable.Entry> entries = MountTable.parseFreeBsdLibxo(json);
56          assertThat(entries, hasSize(1));
57          assertEntry(entries.get(0), "/dev/da1p1", "", "/mnt/a\\b\tc/déA", "ufs", "rw");
58      }

68      @Test
69      void testParseFstab() {
70          // mount -p on DragonFly 6.4.2 in CI, as root, with a mount point holding spaces: one tab between fields, and
71          // a space between the type and the options
72          List<MountTable.Entry> dragonfly = MountTable.parseFstab(List.of("serno/QM00001.s1d\t/\thammer2 rw\t1 1",
73                  "tmpfs\t/mnt/plain\ttmpfs rw\t0 0", "tmpfs\t/mnt/My External Backup Drive\ttmpfs rw\t0 0"));
74          assertThat(dragonfly, hasSize(3));
75          assertEntry(dragonfly.get(0), "serno/QM00001.s1d", "", "/", "hammer2", "rw");
76          assertEntry(dragonfly.get(2), "tmpfs", "", "/mnt/My External Backup Drive", "tmpfs", "rw");
77          // FreeBSD pads with tabs, the type and options apart
78          List<MountTable.Entry> freebsd = MountTable.parseFstab(
79                  List.of("/dev/da0p2\t\t/\t\t\tufs\trw,noatime \t1 1", "devfs\t\t\t/dev\t\t\tdevfs\trw\t\t0 0"));
80          assertThat(freebsd, hasSize(2));
81          assertEntry(freebsd.get(0), "/dev/da0p2", "", "/", "ufs", "rw,noatime");
82          assertEntry(freebsd.get(1), "devfs", "", "/dev", "devfs", "rw");
83          // FreeBSD 15.1 in CI, as root: a mount point of 24 characters or more is followed by a single space
84          List<MountTable.Entry> longPath = MountTable
85                  .parseFstab(List.of("/dev/md0\t\t/mnt/My External Backup Drive ufs\trw\t\t2 2"));
86          assertThat(longPath, hasSize(1));
87          assertEntry(longPath.get(0), "/dev/md0", "", "/mnt/My External Backup Drive", "ufs", "rw");
88          // The same, with options long enough that FreeBSD follows them with a space before the dump field
89          List<MountTable.Entry> longOpts = MountTable
90                  .parseFstab(List.of("/dev/md0\t\t/mnt/My External Backup Drive ufs\trw,noatime \t2 2"));
91          assertEntry(longOpts.get(0), "/dev/md0", "", "/mnt/My External Backup Drive", "ufs", "rw,noatime");
92          // No tabs at all falls back to whitespace
93          List<MountTable.Entry> spaced = MountTable.parseFstab(List.of("/dev/md0 /mnt/a ufs rw 2 2"));
94          assertEntry(spaced.get(0), "/dev/md0", "", "/mnt/a", "ufs", "rw");
95      }

97      @Test
98      void testParseNetBsd() {
99          // mount on NetBSD 10.1 (cfarm428), with an invented mount point holding spaces
100         List<String> lines = List.of("/dev/dk1 on / type ffs (noatime, local)", "tmpfs on /tmp type tmpfs (local)",
101                 "tmpfs on /var/shm type tmpfs (local)", "/dev/dk2 on /home/my files type ffs (noatime, local)");
102         List<MountTable.Entry> entries = MountTable.parseNetBsd(lines);
103         assertThat(entries, hasSize(4));
104         assertEntry(entries.get(0), "/dev/dk1", "", "/", "ffs", "(noatime, local)");
105         assertEntry(entries.get(2), "tmpfs", "", "/var/shm", "tmpfs", "(local)");
106         assertEntry(entries.get(3), "/dev/dk2", "", "/home/my files", "ffs", "(noatime, local)");
107     }

109     @Test
110     void testParseOpenBsdVerbose() {
111         // mount -v on OpenBSD 7.9 (cfarm220), then two file systems with no disk and so no DUID, which a split on
112         // whitespace read with "on" as the identifier and "type" as the path
113         List<String> lines = List.of(
114                 "/dev/sd0a (ed2d8ecd4d5492ee.a) on / type ffs (rw, local, ctime=Wed Oct  7 22:17:15 2026)",
115                 "/dev/sd0k (ed2d8ecd4d5492ee.k) on /home/my files type ffs (rw, local, nodev, nosuid)",
116                 "tmpfs on /tmp type tmpfs (rw, local, nodev, nosuid)",
117                 "server:/export on /mnt/nfs type nfs (rw, v3, udp, timeo=100, retrans=101)");
118         List<MountTable.Entry> entries = MountTable.parseOpenBsdVerbose(lines);
119         assertThat(entries, hasSize(4));
120         assertEntry(entries.get(0), "/dev/sd0a", "ed2d8ecd4d5492ee.a", "/", "ffs",
121                 "(rw, local, ctime=Wed Oct  7 22:17:15 2026)");
122         assertEntry(entries.get(1), "/dev/sd0k", "ed2d8ecd4d5492ee.k", "/home/my files", "ffs",
123                 "(rw, local, nodev, nosuid)");
124         assertEntry(entries.get(2), "tmpfs", "", "/tmp", "tmpfs", "(rw, local, nodev, nosuid)");
125         assertEntry(entries.get(3), "server:/export", "", "/mnt/nfs", "nfs", "(rw, v3, udp, timeo=100, retrans=101)");
126     }

128     @Test
129     void testParseDfInodes() {
130         // df -i from FreeBSD 16.0-CURRENT, NetBSD 10.1 and OpenBSD 7.9, header lines included. FreeBSD prints - for
131         // the percentage of a file system without inodes. NetBSD mounts tmpfs twice, which a key by volume collapsed
132         List<String> lines = """
133                 Filesystem  1K-blocks      Used     Avail Capacity iused     ifree %iused  Mounted on
134                 /dev/da0p2   80987124  57459148  17049008    77%  958125   9538641    9%   /
135                 devfs               1         0         1     0%       0         0     -   /dev
136                 Filesystem     512-blocks         Used        Avail %Cap      iUsed     iAvail %iCap Mounted on
137                 tmpfs             8375792        37504      8338288   0%        481    4187421    0% /tmp
138                 tmpfs             8375792            0      8375792   0%          1    4187901    0% /var/shm
139                 Filesystem  512-blocks      Used     Avail Capacity  iused    ifree %iused  Mounted on
140                 /dev/sd0k    619226672 377471440 210793904    65%  4246584 15494086   22%   /home
141                 /dev/sd0h     40614428  14885588 -23698120   100%   154725  2489113    6%   /usr/local stuff
142                 """.lines().toList();
143         Pair<Map<String, Long>, Map<String, Long>> inodes = MountTable.parseDfInodes(lines);
144         Map<String, Long> free = inodes.getA();
145         Map<String, Long> total = inodes.getB();
146         assertThat(free, is(aMapWithSize(6)));
147         assertThat(free.get("/"), is(9538641L));
148         assertThat(total.get("/"), is(958125L + 9538641L));
149         assertThat(total.get("/dev"), is(0L));
150         assertThat(free.get("/tmp"), is(4187421L));
151         assertThat(free.get("/var/shm"), is(4187901L));
152         assertThat(total.get("/home"), is(4246584L + 15494086L));
153         // Negative avail, a full file system, and a mount point holding a space
154         assertThat(free.get("/usr/local stuff"), is(2489113L));
155         assertThat(MountTable.parseDfInodes(Collections.emptyList()).getA(), is(anEmptyMap()));
156     }
