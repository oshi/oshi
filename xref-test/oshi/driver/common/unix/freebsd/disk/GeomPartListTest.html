106     private static final Map<String, String> MOUNTS = Map.of("da0p2", "/", "da0p1", "/boot/efi", "da1p1", "/home");
107 
108     private static List<String> names(List<HWPartition> partitions) {
109         return partitions.stream().map(HWPartition::getName).toList();
110     }
111 
112     @Test
113     void testParsePartitions() {
114         Map<String, List<HWPartition>> map = GeomPartList.parsePartitions(GEOM_PART_LIST.lines().toList(), MOUNTS,
115                 part -> part.equals("da0p2") ? 119 : 0);
116         assertThat(map.keySet(), containsInAnyOrder("da0", "da1", "da2"));
117 
118         // Only providers, each under its own disk: no consumer becomes a partition of the next geom
119         List<HWPartition> da0 = map.get("da0");
120         List<HWPartition> da1 = map.get("da1");
121         List<HWPartition> da2 = map.get("da2");
122         assertNotNull(da0);
123         assertNotNull(da1);
124         assertNotNull(da2);
125         assertThat(names(da0), contains("da0p1", "da0p2"));
126         assertThat(names(da1), contains("da1p1"));
127         assertThat(names(da2), contains("da2p1"));
128 
129         HWPartition root = da0.get(1);
130         assertThat(root.getIdentification(), is("da0p2"));
131         assertThat(root.getType(), is("freebsd-ufs"));
132         assertThat(root.getUuid(), is("d4553393-8f91-11ef-8e4d-00155dc40030"));
133         assertThat(root.getLabel(), is("rootfs"));
134         assertThat(root.getSize(), is(85626675200L));
135         assertThat(root.getMajor(), is(0));
136         assertThat(root.getMinor(), is(119));
137         assertThat(root.getMountPoint(), is("/"));
138 
139         HWPartition efi = da0.get(0);
140         assertThat(efi.getLabel(), is(""));
141         assertThat(efi.getMountPoint(), is("/boot/efi"));
142         HWPartition swap = da2.get(0);
143         assertThat(swap.getType(), is("freebsd-swap"));
144         assertThat(swap.getMountPoint(), is(""));
145         assertThat(da1.get(0).getLabel(), is("cfarm430"));
146     }

153     @Test
154     void testPartitionWithoutDetails() {
155         Map<String, List<HWPartition>> map = GeomPartList.parsePartitions(
156                 List.of("Geom name: ada0", "Providers:", "1. Name: ada0p1", "Consumers:", "1. Name: ada0"),
157                 Collections.emptyMap(), part -> 0);
158         List<HWPartition> ada0 = map.get("ada0");
159         assertNotNull(ada0);
160         assertThat(ada0.size(), is(1));
161         assertThat(ada0.get(0).getType(), is(Constants.UNKNOWN));
162         assertThat(ada0.get(0).getSize(), is(0L));

200     @Test
201     void testBsdLabelPartitionsBelongToTheDisk() {
202         Map<String, List<HWPartition>> map = GeomPartList.parsePartitions(MBR_WITH_BSD_LABEL.lines().toList(),
203                 Map.of("ada0s1a", "/"), part -> 0);
204         assertThat(map.keySet(), contains("ada0"));
205         List<HWPartition> ada0 = map.get("ada0");
206         assertNotNull(ada0);
207         assertThat(names(ada0), contains("ada0s1", "ada0s1a", "ada0s1b"));
208         assertThat(ada0.get(1).getMountPoint(), is("/"));
209         assertThat(ada0.get(1).getType(), is("freebsd-ufs"));
210     }

212     @Test
213     void testMountedByGptLabelOrId() {
214         // da0p2 mounted as /dev/gpt/rootfs, da1p1 as /dev/gptid/<rawuuid>
215         Map<String, String> mounts = Map.of("gpt/rootfs", "/", "gptid/deb7728b-8f91-11ef-8e4d-00155dc40030", "/home");
216         Map<String, List<HWPartition>> map = GeomPartList.parsePartitions(GEOM_PART_LIST.lines().toList(), mounts,
217                 part -> 0);
218         List<HWPartition> da0 = map.get("da0");
219         List<HWPartition> da1 = map.get("da1");
220         assertNotNull(da0);
221         assertNotNull(da1);
222         assertThat(da0.get(1).getMountPoint(), is("/"));
223         assertThat(da0.get(0).getMountPoint(), is(""));
224         assertThat(da1.get(0).getMountPoint(), is("/home"));
225     }
