8   import static org.hamcrest.Matchers.anEmptyMap;
9   import static org.hamcrest.Matchers.is;

12  import java.util.Collections;
13  import java.util.List;
14  import java.util.Map;

16  import org.junit.jupiter.api.Test;

18  import oshi.util.Constants;
19  import oshi.util.tuples.Triplet;
20  
21  class GeomDiskListTest {
22  
23      @Test
24      void testParseGeomDiskListTypicalOutput() {
25          List<String> geom = """
26                  Geom name: ada0
27                  Providers:
28                  1. Name: ada0
29                     Mediasize: 500107862016 (465G)
30                     Sectorsize: 512
31                     Mode: r2w2e3
32                     descr: Samsung SSD 860 EVO 500GB
33                     ident: S3Z2NB0K123456X
34                     rotationrate: 0
35  
36                  Geom name: da0
37                  Providers:
38                  1. Name: da0
39                     Mediasize: 8053063680 (7.5G)
40                     Sectorsize: 512
41                     Mode: r0w0e0
42                     descr: USB Flash Drive
43                     ident: (null)
44                     rotationrate: unknown
45                  """.lines().toList();
46  
47          Map<String, Triplet<String, String, Long>> result = GeomDiskList.parseGeomDiskList(geom);
48  
49          assertThat(result.size(), is(2));
50  
51          Triplet<String, String, Long> ada0 = result.get("ada0");
52          assertNotNull(ada0, "ada0 missing");
53          assertThat(ada0.getA(), is("Samsung SSD 860 EVO 500GB"));
54          assertThat(ada0.getB(), is("S3Z2NB0K123456X"));
55          assertThat(ada0.getC(), is(500107862016L));
56  
57          Triplet<String, String, Long> da0 = result.get("da0");
58          assertNotNull(da0, "da0 missing");
59          assertThat(da0.getA(), is("USB Flash Drive"));
60          assertThat(da0.getB(), is(""));
61          assertThat(da0.getC(), is(8053063680L));
62      }
63  
64      @Test
65      void testParseGeomDiskListEmptyInput() {
66          Map<String, Triplet<String, String, Long>> result = GeomDiskList.parseGeomDiskList(Collections.emptyList());
67          assertThat(result, is(anEmptyMap()));
68      }
69  
70      @Test
71      void testParseGeomDiskListNoMediasizeDefaultsToZero() {
72          List<String> geom = List.of("Geom name: vtbd0", "   descr: VirtIO Block Device", "   ident: serial123");
73  
74          Map<String, Triplet<String, String, Long>> result = GeomDiskList.parseGeomDiskList(geom);
75  
76          assertThat(result.size(), is(1));
77          Triplet<String, String, Long> vtbd0 = result.get("vtbd0");
78          assertNotNull(vtbd0, "vtbd0 missing");
79          assertThat(vtbd0.getA(), is("VirtIO Block Device"));
80          assertThat(vtbd0.getB(), is("serial123"));
81          assertThat(vtbd0.getC(), is(0L));
82      }
83  
84      @Test
85      void testParseGeomDiskListMissingFieldsDefaultToUnknown() {
86          List<String> geom = List.of("Geom name: nvd0", "   Mediasize: 1024000000000 (953G)");
87  
88          Map<String, Triplet<String, String, Long>> result = GeomDiskList.parseGeomDiskList(geom);
89  
90          assertThat(result.size(), is(1));
91          Triplet<String, String, Long> nvd0 = result.get("nvd0");
92          assertNotNull(nvd0, "nvd0 missing");
93          assertThat(nvd0.getA(), is(Constants.UNKNOWN));
94          assertThat(nvd0.getB(), is(Constants.UNKNOWN));
95          assertThat(nvd0.getC(), is(1024000000000L));
96      }
97  }
