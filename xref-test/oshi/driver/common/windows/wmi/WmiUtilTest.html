20      @Test
21      void testQueryToString() {
22          WmiQuery<TestProp> query = new WmiQuery<>("TestClass", TestProp.class);

105     @Test
106     // The constructor's parameters are non-null and it enforces that with requireNonNull. This test exists to prove
107     // the enforcement, so it has to pass the nulls the type system otherwise rules out.
108     @SuppressWarnings("NullAway")
109     void testWmiQueryNullValidation() {
110         assertThrows(NullPointerException.class, () -> new WmiQuery<>(null, "Class", TestProp.class));
111         assertThrows(NullPointerException.class, () -> new WmiQuery<>("NS", null, TestProp.class));
112         assertThrows(NullPointerException.class, () -> new WmiQuery<>("NS", "Class", null));
113     }

115     private static class MockWmiResult<T extends Enum<T>> implements WmiResult<T> {
116         private final int cimType;
117         private final int vtType;
118         private final @Nullable Object value;
119 
120         MockWmiResult(int cimType, int vtType, @Nullable Object value) {
121             this.cimType = cimType;
122             this.vtType = vtType;
123             this.value = value;
124         }
125 
126         @Override
127         public int getResultCount() {
128             return 1;
129         }
130 
131         @Override
132         public @Nullable Object getValue(T property, int index) {
133             return value;
134         }
135 
136         @Override
137         public int getVtType(T property) {
138             return vtType;
139         }
140 
141         @Override
142         public int getCIMType(T property) {
143             return cimType;
144         }
145     }
