5   package oshi.driver.common.windows;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import oshi.util.Constants;
13  
14  class DisplayConnectorTest {
15  
16      @Test
17      void testConnectorNameSingleConnectorHasNoIndex() {
18          // connectorInstance 0 means the adapter has a single connector of that type
19          assertThat(DisplayConnector.connectorName(5, 0), is("HDMI"));
20          assertThat(DisplayConnector.connectorName(10, 0), is("DisplayPort"));
21          assertThat(DisplayConnector.connectorName(11, 0), is("eDP"));
22          assertThat(DisplayConnector.connectorName(0x80000000, 0), is("Internal"));
23      }
24  
25      @Test
26      void testConnectorNameMultipleConnectorsAppendInstance() {
27          assertThat(DisplayConnector.connectorName(5, 1), is("HDMI-1"));
28          assertThat(DisplayConnector.connectorName(10, 2), is("DisplayPort-2"));
29      }
30  
31      @Test
32      void testConnectorNameKnownTechnologies() {
33          assertThat(DisplayConnector.connectorName(0, 0), is("VGA"));
34          assertThat(DisplayConnector.connectorName(4, 0), is("DVI"));
35          assertThat(DisplayConnector.connectorName(6, 0), is("LVDS"));
36      }
37  
38      @Test
39      void testConnectorNameUnknownTechnology() {
40          assertThat(DisplayConnector.connectorName(-1, 0), is("Other"));
41          assertThat(DisplayConnector.connectorName(999, 0), is("Other"));
42      }
43  
44      @Test
45      void testNormalizePathLowerCases() {
46          assertThat(DisplayConnector.normalizePath("\\\\?\\DISPLAY#DELA1CD#5&2a1c8d3&0&UID4353#{E6F07B5F}"),
47                  is("\\\\?\\display#dela1cd#5&2a1c8d3&0&uid4353#{e6f07b5f}"));
48      }
49  
50      @Test
51      void testNormalizePathBlankIsUnknown() {
52          assertThat(DisplayConnector.normalizePath(""), is(Constants.UNKNOWN));
53      }
54  }
