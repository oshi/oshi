57      @Test
58      void testParse() {
59          List<Reading> readings = LhmMetrics.parse(METRICS.lines().toList());
60          assertThat("Comment lines and the NaN sample are skipped", readings, hasSize(5));
61  
62          Reading load = readings.get(0);
63          assertThat(load.getMetric(), is("lhm_cpu_load_percent"));
64          assertThat(load.getHardwareId(), is("/intelcpu/0"));
65          assertThat(load.getSensorName(), is("CPU Total"));
66          assertThat(load.getValue(), closeTo(15.2225, 1e-4));
67  
68          assertThat(readings.get(1).getValue(), closeTo(37.5, 1e-9));
69          assertThat(readings.get(2).getMetric(), is(LhmMetrics.CPU_TEMPERATURE));
70          assertThat(readings.get(3).getMetric(), is(LhmMetrics.MOTHERBOARD_FAN));
71          assertThat(readings.get(3).getHardwareId(), is("/lpc/nct6798d"));
72          Reading vcore = readings.get(4);
73          assertThat(vcore.getMetric(), is(LhmMetrics.MOTHERBOARD_VOLTAGE));
74          assertThat("A trailing timestamp is not taken as the value", vcore.getValue(), closeTo(1.2, 1e-9));
75      }

84      @Test
85      void testParseKeepsTheLatestSampleOfEachSensor() {
86          // 0.9.5 lists up to five samples per sensor, newest first, each with a timestamp in milliseconds
87          List<String> history = List.of(
88                  "lhm_motherboard_fan_rpm {\"sensorName\"=\"Fan 1\", \"sensorId\"=\"/fan/0\", \"hardwareId\"=\"/lpc/nct6798d\"}"
89                          + " 1100 1763170763000",
90                  "lhm_motherboard_fan_rpm {\"sensorName\"=\"Fan 1\", \"sensorId\"=\"/fan/0\", \"hardwareId\"=\"/lpc/nct6798d\"}"
91                          + " 1200 1763170762000",
92                  "lhm_motherboard_fan_rpm {\"sensorName\"=\"Fan 2\", \"sensorId\"=\"/fan/1\", \"hardwareId\"=\"/lpc/nct6798d\"}"
93                          + " 800 1763170761000",
94                  "lhm_motherboard_fan_rpm {\"sensorName\"=\"Fan 2\", \"sensorId\"=\"/fan/1\", \"hardwareId\"=\"/lpc/nct6798d\"}"
95                          + " 850 1763170763000");
96          List<Reading> readings = LhmMetrics.parse(history);
97          assertThat("One reading per sensor", readings, hasSize(2));
98          assertThat(readings.get(0).getSensorName(), is("Fan 1"));
99          assertThat("The newest sample, listed first", readings.get(0).getValue(), closeTo(1100, 1e-9));
100         assertThat("The newest sample, listed last", readings.get(1).getValue(), closeTo(850, 1e-9));
101     }

103     @Test
104     void testReadBounded() throws IOException {
105         long later = System.nanoTime() + 60_000_000_000L;
106         assertThat(LhmMetrics.readBounded(new StringReader("a\nb\n"), 100, later), is("a\nb\n"));
107         assertThat("Too large", LhmMetrics.readBounded(new StringReader("x".repeat(101)), 100, later), is(nullValue()));
108         assertThat("Too slow", LhmMetrics.readBounded(new StringReader("a"), 100, System.nanoTime() - 1L),
109                 is(nullValue()));
110         assertThat("Too slow, checked at the end of the body too",
111                 LhmMetrics.readBounded(new StringReader(""), 100, System.nanoTime() - 1L), is(nullValue()));
112     }

121     @Test
122     void testQueryReadsTheServer() throws IOException, InterruptedException {
123         // Bound to the IPv4 address the client names; the JVM's loopback address may be ::1
124         try (ServerSocket server = new ServerSocket(0, 2, InetAddress.getByAddress(new byte[] { 127, 0, 0, 1 }))) {
125             Thread responder = new Thread(() -> respond(server, 2));
126             responder.setDaemon(true);
127             responder.start();
128             String base = "http://127.0.0.1:" + server.getLocalPort();
129             assertThat(LhmMetrics.query(base), hasSize(5));
130             assertThat("A non-200 status reads as no data", LhmMetrics.query(base + "/denied/metrics"), is(empty()));
131             responder.join(5000L);
132         }
133     }

135     /** Answers {@code count} requests: 401 for a path under {@code /denied}, otherwise the fixture. */
136     private static void respond(ServerSocket server, int count) {
137         for (int i = 0; i < count; i++) {
138             try (Socket socket = server.accept();
139                     BufferedReader in = new BufferedReader(new InputStreamReader(socket.getInputStream(), UTF_8));
140                     OutputStream out = socket.getOutputStream()) {
141                 String requestLine = in.readLine();
142                 // Skip the request headers
143                 String line = in.readLine();
144                 while (line != null && !line.isEmpty()) {
145                     line = in.readLine();
146                 }
147                 byte[] body = METRICS.getBytes(UTF_8);
148                 String head = requestLine != null && requestLine.contains("/denied")
149                         ? "HTTP/1.1 401 Unauthorized\r\nContent-Length: 0\r\nConnection: close\r\n\r\n"
150                         : "HTTP/1.1 200 OK\r\nContent-Type: text/plain\r\nContent-Length: " + body.length
151                                 + "\r\nConnection: close\r\n\r\n";
152                 out.write(head.getBytes(UTF_8));
153                 if (head.startsWith("HTTP/1.1 200")) {
154                     out.write(body);
155                 }
156                 out.flush();
157             } catch (IOException e) {
158                 return;
159             }
160         }
161     }

163     @Test
164     void testQueryUnreachableOrInvalid() {
165         long start = System.nanoTime();
166         // Port 9 (discard) is not served on any CI host
167         assertThat(LhmMetrics.query("http://127.0.0.1:9"), is(empty()));
168         assertThat("The connect timeout bounds a refused connection", (System.nanoTime() - start) / 1_000_000L,
169                 is(lessThan(5000L)));
170         assertThat(LhmMetrics.query("not a url"), is(empty()));
171         assertThat("Only http and https are requested", LhmMetrics.query("file:///C:/lhm"), is(empty()));
172         assertThat(LhmMetrics.query("ftp://127.0.0.1:8085"), is(empty()));
173     }
