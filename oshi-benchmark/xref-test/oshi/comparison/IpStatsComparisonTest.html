27  /**
28   * Compares the TCP and UDP counters read natively, from the {@code net.inet.tcp.stats} and {@code net.inet.udp.stats}
29   * sysctls on macOS and FreeBSD and from {@code perfstat_protocol} on AIX, against the same counters printed by
30   * {@code netstat -s}.
31   * <p>
32   * The native path reads selected counters at fixed offsets in a kernel structure that has no version field, and those
33   * offsets have moved between releases: FreeBSD 14.0 inserted a counter mid-structure. A wrong offset still yields a
34   * plausible number, so only a second reading of the same counter can catch it. {@code netstat} prints each counter with
35   * its own label, so this test parses that output itself rather than through the production parser, whose mapping serves
36   * several platforms' wording.
37   * <p>
38   * The counters only grow, so each is checked to lie between a native read taken before running {@code netstat} and one
39   * taken after.
40   */
41  @DisabledIf("isNotNativeSysctl")
42  class IpStatsComparisonTest {
43  
44      private static final Pattern COUNT_LINE = Pattern.compile("^\\s*(\\d+) (.+)$");
45      private static final Pattern DROPS = Pattern.compile("\\(including (\\d+) drops?\\)");
46  
47      /** Only the platforms reading these counters from a native structure. */
48      static boolean isNotNativeSysctl() {
49          PlatformEnum p = PlatformEnum.getCurrentPlatform();
50          return p != PlatformEnum.MACOS && p != PlatformEnum.FREEBSD && p != PlatformEnum.AIX;
51      }
52  
53      private static boolean isAix() {
54          return PlatformEnum.getCurrentPlatform() == PlatformEnum.AIX;
55      }
56  
57      /**
58       * Parses {@code netstat -s -p <protocol>} into counts keyed by label, with the count and any parenthesized byte
59       * count removed and plurals reduced to the singular, so {@code "1 data packet (36 bytes)"} and
60       * {@code "2 data packets (72 bytes)"} share the key {@code "data packet"}. A count inside the label, as in
61       * {@code "connections closed (including 3 drops)"}, is kept under its own key, here {@code "drop"}.
62       */
63      private static Map<String, Long> netstat(String protocol) {
64          Map<String, Long> counts = new HashMap<>();
65          for (String line : ExecutingCommand.runNative("netstat -s -p " + protocol)) {
66              Matcher m = COUNT_LINE.matcher(line);
67              if (!m.matches()) {
68                  continue;
69              }
70              String label = m.group(2);
71              Matcher drops = DROPS.matcher(label);
72              if (drops.find()) {
73                  counts.put("drop", ParseUtil.parseLongOrDefault(drops.group(1), 0L));
74              }
75              String key = label.replaceAll("\\s*\\([^)]*\\)", "").replaceAll("(\\w)s\\b", "$1").trim();
76              counts.putIfAbsent(key, ParseUtil.parseLongOrDefault(m.group(1), 0L));
77          }
78          return counts;
79      }
80  
81      private static InternetProtocolStats freshStats() {
82          // A new SystemInfo for each read, so neither read is served from the other's memoized value
83          return new SystemInfo().getOperatingSystem().getInternetProtocolStats();
84      }
85  
86      /** A count netstat printed, or -1 if it printed no such line, which no counter range contains. */
87      private static long count(Map<String, Long> netstat, String... keys) {
88          long sum = 0;
89          for (String key : keys) {
90              long value = netstat.getOrDefault(key, -1L);
91              if (value < 0) {
92                  return -1L;
93              }
94              sum += value;
95          }
96          return sum;
97      }
98  
99      /** Checks that a counter read by netstat lies between the native reads taken before and after it. */
100     private static <T> void between(List<String> failures, String name, T before, T after, ToLongFunction<T> field,
101             long command) {
102         long low = field.applyAsLong(before);
103         long high = field.applyAsLong(after);
104         if (command < low || command > high) {
105             failures.add(name + ": netstat " + command + " not within native [" + low + ", " + high + "]");
106         }
107     }

109     @RetryingComparison
110     void testTcpAgreesWithNetstat() {
111         TcpStats before = freshStats().getTCPv4Stats();
112         Map<String, Long> cmd = netstat("tcp");
113         TcpStats after = freshStats().getTCPv4Stats();
114         assertThat(cmd).as("netstat -s -p tcp counters").containsKeys("connection request", "connection accept",
115                 "embryonic connection dropped", "drop", "data packet retransmitted");
116 
117         List<String> failures = new ArrayList<>();
118         between(failures, "active opens", before, after, TcpStats::getConnectionsActive,
119                 count(cmd, "connection request"));
120         between(failures, "passive opens", before, after, TcpStats::getConnectionsPassive,
121                 count(cmd, "connection accept"));
122         between(failures, "resets", before, after, TcpStats::getConnectionsReset, count(cmd, "drop"));
123         if (isAix()) {
124             // perfstat_protocol carries no count of embryonic drops or retransmissions, which OSHI reports as 0, and
125             // its packets sent leave out the retransmitted ones, as the MIB's tcpOutSegs does
126             between(failures, "sent", before, after, TcpStats::getSegmentsSent,
127                     count(cmd, "packet sent") - count(cmd, "data packet retransmitted"));
128             between(failures, "received", before, after, TcpStats::getSegmentsReceived, count(cmd, "packet received"));
129             assertThat(after.getConnectionFailures()).as("failures, which perfstat does not count").isZero();
130             assertThat(after.getSegmentsRetransmitted()).as("retransmitted, which perfstat does not count").isZero();
131         } else {
132             between(failures, "failures", before, after, TcpStats::getConnectionFailures,
133                     count(cmd, "embryonic connection dropped"));
134             between(failures, "retransmitted", before, after, TcpStats::getSegmentsRetransmitted,
135                     count(cmd, "data packet retransmitted"));
136         }
137         // Without root, macOS estimates these three from IP-level counters instead of reading them
138         if (PlatformEnum.getCurrentPlatform() == PlatformEnum.FREEBSD) {
139             between(failures, "sent", before, after, TcpStats::getSegmentsSent, count(cmd, "data packet"));
140             between(failures, "received", before, after, TcpStats::getSegmentsReceived,
141                     count(cmd, "packet received in-sequence"));
142             between(failures, "in errors", before, after, TcpStats::getInErrors,
143                     count(cmd, "discarded for bad checksum", "discarded for bad header offset field",
144                             "discarded because packet too short"));
145         }
146         assertThat(failures).as("TCP counters disagreeing with netstat").isEmpty();
147     }
148 
149     @RetryingComparison
150     void testUdpAgreesWithNetstat() {
151         UdpStats before = freshStats().getUDPv4Stats();
152         Map<String, Long> cmd = netstat("udp");
153         UdpStats after = freshStats().getUDPv4Stats();
154         assertThat(cmd).as("netstat -s -p udp counters").containsKeys("datagram received", "datagram output",
155                 "dropped due to no socket");
156 
157         List<String> failures = new ArrayList<>();
158         between(failures, "received", before, after, UdpStats::getDatagramsReceived, count(cmd, "datagram received"));
159         between(failures, "sent", before, after, UdpStats::getDatagramsSent, count(cmd, "datagram output"));
160         if (isAix()) {
161             // AIX reads no_socket alone, without its broadcast and multicast drops. perfstat's input errors count those
162             // drops too, with no way to take them back out, so the receive errors are reported as 0
163             between(failures, "no port", before, after, UdpStats::getDatagramsNoPort,
164                     count(cmd, "dropped due to no socket"));
165             assertThat(after.getDatagramsReceivedErrors()).as("receive errors, which perfstat cannot separate")
166                     .isZero();
167             assertThat(failures).as("UDP counters disagreeing with netstat").isEmpty();
168             return;
169         }
170         assertThat(cmd).as("netstat -s -p udp counters").containsKeys("with incomplete header",
171                 "with bad data length field", "with bad checksum", "broadcast/multicast datagram undelivered");
172         between(failures, "receive errors", before, after, UdpStats::getDatagramsReceivedErrors,
173                 count(cmd, "with incomplete header", "with bad data length field", "with bad checksum"));
174         between(failures, "no port", before, after, UdpStats::getDatagramsNoPort,
175                 count(cmd, "dropped due to no socket", "broadcast/multicast datagram undelivered"));
176         assertThat(failures).as("UDP counters disagreeing with netstat").isEmpty();
177     }
