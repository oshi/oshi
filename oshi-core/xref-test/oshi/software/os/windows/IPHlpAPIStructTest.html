7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;

10  import org.junit.jupiter.api.Test;
11  import org.junit.jupiter.api.condition.DisabledIfSystemProperty;
12  import org.junit.jupiter.api.condition.EnabledOnOs;
13  import org.junit.jupiter.api.condition.OS;
14  
15  import com.sun.jna.Memory;
16  
17  import oshi.jna.platform.windows.IPHlpAPI;
18  import oshi.jna.platform.windows.IPHlpAPI.IP_ADDRESS_PREFIX;
19  import oshi.jna.platform.windows.IPHlpAPI.MIB_IPFORWARD_ROW2;
20  import oshi.jna.platform.windows.IPHlpAPI.SOCKADDR_INET;

22  /**
23   * Asserts the routing table structure layout, which is hand-mapped from {@code netioapi.h} rather than generated.
24   * <p>
25   * JNA derives the size and field placement independently of that hand mapping, which makes this test the real check on
26   * the equivalent constants the FFM twin hardcodes in {@code IPHlpAPIFFM}: {@code MIB_IPFORWARD_ROW2_SIZE} and the
27   * {@code OFFSET_ROUTE_*} values. If anything here changes, the twin must change with it.
28   * <p>
29   * The structures use only fixed-width members, so the layout depends on the address size rather than the operating
30   * system. This still has to run on Windows, because JNA reads a structure's alignment and type-mapper options from its
31   * enclosing class, which loads {@code IPHlpAPI} and fails everywhere else.
32   */
33  @EnabledOnOs(OS.WINDOWS)
34  @DisabledIfSystemProperty(named = "os.name", matches = "(?i).*netbsd.*")
35  class IPHlpAPIStructTest {
36  
37      @Test
38      void testStructureSizes() {
39          // SOCKADDR_INET is a union of SOCKADDR_IN (16) and SOCKADDR_IN6 (28), mapped as the larger arm
40          assertThat(new SOCKADDR_INET().size(), is(28));
41          // IP_ADDRESS_PREFIX adds a UINT8 PrefixLength, padded to the structure's 4-byte alignment
42          assertThat(new IP_ADDRESS_PREFIX().size(), is(32));
43          // The FFM twin hardcodes this value as MIB_IPFORWARD_ROW2_SIZE
44          assertThat("MIB_IPFORWARD_ROW2 is 104 bytes on 64-bit Windows", new MIB_IPFORWARD_ROW2().size(), is(104));
45      }
46  
47      @Test
48      void testFieldOffsetsMatchTheFfmConstants() {
49          // Write a distinct value at each offset the FFM twin declares, then confirm the JNA mapping reads it from
50          // that same place. This checks the offsets themselves, not merely that the total size happens to agree.
51          try (Memory buf = new Memory(104)) {
52              buf.clear();
53              buf.setLong(0, 0x1122334455667788L); // InterfaceLuid
54              buf.setInt(8, 42); // OFFSET_ROUTE_INTERFACE_INDEX
55              buf.setShort(12, (short) IPHlpAPI.AF_INET); // OFFSET_ROUTE_DEST_FAMILY
56              buf.setInt(16, 0x0100007F); // OFFSET_ROUTE_DEST_IPV4
57              buf.setByte(20, (byte) 0x20); // OFFSET_ROUTE_DEST_IPV6
58              buf.setByte(40, (byte) 24); // OFFSET_ROUTE_PREFIX_LENGTH
59              buf.setShort(44, (short) IPHlpAPI.AF_INET6); // OFFSET_ROUTE_NEXTHOP_FAMILY
60              buf.setInt(48, 0x0201A8C0); // OFFSET_ROUTE_NEXTHOP_IPV4
61              buf.setByte(52, (byte) 0xfe); // OFFSET_ROUTE_NEXTHOP_IPV6
62              buf.setInt(84, 256); // OFFSET_ROUTE_METRIC
63  
64              MIB_IPFORWARD_ROW2 row = new MIB_IPFORWARD_ROW2(buf);
65              row.read();
66  
67              assertThat(row.InterfaceIndex, is(42));
68              assertThat(row.DestinationPrefix.Prefix.si_family, is((short) IPHlpAPI.AF_INET));
69              assertThat(row.DestinationPrefix.Prefix.ipv4AddrOrFlowInfo, is(0x0100007F));
70              assertThat("PrefixLength sits after the 28-byte union", row.DestinationPrefix.PrefixLength, is((byte) 24));
71              assertThat("NextHop begins after the 32-byte IP_ADDRESS_PREFIX", row.NextHop.si_family,
72                      is((short) IPHlpAPI.AF_INET6));
73              assertThat("The IPv6 arm's address begins 8 bytes into the union", row.NextHop.ipv6Addr[0],
74                      is((byte) 0xfe));
75              assertThat(row.Metric, is(256));
76  
77              // Both unions carry both arms whatever the family says, so the two offsets the family selection does not
78              // exercise above are checked here. Reading them directly is what pins OFFSET_ROUTE_DEST_IPV6 and
79              // OFFSET_ROUTE_NEXTHOP_IPV4, which the backends use for the opposite family.
80              assertThat("The destination union's IPv6 arm sits at offset 20", row.DestinationPrefix.Prefix.ipv6Addr[0],
81                      is((byte) 0x20));
82              assertThat("The next hop union's IPv4 arm sits at offset 48", row.NextHop.ipv4AddrOrFlowInfo,
83                      is(0x0201A8C0));
84          }
85      }
86  }
