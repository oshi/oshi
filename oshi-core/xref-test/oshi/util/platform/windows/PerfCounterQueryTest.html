32  /**
33   * Tests the CIM type conversions in the WMI value mapping.
34   * <p>
35   * A real query returns whichever CIM type the queried performance class happens to use, so on hardware only one of the
36   * four conversions is ever reached. Stubbing the result reaches all of them, and the unsupported-type path too.
37   * <p>
38   * Windows only, unlike the wildcard equivalent: {@link PerfCounterQuery} caches
39   * {@code VersionHelpers.IsWindowsVistaOrGreater()} in a static field, so the class cannot be loaded on another OS.
40   */
41  @EnabledOnOs(OS.WINDOWS)
42  class PerfCounterQueryTest {
43  
44      private enum Prop {
45          U16, U32, U64, WHEN;
46      }
47  
48      /** A single-row WMI result stub. */
49      private static final class StubResult implements WmiResult<Prop> {
50          private final Map<Prop, Integer> cimTypes = new EnumMap<>(Prop.class);
51          private final Map<Prop, Integer> vtTypes = new EnumMap<>(Prop.class);
52          private final Map<Prop, Object> values = new EnumMap<>(Prop.class);
53          private int rows;
54  
55          private StubResult put(Prop p, int cim, int vt, Object value) {
56              cimTypes.put(p, cim);
57              vtTypes.put(p, vt);
58              values.put(p, value);
59              rows = 1;
60              return this;
61          }
62  
63          @Override
64          public int getResultCount() {
65              return rows;
66          }
67  
68          @Override
69          public @Nullable Object getValue(Prop property, int index) {
70              return values.get(property);
71          }
72  
73          @Override
74          public int getVtType(Prop property) {
75              return vtTypes.getOrDefault(property, 0);
76          }
77  
78          @Override
79          public int getCIMType(Prop property) {
80              return cimTypes.getOrDefault(property, 0);
81          }
82      }
83  
84      // One value per supported CIM type. WMI reports UINT64 as a string and datetimes in CIM format.
85      private static StubResult oneRow() {
86          return new StubResult().put(Prop.U16, CIM_UINT16, VT_I4, 16).put(Prop.U32, CIM_UINT32, VT_I4, 32)
87                  .put(Prop.U64, CIM_UINT64, VT_BSTR, "64")
88                  .put(Prop.WHEN, CIM_DATETIME, VT_BSTR, "20240115143000.000000+000");
89      }
90  
91      @Test
92      void testEachCimTypeIsConverted() {
93          Map<Prop, Long> values = PerfCounterQuery.mapValuesFromResult(Prop.class, oneRow());
94          assertThat("UINT16 widens to long", values.get(Prop.U16), is(16L));
95          assertThat("UINT32 widens unsigned", values.get(Prop.U32), is(32L));
96          assertThat("UINT64 is parsed from its string form", values.get(Prop.U64), is(64L));
97          assertThat("DATETIME becomes epoch milliseconds", values.get(Prop.WHEN), is(1705329000000L));
98      }
99  
100     @Test
101     void testUnsignedIntegersAreNotSignExtended() {
102         // 0xFFFFFFFF arrives as -1 in a 32-bit slot; as an unsigned CIM_UINT32 it must read as 4294967295.
103         StubResult result = new StubResult().put(Prop.U16, CIM_UINT16, VT_I4, 0).put(Prop.U32, CIM_UINT32, VT_I4, -1)
104                 .put(Prop.U64, CIM_UINT64, VT_BSTR, "0")
105                 .put(Prop.WHEN, CIM_DATETIME, VT_BSTR, "20240115143000.000000+000");
106         assertThat(PerfCounterQuery.mapValuesFromResult(Prop.class, result).get(Prop.U32), is(4294967295L));
107     }
108 
109     @Test
110     void testAnUnsupportedCimTypeIsRejected() {
111         // CIM_SINT32 is a type the mapping does not convert; it must fail loudly rather than silently drop the value.
112         StubResult result = new StubResult().put(Prop.U16, 3, VT_I4, 1);
113         assertThrows(ClassCastException.class, () -> PerfCounterQuery.mapValuesFromResult(Prop.class, result));
114     }
115 
116     @Test
117     void testNoRowsYieldsAnEmptyMap() {
118         assertThat(PerfCounterQuery.mapValuesFromResult(Prop.class, new StubResult()).isEmpty(), is(true));
119     }
120 
121     /**
122      * The properties of {@code Win32_OperatingSystem} that this mapping converts, one per supported CIM type. Chosen
123      * because a single real class covers all four, so no string property is present to hit the unsupported-type throw.
124      */
125     private enum OsProp {
126         OSTYPE, NUMBEROFPROCESSES, FREEPHYSICALMEMORY, LASTBOOTUPTIME;
127     }
128 
129     /**
130      * Runs the real query, rather than a stub, against a class that genuinely holds all four types.
131      * <p>
132      * The stubbed tests above encode assumptions about how WMI represents each type -- that a UINT64 arrives as a
133      * string in a BSTR, that a DATETIME arrives in CIM format. The production code assumes the same, so those tests and
134      * the code could agree with each other and both be wrong. This one takes the assumption out of the loop.
135      */
136     @Test
137     void testAgainstRealWmiDataCoveringEveryType() {
138         Map<OsProp, Long> values = PerfCounterQuery.queryValuesFromWMI(OsProp.class, "Win32_OperatingSystem");
139         assertThat("Win32_OperatingSystem should always return a row", values.keySet(),
140                 containsInAnyOrder(OsProp.values()));
141         assertThat("OSType is a small enumeration value", values.get(OsProp.OSTYPE), is(greaterThan(0L)));
142         assertThat("A running system has processes", values.get(OsProp.NUMBEROFPROCESSES), is(greaterThan(0L)));
143         assertThat("Free memory is reported in kilobytes", values.get(OsProp.FREEPHYSICALMEMORY), is(greaterThan(0L)));
144         Long boot = values.get(OsProp.LASTBOOTUPTIME);
145         assertNotNull(boot);
146         assertThat("Boot time is a plausible epoch millisecond value in the past", boot,
147                 is(both(greaterThan(946684800000L)).and(lessThanOrEqualTo(System.currentTimeMillis()))));
148     }
