5   package oshi.hardware.platform.unix.dragonflybsd;

7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.instanceOf;
9   import static org.hamcrest.Matchers.is;
10  import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
11  import static org.junit.jupiter.api.Assertions.assertThrows;
12  
13  import org.junit.jupiter.api.Test;
14  
15  import oshi.hardware.platform.unix.freebsd.FreeBsdHardwareAbstractionLayerJNA;
16  
17  /**
18   * DragonFly BSD wires every hardware component the way FreeBSD does except the processor, so its hardware abstraction
19   * layer inherits the FreeBSD one. These guard that relationship, which no runtime test can: the components themselves
20   * need a BSD to construct.
21   */
22  class DragonFlyBsdHardwareAbstractionLayerJNATest {
23  
24      @Test
25      void testInheritsTheFreeBsdWiring() {
26          assertThat(new DragonFlyBsdHardwareAbstractionLayerJNA(),
27                  is(instanceOf(FreeBsdHardwareAbstractionLayerJNA.class)));
28      }
29  
30      /**
31       * The one component DragonFly must not inherit. Its processor reads tick counters from {@code kern.cputime} where
32       * FreeBSD reads {@code kern.cp_time}, so dropping this override would silently report FreeBSD's counters.
33       */
34      @Test
35      void testProcessorIsOverridden() {
36          assertDoesNotThrow(() -> DragonFlyBsdHardwareAbstractionLayerJNA.class.getDeclaredMethod("createProcessor"),
37                  "DragonFly must declare its own createProcessor, not inherit FreeBSD's");
38      }
39  
40      /**
41       * The point of inheriting: these must not be copied back in. A re-declared method here would silently diverge from
42       * the FreeBSD one it was copied from.
43       */
44      @Test
45      void testEverythingElseIsInherited() {
46          for (String method : new String[] { "createComputerSystem", "createMemory", "createSensors", "createDisplays",
47                  "createUsbDevices", "createSoundCards", "createGraphicsCards" }) {
48              assertThrows(NoSuchMethodException.class,
49                      () -> DragonFlyBsdHardwareAbstractionLayerJNA.class.getDeclaredMethod(method),
50                      method + " must be inherited from FreeBSD, not redeclared");
51          }
52      }
53  }
