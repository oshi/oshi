61      /**
62       * Gets the disks on this machine
63       *
64       * @return a list of {@link HWDiskStore} objects representing the disks
65       */
66      public static List<HWDiskStore> getDisks() {
67          WmiQueryHandler h = Objects.requireNonNull(WmiQueryHandler.createInstance());
68          boolean comInit = false;
69          try {
70              comInit = h.initCOM();
71              DiskStats stats = populateDiskStats(null, PhysicalDiskJNA.queryDiskCounters());
72              PartitionMaps maps = queryPartitionMaps(h);
73  
74              WmiResult<DiskDriveProperty> vals = Win32DiskDrive.queryDiskDrive(h);
75              int count = vals.getResultCount();
76  
77              // Ask the storage driver directly first: it needs no privilege, no COM, and is available on
78              // every supported Windows version. Only if it leaves a disk unclassified is the Storage
79              // namespace queried, which is slower and requires Windows 8.
80              String[] diskTypes = classifyDiskTypes(vals, WindowsHWDiskStoreJNA::queryStorageProperties);
81              if (anyDiskTypeUnknown(diskTypes)) {
82                  int savedTimeout = h.getWmiTimeout();
83                  if (savedTimeout < 0) {
84                      h.setWmiTimeout(MSFT_PHYSICAL_DISK_TIMEOUT_MS);
85                  }
86                  try {
87                      fillUnknownDiskTypes(diskTypes, vals, MSFTStorage.queryPhysicalDiskMediaTypes(h));
88                  } finally {
89                      h.setWmiTimeout(savedTimeout);
90                  }
91              }
92  
93              List<HWDiskStore> result = new ArrayList<>(count);
94              for (int i = 0; i < count; i++) {
95                  WindowsHWDiskStoreJNA ds = new WindowsHWDiskStoreJNA(WmiUtil.getString(vals, DiskDriveProperty.NAME, i),
96                          (WmiUtil.getString(vals, DiskDriveProperty.MODEL, i) + " "
97                                  + WmiUtil.getString(vals, DiskDriveProperty.MANUFACTURER, i)).trim(),
98                          ParseUtil.hexStringToString(WmiUtil.getString(vals, DiskDriveProperty.SERIALNUMBER, i)),
99                          WmiUtil.getUint64(vals, DiskDriveProperty.SIZE, i), diskTypes[i]);
100 
101                 String index = Integer.toString(WmiUtil.getUint32(vals, DiskDriveProperty.INDEX, i));
102                 ds.setDiskStats(stats, index);
103                 ds.setPartitionList(buildPartitionList(maps, ds.getName()));
104                 result.add(ds);
105             }
106             return Collections.unmodifiableList(result);
107         } catch (COMException e) {
108             LOG.warn("COM exception", e);
109             return Collections.emptyList();
110         } finally {
111             if (comInit) {
112                 h.unInitCOM();
113             }
114         }
115     }

117     /**
118      * Reads a physical drive's storage properties with {@code IOCTL_STORAGE_QUERY_PROPERTY}.
119      *
120      * @param name the device path from {@code Win32_DiskDrive}'s Name
121      * @return the storage properties, or {@code null} if the device could not be opened or did not answer
122      */
123     private static @Nullable StorageProperties queryStorageProperties(String name) {
124         if (!name.startsWith(PHYSICALDRIVE_PREFIX)) {
125             return null;
126         }
127         // Zero desired access is enough for IOCTL_STORAGE_QUERY_PROPERTY and needs no privilege
128         HANDLE handle = Kernel32.INSTANCE.CreateFile(name, 0,
129                 WinNT.FILE_SHARE_READ | WinNT.FILE_SHARE_WRITE | WinNT.FILE_SHARE_DELETE, null, WinNT.OPEN_EXISTING, 0,
130                 null);
131         if (handle == null || WinBase.INVALID_HANDLE_VALUE.equals(handle)) {
132             LOG.debug("Couldn't open {} to query storage properties.", name);
133             return null;
134         }
135         try (Memory query = new Memory(STORAGE_PROPERTY_QUERY_SIZE);
136                 Memory deviceDescriptor = new Memory(STORAGE_DEVICE_DESCRIPTOR_BUFSIZE);
137                 Memory seekPenaltyDescriptor = new Memory(SEEK_PENALTY_DESCRIPTOR_SIZE);
138                 CloseableIntByReference bytesReturned = new CloseableIntByReference()) {
139             query.clear();
140             query.setInt(0, STORAGE_DEVICE_PROPERTY);
141             query.setInt(Integer.BYTES, PROPERTY_STANDARD_QUERY);
142             deviceDescriptor.clear();
143             if (!Kernel32.INSTANCE.DeviceIoControl(handle, IOCTL_STORAGE_QUERY_PROPERTY, query,
144                     STORAGE_PROPERTY_QUERY_SIZE, deviceDescriptor, STORAGE_DEVICE_DESCRIPTOR_BUFSIZE, bytesReturned,
145                     null)) {
146                 LOG.debug("Couldn't read the device descriptor for {}.", name);
147                 return null;
148             }
149             boolean removableMedia = deviceDescriptor.getByte(SDD_REMOVABLE_MEDIA_OFFSET) != 0;
150             int busType = deviceDescriptor.getInt(SDD_BUS_TYPE_OFFSET);
151 
152             // Not every storage driver implements the seek penalty property. A failure here is left as an
153             // unknown rather than a guess, so the Storage namespace fallback can answer instead.
154             Boolean incursSeekPenalty = null;
155             query.setInt(0, STORAGE_DEVICE_SEEK_PENALTY_PROPERTY);
156             seekPenaltyDescriptor.clear();
157             if (Kernel32.INSTANCE.DeviceIoControl(handle, IOCTL_STORAGE_QUERY_PROPERTY, query,
158                     STORAGE_PROPERTY_QUERY_SIZE, seekPenaltyDescriptor, SEEK_PENALTY_DESCRIPTOR_SIZE, bytesReturned,
159                     null)) {
160                 incursSeekPenalty = seekPenaltyDescriptor.getByte(SPD_INCURS_SEEK_PENALTY_OFFSET) != 0;
161             }
162             return new StorageProperties(busType, removableMedia, incursSeekPenalty);
163         } finally {
164             Kernel32.INSTANCE.CloseHandle(handle);
165         }
166     }

168     private static PartitionMaps queryPartitionMaps(WmiQueryHandler h) {
169         PartitionMaps maps = new PartitionMaps();
170 
171         // Map drives to partitions
172         WmiResult<DriveToPartitionProperty> drivePartitionMap = Win32DiskDriveToDiskPartition.queryDriveToPartition(h);
173         for (int i = 0; i < drivePartitionMap.getResultCount(); i++) {
174             mapDriveToPartition(maps, WmiUtil.getRefString(drivePartitionMap, DriveToPartitionProperty.ANTECEDENT, i),
175                     WmiUtil.getRefString(drivePartitionMap, DriveToPartitionProperty.DEPENDENT, i));
176         }
177 
178         // Map partitions to logical disks
179         WmiResult<DiskToPartitionProperty> diskPartitionMap = Win32LogicalDiskToPartition.queryDiskToPartition(h);
180         for (int i = 0; i < diskPartitionMap.getResultCount(); i++) {
181             long size = WmiUtil.getUint64(diskPartitionMap, DiskToPartitionProperty.ENDINGADDRESS, i)
182                     - WmiUtil.getUint64(diskPartitionMap, DiskToPartitionProperty.STARTINGADDRESS, i) + 1L;
183             mapPartitionToLogicalDrive(maps,
184                     WmiUtil.getRefString(diskPartitionMap, DiskToPartitionProperty.ANTECEDENT, i),
185                     WmiUtil.getRefString(diskPartitionMap, DiskToPartitionProperty.DEPENDENT, i), size);
186         }
187 
188         // Get all partitions and create objects
189         WmiResult<DiskPartitionProperty> hwPartitionQueryMap = Win32DiskPartition.queryPartition(h);
190         for (int i = 0; i < hwPartitionQueryMap.getResultCount(); i++) {
191             String deviceID = WmiUtil.getString(hwPartitionQueryMap, DiskPartitionProperty.DEVICEID, i);
192             List<Pair<String, Long>> logicalDrives = maps.getPartitionToLogicalDriveMap().get(deviceID);
193             if (logicalDrives == null) {
194                 continue;
195             }
196             for (int j = 0; j < logicalDrives.size(); j++) {
197                 Pair<String, Long> logicalDrive = logicalDrives.get(j);
198                 if (logicalDrive != null && !logicalDrive.getA().isEmpty()) {
199                     char[] volumeChr = new char[GUID_BUFSIZE];
200                     Kernel32.INSTANCE.GetVolumeNameForVolumeMountPoint(logicalDrive.getA(), volumeChr, GUID_BUFSIZE);
201                     String uuid = ParseUtil.parseUuidOrDefault(new String(volumeChr).trim(), "");
202                     char[] labelChr = new char[LABEL_BUFSIZE];
203                     String label = "";
204                     if (Kernel32.INSTANCE.GetVolumeInformation(logicalDrive.getA(), labelChr, LABEL_BUFSIZE, null, null,
205                             null, null, 0)) {
206                         label = new String(labelChr).trim();
207                     } else {
208                         int error = Kernel32.INSTANCE.GetLastError();
209                         LOG.debug("Failed to get volume label for {}: error code {}", logicalDrive.getA(), error);
210                     }
211                     HWPartition pt = new HWPartition(
212                             WmiUtil.getString(hwPartitionQueryMap, DiskPartitionProperty.NAME, i),
213                             WmiUtil.getString(hwPartitionQueryMap, DiskPartitionProperty.TYPE, i),
214                             WmiUtil.getString(hwPartitionQueryMap, DiskPartitionProperty.DESCRIPTION, i), uuid, label,
215                             logicalDrive.getB(),
216                             WmiUtil.getUint32(hwPartitionQueryMap, DiskPartitionProperty.DISKINDEX, i),
217                             WmiUtil.getUint32(hwPartitionQueryMap, DiskPartitionProperty.INDEX, i),
218                             logicalDrive.getA());
219                     maps.getPartitionMap().computeIfAbsent(deviceID, x -> new ArrayList<>()).add(pt);
220                 }
221             }
222         }
223         return maps;
224     }
