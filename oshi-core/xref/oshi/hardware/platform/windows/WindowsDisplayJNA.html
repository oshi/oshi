86      /**
87       * Gets Display Information
88       *
89       * @return A list of Display objects representing monitors, etc.
90       */
91      public static List<Display> getDisplays() {
92          List<Display> displays = new ArrayList<>();
93  
94          // Map every active connector's device interface path to its connector name (e.g. "HDMI", "DisplayPort-1").
95          Map<String, String> portByPath = queryConnectorPorts();
96  
97          HANDLE hDevInfo = SU.SetupDiGetClassDevs(GUID_DEVINTERFACE_MONITOR, null, null,
98                  SetupApi.DIGCF_PRESENT | SetupApi.DIGCF_DEVICEINTERFACE);
99          if (!hDevInfo.equals(WinBase.INVALID_HANDLE_VALUE)) {
100             try (CloseableSpDeviceInterfaceData deviceInterfaceData = new CloseableSpDeviceInterfaceData();
101                     CloseableSpDevinfoData info = new CloseableSpDevinfoData()) {
102                 deviceInterfaceData.cbSize = deviceInterfaceData.size();
103 
104                 for (int memberIndex = 0; SU.SetupDiEnumDeviceInfo(hDevInfo, memberIndex, info); memberIndex++) {
105                     HKEY key = SU.SetupDiOpenDevRegKey(hDevInfo, info, SetupApi.DICS_FLAG_GLOBAL, 0, SetupApi.DIREG_DEV,
106                             WinNT.KEY_QUERY_VALUE);
107                     try {
108                         byte[] edid = new byte[1];
109 
110                         try (CloseableIntByReference pType = new CloseableIntByReference();
111                                 CloseableIntByReference lpcbData = new CloseableIntByReference()) {
112                             if (ADV.RegQueryValueEx(key, "EDID", 0, pType, edid,
113                                     lpcbData) == WinError.ERROR_MORE_DATA) {
114                                 edid = new byte[lpcbData.getValue()];
115                                 if (ADV.RegQueryValueEx(key, "EDID", 0, pType, edid,
116                                         lpcbData) == WinError.ERROR_SUCCESS) {
117                                     String port = lookupPort(hDevInfo, info, deviceInterfaceData, portByPath);
118                                     displays.add(new WindowsDisplayJNA(edid, port));
119                                 }
120                             }
121                         }
122                     } finally {
123                         Advapi32.INSTANCE.RegCloseKey(key);
124                     }
125                 }
126             } finally {
127                 SU.SetupDiDestroyDeviceInfoList(hDevInfo);
128             }
129         }
130         return displays;
131     }

133     // Resolves the connector name for the current device by fetching its device interface path and looking it up in the
134     // CCD-derived map. Returns the sentinel if the interface or path cannot be obtained.
135     private static String lookupPort(HANDLE hDevInfo, CloseableSpDevinfoData info,
136             CloseableSpDeviceInterfaceData deviceInterfaceData, Map<String, String> portByPath) {
137         if (!SU.SetupDiEnumDeviceInterfaces(hDevInfo, info.getPointer(), GUID_DEVINTERFACE_MONITOR, 0,
138                 deviceInterfaceData)) {
139             return Constants.UNKNOWN;
140         }
141         String path = getDeviceInterfacePath(hDevInfo, deviceInterfaceData);
142         if (path == null) {
143             return Constants.UNKNOWN;
144         }
145         return portByPath.getOrDefault(DisplayConnector.normalizePath(path), Constants.UNKNOWN);
146     }

148     // Two-call SetupDiGetDeviceInterfaceDetail: first for the required size, then to read the device path.
149     private static @Nullable String getDeviceInterfacePath(HANDLE hDevInfo,
150             CloseableSpDeviceInterfaceData deviceInterfaceData) {
151         try (CloseableIntByReference requiredSize = new CloseableIntByReference()) {
152             SU.SetupDiGetDeviceInterfaceDetail(hDevInfo, deviceInterfaceData, null, 0, requiredSize, null);
153             int size = requiredSize.getValue();
154             if (size <= DETAIL_PATH_OFFSET) {
155                 return null;
156             }
157             try (Memory detail = new Memory(size)) {
158                 detail.clear();
159                 detail.setInt(0, DETAIL_CBSIZE);
160                 if (SU.SetupDiGetDeviceInterfaceDetail(hDevInfo, deviceInterfaceData, detail, size, requiredSize,
161                         null)) {
162                     return detail.getWideString(DETAIL_PATH_OFFSET);
163                 }
164             }
165         }
166         return null;
167     }

169     // Builds a map from normalized monitor device interface path to connector name, from the CCD active paths. A
170     // topology change between sizing and querying the buffers makes QueryDisplayConfig fail with
171     // ERROR_INSUFFICIENT_BUFFER, which is retryable by re-sizing.
172     private static Map<String, String> queryConnectorPorts() {
173         User32 u32 = User32.INSTANCE;
174         for (int attempt = 0; attempt < QDC_ATTEMPTS; attempt++) {
175             Map<String, String> map = queryConnectorPortsOnce(u32);
176             if (map != null) {
177                 return map;
178             }
179         }
180         LOG.debug("Display configuration kept changing; unable to map connectors.");
181         return new HashMap<>();
182     }
183 
184     // Returns null if the buffers were too small and the caller should re-size and retry.
185     private static @Nullable Map<String, String> queryConnectorPortsOnce(User32 u32) {
186         Map<String, String> map = new HashMap<>();
187         try (CloseableIntByReference numPaths = new CloseableIntByReference();
188                 CloseableIntByReference numModes = new CloseableIntByReference()) {
189             if (u32.GetDisplayConfigBufferSizes(DisplayConnector.QDC_ONLY_ACTIVE_PATHS, numPaths,
190                     numModes) != WinError.ERROR_SUCCESS) {
191                 return map;
192             }
193             int pathCount = numPaths.getValue();
194             int modeCount = numModes.getValue();
195             if (pathCount <= 0) {
196                 return map;
197             }
198             try (Memory paths = new Memory((long) pathCount * DisplayConnector.PATH_INFO_SIZE);
199                     Memory modes = new Memory(Math.max(1L, (long) modeCount * DisplayConnector.MODE_INFO_SIZE))) {
200                 paths.clear();
201                 modes.clear();
202                 int rc = u32.QueryDisplayConfig(DisplayConnector.QDC_ONLY_ACTIVE_PATHS, numPaths, paths, numModes,
203                         modes, null);
204                 if (rc == WinError.ERROR_INSUFFICIENT_BUFFER) {
205                     return null;
206                 }
207                 if (rc != WinError.ERROR_SUCCESS) {
208                     return map;
209                 }
210                 int actualPaths = numPaths.getValue();
211                 for (int i = 0; i < actualPaths; i++) {
212                     long base = (long) i * DisplayConnector.PATH_INFO_SIZE;
213                     int flags = paths.getInt(base + DisplayConnector.PATH_FLAGS_OFFSET);
214                     if ((flags & DisplayConnector.PATH_ACTIVE) == 0) {
215                         continue;
216                     }
217                     long adapterId = paths.getLong(base + DisplayConnector.PATH_TARGET_ADAPTER_ID_OFFSET);
218                     int targetId = paths.getInt(base + DisplayConnector.PATH_TARGET_ID_OFFSET);
219                     addConnector(map, u32, adapterId, targetId);
220                 }
221             }
222         }
223         return map;
224     }

226     // Fetches one target's DISPLAYCONFIG_TARGET_DEVICE_NAME and records its device path -> connector name.
227     private static void addConnector(Map<String, String> map, User32 u32, long adapterId, int targetId) {
228         try (Memory tdn = new Memory(DisplayConnector.TARGET_DEVICE_NAME_SIZE)) {
229             tdn.clear();
230             tdn.setInt(0, DisplayConnector.DEVICE_INFO_GET_TARGET_NAME);
231             tdn.setInt(DisplayConnector.TDN_HEADER_SIZE_OFFSET, DisplayConnector.TARGET_DEVICE_NAME_SIZE);
232             tdn.setLong(DisplayConnector.TDN_HEADER_ADAPTER_ID_OFFSET, adapterId);
233             tdn.setInt(DisplayConnector.TDN_HEADER_ID_OFFSET, targetId);
234             if (u32.DisplayConfigGetDeviceInfo(tdn) != WinError.ERROR_SUCCESS) {
235                 return;
236             }
237             int outputTechnology = tdn.getInt(DisplayConnector.TDN_OUTPUT_TECHNOLOGY_OFFSET);
238             int connectorInstance = tdn.getInt(DisplayConnector.TDN_CONNECTOR_INSTANCE_OFFSET);
239             String key = DisplayConnector
240                     .normalizePath(tdn.getWideString(DisplayConnector.TDN_MONITOR_DEVICE_PATH_OFFSET));
241             if (!Constants.UNKNOWN.equals(key)) {
242                 map.put(key, DisplayConnector.connectorName(outputTechnology, connectorInstance));
243             }
244         }
245     }
