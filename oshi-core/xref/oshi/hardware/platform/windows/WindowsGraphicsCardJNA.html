43  /**
44   * Graphics Card obtained from the Windows registry, with VRAM sourced from DXGI.
45   *
46   * <p>
47   * VRAM detection priority:
48   * <ol>
49   * <li>{@code DXGI_ADAPTER_DESC.DedicatedVideoMemory}: the authoritative Windows API value, not subject to the 2 GiB cap
50   * that affects the 32-bit registry field.</li>
51   * <li>{@code HardwareInformation.qwMemorySize} (64-bit registry value): used when DXGI enumeration is unavailable or no
52   * adapter match is found.</li>
53   * </ol>
54   *
55   * <p>
56   * {@code HardwareInformation.MemorySize} (32-bit) is intentionally not used: Windows writes the sentinel value
57   * {@code 0x7FFFF000} (~2 GiB) into this field for GPUs with more than 2 GiB of dedicated VRAM, making it unreliable for
58   * modern discrete GPUs.
59   *
60   * <p>
61   * When DXGI is available, ghost adapters (stale registry entries from hardware no longer present) are excluded because
62   * {@code IDXGIFactory::EnumAdapters} only enumerates physically present adapters. The returned list is ordered to match
63   * DXGI enumeration order, which places the primary desktop adapter first.
64   *
65   * <p>
66   * Dynamic metrics (utilization, VRAM used, shared memory, ticks) are sourced from PDH GPU Engine / GPU Adapter Memory
67   * counters (Windows 10 1709+) and optionally from LibreHardwareMonitor WMI sensors when LHM is running.
68   */
69  @ThreadSafe
70  final class WindowsGraphicsCardJNA extends WindowsGraphicsCard {
71

108     /**
109      * public method used by {@code AbstractHardwareAbstractionLayer} to access the graphics cards.
110      *
111      * <p>
112      * When DXGI is available, ghost adapters are excluded and the list is ordered with the primary desktop adapter
113      * first. On systems without DXGI, all registry entries are returned in registry key order.
114      *
115      * @return List of {@link WindowsGraphicsCardJNA} objects.
116      */
117     public static List<GraphicsCard> getGraphicsCards() {
118         // Query DXGI once. Fails gracefully to empty list if unavailable.
119         List<DxgiAdapterInfo> dxgiAdapters = DxgiUtilJNA.queryAdapters();
120         boolean dxgiAvailable = !dxgiAdapters.isEmpty();
121         // Mutable copy for match-and-consume (prevents same adapter matching two registry entries).
122         List<DxgiAdapterInfo> remainingDxgi = new ArrayList<>(dxgiAdapters);
123 
124         // Build LHM parent map: GPU name (normalized) -> LHM identifier
125         Map<String, String> lhmParentMap = buildLhmParentMap();
126 
127         // When DXGI is available, collect cards keyed by their DXGI enumeration index so the
128         // final list is ordered primary-adapter-first (DXGI guarantees adapter 0 is the primary
129         // desktop adapter). Registry entries with no DXGI match are ghost adapters and excluded.
130         // When DXGI is unavailable, fall back to simple insertion order.
131         TreeMap<Integer, GraphicsCard> dxgiOrdered = new TreeMap<>();
132         List<GraphicsCard> cardList = new ArrayList<>();
133 
134         int index = 1;
135         String[] keys = Advapi32Util.registryGetKeys(WinReg.HKEY_LOCAL_MACHINE, DISPLAY_DEVICES_REGISTRY_PATH);
136         for (String key : keys) {
137             if (!key.startsWith("0")) {
138                 continue;
139             }
140 
141             try {
142                 String fullKey = DISPLAY_DEVICES_REGISTRY_PATH + key;
143                 if (!Advapi32Util.registryValueExists(WinReg.HKEY_LOCAL_MACHINE, fullKey, ADAPTER_STRING)) {
144                     continue;
145                 }
146 
147                 String name = RegistryUtil.getStringValue(WinReg.HKEY_LOCAL_MACHINE, fullKey, DRIVER_DESC, 0);
148                 String deviceId = "VideoController" + index++;
149                 String vendor = RegistryUtil.getStringValue(WinReg.HKEY_LOCAL_MACHINE, fullKey, VENDOR, 0);
150                 String versionInfo = RegistryUtil.getStringValue(WinReg.HKEY_LOCAL_MACHINE, fullKey, DRIVER_VERSION, 0);
151 
152                 // Parse PCI vendor/device IDs from MatchingDeviceId (e.g. "pci\ven_8086&dev_56a0&...")
153                 String matchingDeviceId = RegistryUtil.getStringValue(WinReg.HKEY_LOCAL_MACHINE, fullKey,
154                         MATCHING_DEVICE_ID, 0);
155                 Pair<Integer, Integer> pciIds = ParseUtil.parseDeviceIdToVendorProductIds(matchingDeviceId);
156                 int pciVendorId = pciIds == null ? 0 : pciIds.getA();
157                 int pciDeviceId = pciIds == null ? 0 : pciIds.getB();
158 
159                 // Primary: DXGI DedicatedVideoMemory.
160                 // Track whether a DXGI match was found separately from the vram value, so that a
161                 // legitimate DedicatedVideoMemory == 0 (e.g. a software/render-only adapter) is
162                 // preserved and does not trigger the registry fallback.
163                 long vram = -1L;
164                 int dxgiIndex = -1;
165                 String luidPrefix = "";
166                 int pciBusNumber = -1;
167                 String pciBusId = "";
168                 String locationInfo = RegistryUtil.getStringValue(WinReg.HKEY_LOCAL_MACHINE, fullKey,
169                         LOCATION_INFORMATION, 0);
170                 DxgiAdapterInfo dxgiMatch = DxgiUtilJNA.findMatch(remainingDxgi, pciVendorId, pciDeviceId, name);
171                 if (dxgiMatch != null) {
172                     vram = dxgiMatch.getDedicatedVideoMemory();
173                     dxgiIndex = dxgiAdapters.indexOf(dxgiMatch);
174                     luidPrefix = buildLuidPrefix(dxgiMatch);
175                     pciBusNumber = parsePciBusNumber(locationInfo);
176                     pciBusId = buildPciBusId(locationInfo);
177                 } else if (dxgiAvailable) {
178                     // DXGI is available but this registry entry has no matching adapter:
179                     // it is a ghost device (stale driver from hardware no longer present). Skip it.
180                     continue;
181                 }
182 
183                 // Fallback: 64-bit registry value qwMemorySize, only when DXGI had no match.
184                 if (vram < 0 && Advapi32Util.registryValueExists(WinReg.HKEY_LOCAL_MACHINE, fullKey, QW_MEMORY_SIZE)) {
185                     Object regValue = Advapi32Util.registryGetValue(WinReg.HKEY_LOCAL_MACHINE, fullKey, QW_MEMORY_SIZE);
186                     vram = registryValueToVram(regValue);
187                 }
188 
189                 // Normalise sentinel: if still unresolved report 0.
190                 if (vram < 0) {
191                     vram = 0L;
192                 }
193 
194                 // HardwareInformation.MemorySize (32-bit) is intentionally omitted: Windows caps
195                 // it at 0x7FFFF000 (~2 GiB) for GPUs with more VRAM, making it unreliable.
196 
197                 String lhmParent = lhmParentMap.getOrDefault(DxgiUtilJNA.normalizeName(Util.isBlank(name) ? "" : name),
198                         "");
199 
200                 GraphicsCard card = new WindowsGraphicsCardJNA(ParseUtil.getStringValueOrUnknown(name),
201                         ParseUtil.getStringValueOrUnknown(deviceId), ParseUtil.getStringValueOrUnknown(vendor),
202                         ParseUtil.getStringValueOrUnknown(versionInfo), vram, luidPrefix, lhmParent, pciBusNumber,
203                         pciBusId);
204                 // Remove dxgiMatch from remainingDxgi only after the card is successfully
205                 // constructed. This ensures that if earlier registry reads in this try block
206                 // throw a Win32Exception, dxgiMatch remains in remainingDxgi and is still
207                 // available for subsequent iterations or WMI fallback processing.
208                 if (dxgiMatch != null) {
209                     remainingDxgi.remove(dxgiMatch);
210                 }
211                 if (dxgiIndex >= 0) {
212                     dxgiOrdered.put(dxgiIndex, card);
213                 } else {
214                     cardList.add(card);
215                 }
216             } catch (Win32Exception e) {
217                 if (e.getErrorCode() != WinError.ERROR_ACCESS_DENIED) {
218                     // Ignore access denied errors, re-throw others
219                     throw e;
220                 }
221             }
222         }
223 
224         // Merge: DXGI-ordered cards first (primary adapter at index 0), then any non-DXGI cards.
225         List<GraphicsCard> result = new ArrayList<>(dxgiOrdered.values());
226         result.addAll(cardList);
227 
228         if (result.isEmpty()) {
229             return getGraphicsCardsFromWmi(remainingDxgi, lhmParentMap);
230         }
231         return result;
232     }

245     // fall back if something went wrong
246     private static List<GraphicsCard> getGraphicsCardsFromWmi(List<DxgiAdapterInfo> dxgiAdapters,
247             Map<String, String> lhmParentMap) {
248         if (!IS_VISTA_OR_GREATER) {
249             return Collections.emptyList();
250         }
251         return buildFromWmi(dxgiAdapters, lhmParentMap, Win32VideoControllerJNA.queryVideoController(),
252                 WindowsGraphicsCardJNA::buildLuidPrefix, WindowsGraphicsCardJNA::new);
253     }

255     /**
256      * Queries LHM WMI for GPU hardware entries and returns a map from normalized GPU name to LHM parent identifier.
257      * Returns an empty map if LHM is not running. If two GPU entries normalize to the same name the key is mapped to an
258      * empty string so callers skip LHM for that ambiguous name rather than returning the wrong adapter's metrics.
259      *
260      * @return map of normalized GPU name to LHM hardware identifier
261      */
262     private static Map<String, String> buildLhmParentMap() {
263         Map<String, String> map = new HashMap<>();
264         try {
265             WmiResult<LhmHardwareProperty> hw = LhmSensorJNA.queryGpuHardware();
266             for (int i = 0; i < hw.getResultCount(); i++) {
267                 String identifier = WmiUtil.getString(hw, LhmHardwareProperty.IDENTIFIER, i);
268                 String hwName = WmiUtil.getString(hw, LhmHardwareProperty.NAME, i);
269                 if (!identifier.isEmpty() && !hwName.isEmpty()) {
270                     String norm = DxgiUtilJNA.normalizeName(hwName);
271                     if (map.containsKey(norm)) {
272                         // Two adapters with the same normalized name: mark ambiguous so neither
273                         // is used (empty string is the "skip LHM" sentinel for callers).
274                         map.put(norm, "");
275                     } else {
276                         map.put(norm, identifier);
277                     }
278                 }
279             }
280         } catch (Exception e) {
281             LOG.debug("LHM GPU hardware query failed (LHM may not be running)", e);
282         }
283         return map;
284     }

328     /**
329      * Builds the PDH LUID instance prefix for the given DXGI adapter. The prefix has the form
330      * {@code luid_0xHHHHHHHH_0xLLLLLLLL_phys_0} matching the Windows PDH GPU Engine and GPU Adapter Memory counter
331      * instance names.
332      *
333      * <p>
334      * The LUID is read directly from {@code DXGI_ADAPTER_DESC.AdapterLuid}, so this method works correctly on multi-GPU
335      * systems. A zero LUID (both parts zero) indicates the adapter did not supply a valid LUID; in that case an empty
336      * string is returned and PDH metrics will report {@code -1}.
337      *
338      * @param adapter the DXGI adapter info containing the LUID
339      * @return PDH LUID instance prefix string, or empty string if the LUID is zero
340      */
341     private static String buildLuidPrefix(DxgiAdapterInfo adapter) {
342         String prefix = DxgiUtil.buildLuidPrefix(adapter);
343         if (!prefix.isEmpty()) {
344             return prefix;
345         }
346         // Zero LUID is invalid; fall back to PDH enumeration for single-GPU case.
347         return buildLuidPrefixFromPdh();
348     }
349 
350     /**
351      * Fallback LUID prefix discovery by enumerating PDH GPU Adapter Memory instances. Used when the DXGI adapter
352      * reports a zero LUID.
353      *
354      * <p>
355      * Only reliable on single-GPU systems: when multiple GPU Adapter Memory instances are present, the correct
356      * per-adapter mapping cannot be determined without a valid LUID, so an empty string is returned and PDH metrics
357      * will report {@code -1}.
358      *
359      * @return PDH LUID instance prefix string, or empty string if not determinable
360      */
361     private static String buildLuidPrefixFromPdh() {
362         Pair<List<String>, Map<GpuAdapterMemoryProperty, List<Long>>> adapterData = GpuInformationJNA
363                 .queryGpuAdapterMemoryCounters();
364         List<String> instances = adapterData.getA();
365         if (instances.isEmpty()) {
366             return "";
367         }
368         // GPU Adapter Memory instance names have the form: luid_0xHHHH_0xLLLL_phys_0
369         // There is one instance per physical adapter. We return the full instance name as the prefix.
370         // If there is only one adapter, use it directly.
371         if (instances.size() == 1) {
372             return instances.get(0);
373         }
374         // Multiple adapters: we cannot reliably match by name here since GPU Adapter Memory
375         // instances do not carry a name. Return empty; callers will get -1 for PDH metrics.
376         // A future improvement could correlate via DXGI LUID enumeration.
377         LOG.debug("Multiple GPU Adapter Memory instances found ({}); LUID matching not yet implemented for multi-GPU",
378                 instances.size());
379         return "";
380     }
