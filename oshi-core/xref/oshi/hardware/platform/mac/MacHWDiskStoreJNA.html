61      private MacHWDiskStoreJNA(String name, String model, String serial, long size, String diskType,
62              Optional<List<String>> backingStores, DASessionRef session, Map<String, String> mountPointMap,
63              Map<CFKey, CFStringRef> cfKeyMap) {
64          super(name, model, serial, size, diskType, backingStores);
65          updateDiskStats(session, mountPointMap, cfKeyMap);
66      }

68      @Override
69      public synchronized boolean updateAttributes() {
70          // Open a session and create CFStrings
71          DASessionRef session = DA.DASessionCreate(CF.CFAllocatorGetDefault());
72          if (session == null) {
73              LOG.error("Unable to open session to DiskArbitration framework.");
74              return false;
75          }
76          try {
77              Map<CFKey, CFStringRef> cfKeyMap = mapCFKeys();
78              try {
79                  // Execute the update
80                  return updateDiskStats(session, Fsstat.queryPartitionToMountMap(), cfKeyMap);
81              } finally {
82                  for (CFTypeRef value : cfKeyMap.values()) {
83                      value.release();
84                  }
85              }
86          } finally {
87              session.release();
88          }
89      }

91      private boolean updateDiskStats(DASessionRef session, Map<String, String> mountPointMap,
92              Map<CFKey, CFStringRef> cfKeyMap) {
93          // Now look up the device using the BSD Name to get its
94          // statistics
95          String bsdName = getName();
96          CFMutableDictionaryRef matchingDict = IOKitUtil.getBSDNameMatchingDict(bsdName);
97          if (matchingDict != null) {
98              // search for all IOservices that match the bsd name
99              IOIterator driveListIter = IOKitUtil.getMatchingServices(matchingDict);
100             if (driveListIter != null) {
101                 // getMatchingServices releases matchingDict
102                 boolean updated = false;
103                 IORegistryEntry drive = driveListIter.next();
104                 // Should only match one drive
105                 if (drive != null) {
106                     // Should be an IOMedia object with a parent
107                     // IOBlockStorageDriver or AppleAPFSContainerScheme object
108                     // Get the properties from the parent
109                     if (drive.conformsTo(IO_MEDIA)) {
110                         IORegistryEntry parent = drive.getParentEntry("IOService");
111                         if (parent != null && (parent.conformsTo("IOBlockStorageDriver")
112                                 || parent.conformsTo("AppleAPFSContainerScheme"))) {
113                             CFMutableDictionaryRef properties = parent.createCFProperties();
114                             // We now have a properties object with the
115                             // statistics we need on it. Fetch them
116                             Pointer result = properties.getValue(cfKeyMap.get(CFKey.STATISTICS));
117                             CFDictionaryRef statistics = new CFDictionaryRef(result);
118                             this.timeStamp = System.currentTimeMillis();
119 
120                             // Now get the stats we want
121                             result = statistics.getValue(cfKeyMap.get(CFKey.READ_OPS));
122                             CFNumberRef stat = new CFNumberRef(result);
123                             this.reads = stat.longValue();
124                             result = statistics.getValue(cfKeyMap.get(CFKey.READ_BYTES));
125                             stat.setPointer(result);
126                             this.readBytes = stat.longValue();
127 
128                             result = statistics.getValue(cfKeyMap.get(CFKey.WRITE_OPS));
129                             stat.setPointer(result);
130                             this.writes = stat.longValue();
131                             result = statistics.getValue(cfKeyMap.get(CFKey.WRITE_BYTES));
132                             stat.setPointer(result);
133                             this.writeBytes = stat.longValue();
134 
135                             // Total time is in nanoseconds. Add read+write
136                             // and convert total to ms
137                             final Pointer readTimeResult = statistics.getValue(cfKeyMap.get(CFKey.READ_TIME));
138                             final Pointer writeTimeResult = statistics.getValue(cfKeyMap.get(CFKey.WRITE_TIME));
139                             // AppleAPFSContainerScheme does not have timer statistics
140                             if (readTimeResult != null && writeTimeResult != null) {
141                                 stat.setPointer(readTimeResult);
142                                 long xferTime = stat.longValue();
143                                 stat.setPointer(writeTimeResult);
144                                 xferTime += stat.longValue();
145                                 this.transferTime = xferTime / 1_000_000L;
146                             }
147 
148                             properties.release();
149                         } else {
150                             // This is normal for FileVault drives, Fusion
151                             // drives, and other virtual bsd names
152                             LOG.debug("Unable to find block storage driver properties for {}", bsdName);
153                         }
154                         // Now get partitions for this disk.
155                         List<HWPartition> partitions = new ArrayList<>();
156 
157                         CFMutableDictionaryRef properties = drive.createCFProperties();
158                         // Partitions will match BSD Unit property
159                         Pointer result = properties.getValue(cfKeyMap.get(CFKey.BSD_UNIT));
160                         CFNumberRef bsdUnit = new CFNumberRef(result);
161                         // We need a CFBoolean that's false.
162                         // Whole disk has 'true' for Whole and 'false'
163                         // for leaf; store the boolean false
164                         result = properties.getValue(cfKeyMap.get(CFKey.LEAF));
165                         CFBooleanRef cfFalse = new CFBooleanRef(result);
166                         // create a matching dict for BSD Unit
167                         CFMutableDictionaryRef propertyDict = CF.CFDictionaryCreateMutable(CF.CFAllocatorGetDefault(),
168                                 new CFIndex(0), null, null);
169                         propertyDict.setValue(cfKeyMap.get(CFKey.BSD_UNIT), bsdUnit);
170                         propertyDict.setValue(cfKeyMap.get(CFKey.WHOLE), cfFalse);
171                         matchingDict = CF.CFDictionaryCreateMutable(CF.CFAllocatorGetDefault(), new CFIndex(0), null,
172                                 null);
173                         matchingDict.setValue(cfKeyMap.get(CFKey.IO_PROPERTY_MATCH), propertyDict);
174 
175                         // search for IOservices that match the BSD Unit
176                         // with whole=false; these are partitions
177                         IOIterator serviceIterator = IOKitUtil.getMatchingServices(matchingDict);
178                         // getMatchingServices releases matchingDict
179                         properties.release();
180                         propertyDict.release();
181 
182                         if (serviceIterator != null) {
183                             // Iterate disks
184                             IORegistryEntry sdService = IOKit.INSTANCE.IOIteratorNext(serviceIterator);
185                             while (sdService != null) {
186                                 // look up the BSD Name
187                                 String partBsdName = sdService.getStringProperty("BSD Name");
188                                 String name = partBsdName;
189                                 String type = "";
190                                 String label = "";
191                                 // Get the DiskArbitration dictionary for
192                                 // this partition
193                                 DADiskRef disk = DA.DADiskCreateFromBSDName(CF.CFAllocatorGetDefault(), session,
194                                         partBsdName);
195                                 if (disk != null) {
196                                     CFDictionaryRef diskInfo = DA.DADiskCopyDescription(disk);
197                                     if (diskInfo != null) {
198                                         // get volume name from its key
199                                         result = diskInfo.getValue(cfKeyMap.get(CFKey.DA_MEDIA_NAME));
200                                         type = CFUtil.cfPointerToString(result);
201                                         result = diskInfo.getValue(cfKeyMap.get(CFKey.DA_VOLUME_NAME));
202                                         if (result == null) {
203                                             name = type;
204                                         } else {
205                                             String volumeName = CFUtil.cfPointerToString(result);
206                                             name = volumeName;
207                                             label = volumeName;
208                                         }
209                                         diskInfo.release();
210                                     }
211                                     disk.release();
212                                 }
213                                 String mountPoint = mountPointMap.getOrDefault(partBsdName, "");
214                                 Long size = sdService.getLongProperty("Size");
215                                 Integer bsdMajor = sdService.getIntegerProperty("BSD Major");
216                                 Integer bsdMinor = sdService.getIntegerProperty("BSD Minor");
217                                 String uuid = sdService.getStringProperty("UUID");
218                                 partitions.add(new HWPartition(partBsdName, name, type,
219                                         ParseUtil.getStringValueOrUnknown(uuid), label, size == null ? 0L : size,
220                                         bsdMajor == null ? 0 : bsdMajor, bsdMinor == null ? 0 : bsdMinor, mountPoint));
221                                 // iterate
222                                 sdService.release();
223                                 sdService = IOKit.INSTANCE.IOIteratorNext(serviceIterator);
224                             }
225                             serviceIterator.release();
226                         }
227                         setPartitionList(Collections.unmodifiableList(partitions.stream()
228                                 .sorted(Comparator.comparing(HWPartition::getName)).collect(Collectors.toList())));
229                         if (parent != null) {
230                             parent.release();
231                         }
232                         updated = true;
233                     } else {
234                         LOG.error("Unable to find IOMedia device or parent for {}", bsdName);
235                     }
236                     drive.release();
237                 }
238                 driveListIter.release();
239                 return updated;
240             }
241         }
242         return false;
243     }

245     /**
246      * Gets the disks on this machine
247      *
248      * @return a list of {@link HWDiskStore} objects representing the disks
249      */
250     public static List<HWDiskStore> getDisks() {
251         Map<String, String> mountPointMap = Fsstat.queryPartitionToMountMap();
252         Map<CFKey, CFStringRef> cfKeyMap = mapCFKeys();
253 
254         List<HWDiskStore> diskList = new ArrayList<>();
255 
256         try {
257             // Open a DiskArbitration session
258             DASessionRef session = DA.DASessionCreate(CF.CFAllocatorGetDefault());
259             if (session == null) {
260                 LOG.error("Unable to open session to DiskArbitration framework.");
261                 return Collections.emptyList();
262             }
263             try {
264 
265                 // Get IOMedia objects representing whole drives
266                 List<String> bsdNames = new ArrayList<>();
267                 IOIterator iter = IOKitUtil.getMatchingServices(IO_MEDIA);
268                 if (iter != null) {
269                     IORegistryEntry media = iter.next();
270                     while (media != null) {
271                         Boolean whole = media.getBooleanProperty(WHOLE);
272                         if (whole != null && whole) {
273                             DADiskRef disk = DA.DADiskCreateFromIOMedia(CF.CFAllocatorGetDefault(), session, media);
274                             if (disk != null) {
275                                 String bsdName = DA.DADiskGetBSDName(disk);
276                                 if (bsdName != null) {
277                                     bsdNames.add(bsdName);
278                                 }
279                                 disk.release();
280                             }
281                         }
282                         media.release();
283                         media = iter.next();
284                     }
285                     iter.release();
286                 }
287 
288                 // Now iterate the bsdNames
289                 for (String bsdName : bsdNames) {
290                     String model = "";
291                     String serial = "";
292                     long size = 0L;
293 
294                     // Get a reference to the disk - only matching /dev/disk*
295                     String path = "/dev/" + bsdName;
296 
297                     // Get the DiskArbitration dictionary for this disk, which has model
298                     // and size (capacity)
299                     DADiskRef disk = DA.DADiskCreateFromBSDName(CF.CFAllocatorGetDefault(), session, path);
300                     if (disk != null) {
301                         CFDictionaryRef diskInfo = DA.DADiskCopyDescription(disk);
302                         if (diskInfo != null) {
303                             // Parse out model and size from their respective keys
304                             Pointer result = diskInfo.getValue(cfKeyMap.get(CFKey.DA_DEVICE_MODEL));
305                             model = CFUtil.cfPointerToString(result);
306                             result = diskInfo.getValue(cfKeyMap.get(CFKey.DA_MEDIA_SIZE));
307                             CFNumberRef sizePtr = new CFNumberRef(result);
308                             size = sizePtr.longValue();
309                             diskInfo.release();
310 
311                             // Read the serial from this disk's own block storage device; fall back to
312                             // matching on the model for devices that do not publish it there
313                             serial = queryDeviceSerial(bsdName);
314                             if (serial.isEmpty() && !"Disk Image".equals(model)) {
315                                 CFStringRef modelNameRef = CFStringRef.createCFString(model);
316                                 CFMutableDictionaryRef propertyDict = CF.CFDictionaryCreateMutable(
317                                         CF.CFAllocatorGetDefault(), new CFIndex(0), null, null);
318                                 propertyDict.setValue(cfKeyMap.get(CFKey.MODEL), modelNameRef);
319                                 CFMutableDictionaryRef matchingDict = CF.CFDictionaryCreateMutable(
320                                         CF.CFAllocatorGetDefault(), new CFIndex(0), null, null);
321                                 matchingDict.setValue(cfKeyMap.get(CFKey.IO_PROPERTY_MATCH), propertyDict);
322 
323                                 // search for all IOservices that match the model
324                                 IOIterator serviceIterator = IOKitUtil.getMatchingServices(matchingDict);
325                                 // getMatchingServices releases matchingDict
326                                 modelNameRef.release();
327                                 propertyDict.release();
328 
329                                 if (serviceIterator != null) {
330                                     IORegistryEntry sdService = serviceIterator.next();
331                                     while (sdService != null) {
332                                         // look up the serial number
333                                         serial = sdService.getStringProperty("Serial Number");
334                                         sdService.release();
335                                         if (serial != null) {
336                                             break;
337                                         }
338                                         sdService = serviceIterator.next();
339                                     }
340                                     serviceIterator.release();
341                                 }
342                                 if (serial == null) {
343                                     serial = "";
344                                 }
345                             }
346                         }
347                         disk.release();
348 
349                         // If empty, ignore
350                         if (size <= 0) {
351                             continue;
352                         }
353                         HWDiskStore diskStore = new MacHWDiskStoreJNA(bsdName, model.trim(), serial.trim(), size,
354                                 detectDiskType(bsdName), queryBackingStores(bsdName), session, mountPointMap, cfKeyMap);
355                         diskList.add(diskStore);
356                     }
357                 }
358                 return Collections.unmodifiableList(diskList);
359             } finally {
360                 session.release();
361             }
362         } finally {
363             for (CFTypeRef value : cfKeyMap.values()) {
364                 value.release();
365             }
366         }
367     }

369     /**
370      * Temporarily cache pointers to keys. The values from this map must be released after use.}
371      *
372      * @return A map of keys in the {@link CFKey} enum to corresponding {@link CFStringRef}.
373      */
374     private static Map<CFKey, CFStringRef> mapCFKeys() {
375         Map<CFKey, CFStringRef> keyMap = new EnumMap<>(CFKey.class);
376         for (CFKey cfKey : CFKey.values()) {
377             keyMap.put(cfKey, CFStringRef.createCFString(cfKey.getKey()));
378         }
379         return keyMap;
380     }

382     /**
383      * Finds what a disk is built on by walking up the IOService plane from its IOMedia object to the first ancestor
384      * that is either another media object or a disk image device. Reaching the platform expert without finding one
385      * means the disk is not layered.
386      *
387      * @param bsdName the disk's BSD name
388      * @return the backing stores, an empty list for a disk that is not layered, or empty if they could not be read
389      */
390     private static Optional<List<String>> queryBackingStores(String bsdName) {
391         CFMutableDictionaryRef matchingDict = IOKitUtil.getBSDNameMatchingDict(bsdName);
392         if (matchingDict == null) {
393             return Optional.empty();
394         }
395         IOIterator iter = IOKitUtil.getMatchingServices(matchingDict);
396         if (iter == null) {
397             return Optional.empty();
398         }
399         try {
400             IORegistryEntry media = iter.next();
401             if (media == null) {
402                 return Optional.empty();
403             }
404             IORegistryEntry entry;
405             try {
406                 entry = media.getParentEntry("IOService");
407             } finally {
408                 media.release();
409             }
410             while (entry != null) {
411                 IORegistryEntry current = entry;
412                 try {
413                     if (current.conformsTo(IO_MEDIA)) {
414                         return mediaBackingStore(current.getStringProperty(BSD_NAME),
415                                 current.getBooleanProperty(WHOLE));
416                     }
417                     if (current.conformsTo(DISK_IMAGE_DEVICE)) {
418                         String imageUrl = current.getStringProperty(DISK_IMAGE_URL);
419                         return imageUrl == null ? Optional.empty()
420                                 : imageBackingStore(imageUrl, MacHWDiskStoreJNA::queryMountPoint);
421                     }
422                     if (current.conformsTo(LEGACY_DISK_IMAGE_DEVICE)) {
423                         return Optional.empty();
424                     }
425                     if (current.conformsTo(PLATFORM_EXPERT)) {
426                         return Optional.of(Collections.emptyList());
427                     }
428                     entry = current.getParentEntry("IOService");
429                 } finally {
430                     current.release();
431                 }
432             }
433             // The walk ended below the platform expert, so a parent could not be read
434             return Optional.empty();
435         } catch (IOReturnException e) {
436             LOG.debug("Unable to walk the IORegistry for {}", bsdName, e);
437             return Optional.empty();
438         } finally {
439             iter.release();
440         }
441     }

443     private static String queryMountPoint(String path) {
444         Statfs fs = new Statfs();
445         return SystemB.INSTANCE.statfs64(path, fs) == 0 ? Native.toString(fs.f_mntonname, StandardCharsets.UTF_8) : "";
446     }

448     private static String detectDiskType(String bsdName) {
449         CFMutableDictionaryRef matchingDict = IOKitUtil.getBSDNameMatchingDict(bsdName);
450         if (matchingDict == null) {
451             return "Unknown";
452         }
453         IOIterator iter = IOKitUtil.getMatchingServices(matchingDict);
454         if (iter == null) {
455             return "Unknown";
456         }
457         try {
458             IORegistryEntry media = iter.next();
459             if (media == null) {
460                 return "Unknown";
461             }
462             try {
463                 // Check removable at the IOMedia level
464                 Boolean removable = media.getBooleanProperty("Removable");
465                 if (removable != null && removable) {
466                     return "Removable";
467                 }
468                 return parseMediumType(queryDeviceCharacteristic(media, "Medium Type"));
469             } finally {
470                 media.release();
471             }
472         } finally {
473             iter.release();
474         }
475     }

477     /**
478      * Reads the {@code Serial Number} device characteristic of the block storage device backing a whole disk.
479      *
480      * @param bsdName the BSD name of the whole disk, such as {@code disk0}
481      * @return the serial number, or an empty string if the device does not publish one
482      */
483     private static String queryDeviceSerial(String bsdName) {
484         CFMutableDictionaryRef matchingDict = IOKitUtil.getBSDNameMatchingDict(bsdName);
485         if (matchingDict == null) {
486             return "";
487         }
488         IOIterator iter = IOKitUtil.getMatchingServices(matchingDict);
489         if (iter == null) {
490             return "";
491         }
492         try {
493             IORegistryEntry media = iter.next();
494             if (media == null) {
495                 return "";
496             }
497             try {
498                 return ParseUtil.getStringValueOrEmpty(queryDeviceCharacteristic(media, "Serial Number"));
499             } finally {
500                 media.release();
501             }
502         } finally {
503             iter.release();
504         }
505     }

507     /**
508      * Reads a device characteristic, such as {@code Medium Type} or {@code Serial Number}, by traversing up from the
509      * IOMedia node to the block storage device, which is where the characteristics are published.
510      *
511      * @param media the IOMedia registry entry
512      * @param key   the key in the device's {@code Device Characteristics} dictionary
513      * @return the characteristic's string value, or {@code null} if any step of the traversal finds nothing
514      */
515     private static @Nullable String queryDeviceCharacteristic(IORegistryEntry media, String key) {
516         // Traverse up: IOMedia -> IOBlockStorageDriver -> IOBlockStorageDevice
517         IORegistryEntry driver = media.getParentEntry("IOService");
518         if (driver == null) {
519             return null;
520         }
521         try {
522             IORegistryEntry device = driver.getParentEntry("IOService");
523             if (device == null) {
524                 return null;
525             }
526             try {
527                 CFMutableDictionaryRef props = device.createCFProperties();
528                 if (props == null) {
529                     return null;
530                 }
531                 try {
532                     CFDictionaryRef characteristics = CFUtil.getDictionary(props, "Device Characteristics");
533                     return characteristics == null ? null : CFUtil.getString(characteristics, key);
534                 } finally {
535                     props.release();
536                 }
537             } finally {
538                 device.release();
539             }
540         } finally {
541             driver.release();
542         }
