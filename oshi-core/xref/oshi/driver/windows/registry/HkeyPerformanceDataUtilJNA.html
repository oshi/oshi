31   * Utility to read HKEY_PERFORMANCE_DATA information.

36      private static final Logger LOG = LoggerFactory.getLogger(HkeyPerformanceDataUtilJNA.class);

38      private static final Map<String, Integer> COUNTER_INDEX_MAP = mapCounterIndicesFromRegistry();
39  
40      private static int maxPerfBufferSize = 16384;
41  
42      private HkeyPerformanceDataUtilJNA() {
43      }
44  
45      /**
46       * Reads and parses a block of performance data from the registry.
47       *
48       * @param <T>         PDH Counters use an Enum to identify the fields to query in either the counter or WMI backup,
49       *                    and use the enum values as keys to retrieve the results.
50       * @param objectName  The counter object for which to fetch data
51       * @param counterEnum Which counters to return data for
52       * @return A triplet containing the results. The first element maps the input enum to the counter values where the
53       *         first enum will contain the instance name as a {@link String}, and the remaining values will either be
54       *         {@link Long}, {@link Integer}, or {@code null} depending on whether the specified enum counter was
55       *         present and the size of the counter value, or {@code null} if the object could not be read. The second
56       *         element is a timestamp in 100nSec increments (Windows 1601 Epoch) while the third element is a timestamp
57       *         in milliseconds since the 1970 Epoch.
58       */
59      public static <T extends Enum<T> & PdhCounterWildcardProperty> @Nullable Triplet<List<Map<T, Object>>, Long, Long> readPerfDataFromRegistry(
60              String objectName, Class<T> counterEnum) {
61          // Load indices
62          Pair<Integer, EnumMap<T, Integer>> indices = getCounterIndices(objectName, counterEnum, COUNTER_INDEX_MAP);
63          if (indices == null) {
64              return null;
65          }
66          int objectIndex = indices.getA();
67          EnumMap<T, Integer> enumIndexMap = indices.getB();
68          // The above test checks validity of objectName as an index but it could still
69          // fail to read
70          try (Memory pPerfData = readPerfDataBuffer(objectName)) {
71              if (pPerfData == null) {
72                  return null;
73              }
74              return parsePerfData(new MemoryPerfDataBuffer(pPerfData), objectIndex, enumIndexMap, counterEnum);
75          }
76      }
77  
78      /**
79       * Reads the performance data block through JNA's {@link Memory}.
80       */
81      private static final class MemoryPerfDataBuffer implements PerfDataBuffer {
82  
83          private final Memory memory;
84  
85          private MemoryPerfDataBuffer(Memory memory) {
86              this.memory = memory;
87          }
88  
89          @Override
90          public int getInt(long offset) {
91              return this.memory.getInt(offset);
92          }
93  
94          @Override
95          public long getLong(long offset) {
96              return this.memory.getLong(offset);
97          }
98  
99          @Override
100         public String getWideString(long offset) {
101             return this.memory.getWideString(offset);
102         }
103     }

105     /**
106      * Read the performance data for a counter object from the registry.
107      *
108      * @param objectName The counter object for which to fetch data.
109      * @return A buffer containing the data if successful, null otherwise.
110      */
111     private static synchronized @Nullable Memory readPerfDataBuffer(String objectName) {
112         Integer objectIndex = COUNTER_INDEX_MAP.get(objectName);
113         if (objectIndex == null) {
114             LOG.error("No counter index for performance object {}.", objectName);
115             return null;
116         }
117         String objectIndexStr = objectIndex.toString();
118 
119         try (CloseableIntByReference lpcbData = new CloseableIntByReference(maxPerfBufferSize)) {
120             Memory pPerfData = new Memory(maxPerfBufferSize);
121             int ret = Advapi32.INSTANCE.RegQueryValueEx(WinReg.HKEY_PERFORMANCE_DATA, objectIndexStr, 0, null,
122                     pPerfData, lpcbData);
123             if (ret != WinError.ERROR_SUCCESS && ret != WinError.ERROR_MORE_DATA) {
124                 LOG.error("Error reading performance data from registry for {}.", objectName);
125                 pPerfData.close();
126                 return null;
127             }
128             while (ret == WinError.ERROR_MORE_DATA) {
129                 maxPerfBufferSize += 8192;
130                 lpcbData.setValue(maxPerfBufferSize);
131                 pPerfData.close();
132                 pPerfData = new Memory(maxPerfBufferSize);
133                 ret = Advapi32.INSTANCE.RegQueryValueEx(WinReg.HKEY_PERFORMANCE_DATA, objectIndexStr, 0, null,
134                         pPerfData, lpcbData);
135             }
136             // The buffer-grow retry can exit on a hard error rather than ERROR_SUCCESS; return null instead of an
137             // unvalidated buffer. Matches HkeyPerformanceDataUtilFFM.
138             if (ret != WinError.ERROR_SUCCESS) {
139                 LOG.error("Error reading performance data from registry for {}.", objectName);
140                 pPerfData.close();
141                 return null;
142             }
143             return pPerfData;
144         }
145     }

147     private static Map<String, Integer> mapCounterIndicesFromRegistry() {
148         try {
149             String[] counterText = Advapi32Util.registryGetStringArray(WinReg.HKEY_LOCAL_MACHINE, HKEY_PERFORMANCE_TEXT,
150                     COUNTER);
151             return buildCounterIndexMap(counterText);
152         } catch (Win32Exception we) {
153             LOG.error(
154                     "Unable to locate English counter names in registry Perflib 009. Counters may need to be rebuilt: ",
155                     we);
156         }
157         return buildCounterIndexMap(null);
158     }
