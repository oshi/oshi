5   package oshi.jna.platform.unix;
6   
7   import com.sun.jna.Library;
8   import com.sun.jna.Native;
9   import com.sun.jna.Pointer;
10  import com.sun.jna.platform.unix.solaris.LibKstat;
11  import com.sun.jna.platform.unix.solaris.LibKstat.Kstat;
12  import com.sun.jna.platform.unix.solaris.LibKstat.KstatCtl;
13  
14  /**
15   * Re-declares the {@code libkstat} entry points that take a {@code kstat_ctl_t *}, mapping that argument as an opaque
16   * {@link Pointer} rather than as JNA's {@link KstatCtl} structure.
17   * <p>
18   * JNA's {@code KstatCtl} declares {@code kc_chain} as an inline {@link Kstat} structure instead of a pointer to one, so
19   * JNA computes {@code sizeof(kstat_ctl_t)} as 200 bytes where the real struct, from {@code <kstat.h>}, is 24:
20   *
21   * <pre>
22   * typedef struct kstat_ctl {
23   *     kid_t    kc_chain_id;
24   *     kstat_t *kc_chain;
25   *     int      kc_kd;
26   * } kstat_ctl_t;
27   * </pre>
28   *
29   * A {@code Structure} passed by reference is auto-written to native memory before every call and auto-read after it, so
30   * each {@code kstat_chain_update}, {@code kstat_lookup}, or {@code kstat_read} wrote 176 bytes of a stale snapshot past
31   * the end of the 24-byte block {@code kstat_open} allocated, and read 176 bytes past it. On illumos that silently
32   * rewrites whichever neighbouring heap buffers share the allocator's slab, which shows up far from the kstat code as a
33   * {@code UMEM_SLAB_MEMBER} assertion failure in {@code umem_slab_alloc} (a freed neighbour's free-list link
34   * overwritten) or as a SIGSEGV on a half-overwritten pointer.
35   * <p>
36   * Treating the control structure as opaque, as the FFM implementation already does, removes both the overrunning write
37   * and the overrunning read. {@code kstat_data_lookup} needs no redeclaration because it does not take a
38   * {@code kstat_ctl_t *}, and {@link Kstat} itself is mapped correctly at its real 184 bytes.
39   * <p>
40   * This class should be considered non-API as it may be removed if/when its code is incorporated into the JNA project.
41   */
42  public interface SolarisLibKstat extends Library {
43  
44      /** The singleton instance of this library. */
45      SolarisLibKstat INSTANCE = Native.load("kstat", SolarisLibKstat.class);
46  
47      /**
48       * Initializes a kstat control structure, which provides access to the kernel statistics library.
49       *
50       * @return An opaque pointer to the {@code kstat_ctl_t}, which must be supplied as the {@code kc} argument in
51       *         subsequent libkstat calls, or {@code null} on failure.
52       */
53      Pointer kstat_open();
54  
55      /**
56       * Frees all resources that were associated with {@code kc}.
57       *
58       * @param kc A pointer to a kstat control structure
59       * @return 0 on success and -1 on failure.
60       */
61      int kstat_close(Pointer kc);
62  
63      /**
64       * Brings the user's kstat header chain in sync with that of the kernel.
65       *
66       * @param kc A pointer to a kstat control structure
67       * @return the new KCID if the kstat chain has changed, 0 if it hasn't, or -1 on failure.
68       */
69      int kstat_chain_update(Pointer kc);
70  
71      /**
72       * Traverses the kstat chain searching for a kstat with the given {@code ks_module}, {@code ks_instance}, and
73       * {@code ks_name}. If {@code ks_module} is {@code null}, {@code ks_instance} is -1, or {@code ks_name} is
74       * {@code null}, those fields are ignored in the search.
75       *
76       * @param kc          A pointer to a kstat control structure
77       * @param ks_module   The kstat module to search
78       * @param ks_instance The kstat instance number
79       * @param ks_name     The kstat name to search
80       * @return the requested kstat if it is found, or {@code null} if it is not.
81       */
82      Kstat kstat_lookup(Pointer kc, String ks_module, int ks_instance, String ks_name);
83  
84      /**
85       * Gets data from the kernel for the kstat pointed to by {@code ksp}.
86       *
87       * @param kc  A pointer to a kstat control structure
88       * @param ksp The kstat from which to retrieve data
89       * @param p   If non-{@code null}, the data is copied from {@code ksp.ks_data} into it
90       * @return the current kstat chain ID (KCID) on success, or -1 on failure.
91       */
92      int kstat_read(Pointer kc, Kstat ksp, Pointer p);
93  
94      /**
95       * Searches the kstat's data section for the record with the specified name. Valid only for
96       * {@link LibKstat#KSTAT_TYPE_NAMED} and {@link LibKstat#KSTAT_TYPE_TIMER} kstats.
97       *
98       * @param ksp  The kstat to search
99       * @param name The key for the name-value pair, or name of the timer as applicable
100      * @return a pointer to the requested data record, or {@code null} if it is not found or the kstat type is invalid.
101      */
102     Pointer kstat_data_lookup(Kstat ksp, String name);
103 }
