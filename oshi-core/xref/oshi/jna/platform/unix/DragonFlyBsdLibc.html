5   package oshi.jna.platform.unix;
6   
7   import com.sun.jna.Pointer;
8   import com.sun.jna.platform.unix.LibCAPI.size_t;
9   
10  /**
11   * DragonFly BSD C library. Extends {@link FreeBsdLibc} since DragonFly shares the same sysctl and CPU tick interfaces;
12   * its {@code struct utmpx} differs, so the inherited utmpx functions do not apply. Adds the DragonFly-specific
13   * {@link #lwp_gettid()} thread ID function and {@link #sched_getaffinity(int, size_t, Pointer)}.
14   */
15  public interface DragonFlyBsdLibc extends FreeBsdLibc {
16      DragonFlyBsdLibc INSTANCE = BsdLibcLoader.loadLibc(DragonFlyBsdLibc.class);
17  
18      /**
19       * Returns the light weight process thread identifier for the calling thread.
20       *
21       * @return The LWP thread ID. Returns -1 on error.
22       */
23      int lwp_gettid();
24  
25      /**
26       * The size of DragonFly's {@code cpu_set_t}, a {@code cpumask_t} of four 64-bit words on x86_64.
27       */
28      int CPU_SET_SIZE = 32;
29  
30      /**
31       * Gets the CPU affinity mask of a process, that of its first LWP.
32       *
33       * @param pid        the process ID, or 0 for the calling process
34       * @param cpusetsize the size of the buffer, at most {@link #CPU_SET_SIZE} of which is written
35       * @param mask       the buffer that receives the mask, CPU 0 in the lowest bit of the first word
36       * @return 0 on success, -1 on error
37       */
38      int sched_getaffinity(int pid, size_t cpusetsize, Pointer mask);
39  }
