86      /**
87       * Fallback Apple Silicon GPU temperature keys, used only when runtime discovery cannot complete. The hottest
88       * plausible reading among them is reported.
89       * <p>
90       * GPU sensor keys are chip-specific, so no fixed list is complete: on an M2 Max only {@code Tg0f} and {@code Tg0j}

138     /**
139      * Open a connection to SMC.
140      *
141      * @return The connection if successful, null if failure
142      */
143     public static @Nullable IOConnect smcOpen() {
144         IOService smcService = IOKitUtil.getMatchingService("AppleSMC");
145         if (smcService != null) {
146             try (CloseablePointerByReference connPtr = new CloseablePointerByReference()) {
147                 int result = IO.IOServiceOpen(smcService, SystemB.INSTANCE.mach_task_self(), 0, connPtr);
148                 if (result == 0) {
149                     Pointer conn = connPtr.getValue();
150                     if (conn == null) {
151                         OPEN_FAILURE.nullConnection();
152                     } else {
153                         return new IOConnect(conn);
154                     }
155                 } else {
156                     OPEN_FAILURE.openFailed(result);
157                 }
158             } finally {
159                 smcService.release();
160             }
161         } else {
162             OPEN_FAILURE.serviceNotFound();
163         }
164         return null;
165     }

178     /**
179      * Get a value from SMC which is in a floating point datatype (SP78, FPE2, FLT)
180      *
181      * @param conn The connection
182      * @param key  The key to retrieve
183      * @return Double representing the value
184      */
185     public static double smcGetFloat(IOConnect conn, String key) {
186         try (SMCVal val = new SMCVal()) {
187             int result = smcReadKey(conn, key, val);
188             if (result == 0 && val.dataSize > 0) {
189                 if (Arrays.equals(val.dataType, DATATYPE_SP78) && val.dataSize == 2) {
190                     // First bit is sign, next 7 bits are integer portion, last 8 bits are the
191                     // (unsigned) fractional portion
192                     return val.bytes[0] + (val.bytes[1] & 0xFF) / 256d;
193                 } else if (Arrays.equals(val.dataType, DATATYPE_FPE2) && val.dataSize == 2) {
194                     // First E (14) bits are integer portion last 2 bits are fractional portion
195                     return ParseUtil.byteArrayToFloat(val.bytes, val.dataSize, 2);
196                 } else if (Arrays.equals(val.dataType, DATATYPE_FLT) && val.dataSize == 4) {
197                     // Standard 32-bit floating point
198                     return ByteBuffer.wrap(val.bytes).order(ByteOrder.LITTLE_ENDIAN).getFloat();
199                 }
200             }
201         }
202         // Read failed
203         return 0d;
204     }

206     /**
207      * The lowest reading accepted as a genuine temperature, in degrees Celsius.
208      * <p>
209      * Apple Silicon power-gates a CPU core cluster when it is idle, and the SMC then reports a fixed sentinel for each
210      * die sensor in that cluster instead of a reading. Observed sentinels are 6.7 and 4.633 on an M2 Max (Tp01/Tp09 and
211      * Tp05/Tp0D), up to 8.425 across all of that machine's sensors, and -4.0 through 5.2 in the
212      * <a href="https://github.com/dkorunic/iSMC">iSMC</a> sample reports, which independently describe them as
213      * "firmware sentinels from inactive sensor slots". They are below room ambient and so cannot be real die
214      * temperatures, but most are positive, so a simple {@code > 0} test accepts them.
215      * <p>
216      * Across those reports, covering M1 through M5, A18 and Intel T2 machines, no genuine sensor read between 8.425 and
217      * 21, so this floor sits in an empty band with margin on both sides. It is a plausibility guard, not a hardware
218      * specification.
219      */
220     public static final double MIN_PLAUSIBLE_TEMPERATURE = 15d;
221 
222     /**
223      * Tests whether a reading is plausible as a temperature, rejecting the sentinel an idle-gated sensor reports.
224      *
225      * @param celsius the reading to test, in degrees Celsius
226      * @return true if the reading is at least {@link #MIN_PLAUSIBLE_TEMPERATURE}
227      */
228     public static boolean isPlausibleTemperature(double celsius) {
229         return celsius >= MIN_PLAUSIBLE_TEMPERATURE;
230     }

233      * Get the first plausible temperature from a list of SMC keys, skipping sensors that are reporting a parked value

235      *
236      * @param conn The connection
237      * @param keys The keys to try in order
238      * @return The first reading at or above {@link #MIN_PLAUSIBLE_TEMPERATURE}, or 0 if no key returned one
239      */
240     public static double smcGetFirstTemperature(IOConnect conn, List<String> keys) {
241         return SmcKeyIndex.firstPlausible(keys, key -> smcGetFloat(conn, key), SmcUtil::isPlausibleTemperature,
242                 "temperature");
243     }

245     /**
246      * Get the highest plausible temperature among a list of SMC keys.
247      * <p>
248      * Plausibility is applied per read rather than when the keys were discovered, because whether a key exists is a
249      * property of the hardware while whether it currently reports a usable value is not.
250      * <p>
251      * Clusters expose their sensors in pairs at different locations on the die, offset by a few degrees and tracking
252      * each other as load changes; verified on an M2 Max where a pair moved 44.8/51.4 to 56.1/62.6 under load and fell
253      * together afterwards. They are not an instantaneous/peak-hold pair, so taking the maximum consistently reports the
254      * hotter location rather than a latched peak.
255      *
256      * @param conn The connection
257      * @param keys The keys to read
258      * @return The highest reading at or above {@link #MIN_PLAUSIBLE_TEMPERATURE}, or 0 if none were plausible
259      */
260     public static double smcGetMaxTemperature(IOConnect conn, List<String> keys) {
261         return SmcKeyIndex.maxPlausible(keys, key -> smcGetFloat(conn, key), SmcUtil::isPlausibleTemperature);
262     }

264     /**
265      * Get the name of the key at an index in the SMC key index.
266      *
267      * @param conn  The connection
268      * @param index The index, from 0 to the value of {@link #SMC_KEY_COUNT}
269      * @return The four-character key name, or null if it could not be read
270      */
271     public static @Nullable String smcReadKeyAtIndex(IOConnect conn, int index) {
272         try (SMCKeyData input = new SMCKeyData(); SMCKeyData output = new SMCKeyData()) {
273             input.data8 = SMC_CMD_READ_INDEX;
274             input.data32 = index;
275             if (smcCall(conn, KERNEL_INDEX_SMC, input, output) != 0) {
276                 return null;
277             }
278             byte[] keyBytes = ParseUtil.longToByteArray(output.key, 4, 4);
279             StringBuilder sb = new StringBuilder(4);
280             for (byte b : keyBytes) {
281                 sb.append((char) (b & 0xFF));
282             }
283             return sb.toString();
284         }
285     }

287     /**
288      * Get the SMC data type of a key, e.g. {@code flt} or {@code sp78}.
289      *
290      * @param conn The connection
291      * @param key  The key to query
292      * @return The data type, or an empty string if it could not be read
293      */
294     public static String smcGetDataType(IOConnect conn, String key) {
295         try (SMCVal val = new SMCVal()) {
296             if (smcReadKey(conn, key, val) != 0 || val.dataType == null) {
297                 return "";
298             }
299             StringBuilder sb = new StringBuilder(4);
300             for (byte b : val.dataType) {
301                 if (b == 0) {
302                     break;
303                 }
304                 sb.append((char) b);
305             }
306             return sb.toString().trim();
307         }
308     }

310     /**
311      * Get the Apple Silicon GPU temperature keys for this machine, discovering them from the SMC on first use and
312      * caching the result.
313      * <p>
314      * Discovery binary searches the sorted key index for the {@code Tg} block and keeps the keys matching the GPU
315      * naming convention. It does not filter by plausibility: an idle sensor reports a sentinel below ambient, so
316      * filtering here would let one unlucky first call cache an empty set and disable GPU temperature permanently.
317      * Plausibility is applied per read by {@link #smcGetMaxTemperature}.
318      * <p>
319      * Can be overridden with the {@link GlobalConfig#OSHI_OS_MAC_SENSORS_GPUTEMPERATURE_KEYS} configuration property,
320      * which bypasses discovery entirely. Falls back to {@link #SMC_KEYS_GPU_TEMP_AS} if discovery cannot complete,
321      * without caching, so a later call retries.
322      *
323      * @return The keys to read for GPU temperature, never null
324      */
325     public static List<String> getGpuTemperatureKeys() {
326         return GPU_KEYS.get(SmcUtil::discoverGpuTemperatureKeys);
327     }

329     /**
330      * Get the keys to read for fan speeds, discovering them from the SMC key index on first use and caching the result.
331      * <p>
332      * The key index is authoritative about which keys exist, so it is preferred over the count {@code FNum} reports. A
333      * positive {@code FNum} still implies the conventionally named keys where discovery finds none, so this never
334      * reports fewer fans than reading {@code FNum} directly did. See
335      * {@link SmcKeyIndex#reconcileFanKeys(java.util.List, long)}.
336      * <p>
337      * Can be overridden with the {@link GlobalConfig#OSHI_OS_MAC_SENSORS_FANSPEED_KEYS} configuration property, which
338      * bypasses discovery entirely.
339      *
340      * @return The keys to read for fan speeds, never null
341      */
342     public static List<String> getFanSpeedKeys() {
343         return FAN_KEYS.get(SmcUtil::discoverFanSpeedKeys);
344     }

346     /**
347      * @return the keys to read, or null if neither the key index nor {@code FNum} could be read
348      */
349     private static @Nullable List<String> discoverFanSpeedKeys() {
350         IOConnect conn = smcOpen();
351         if (conn == null) {
352             return null; // NOSONAR java:S1168 - null means "could not read", which the caller must not cache
353         }
354         try {
355             int keyCount = (int) smcGetLong(conn, SMC_KEY_COUNT);
356             // Scanning the index rather than probing F0Ac through F9Ac directly, because smcGetDataType cannot tell an
357             // absent key from a failed read: both return an empty string. findKeys tracks read failures and so can
358             // distinguish "this machine has no fans" from "ask again later", which a direct probe cannot.
359             List<String> discovered = SmcKeyIndex.findKeys(keyCount, i -> smcReadKeyAtIndex(conn, i), FAN_KEY_PREFIX,
360                     SmcKeyIndex::isFanSpeedKey);
361             // Read FNum on the same connection, so a machine whose index is unreadable still reports its fans.
362             return SmcKeyIndex.reconcileFanKeys(discovered, smcGetLong(conn, SMC_KEY_FAN_NUM));
363         } finally {
364             smcClose(conn);
365         }
366     }

368     /**
369      * Get the keys to read for CPU voltage, in the order they should be tried.
370      * <p>
371      * Unlike the GPU temperature keys these are not discovered from the key index, because the {@code V} prefix is not
372      * specific to the CPU: an M3 Pro reports a 0.75 V core voltage from {@code VP0C} alongside a 20 V supply rail from
373      * {@code VD0R}, with no naming convention separating them. A prefix scan could therefore report a rail voltage as
374      * the CPU's, which is worse than reporting nothing. Use {@link GlobalConfig#OSHI_OS_MAC_SENSORS_CPUVOLTAGE_KEYS} to
375      * name the key on hardware where neither default works.
376      *
377      * @return The keys to read for CPU voltage, never null
378      */
379     public static List<String> getCpuVoltageKeys() {
380         List<String> configured = SmcKeyIndex
381                 .parseConfiguredKeys(GlobalConfig.get(GlobalConfig.OSHI_OS_MAC_SENSORS_CPUVOLTAGE_KEYS, ""));
382         return configured.isEmpty() ? SMC_KEYS_CPU_VOLTAGE : configured;
383     }

415     /**
416      * @return the discovered keys, or null if the SMC key index could not be read
417      */
418     private static @Nullable List<String> discoverGpuTemperatureKeys() {
419         IOConnect conn = smcOpen();
420         if (conn == null) {
421             return null; // NOSONAR java:S1168 - null means "could not read", which the caller must not cache
422         }
423         try {
424             int keyCount = (int) smcGetLong(conn, SMC_KEY_COUNT);
425             // No data-type filter: smcGetFloat already decodes every temperature encoding the SMC uses (flt, sp78,
426             // fpe2) and returns 0 for anything else, which the plausibility floor then rejects. Filtering on "flt"
427             // here would cost a second read per key and would wrongly exclude a sensor reported as sp78.
428             return SmcKeyIndex.findKeys(keyCount, i -> smcReadKeyAtIndex(conn, i), GPU_KEY_PREFIX,
429                     SmcKeyIndex::isGpuTemperatureKey);
430         } finally {
431             smcClose(conn);
432         }
433     }

453     /**
454      * Get cached keyInfo if it exists, or generate new keyInfo
455      *
456      * @param conn            The connection
457      * @param inputStructure  Key data input
458      * @param outputStructure Key data output
459      * @return 0 if successful, nonzero if failure
460      */
461     public static int smcGetKeyInfo(IOConnect conn, SMCKeyData inputStructure, SMCKeyData outputStructure) {
462         if (keyInfoCache.containsKey(inputStructure.key)) {
463             SMCKeyDataKeyInfo keyInfo = keyInfoCache.get(inputStructure.key);
464             outputStructure.keyInfo.dataSize = keyInfo.dataSize;
465             outputStructure.keyInfo.dataType = keyInfo.dataType;
466             outputStructure.keyInfo.dataAttributes = keyInfo.dataAttributes;
467         } else {
468             inputStructure.data8 = SMC_CMD_READ_KEYINFO;
469             int result = smcCall(conn, KERNEL_INDEX_SMC, inputStructure, outputStructure);
470             if (result != 0) {
471                 return result;
472             }
473             SMCKeyDataKeyInfo keyInfo = new SMCKeyDataKeyInfo();
474             keyInfo.dataSize = outputStructure.keyInfo.dataSize;
475             keyInfo.dataType = outputStructure.keyInfo.dataType;
476             keyInfo.dataAttributes = outputStructure.keyInfo.dataAttributes;
477             keyInfoCache.put(inputStructure.key, keyInfo);
478         }
479         return 0;
480     }

482     /**
483      * Read a key from SMC
484      *
485      * @param conn The connection
486      * @param key  Key to read
487      * @param val  Structure to receive the result
488      * @return 0 if successful, nonzero if failure
489      */
490     public static int smcReadKey(IOConnect conn, String key, SMCVal val) {
491         try (SMCKeyData inputStructure = new SMCKeyData(); SMCKeyData outputStructure = new SMCKeyData()) {
492             inputStructure.key = (int) ParseUtil.strToLong(key, 4);
493             int result = smcGetKeyInfo(conn, inputStructure, outputStructure);
494             if (result == 0) {
495                 val.dataSize = outputStructure.keyInfo.dataSize;
496                 val.dataType = ParseUtil.longToByteArray(outputStructure.keyInfo.dataType, 4, 5);
497 
498                 inputStructure.keyInfo.dataSize = val.dataSize;
499                 inputStructure.data8 = SMC_CMD_READ_BYTES;
500 
501                 result = smcCall(conn, KERNEL_INDEX_SMC, inputStructure, outputStructure);
502                 if (result == 0) {
503                     System.arraycopy(outputStructure.bytes, 0, val.bytes, 0, val.bytes.length);
504                     return 0;
505                 }
506             }
507             return result;
508         }
509     }

511     /**
512      * Call SMC
513      *
514      * @param conn            The connection
515      * @param index           Kernel index
516      * @param inputStructure  Key data input
517      * @param outputStructure Key data output
518      * @return 0 if successful, nonzero if failure
519      */
520     public static int smcCall(IOConnect conn, int index, SMCKeyData inputStructure, SMCKeyData outputStructure) {
521         try (CloseableNativeLongByReference size = new CloseableNativeLongByReference(
522                 new NativeLong(outputStructure.size()))) {
523             return IO.IOConnectCallStructMethod(conn, index, inputStructure, new NativeLong(inputStructure.size()),
524                     outputStructure, size);
525         }
