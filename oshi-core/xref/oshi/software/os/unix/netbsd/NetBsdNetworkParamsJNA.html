5   package oshi.software.os.unix.netbsd;
6   
7   import com.sun.jna.Memory;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.software.common.os.unix.netbsd.NetBsdNetworkParams;
11  import oshi.util.platform.unix.netbsd.NetBsdSysctlUtil;
12  
13  /**
14   * NetBSD network parameters, reading the routing table from the kernel where the base class runs a command.
15   */
16  @ThreadSafe
17  public class NetBsdNetworkParamsJNA extends NetBsdNetworkParams {
18  
19      private static final int CTL_NET = 4;
20      private static final int PF_ROUTE = 17;
21      private static final int NET_RT_DUMP = 1;
22  
23      @Override
24      protected byte[] queryRouteDump() {
25          // Every native call on this platform has to ask first: the runner without java-jna installed still uses this
26          // class, and reaching libc there fails while the library loads rather than by returning anything
27          if (!NetBsdSysctlUtil.JNA_AVAILABLE) {
28              return new byte[0];
29          }
30          int[] mib = { CTL_NET, PF_ROUTE, 0, 0, NET_RT_DUMP, 0 };
31          try (Memory buf = NetBsdSysctlUtil.sysctl(mib)) {
32              if (buf == null) {
33                  return new byte[0];
34              }
35              return buf.getByteArray(0, (int) buf.size());
36          }
37      }
38  }
