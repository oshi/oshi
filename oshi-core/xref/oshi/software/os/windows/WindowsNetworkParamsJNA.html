47      private static final int COMPUTER_NAME_DNS_DOMAIN_FULLY_QUALIFIED = 3;
48  
49      @Override
50      public String getDomainName() {
51          char[] buffer = new char[256];
52          try (CloseableIntByReference bufferSize = new CloseableIntByReference(buffer.length)) {
53              if (!Kernel32.INSTANCE.GetComputerNameEx(COMPUTER_NAME_DNS_DOMAIN_FULLY_QUALIFIED, buffer, bufferSize)) {
54                  LOG.error("Failed to get dns domain name. Error code: {}", Kernel32.INSTANCE.GetLastError());
55                  return "";
56              }
57          }
58          return Native.toString(buffer);
59      }
60  
61      @Override
62      public String[] getDnsServers() {
63          try (CloseableIntByReference bufferSize = new CloseableIntByReference()) {
64              int ret = IPHlpAPI.INSTANCE.GetNetworkParams(null, bufferSize);
65              if (ret != WinError.ERROR_BUFFER_OVERFLOW) {
66                  LOG.error("Failed to get network parameters buffer size. Error code: {}", ret);
67                  return new String[0];
68              }
69  
70              try (Memory buffer = new Memory(bufferSize.getValue())) {
71                  ret = IPHlpAPI.INSTANCE.GetNetworkParams(buffer, bufferSize);
72                  if (ret != 0) {
73                      LOG.error("Failed to get network parameters. Error code: {}", ret);
74                      return new String[0];
75                  }
76                  FIXED_INFO fixedInfo = new FIXED_INFO(buffer);
77  
78                  List<String> list = new ArrayList<>();
79                  IP_ADDR_STRING dns = fixedInfo.DnsServerList;
80                  while (dns != null) {
81                      // a char array of size 16.
82                      // This array holds an IPv4 address in dotted decimal notation.
83                      String addr = Native.toString(dns.IpAddress.String, StandardCharsets.US_ASCII);
84                      int nullPos = addr.indexOf(0);
85                      if (nullPos != -1) {
86                          addr = addr.substring(0, nullPos);
87                      }
88                      list.add(addr);
89                      dns = dns.Next;
90                  }
91                  return list.toArray(new String[0]);
92              }
93          }
94      }

105     @Override
106     protected List<RouteRow> queryRouteRows() {
107         try (CloseablePointerByReference tableRef = new CloseablePointerByReference()) {
108             int ret = IPHlpAPI.INSTANCE.GetIpForwardTable2((short) AF_UNSPEC, tableRef);
109             if (ret != WinError.NO_ERROR) {
110                 LOG.error("Failed to get the IP forward table. Error code: {}", ret);
111                 return new ArrayList<>();
112             }
113             Pointer table = tableRef.getValue();
114             try {
115                 return readRows(table);
116             } finally {
117                 IPHlpAPI.INSTANCE.FreeMibTable(table);
118             }
119         }
120     }
121 
122     private static List<RouteRow> readRows(Pointer table) {
123         int numEntries = table.getInt(0);
124         if (numEntries <= 0) {
125             return new ArrayList<>();
126         }
127         // MIB_IPFORWARD_ROW2 is 8-byte aligned because NET_LUID is a ULONG64, so the row array begins at offset 8
128         // rather than immediately after the 4-byte NumEntries. Let JNA compute the row size rather than hardcoding
129         // it, so this stays correct if the structure is ever remapped.
130         int rowSize = new MIB_IPFORWARD_ROW2().size();
131         List<RouteRow> rows = new ArrayList<>(numEntries);
132         for (int i = 0; i < numEntries; i++) {
133             MIB_IPFORWARD_ROW2 row = Structure.newInstance(MIB_IPFORWARD_ROW2.class,
134                     table.share(8L + (long) i * rowSize));
135             row.read();
136             RouteRow out = new RouteRow();
137             out.destination = addressBytes(row.DestinationPrefix.Prefix);
138             out.prefixLength = row.DestinationPrefix.PrefixLength & 0xff;
139             out.nextHop = addressBytes(row.NextHop);
140             out.interfaceIndex = row.InterfaceIndex;
141             out.metric = ParseUtil.unsignedIntToLong(row.Metric);
142             rows.add(out);
143         }
144         return rows;
145     }
146 
147     private static byte[] addressBytes(SOCKADDR_INET address) {
148         if (address.si_family == AF_INET) {
149             // sin_addr occupies the same four bytes the IPv6 arm uses for sin6_flowinfo, in network order
150             return ParseUtil.parseIntToIP(address.ipv4AddrOrFlowInfo);
151         } else if (address.si_family == AF_INET6) {
152             return Arrays.copyOf(address.ipv6Addr, 16);
153         }
154         return new byte[0];
155     }
