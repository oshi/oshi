5   package oshi.hardware;
6   
7   import java.util.Optional;
8   
9   import oshi.annotation.PublicApi;
10  import oshi.annotation.concurrent.Immutable;
11  import oshi.util.Constants;
12  
13  /**
14   * Display refers to the information regarding a video source and monitor identified by the EDID standard.
15   * <p>
16   * {@link #getDisplayInfo()} returns a {@link DisplayInfo} exposing the display's decoded attributes — manufacturer ID,
17   * product ID, serial number, physical dimensions, preferred resolution, model name, and the raw (or synthesized) EDID
18   * byte array.
19   * <p>
20   * Example: extracting monitor manufacturer and dimensions:
21   *
22   * <pre>{@code
23   * for (Display display : hal.getDisplays()) {
24   *     DisplayInfo info = display.getDisplayInfo();
25   *     System.out.println("Manufacturer: " + info.getManufacturerID());
26   *     System.out.println("Product ID:   " + info.getProductID());
27   *     System.out.printf("Size: %d cm x %d cm%n", info.getHcm(), info.getVcm());
28   * }
29   * }</pre>
30   *
31   * For displays that report their attributes without providing an EDID (such as a built-in macOS Retina panel),
32   * {@link DisplayInfo#isEdidSynthetic()} returns {@code true} and {@link DisplayInfo#getEdid()} returns an EDID
33   * synthesized from those attributes.
34   * <p>
35   * What the display is, as described by its EDID, is on {@link DisplayInfo}. How it is attached and how the windowing
36   * system is currently driving it, such as {@link #getDevicePort()} and {@link #getCurrentMode()}, is on this interface.
37   *
38   * @see DisplayInfo
39   * @see DisplayMode
40   * @see oshi.util.EdidUtil
41   */
42  @PublicApi
43  @Immutable
44  public interface Display {
45      /**
46       * The EDID byte array.
47       *
48       * @return The EDID byte array, either reported by the display or, when {@link DisplayInfo#isEdidSynthetic()} is
49       *         {@code true}, synthesized from the display's reported attributes.
50       * @deprecated As of 7.4.0, use {@link #getDisplayInfo()}.{@link DisplayInfo#getEdid() getEdid()} instead; the
51       *             decoded attributes are also available directly from {@link DisplayInfo}. Scheduled for removal in the
52       *             next major release.
53       */
54      @Deprecated
55      byte[] getEdid();
56  
57      /**
58       * The decoded display information.
59       *
60       * @return A {@link DisplayInfo} holding the display's decoded attributes.
61       */
62      DisplayInfo getDisplayInfo();
63  
64      /**
65       * The system-level device identification for this display. The form of the identifier is platform-specific:
66       * <ul>
67       * <li>Linux: the DRM connector name from sysfs (e.g. {@code HDMI-A-1}, {@code eDP-1}, {@code DP-2}), or the
68       * {@code xrandr} output name when DRM sysfs is not available.</li>
69       * <li>macOS: on Apple Silicon, the port named by an external monitor's framebuffer {@code TransportDescription}
70       * (e.g. {@code Port-HDMI@1}, {@code Port-USB-C@1}), or the built-in panel's device tree name (e.g. {@code disp0}).
71       * Intel Macs do not expose a port.</li>
72       * <li>Windows: the connector derived from the Connecting and Configuring Displays (CCD) API's output technology and
73       * connector instance (e.g. {@code HDMI}, {@code DisplayPort-1}).</li>
74       * <li>Other UNIX platforms: the {@code xrandr} output name, which is the same value {@link #getOutputName()}
75       * returns.</li>
76       * </ul>
77       *
78       * @return The device port identifier, or {@link Constants#UNKNOWN} if not available.
79       */
80      default String getDevicePort() {
81          return Constants.UNKNOWN;
82      }
83  
84      /**
85       * The X11 output name for this display as reported by {@code xrandr} (e.g. {@code HDMI-1}, {@code DP2}). This is
86       * the name to pass to {@code xrandr --output}. Implemented on Linux and the other UNIX platforms, and only
87       * available where an X server with the RandR extension is reachable. On Linux the display is matched to an X output
88       * by its DRM {@code CONNECTOR_ID}, falling back to a comparison of their EDIDs.
89       *
90       * @return An {@link Optional} containing the xrandr output name, or empty if not available.
91       */
92      default Optional<String> getOutputName() {
93          return Optional.empty();
94      }
95  
96      /**
97       * The mode the windowing system is currently driving this display in: its size, pixel resolution, refresh rate,
98       * rotation, and position on the desktop.
99       * <p>
100      * The display must be matched to the windowing system's own view of it, which is done lazily on the first call:
101      * <ul>
102      * <li>macOS: by comparing the EDID's vendor, product and serial numbers with those CoreGraphics reports. Two
103      * connected monitors that report identical numbers cannot be told apart, and neither returns a mode.</li>
104      * <li>Windows: by the monitor's device path, through the Connecting and Configuring Displays (CCD) API.</li>
105      * <li>Linux and the other UNIX platforms: by the {@code xrandr} output that {@link #getOutputName()} names, so an X
106      * server with the RandR extension must be reachable.</li>
107      * </ul>
108      * A {@link Display} is a snapshot, so the mode is read once and not refreshed; query the displays again to see a
109      * change.
110      *
111      * @return An {@link Optional} containing the current mode, or empty if the display is not active on the desktop or
112      *         cannot be matched to the windowing system.
113      */
114     default Optional<DisplayMode> getCurrentMode() {
115         return Optional.empty();
116     }
117 
118     /**
119      * Whether this display is built into the device, such as a laptop panel, rather than an external monitor.
120      * <ul>
121      * <li>macOS: CoreGraphics' own classification for a display it can match, and known directly for the Apple Silicon
122      * built-in panel and external ports.</li>
123      * <li>Windows: from the output technology the CCD API reports for the display's connector, where embedded
124      * DisplayPort, LVDS, embedded UDI and internal connections are built in.</li>
125      * <li>Linux and the other UNIX platforms: from the connector name, where {@code eDP}, {@code LVDS} and {@code DSI}
126      * connectors are built in.</li>
127      * </ul>
128      *
129      * @return An {@link Optional} containing {@code true} if the display is built in, {@code false} if it is external,
130      *         or empty if it cannot be determined.
131      */
132     default Optional<Boolean> isBuiltIn() {
133         return Optional.empty();
134     }
135 
136     /**
137      * Whether this display is the primary display: the one the windowing system treats as the main desktop, placing it
138      * at the desktop origin and, on macOS, giving it the menu bar.
139      * <p>
140      * As for {@link #getCurrentMode()}, the display must be matched to the windowing system's own view of it, which is
141      * done lazily on the first call:
142      * <ul>
143      * <li>macOS: CoreGraphics' main display, as {@code CGDisplayIsMain} reports it. A display is matched by comparing
144      * the EDID's vendor, product and serial numbers with those CoreGraphics reports, and the Apple Silicon built-in
145      * panel by its built-in flag. Two active monitors that report identical numbers cannot be told apart, and neither
146      * returns a value; if CoreGraphics lists only one of them as active, both are matched to it and report its
147      * status.</li>
148      * <li>Windows: the display whose desktop position in its current mode is the origin, which is how Windows defines
149      * the primary display. A display that is attached but not active on the desktop has no current mode, and returns no
150      * value.</li>
151      * <li>Linux and the other UNIX platforms: the {@code xrandr} output marked {@code primary}, matched as
152      * {@link #getOutputName()} describes, so an X server with the RandR extension must be reachable. Under Wayland
153      * without an X server, or on a headless system, no value is returned.</li>
154      * </ul>
155      * Exactly one display reporting {@code true} is not guaranteed. On Windows, a display cloning the primary display
156      * shares its desktop position and also reports {@code true}. X11 need not designate a primary output at all, in
157      * which case every output reports {@code false}.
158      * <p>
159      * A {@link Display} is a snapshot, so the status is read once and not refreshed; query the displays again to see a
160      * change.
161      *
162      * @return An {@link Optional} containing {@code true} if the display is the primary display, {@code false} if it is
163      *         not, or empty if it cannot be determined because the display cannot be matched to the windowing system.
164      */
165     default Optional<Boolean> isPrimary() {
166         return Optional.empty();
167     }
168 }
