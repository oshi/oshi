86      // UdevStat.ordinal() only indexes this class's own lookup table, built from UdevStat.values() itself
87      @SuppressWarnings("EnumOrdinal")
88      private static int[] buildUdevStatOrders() {
89          int[] orders = new int[UdevStat.values().length];
90          for (UdevStat stat : UdevStat.values()) {
91              orders[stat.ordinal()] = stat.getOrder();
92          }
93          return orders;
94      }

142     /**
143      * Reads the stores a disk is built on from its sysfs {@code slaves} directory, which the kernel populates for a
144      * device-mapper, md RAID, or other stacked block device. A slave that has its own {@code /sys/block} entry is a
145      * whole disk; any other is a partition. Package-private for testing.
146      *
147      * Entries are device names below {@link DevPath#NODE_ROOT}, matching the names every backend gives its disks and
148      * partitions.
149      *
150      * @param sysBlock the sysfs block directory, normally {@link SysPath#BLOCK}
151      * @param name     the disk name, a device node path such as {@code /dev/dm-0}
152      * @return the backing stores, sorted, or empty if the disk's sysfs entry could not be read
153      */
154     static Optional<List<String>> readBackingStores(String sysBlock, String name) {
155         if (!name.startsWith(DevPath.NODE_ROOT)) {
156             return Optional.empty();
157         }
158         File[] slaves = new File(sysBlock + toSysfsName(name.substring(DevPath.NODE_ROOT.length())) + "/slaves")
159                 .listFiles();
160         if (slaves == null) {
161             return Optional.empty();
162         }
163         List<String> stores = new ArrayList<>();
164         for (File slave : slaves) {
165             String slaveName = slave.getName();
166             StoreKind kind = new File(sysBlock + slaveName).exists() ? StoreKind.DISK : StoreKind.PARTITION;
167             stores.add(kind.entry(DevPath.NODE_ROOT + toDevName(slaveName)));
168         }
169         stores.sort(null);
170         return Optional.of(Collections.unmodifiableList(stores));
171     }

173     /**
174      * Converts a sysfs block device name to its path below the device root. sysfs spells a slash in a device name as
175      * {@code !}, as in {@code cciss!c0d0} for {@code /dev/cciss/c0d0}.

177      * @param sysfsName the name of the device's sysfs entry
178      * @return the device's path relative to the device root

204     /**
205      * Computes and sets disk statistics from a stat string.
206      *
207      * @param store   the disk store to update
208      * @param devstat the stat string from sysfs or /proc/diskstats
209      */
210     // UdevStat.ordinal() only indexes UDEV_STAT_ORDERS/devstatArray, both built from UdevStat.values() itself
211     @SuppressWarnings("EnumOrdinal")
212     protected static void computeDiskStats(LinuxHWDiskStore store, String devstat) {
213         long[] devstatArray = ParseUtil.parseStringToLongArray(devstat, UDEV_STAT_ORDERS, UDEV_STAT_LENGTH, ' ');
214         store.setDiskStats(devstatArray[UdevStat.READS.ordinal()],
215                 devstatArray[UdevStat.READ_BYTES.ordinal()] * SECTORSIZE, devstatArray[UdevStat.WRITES.ordinal()],
216                 devstatArray[UdevStat.WRITE_BYTES.ordinal()] * SECTORSIZE,
217                 devstatArray[UdevStat.QUEUE_LENGTH.ordinal()], devstatArray[UdevStat.ACTIVE_MS.ordinal()],
218                 System.currentTimeMillis());
219     }

230     /**
231      * Parses {@code /proc/mounts} lines into a map of device to mount point. A device mounted more than once maps to
232      * its first mount, which precedes any bind mounts of it. Package-private for testing.
233      *
234      * @param mounts the lines of {@code /proc/mounts}
235      * @return a map of device paths to mount points, with the kernel's octal escapes decoded
236      */
237     static Map<String, String> parseMountsMap(List<String> mounts) {
238         Map<String, String> mountsMap = new HashMap<>();
239         for (String mount : mounts) {
240             String[] split = ParseUtil.whitespaces.split(mount, -1);
241             if (split.length < 2 || !split[0].startsWith(DevPath.NODE_ROOT)) {
242                 continue;
243             }
244             mountsMap.putIfAbsent(ParseUtil.unescapeOctal(split[0]), ParseUtil.unescapeOctal(split[1]));
245         }
246         return mountsMap;
247     }

249     /**
250      * Splits an LVM device-mapper name into its volume group and logical volume names, as {@code dmsetup splitname}
251      * does for the udev rules that set {@link #DM_VG_NAME} and {@link #DM_LV_NAME}. A single hyphen separates the
252      * names, and a doubled hyphen stands for a hyphen within one, so {@code my--vg-root} is volume group {@code my-vg}
253      * and logical volume {@code root}. Any further component is an LVM layer such as {@code real} or {@code cow}, which
254      * is not part of either name.
255      *
256      * @param dmName the device-mapper name
257      * @return the volume group and logical volume names, either of which is empty if the name does not carry it
258      */
259     protected static String[] splitLvmDmName(String dmName) {
260         String[] names = { "", "" };
261         StringBuilder sb = new StringBuilder();
262         int component = 0;
263         int i = 0;
264         while (i < dmName.length() && component < names.length) {
265             char c = dmName.charAt(i++);
266             if (c != '-') {
267                 sb.append(c);
268             } else if (i < dmName.length() && dmName.charAt(i) == '-') {
269                 sb.append('-');
270                 i++;
271             } else {
272                 names[component++] = sb.toString();
273                 sb.setLength(0);
274             }
275         }
276         if (component < names.length) {
277             names[component] = sb.toString();
278         }
279         return names;
280     }

350     /**
351      * Gets the mount point for a device-mapper device.
352      *
353      * @param mountsMap the map of device paths to mount points
354      * @param dmName    the device-mapper name, or {@code null} if unnamed
355      * @param devnode   the device node path
356      * @param sysPath   the sysfs path for the device
357      * @return the mount point or dependent device names
358      */
359     protected static String getMountPointForDmDevice(Map<String, String> mountsMap, @Nullable String dmName,
360             String devnode, String sysPath) {
361         String devicePath = getDmDevicePath(dmName, devnode);
362         String mountPoint = mountsMap.get(devicePath);
363         if (mountPoint != null) {
364             return mountPoint;
365         }
366         if (!devicePath.equals(devnode)) {
367             mountPoint = mountsMap.get(devnode);
368             if (mountPoint != null) {
369                 return mountPoint;
370             }
371         }
372         return getDependentNamesFromHoldersDirectory(sysPath);
373     }

375     /**
376      * Adds a partition entry for a supported device-mapper device.
377      *
378      * @param store     the disk store to update
379      * @param mountsMap the map of device paths to mount points
380      * @param dmUuid    the device-mapper UUID, or {@code null} if the device has none
381      * @param vgName    the LVM volume group name, or {@code null} if not an LVM volume
382      * @param lvName    the LVM logical volume name, or {@code null} if not an LVM volume
383      * @param dmName    the device-mapper name
384      * @param devnode   the device node path
385      * @param sysname   the sysfs device name
386      * @param sysPath   the sysfs path for the device
387      * @param fsType    the filesystem type, or {@code null} if unknown
388      * @param fsUuid    the filesystem UUID, or {@code null} if unknown
389      * @param fsLabel   the filesystem label, or {@code null} if unknown
390      * @param size      the partition size in bytes
391      * @param major     the major device ID
392      * @param minor     the minor device ID
393      */
394     protected static void addDeviceMapperPartition(LinuxHWDiskStore store, Map<String, String> mountsMap,
395             @Nullable String dmUuid, @Nullable String vgName, @Nullable String lvName, @Nullable String dmName,
396             String devnode, String sysname, String sysPath, @Nullable String fsType, @Nullable String fsUuid,
397             @Nullable String fsLabel, long size, int major, int minor) {
398         if (isLogicalVolume(dmUuid) && !Util.isBlank(vgName) && !Util.isBlank(lvName)) {
399             // The isBlank guards above already establish these, but a predicate does not narrow for the analyzer;
400             // the normalizer's return type does.
401             String vg = ParseUtil.getStringValueOrEmpty(vgName);
402             String lv = ParseUtil.getStringValueOrEmpty(lvName);
403             store.getMutablePartitionList().add(new HWPartition(getPartitionNameForDmDevice(vg, lv), sysname,
404                     fsType == null ? PARTITION : fsType, ParseUtil.getStringValueOrEmpty(fsUuid),
405                     ParseUtil.getStringValueOrEmpty(fsLabel), size, major, minor, getMountPointOfDmDevice(vg, lv)));
406         } else if (isEncryptedVolume(dmUuid)) {
407             String name = getDmDevicePath(dmName, devnode);
408             store.getMutablePartitionList()
409                     .add(new HWPartition(name, sysname, fsType == null ? PARTITION : fsType,
410                             ParseUtil.getStringValueOrEmpty(fsUuid), ParseUtil.getStringValueOrEmpty(fsLabel), size,
411                             major, minor, getMountPointForDmDevice(mountsMap, dmName, devnode, sysPath)));
412         }
413     }

415     /**
416      * Gets dependent device names from the holders directory.
417      *
418      * @param sysPath the sysfs path for the device
419      * @return space-separated holder names, or empty string
420      */
421     protected static String getDependentNamesFromHoldersDirectory(String sysPath) {
422         File holdersDir = new File(sysPath + "/holders");
423         File[] holders = holdersDir.listFiles();
424         if (holders != null) {
425             return Arrays.stream(holders).map(File::getName).collect(Collectors.joining(" "));
426         }
427         return "";
428     }

430     /**
431      * Sorts and makes partition lists unmodifiable for all disk stores.
432      *
433      * @param result the list of disk stores to finalize
434      */
435     protected static void finalizePartitions(List<HWDiskStore> result) {
436         for (HWDiskStore hwds : result) {
437             LinuxHWDiskStore store = (LinuxHWDiskStore) hwds;
438             store.setPartitionList(Collections.unmodifiableList(store.getPartitions().stream()
439                     .sorted(java.util.Comparator.comparing(HWPartition::getName)).collect(Collectors.toList())));
440         }
441     }

443     /**
444      * Field ordering in udev stat output.

446     protected enum UdevStat {
447         /** Number of reads completed. */
448         READS(0),
449         /** Number of sectors read. */
450         READ_BYTES(2),
451         /** Number of writes completed. */
452         WRITES(4),
453         /** Number of sectors written. */
454         WRITE_BYTES(6),
455         /** Number of I/Os currently in progress. */
456         QUEUE_LENGTH(8),
457         /** Time spent doing I/Os in milliseconds. */
458         ACTIVE_MS(9);
