49      /**
50       * Populates the physical volumes map from the {@code pvs} command output. Requires elevated permissions; may return
51       * an empty map if the command fails.
52       *
53       * @return map of VG name to set of PV device paths
54       */
55      protected static Map<String, Set<String>> queryPhysicalVolumes() {
56          return parsePhysicalVolumes(ExecutingCommand.runNative("pvs -o vg_name,pv_name"));
57      }
58  
59      /**
60       * Parses {@code pvs -o vg_name,pv_name} output into a map of volume-group name to physical-volume device paths.
61       * Package-private for testing.
62       *
63       * @param pvs the lines of {@code pvs -o vg_name,pv_name} output
64       * @return map of VG name to set of PV device paths
65       */
66      static Map<String, Set<String>> parsePhysicalVolumes(List<String> pvs) {
67          Map<String, Set<String>> physicalVolumesMap = new HashMap<>();
68          for (String s : pvs) {
69              String[] split = ParseUtil.whitespaces.split(s.trim(), -1);
70              if (split.length == 2 && split[1].startsWith(DevPath.DEV)) {
71                  physicalVolumesMap.computeIfAbsent(split[0], k -> new HashSet<>()).add(split[1]);
72              }
73          }
74          return physicalVolumesMap;
75      }
76  
77      /**
78       * The udev attributes of one block device, read by the bindings so the device-mapper filtering and volume-group
79       * assembly can be shared. Any field may be null if udev did not report that property.
80       */
81      public static final class UdevBlockDevice {
82          private final String syspath;
83          private final String devnode;
84          private final String uuid;
85          private final String vgName;
86          private final String lvName;
87  
88          /**
89           * Creates a block device record.
90           *
91           * @param syspath the sysfs path of the device
92           * @param devnode the device node path, e.g. {@code /dev/dm-0}
93           * @param uuid    the {@code DM_UUID} property
94           * @param vgName  the {@code DM_VG_NAME} property
95           * @param lvName  the {@code DM_LV_NAME} property
96           */
97          public UdevBlockDevice(String syspath, String devnode, String uuid, String vgName, String lvName) {
98              this.syspath = syspath;
99              this.devnode = devnode;
100             this.uuid = uuid;
101             this.vgName = vgName;
102             this.lvName = lvName;
103         }
104     }

106     /**
107      * Creates the binding's volume group type.
108      */
109     @FunctionalInterface
110     public interface LogicalVolumeGroupFactory {
111         /**
112          * Creates a volume group.
113          *
114          * @param name  the volume group name
115          * @param lvMap the logical volume map
116          * @param pvSet the physical volume set
117          * @return the volume group
118          */
119         LogicalVolumeGroup create(String name, Map<String, Set<String>> lvMap, Set<String> pvSet);
120     }
121 
122     /**
123      * Assembles volume groups from enumerated block devices, keeping only the device-mapper devices that LVM owns and
124      * reading each one's physical volumes from its sysfs {@code slaves} directory.
125      *
126      * @param devices the block devices reported by udev
127      * @param factory creates the binding's volume group type
128      * @return the volume groups, never null
129      */
130     protected static List<LogicalVolumeGroup> buildLogicalVolumeGroups(List<UdevBlockDevice> devices,
131             LogicalVolumeGroupFactory factory) {
132         return buildLogicalVolumeGroups(devices, queryPhysicalVolumes(), factory);
133     }
134 
135     /**
136      * Assembles volume groups from enumerated block devices and an already-queried physical volume map. Package-private
137      * so tests can supply the map instead of running {@code pvs}.
138      *
139      * @param devices            the block devices reported by udev
140      * @param physicalVolumesMap map of VG name to set of PV device paths, mutated as devices are processed
141      * @param factory            creates the binding's volume group type
142      * @return the volume groups, never null
143      */
144     static List<LogicalVolumeGroup> buildLogicalVolumeGroups(List<UdevBlockDevice> devices,
145             Map<String, Set<String>> physicalVolumesMap, LogicalVolumeGroupFactory factory) {
146         Map<String, Map<String, Set<String>>> logicalVolumesMap = new HashMap<>();
147         for (UdevBlockDevice device : devices) {
148             if (device.devnode == null || !device.devnode.startsWith(DevPath.DM) || device.uuid == null
149                     || !device.uuid.startsWith("LVM-") || Util.isBlank(device.vgName) || Util.isBlank(device.lvName)) {
150                 continue;
151             }
152             Map<String, Set<String>> lvMapForGroup = logicalVolumesMap.computeIfAbsent(device.vgName,
153                     k -> new HashMap<>());
154             Set<String> pvSetForGroup = physicalVolumesMap.computeIfAbsent(device.vgName, k -> new HashSet<>());
155             File[] slaves = new File(device.syspath + "/slaves").listFiles();
156             if (slaves != null) {
157                 for (File f : slaves) {
158                     String pvName = DevPath.DEV + f.getName();
159                     lvMapForGroup.computeIfAbsent(device.lvName, k -> new HashSet<>()).add(pvName);
160                     pvSetForGroup.add(pvName);
161                 }
162             }
163         }
164         List<LogicalVolumeGroup> lvgList = new ArrayList<>();
165         for (Entry<String, Map<String, Set<String>>> entry : logicalVolumesMap.entrySet()) {
166             // Every key here was added to physicalVolumesMap above, but default rather than risk a null set reaching
167             // the immutable-copy constructor.
168             lvgList.add(factory.create(entry.getKey(), entry.getValue(),
169                     physicalVolumesMap.getOrDefault(entry.getKey(), Collections.emptySet())));
170         }
171         return lvgList;
172     }
