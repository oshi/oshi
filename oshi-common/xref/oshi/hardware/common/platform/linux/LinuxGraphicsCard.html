198     /**
199      * Returns the total VRAM for a card from sysfs when the driver publishes it, falling back to the size parsed from
200      * lspci or lshw.
201      *
202      * <p>
203      * The lspci and lshw values are memory BAR sizes, which are the aperture the card exposes to the host rather than
204      * the memory it has. amdgpu publishes the real total, so read that when it is there. NVML takes precedence on
205      * NVIDIA cards, so a backend that has an NVML lookup calls this only after that lookup comes back empty.
206      *
207      * @param attrs the parsed attributes for the card
208      * @return VRAM in bytes, or 0 if no source reported it
209      */
210     protected static long vramTotal(Attrs attrs) {
211         if (!attrs.getDrmDevicePath().isEmpty() && "amdgpu".equals(attrs.getDriverName().toLowerCase(Locale.ROOT))) {
212             long total = FileUtil.getLongFromFile(attrs.getDrmDevicePath() + "/mem_info_vram_total", -1L);
213             if (total > 0) {
214                 return total;
215             }
216         }
217         return attrs.getVram();
218     }

220     /**
221      * Queries graphics cards from lspci/lshw and constructs instances using the provided factory.
222      *
223      * @param factory function that creates a concrete {@link GraphicsCard} from parsed attributes
224      * @return list of graphics cards
225      */
226     public static List<GraphicsCard> getGraphicsCards(Function<Attrs, GraphicsCard> factory) {
227         List<GraphicsCard> cardList = getGraphicsCardsFromLspci(factory);
228         if (cardList.isEmpty()) {
229             cardList = getGraphicsCardsFromLshw(factory);
230         }
231         if (cardList.isEmpty()) {
232             cardList = getGraphicsCardsFromDrm(factory, DRM_PATH, UdevData.RUN_UDEV_DATA);
233         }
234         return cardList;
235     }

237     /**
238      * Lists the graphics cards the kernel's DRM subsystem drives, for hosts with neither lspci nor lshw, and for the
239      * GPUs of an ARM system-on-chip, which are not PCI devices. A PCI card is named from udev's hardware database, as
240      * lspci would name it; a system-on-chip GPU by its devicetree {@code compatible} string.
241      *
242      * @param factory     function that creates a concrete {@link GraphicsCard} from parsed attributes
243      * @param drmPath     the DRM class directory, ending in a slash
244      * @param udevDataDir the udev database directory, ending in a slash
245      * @return list of graphics cards
246      */
247     static List<GraphicsCard> getGraphicsCardsFromDrm(Function<Attrs, GraphicsCard> factory, String drmPath,
248             String udevDataDir) {
249         List<GraphicsCard> cardList = new ArrayList<>();
250         File[] cards = new File(drmPath).listFiles(f -> f.getName().matches("card\\d+"));
251         if (cards == null) {
252             return cardList;
253         }
254         Arrays.sort(cards, Comparator.comparingInt(f -> ParseUtil.getFirstIntValue(f.getName())));
255         for (File card : cards) {
256             String devicePath = card.getAbsolutePath() + DEVICE;
257             String driver = readDriverName(devicePath + "/driver");
258             // simpledrm only drives the framebuffer the firmware left behind
259             if (driver.isEmpty() || "simple-framebuffer".equals(driver) || "simpledrm".equals(driver)) {
260                 continue;
261             }
262             Map<String, String> uevent = FileUtil.getKeyValueMapFromFile(devicePath + "/uevent", "=");
263             String slot = uevent.getOrDefault("PCI_SLOT_NAME", "");
264             String name = Constants.UNKNOWN;
265             String deviceId = Constants.UNKNOWN;
266             String vendor = Constants.UNKNOWN;
267             String versionInfo = Constants.UNKNOWN;
268             if (!slot.isEmpty()) {
269                 Map<String, String> udev = UdevData.queryDevice(udevDataDir, "pci", slot);
270                 String vendorId = FileUtil.getStringFromFile(devicePath + "/vendor").trim();
271                 deviceId = ParseUtil.getStringValueOrUnknown(FileUtil.getStringFromFile(devicePath + DEVICE).trim());
272                 name = ParseUtil.getStringValueOrUnknown(udev.get("ID_MODEL_FROM_DATABASE"));
273                 String vendorName = ParseUtil.getStringValueOrEmpty(udev.get("ID_VENDOR_FROM_DATABASE")).trim();
274                 if (!vendorName.isEmpty()) {
275                     vendor = vendorId.isEmpty() ? vendorName : vendorName + " (" + vendorId + ")";
276                 } else {
277                     vendor = ParseUtil.getStringValueOrUnknown(vendorId);
278                 }
279                 String revision = FileUtil.getStringFromFile(devicePath + "/revision").trim();
280                 if (revision.startsWith("0x")) {
281                     versionInfo = "Rev:\t" + revision.substring(2);
282                 }
283             } else {
284                 String compatible = ParseUtil.getStringValueOrEmpty(uevent.get("OF_COMPATIBLE_0")).trim();
285                 if (!compatible.isEmpty()) {
286                     name = compatible;
287                     vendor = ParseUtil.getStringValueOrUnknown(Devicetree.queryVendor(compatible));
288                 }
289             }
290             cardList.add(factory.apply(new Attrs(name, deviceId, vendor, versionInfo, 0L, devicePath, driver, slot)));
291         }
292         return cardList;
293     }

302     /**
303      * Parse graphics card information from lspci machine-readable output.
304      *
305      * @param lspci      output of {@code lspci -vnnmm}
306      * @param factory    function that creates a concrete {@link GraphicsCard} from parsed attributes
307      * @param vramLookup function to look up VRAM for a PCI slot address
308      * @param drmLookup  function to look up DRM info for a PCI slot address
309      * @return list of graphics cards
310      */
311     static List<GraphicsCard> getGraphicsCardsFromLspci(List<String> lspci, Function<Attrs, GraphicsCard> factory,
312             ToLongFunction<String> vramLookup, Function<@Nullable String, Triplet<String, String, String>> drmLookup) {
313         List<GraphicsCard> cardList = new ArrayList<>();
314         String name = Constants.UNKNOWN;
315         String deviceId = Constants.UNKNOWN;
316         String vendor = Constants.UNKNOWN;
317         List<String> versionInfoList = new ArrayList<>();
318         boolean found = false;
319         String slot = null;
320         String lookupDevice = null;
321         for (String line : lspci) {
322             String[] split = line.trim().split(":", 2);
323             String prefix = split[0];
324             // Skip until we reach a display controller class
325             if (prefix.equals("Class") && isDisplayClass(split.length > 1 ? split[1].trim() : "")) {
326                 found = true;
327                 lookupDevice = slot;
328                 name = Constants.UNKNOWN;
329                 deviceId = Constants.UNKNOWN;
330                 vendor = Constants.UNKNOWN;
331                 versionInfoList.clear();
332             } else if (prefix.equals("Slot") && split.length > 1) {
333                 // Capture PCI slot address (e.g. "01:00.0") for use with lspci -s
334                 slot = split[1].trim();
335             }
336             if (found) {
337                 if (split.length < 2) {
338                     // Save previous card
339                     Triplet<String, String, String> drmInfo = drmLookup.apply(lookupDevice);
340                     cardList.add(factory.apply(new Attrs(name, deviceId, vendor,
341                             versionInfoList.isEmpty() ? Constants.UNKNOWN : String.join(", ", versionInfoList),
342                             lookupDevice != null ? vramLookup.applyAsLong(lookupDevice) : 0L, drmInfo.getA(),
343                             drmInfo.getB(), drmInfo.getC())));
344                     versionInfoList.clear();
345                     found = false;
346                 } else {
347                     if (prefix.equals("Device")) {
348                         Pair<String, String> pair = ParseUtil.parseLspciMachineReadable(split[1].trim());
349                         if (pair != null) {
350                             name = pair.getA();
351                             deviceId = "0x" + pair.getB();
352                         }
353                     } else if (prefix.equals("Vendor")) {
354                         Pair<String, String> pair = ParseUtil.parseLspciMachineReadable(split[1].trim());
355                         if (pair != null) {
356                             vendor = pair.getA() + " (0x" + pair.getB() + ")";
357                         } else {
358                             vendor = split[1].trim();
359                         }
360                     } else if (prefix.equals("Rev")) {
361                         versionInfoList.add(line.trim());
362                     }
363                 }
364             }
365         }
366         // If we haven't yet written the last card do so now
367         if (found) {
368             Triplet<String, String, String> drmInfo = drmLookup.apply(lookupDevice);
369             cardList.add(factory.apply(new Attrs(name, deviceId, vendor,
370                     versionInfoList.isEmpty() ? Constants.UNKNOWN : String.join(", ", versionInfoList),
371                     lookupDevice != null ? vramLookup.applyAsLong(lookupDevice) : 0L, drmInfo.getA(), drmInfo.getB(),
372                     drmInfo.getC())));
373         }
374         return cardList;
375     }

377     /**
378      * Parse prefetchable memory size from lspci verbose output.
379      *
380      * @param lspciMem output of {@code lspci -v -s <device>}
381      * @return total prefetchable memory size in bytes
382      */
383     static long queryLspciMemorySize(List<String> lspciMem) {
384         long vram = 0L;
385         for (String mem : lspciMem) {
386             if (mem.contains(" prefetchable")) {
387                 vram += ParseUtil.parseLspciMemorySize(mem);
388             }
389         }
390         return vram;
391     }

399     /**
400      * Parse graphics card information from lshw output.
401      *
402      * @param lshw      output of {@code lshw -C display}
403      * @param factory   function that creates a concrete {@link GraphicsCard} from parsed attributes
404      * @param drmLookup function to look up DRM info for a PCI slot address
405      * @return list of graphics cards
406      */
407     static List<GraphicsCard> getGraphicsCardsFromLshw(List<String> lshw, Function<Attrs, GraphicsCard> factory,
408             Function<@Nullable String, Triplet<String, String, String>> drmLookup) {
409         List<GraphicsCard> cardList = new ArrayList<>();
410         String name = Constants.UNKNOWN;
411         String deviceId = Constants.UNKNOWN;
412         String vendor = Constants.UNKNOWN;
413         List<String> versionInfoList = new ArrayList<>();
414         long vram = 0;
415         int cardNum = 0;
416         String busInfo = null;
417         for (String line : lshw) {
418             String[] split = line.trim().split(":", 2);
419             if (split[0].startsWith("*-display")) {
420                 // Save previous card
421                 if (cardNum++ > 0) {
422                     Triplet<String, String, String> drmInfo = drmLookup.apply(busInfo);
423                     cardList.add(factory.apply(new Attrs(name, deviceId, vendor,
424                             versionInfoList.isEmpty() ? Constants.UNKNOWN : String.join(", ", versionInfoList), vram,
425                             drmInfo.getA(), drmInfo.getB(), drmInfo.getC())));
426                 }
427                 name = Constants.UNKNOWN;
428                 deviceId = Constants.UNKNOWN;
429                 vendor = Constants.UNKNOWN;
430                 vram = 0;
431                 versionInfoList.clear();
432                 busInfo = null;
433             } else if (split.length == 2) {
434                 String prefix = split[0];
435                 if (prefix.equals("product")) {
436                     name = split[1].trim();
437                 } else if (prefix.equals("vendor")) {
438                     vendor = split[1].trim();
439                 } else if (prefix.equals("version")) {
440                     versionInfoList.add(line.trim());
441                 } else if (prefix.startsWith("resources")) {
442                     vram = ParseUtil.parseLshwResourceString(split[1].trim());
443                 } else if (prefix.equals("bus info")) {
444                     // lshw reports PCI slot as "pci@0000:01:00.0"; the value contains multiple
445                     // colons so we locate the first colon in the original line to get the full value.
446                     int colonIdx = line.indexOf(':');
447                     String raw = colonIdx >= 0 ? line.substring(colonIdx + 1).trim() : "";
448                     busInfo = raw.startsWith("pci@") ? raw.substring(4) : raw;
449                 }
450             }
451         }
452         if (cardNum > 0) {
453             Triplet<String, String, String> drmInfo = drmLookup.apply(busInfo);
454             cardList.add(factory.apply(new Attrs(name, deviceId, vendor,
455                     versionInfoList.isEmpty() ? Constants.UNKNOWN : String.join(", ", versionInfoList), vram,
456                     drmInfo.getA(), drmInfo.getB(), drmInfo.getC())));
457         }
458         return cardList;
459     }

461     /**
462      * Tests whether an lspci {@code Class:} field value denotes a display controller.
463      * <p>
464      * A plain substring test for "VGA" is not sufficient: PCI class 0x0000 is rendered as "Non-VGA unclassified
465      * device", which contains "VGA" but is not a graphics card. When the numeric class code is present (lspci
466      * {@code -nn}) it is authoritative; PCI base class 0x03 is Display controller.
467      *
468      * @param classValue the trimmed value of an lspci {@code Class:} line
469      * @return true if the class denotes a display controller
470      */
471     static boolean isDisplayClass(String classValue) {
472         Pair<String, String> pair = ParseUtil.parseLspciMachineReadable(classValue);
473         if (pair != null && pair.getB().length() >= 2) {
474             return pair.getB().startsWith("03");
475         }
476         return classValue.startsWith("VGA compatible controller") || classValue.startsWith("3D controller")
477                 || classValue.startsWith("Display controller");
478     }

480     /**
481      * Finds the sysfs DRM device path, driver name, and PCI bus ID for a GPU by matching against the PCI slot address
482      * from the uevent file under each DRM card's device directory.
483      *
484      * <p>
485      * When {@code pciSlot} is non-null, each card's {@code device/uevent} file is read and the {@code PCI_SLOT_NAME}
486      * key is compared against the supplied slot (e.g. {@code "0000:01:00.0"} or {@code "01:00.0"}). The first card
487      * whose slot matches is returned. If no match is found, or if {@code pciSlot} is null (lshw path), the first card
488      * with a non-empty driver symlink is returned as a best-effort fallback.
489      *
490      * @param pciSlot the PCI slot address from lspci (e.g. {@code "01:00.0"}), or {@code null} to use first-match
491      * @return triplet of (drmDevicePath, driverName, pciBusId), all empty strings if not found
492      */
493     private static Triplet<String, String, String> findDrmInfo(@Nullable String pciSlot) {
494         return findDrmInfo(pciSlot, DRM_PATH);
495     }

497     /**
498      * Finds the sysfs DRM device path, driver name, and PCI bus ID for a GPU.
499      *
500      * @param pciSlot the PCI slot address, or {@code null} to use first-match
501      * @param drmPath the base DRM sysfs directory path
502      * @return triplet of (drmDevicePath, driverName, pciBusId), all empty strings if not found
503      */
504     static Triplet<String, String, String> findDrmInfo(@Nullable String pciSlot, String drmPath) {
505         File drmDir = new File(drmPath);
506         File[] cards = drmDir.listFiles(f -> f.getName().matches("card\\d+"));
507         if (cards == null) {
508             return new Triplet<>("", "", "");
509         }
510         Triplet<String, String, String> firstWithDriver = null;
511         for (File card : cards) {
512             String devicePath = card.getAbsolutePath() + DEVICE;
513             String driver = readDriverName(devicePath + "/driver");
514             if (driver.isEmpty()) {
515                 continue;
516             }
517             String slotName = readUeventValue(devicePath + "/uevent", "PCI_SLOT_NAME");
518             if (firstWithDriver == null) {
519                 firstWithDriver = new Triplet<>(devicePath, driver, slotName);
520             }
521             // Attempt PCI slot match via uevent
522             if (pciSlot != null && slotName.endsWith(pciSlot)) {
523                 return new Triplet<>(devicePath, driver, slotName);
524             }
525         }
526         if (pciSlot != null) {
527             // We had a slot to match on and no DRM card claimed it, so this card has no DRM node (e.g. a GPU bound to
528             // vfio-pci). Returning another card's path would attribute that card's driver and metrics to this one.
529             return new Triplet<>("", "", "");
530         }
531         // No slot to match on: fall back to the first card with a driver symlink
532         return firstWithDriver != null ? firstWithDriver : new Triplet<>("", "", "");
533     }

535     /**
536      * Reads a key=value entry from a sysfs uevent file.
537      *
538      * @param ueventPath absolute path to the uevent file
539      * @param key        the key to look up (e.g. {@code "PCI_SLOT_NAME"})
540      * @return the value string, or empty string if not found
541      */
542     static String readUeventValue(String ueventPath, String key) {
543         List<String> lines = FileUtil.readFile(ueventPath);
544         String prefix = key + "=";
545         for (String line : lines) {
546             if (line.startsWith(prefix)) {
547                 return line.substring(prefix.length()).trim();
548             }
549         }
550         return "";
551     }

553     static String readDriverName(String driverSymlink) {
554         String target = FileUtil.readSymlinkTarget(new File(driverSymlink));
555         if (target == null || target.isEmpty()) {
556             return "";
557         }
558         // The symlink target resolves to a driver directory,
559         // e.g. "../../../bus/pci/drivers/amdgpu"; the last path segment is the driver name.
560         String name = FileUtil.getFileName(target);
561         return name.isEmpty() ? target : name;
562     }
