5   package oshi.hardware.common.platform.mac;
6   
7   import org.jspecify.annotations.Nullable;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  
11  /**
12   * A subscription to the macOS IOReport CPU performance state channels, implemented once per binding.
13   * <p>
14   * The two implementations wrap genuinely different native APIs, so only this shape is shared. It exists so
15   * {@link MacCentralProcessor} can hold the frequency logic once rather than duplicating it per backend.
16   * <p>
17   * These channels exist only on Apple Silicon, so a sampler is created only there.
18   */
19  @ThreadSafe
20  public interface IOReportCpuSampler extends AutoCloseable {
21  
22      /**
23       * Samples how long each CPU core, and each cluster of cores, spent in each of its performance states.
24       *
25       * @return the residency of both since the previous sample, or null if no previous sample exists, as on the first
26       *         call, or if the sample could not be taken
27       */
28      @Nullable
29      CpuResidencySample sampleResidencyDelta();
30  
31      /**
32       * Releases the subscription. Overridden to drop {@code AutoCloseable}'s checked exception.
33       */
34      @Override
35      void close();
36  }
