202     /**
203      * Queries the per-core IORegistry properties used to identify the CPU. Walks the {@code IOPlatformDevice} nodes
204      * named {@code cpuN} once, reading both the {@code compatible} property (which names the core's microarchitecture,
205      * e.g. {@code apple,avalanche arm,v8}) and the {@code cluster-type} property (a single letter, {@code P} for a
206      * performance cluster or {@code E} for an efficiency cluster).
207      *
208      * @return a map of physical processor numbers to a pair of ({@code compatible}, {@code cluster-type}). Either
209      *         element may be {@code null} if that property is absent from the node.
210      */
211     protected Map<Integer, Pair<@Nullable String, @Nullable String>> queryCoreProperties() {
212         Map<Integer, Pair<@Nullable String, @Nullable String>> coreProperties = new HashMap<>();
213         ioKitProvider().forEachMatchingService("IOPlatformDevice", entry -> {
214             String name = entry.getName();
215             if (name != null) {
216                 Matcher m = CPU_N.matcher(name.toLowerCase(Locale.ROOT));
217                 if (m.matches()) {
218                     int procId = ParseUtil.parseIntOrDefault(m.group(1), 0);
219                     coreProperties.put(procId, new Pair<>(ioRegString(entry.getByteArrayProperty("compatible")),
220                             ioRegString(entry.getByteArrayProperty("cluster-type"))));
221                 }
222             }
223         });
224         return coreProperties;
225     }

238     /**
239      * Derives a performance/efficiency class for each physical core, so that hybrid Apple Silicon cores are classified
240      * without a hardcoded table of core codenames. A higher value indicates a higher-performing core, matching
241      * {@link PhysicalProcessor#getEfficiency()}.
242      * <p>
243      * The first of these strategies that classifies every core wins:
244      * <ol>
245      * <li>The number of cores at each performance level, from {@code hw.nperflevels} and
246      * {@code hw.perflevel<n>.physicalcpu}. Level 0 is always the highest-performing level and macOS numbers the
247      * lowest-performing cores first, so the counts partition the cores in descending class order from the end. This is
248      * a documented Apple API, it produces dense classes for any number of levels, and it is the only source that can
249      * separate two levels whose clusters report the same {@code cluster-type}.</li>
250      * <li>The {@code cluster-type} property read from each core: {@code E} is an efficiency cluster, {@code P} a
251      * performance cluster, and {@code S} a higher-performing still cluster. Chip-independent and exact, and the only
252      * source available before macOS 12 introduced {@code hw.nperflevels}. A value outside that ranking abandons this
253      * strategy for the whole machine rather than only for the core reporting it, since it shows the letters here are
254      * not the letters being ranked.</li>
255      * <li>If only some cores report a recognized {@code cluster-type}, the mapping learned from those cores is applied
256      * to the other cores sharing the same codename.</li>
257      * <li>Grouping cores by codename and ordering the groups by their lowest core number, since macOS numbers
258      * efficiency cores first. Where there are exactly two groups and {@code hw.perflevel0.physicalcpu} matches the size
259      * of exactly one of them, that count decides which group is the performance group, overriding the ordering. This
260      * applies only when {@code cluster-type} classified no core at all: a {@code cluster-type} reading is exact, so
261      * where some cores reported one those values are kept and only the remainder falls to the default below.</li>
262      * <li>Failing all of those, every core is class 0, which is the historical behavior for a chip whose cores were not
263      * recognized, and for non-hybrid and Intel processors.</li>
264      * </ol>
265      * Note that {@code hw.nperflevels} gates only the first strategy, and does not gate the rest: it does not exist
266      * before macOS 12, so requiring it would misclassify an M1 running macOS 11 as non-hybrid.
267      *
268      * @param coreKeys       the physical core keys, which sort in core number order
269      * @param coreProperties per-core ({@code compatible}, {@code cluster-type}) values from
270      *                       {@link #queryCoreProperties()}; may be empty, and either element of any pair may be null
271      * @param perfLevelCores the value of {@code hw.perflevel<n>.physicalcpu} for each level, highest-performing level
272      *                       first. May be empty, or hold zeros, if the sysctls are unavailable.
273      * @return a map from core key to efficiency class, containing every key in {@code coreKeys}
274      */
275     static Map<Integer, Integer> deriveEfficiencyClasses(List<Integer> coreKeys,
276             Map<Integer, Pair<@Nullable String, @Nullable String>> coreProperties, int[] perfLevelCores) {
277         // Strategy 1: the per-performance-level core counts
278         Map<Integer, Integer> efficiencyMap = perfLevelClasses(coreKeys, perfLevelCores);
279         if (!efficiencyMap.isEmpty()) {
280             return efficiencyMap;
281         }
282         // Strategy 2: the cluster-type each core reports, ranked
283         Map<Integer, Integer> rankByKey = new HashMap<>();
284         Map<String, Integer> rankByCodename = new HashMap<>();
285         boolean unrankedClusterType = false;
286         for (Integer key : coreKeys) {
287             String clusterType = clusterType(coreProperties.get(key));
288             Integer rank = clusterType == null ? null : clusterTypeRank(clusterType);
289             if (rank != null) {
290                 rankByKey.put(key, rank);
291                 String codename = codename(coreProperties.get(key));
292                 if (codename != null) {
293                     rankByCodename.put(codename, rank);
294                 }
295             } else if (clusterType != null) {
296                 unrankedClusterType = true;
297             }
298         }
299         if (unrankedClusterType) {
300             // A cluster type present but outside the known ranking means this machine's letters are not the ones being
301             // ranked, which discredits the letters it did report rather than only the ones it did not. Ranking the
302             // rest would place an unknown top tier at the same class as the efficiency cores, so the codename groups
303             // below, which need no vocabulary, decide instead.
304             rankByKey.clear();
305             rankByCodename.clear();
306         }
307         // Strategy 3: propagate those readings to cores sharing a codename with a core that reported one
308         if (!rankByKey.isEmpty() && rankByKey.size() < coreKeys.size()) {
309             for (Integer key : coreKeys) {
310                 // Only the cores still unclassified: a core that reported its own cluster-type keeps that exact
311                 // reading rather than one inferred from another core sharing its codename.
312                 if (rankByKey.containsKey(key)) {
313                     continue;
314                 }
315                 String codename = codename(coreProperties.get(key));
316                 Integer rank = codename == null ? null : rankByCodename.get(codename);
317                 if (rank != null) {
318                     rankByKey.put(key, rank);
319                 }
320             }
321         }
322         // Only the cluster types actually present are ranked, so that a chip with performance and higher-performing
323         // clusters but no efficiency cluster reports classes 0 and 1 rather than 1 and 2. The efficiency class is
324         // relative within a machine, as it is on Windows.
325         Set<Integer> ranks = new TreeSet<>(rankByKey.values());
326         if (rankByKey.size() < coreKeys.size()) {
327             // Cores this strategy could not classify default to class 0 below, so that class is occupied whether or not
328             // any core reported an efficiency cluster type. Compressing without it would collapse a partial reading of
329             // one cluster type onto the very default it has to remain distinguishable from.
330             ranks.add(0);
331         }
332         List<Integer> presentRanks = new ArrayList<>(ranks);
333         for (Map.Entry<Integer, Integer> rank : rankByKey.entrySet()) {
334             efficiencyMap.put(rank.getKey(), presentRanks.indexOf(rank.getValue()));
335         }
336         if (efficiencyMap.size() == coreKeys.size()) {
337             return efficiencyMap;
338         }
339         int topPerfLevelCores = perfLevelCores.length > 0 ? perfLevelCores[0] : 0;
340         // Strategy 4: group by codename, ordered by lowest core number, cross-checked against the perf level count.
341         // Only when cluster-type classified nothing at all: where it classified some cores, those readings are exact
342         // and are kept, with strategy 5 below defaulting the remainder, rather than discarded in favor of an
343         // inference over the whole set.
344         Map<String, List<Integer>> groups = new LinkedHashMap<>();
345         if (efficiencyMap.isEmpty()) {
346             for (Integer key : coreKeys) {
347                 String codename = codename(coreProperties.get(key));
348                 if (codename == null) {
349                     // A core with no codename cannot be grouped, so this strategy cannot classify every core
350                     groups.clear();
351                     break;
352                 }
353                 List<Integer> group = groups.get(codename);
354                 if (group == null) {
355                     group = new ArrayList<>();
356                     groups.put(codename, group);
357                 }
358                 group.add(key);
359             }
360         }
361         List<List<Integer>> ordered = new ArrayList<>(groups.values());
362         // Only decisive when the first group alone matches the highest perf level's core count; an even split such as
363         // the base M1's 4+4 matches both groups and leaves the ordering to decide instead.
364         if (ordered.size() == 2 && topPerfLevelCores > 0 && ordered.get(0).size() == topPerfLevelCores
365                 && ordered.get(1).size() != topPerfLevelCores) {
366             Collections.reverse(ordered);
367         }
368         for (int i = 0; i < ordered.size(); i++) {
369             for (Integer key : ordered.get(i)) {
370                 efficiencyMap.put(key, i);
371             }
372         }
373         // Strategy 5: anything still unclassified is class 0
374         for (Integer key : coreKeys) {
375             if (!efficiencyMap.containsKey(key)) {
376                 efficiencyMap.put(key, 0);
377             }
378         }
379         return efficiencyMap;
380     }

382     /**
383      * Assigns an efficiency class to every core from the number of cores at each performance level.
384      * <p>
385      * Requires at least two levels, a positive count for each, and a total matching the number of cores. Anything else
386      * means the sysctls are absent, this is not a hybrid chip, or the values disagree with the topology, in each of
387      * which cases another strategy should decide instead of this one guessing.
388      *
389      * @param coreKeys       the physical core keys, in core number order
390      * @param perfLevelCores the core count of each performance level, highest-performing level first
391      * @return a map from core key to efficiency class, or an empty map if the counts cannot be trusted
392      */
393     private static Map<Integer, Integer> perfLevelClasses(List<Integer> coreKeys, int[] perfLevelCores) {
394         Map<Integer, Integer> efficiencyMap = new HashMap<>();
395         if (perfLevelCores.length < 2) {
396             return efficiencyMap;
397         }
398         int total = 0;
399         for (int levelCores : perfLevelCores) {
400             if (levelCores < 1) {
401                 return efficiencyMap;
402             }
403             total += levelCores;
404         }
405         if (total != coreKeys.size()) {
406             return efficiencyMap;
407         }
408         // Level 0 is the highest-performing level and holds the highest-numbered cores, so fill from the end
409         int index = coreKeys.size();
410         for (int level = 0; level < perfLevelCores.length; level++) {
411             int efficiency = perfLevelCores.length - 1 - level;
412             for (int i = 0; i < perfLevelCores[level]; i++) {
413                 efficiencyMap.put(coreKeys.get(--index), efficiency);
414             }
415         }
416         return efficiencyMap;
417     }

430     /**
431      * Ranks a core's {@code cluster-type} against the other cluster types. The value is an absolute ranking rather than
432      * an efficiency class, because a class must be dense across the types a given chip actually has.
433      *
434      * @param clusterType the core's non-empty {@code cluster-type} value
435      * @return 0 for an efficiency cluster, 1 for a performance cluster, 2 for a higher-performing still cluster, or
436      *         null if not recognized
437      */
438     private static @Nullable Integer clusterTypeRank(String clusterType) {
439         switch (Character.toUpperCase(clusterType.charAt(0))) {
440             case 'E':
441                 return 0;
442             case 'P':
443                 return 1;
444             case 'S':
445                 return 2;
446             default:
447                 return null;
448         }
449     }

453      *
454      * @param properties the core's properties, possibly null
455      * @return the codename, e.g. {@code avalanche}, or null if absent

457     private static @Nullable String codename(@Nullable Pair<@Nullable String, @Nullable String> properties) {
458         String compatible = properties == null ? null : properties.getA();
459         if (compatible == null) {

463         return m.find() ? m.group(1) : null;

466     /**
467      * Derives a microarchitecture description from the core codenames, highest-performing first, for example
468      * {@code ARM64 SoC: Avalanche + Blizzard}. This lets a chip absent from the architecture table still be described,
469      * rather than reported as unknown.
470      *
471      * @param physicalProcessors the physical processors, whose ID strings hold the IORegistry {@code compatible} value
472      * @return the description, or null if no Apple core codename was found
473      */
474     static @Nullable String deriveMicroarchitecture(List<PhysicalProcessor> physicalProcessors) {
475         // Sorting keys, ordering distinct codenames by descending efficiency class and then by core number
476         Map<String, Pair<Integer, Integer>> codenames = new LinkedHashMap<>();
477         for (PhysicalProcessor processor : physicalProcessors) {
478             Matcher m = APPLE_CORE.matcher(processor.getIdString().toLowerCase(Locale.ROOT));
479             if (m.find() && !codenames.containsKey(m.group(1))) {
480                 codenames.put(m.group(1),
481                         new Pair<>(processor.getEfficiency(), processor.getPhysicalProcessorNumber()));
482             }
483         }
484         if (codenames.isEmpty()) {
485             return null;
486         }
487         List<Map.Entry<String, Pair<Integer, Integer>>> entries = new ArrayList<>(codenames.entrySet());
488         Collections.sort(entries, (a, b) -> {
489             int byEfficiency = b.getValue().getA().compareTo(a.getValue().getA());
490             return byEfficiency == 0 ? a.getValue().getB().compareTo(b.getValue().getB()) : byEfficiency;
491         });
492         StringBuilder sb = new StringBuilder(MICROARCH_PREFIX);
493         for (int i = 0; i < entries.size(); i++) {
494             String codename = entries.get(i).getKey();
495             if (i > 0) {
496                 sb.append(" + ");
497             }
498             sb.append(codename.substring(0, 1).toUpperCase(Locale.ROOT)).append(codename.substring(1));
499         }
500         return sb.toString();
501     }

516         return byClass;

544     /**
545      * Reads the complete voltage state table of every CPU cluster from the power manager.
546      * <p>
547      * Which tables hold the CPU clusters is chip-dependent. Since the M5 the power manager lists them in its
548      * {@code acc-clusters} property; before that the numbering was fixed, so those two tables are the fallback.
549      * <p>
550      * Clusters of the same core type publish identical tables, so there are usually fewer tables returned than the chip
551      * has clusters.
552      *
553      * @return the distinct cluster frequency tables in Hz, each in ascending order and ordered by their maxima, or an
554      *         empty array if none could be read. No returned table is empty.
555      */
556     protected long[][] queryClusterFrequencyTables() {
557         List<long[]> tables = new ArrayList<>();
558         ioKitProvider().forEachMatchingServiceUntil("AppleARMIODevice", entry -> {
559             if ("pmgr".equalsIgnoreCase(entry.getName())) {
560                 for (int table : parseClusterTables(entry.getByteArrayProperty(ACC_CLUSTERS))) {
561                     addTable(tables, entry, "voltage-states" + table + "-sram");
562                 }
563                 if (tables.isEmpty()) {
564                     addTable(tables, entry, LEGACY_EFFICIENCY_TABLE);
565                     addTable(tables, entry, LEGACY_PERFORMANCE_TABLE);
566                 }
567                 return true;
568             }
569             return false;
570         });
571         Collections.sort(tables, (a, b) -> Long.compare(a[a.length - 1], b[b.length - 1]));
572         List<long[]> distinct = new ArrayList<>(tables.size());
573         for (long[] table : tables) {
574             long[] previous = distinct.isEmpty() ? null : distinct.get(distinct.size() - 1);
575             if (previous == null || previous[previous.length - 1] != table[table.length - 1]) {
576                 distinct.add(table);
577             }
578         }
579         return distinct.toArray(new long[0][]);
580     }

582     /**
583      * Reads one voltage state table and adds it to a list, ignoring a table that is absent, too short to hold an entry,
584      * or whose maximum is not a positive frequency.
585      *
586      * @param tables the list to add to
587      * @param entry  the power manager registry entry
588      * @param key    the voltage state table property name
589      */
590     private void addTable(List<long[]> tables, RegistryEntry entry, String key) {
591         long[] table = parseFrequencyTable(entry.getByteArrayProperty(key));
592         if (table.length > 0 && table[table.length - 1] > 0) {
593             // A zero-filled table is long enough to parse but says nothing, so leave it out rather than report a
594             // cluster running at 0 Hz. Dropping it also keeps the list empty, so the legacy tables are still read.
595             tables.add(table);
596         }
597     }

599     /**
600      * Parses a voltage state table property. Each entry pairs a frequency with the voltage needed to reach it, in
601      * ascending order.
602      *
603      * @param data the raw property value, or null if the property is absent
604      * @return every frequency in the table in Hz, in ascending order, or an empty array if the property is absent or
605      *         too short to hold an entry
606      */
607     static long[] parseFrequencyTable(byte @Nullable [] data) {
608         if (data == null || data.length < ENTRY_SIZE) {
609             return new long[0];
610         }
611         long[] frequencies = new long[data.length / ENTRY_SIZE];
612         for (int i = 0; i < frequencies.length; i++) {
613             frequencies[i] = toHz(
614                     ParseUtil.byteArrayToLong(Arrays.copyOfRange(data, i * ENTRY_SIZE, i * ENTRY_SIZE + 4), 4, false));
615         }
616         return frequencies;
617     }

619     /**
620      * Parses the power manager's {@code acc-clusters} property, which lists the CPU clusters. Each entry holds the
621      * number of the cluster's {@code voltage-states} table in its first byte and the cluster's tier in its second, so
622      * an M5 Max reporting {@code 16 00 …, 17 01 …, 05 02 …} has three clusters described by
623      * {@code voltage-states22-sram}, {@code voltage-states23-sram} and {@code voltage-states5-sram}.
624      *
625      * @param data the raw property value, or null on a chip that does not publish it
626      * @return the voltage state table numbers, in ascending tier order, or an empty array if the property is absent or
627      *         too short to hold an entry
628      */
629     static int[] parseClusterTables(byte @Nullable [] data) {
630         if (data == null || data.length < ENTRY_SIZE) {
631             return new int[0];
632         }
633         int clusters = data.length / ENTRY_SIZE;
634         int[] tables = new int[clusters];
635         Integer[] order = new Integer[clusters];
636         for (int i = 0; i < clusters; i++) {
637             order[i] = i;
638         }
639         // Sorted by tier rather than assuming the entries are already in tier order, and stably, so that two clusters
640         // reporting the same tier keep their listed order
641         Arrays.sort(order, (a, b) -> Integer.compare(data[a * ENTRY_SIZE + 1] & 0xff, data[b * ENTRY_SIZE + 1] & 0xff));
642         for (int i = 0; i < clusters; i++) {
643             tables[i] = data[order[i] * ENTRY_SIZE] & 0xff;
644         }
645         return tables;
646     }

648     /**
649      * Distributes the cluster frequencies over the efficiency classes. The two lists are aligned at the top, so the
650      * highest class always reports the highest frequency the chip publishes, and a class with no frequency of its own
651      * borrows the nearest one rather than reporting a placeholder.
652      *
653      * @param clusterFrequencies the distinct cluster frequencies in Hz, in ascending order
654      * @param classCount         the number of efficiency classes, at least 1
655      * @return the frequency in Hz of each efficiency class, indexed by class
656      */
657     static long[] mapClusterFrequencies(long[] clusterFrequencies, int classCount) {
658         int[] indices = CpuFrequencyResidency.alignAtTop(clusterFrequencies.length, classCount);
659         if (indices.length == 0) {
660             long[] byClass = new long[Math.max(classCount, 1)];
661             Arrays.fill(byClass, DEFAULT_FREQUENCY);
662             return byClass;
663         }
664         long[] byClass = new long[indices.length];
665         for (int i = 0; i < indices.length; i++) {
666             byClass[i] = clusterFrequencies[indices[i]];
667         }
668         return byClass;
669     }

671     /**
672      * Distributes the cluster frequency tables over the efficiency classes, aligned at the top exactly as
673      * {@link #mapClusterFrequencies} aligns their maxima, so that a class reports the table whose maximum it reports as
674      * its nominal frequency.
675      *
676      * @param clusterTables the distinct cluster frequency tables, ordered by their maxima
677      * @param classCount    the number of efficiency classes, at least 1
678      * @return the frequency table of each efficiency class, indexed by class. Every element is empty if no table could
679      *         be read.
680      */
681     static long[][] mapClusterTables(long[][] clusterTables, int classCount) {
682         int[] indices = CpuFrequencyResidency.alignAtTop(clusterTables.length, classCount);
683         if (indices.length == 0) {
684             long[][] byClass = new long[Math.max(classCount, 1)][];
685             Arrays.fill(byClass, new long[0]);
686             return byClass;
687         }
688         long[][] byClass = new long[indices.length][];
689         for (int i = 0; i < indices.length; i++) {
690             byClass[i] = clusterTables[indices[i]];
691         }
692         return byClass;
693     }

728     protected ProcessorIdentifier queryProcessorId() {
729         String cpuName = sysctlString("machdep.cpu.brand_string", "");
730         String cpuVendor;
731         String cpuStepping;
732         String cpuModel;
733         String cpuFamily;
734         String processorID;
735         String microarchitecture = null;
736         if (cpuName.startsWith("Apple")) {
737             cpuVendor = vendor.get();
738             cpuStepping = "0";
739             cpuModel = "0";
740             int type;
741             int family;
742             if (isArmCpu) {
743                 type = ARM_CPUTYPE;
744                 // No fallback when this is absent: guessing a family from the brand string reported an unrecognized
745                 // chip as an M1. A zero family simply misses the architecture table, and the derived
746                 // microarchitecture below covers that case instead.
747                 family = sysctlInt("hw.cpufamily", 0);
748                 microarchitecture = deriveMicroarchitecture(getPhysicalProcessors());
749             } else {
750                 type = sysctlInt("hw.cputype", 0);
751                 family = sysctlInt("hw.cpufamily", 0);
752             }
753             cpuFamily = String.format(Locale.ROOT, "0x%08x", family);
754             processorID = String.format(Locale.ROOT, "%08x%08x", type, family);
755         } else {
756             cpuVendor = sysctlString("machdep.cpu.vendor", "");
757             int i = sysctlInt("machdep.cpu.stepping", -1);
758             cpuStepping = i < 0 ? "" : Integer.toString(i);
759             i = sysctlInt("machdep.cpu.model", -1);
760             cpuModel = i < 0 ? "" : Integer.toString(i);
761             i = sysctlInt("machdep.cpu.family", -1);
762             cpuFamily = i < 0 ? "" : Integer.toString(i);
763             long processorIdBits = 0L;
764             processorIdBits |= sysctlInt("machdep.cpu.signature", 0);
765             processorIdBits |= (sysctlLong("machdep.cpu.feature_bits", 0L) & 0xffffffff) << 32;
766             processorID = String.format(Locale.ROOT, "%016x", processorIdBits);
767         }
768         long cpuFreq = isArmCpu ? getPerformanceCoreFrequency() : sysctlLong("hw.cpufrequency", 0L);
769         boolean cpu64bit = sysctlInt("hw.cpu64bit_capable", 0) != 0;
770 
771         return new ProcessorIdentifier(cpuVendor, cpuName, cpuFamily, cpuModel, cpuStepping, processorID, cpu64bit,
772                 cpuFreq, microarchitecture);
773     }

775     @Override
776     protected Quartet<List<LogicalProcessor>, @Nullable List<PhysicalProcessor>, @Nullable List<ProcessorCache>, List<String>> initProcessorCounts() {
777         int logicalProcessorCount = sysctlInt("hw.logicalcpu", 1);
778         int physicalProcessorCount = sysctlInt("hw.physicalcpu", 1);
779         int physicalPackageCount = sysctlInt("hw.packages", 1);
780         List<LogicalProcessor> logProcs = new ArrayList<>(logicalProcessorCount);
781         Set<Integer> pkgCoreKeys = new HashSet<>();
782         for (int i = 0; i < logicalProcessorCount; i++) {
783             int coreId = i * physicalProcessorCount / logicalProcessorCount;
784             int pkgId = i * physicalPackageCount / logicalProcessorCount;
785             logProcs.add(new LogicalProcessor(i, coreId, pkgId));
786             pkgCoreKeys.add((pkgId << 16) + coreId);
787         }
788         Map<Integer, Pair<@Nullable String, @Nullable String>> coreProps = queryCoreProperties();
789         int perflevels = Math.max(1, sysctlIntNoWarn("hw.nperflevels", 1));
790         int[] perfLevelCores = new int[perflevels];
791         for (int i = 0; i < perflevels; i++) {
792             perfLevelCores[i] = sysctlIntNoWarn("hw.perflevel" + i + ".physicalcpu", 0);
793         }
794         List<Integer> coreKeys = pkgCoreKeys.stream().sorted().collect(Collectors.toList());
795         Map<Integer, Integer> efficiencyMap = deriveEfficiencyClasses(coreKeys, coreProps, perfLevelCores);
796         List<PhysicalProcessor> physProcs = new ArrayList<>(coreKeys.size());
797         for (Integer k : coreKeys) {
798             Pair<@Nullable String, @Nullable String> props = coreProps.get(k);
799             String compat = props == null || props.getA() == null ? "" : props.getA().toLowerCase(Locale.ROOT);
800             physProcs.add(new PhysicalProcessor(k >> 16, k & 0xffff, efficiencyMap.getOrDefault(k, 0), compat));
801         }
802         List<ProcessorCache> caches = orderedProcCaches(getCacheValues(perflevels));
803         List<String> featureFlags = getFeatureFlagsFromSysctl();
804         return new Quartet<>(logProcs, physProcs, caches, featureFlags);
805     }

807     private Set<ProcessorCache> getCacheValues(int perflevels) {
808         int linesize = (int) sysctlLong("hw.cachelinesize", 0L);
809         int l1associativity = sysctlIntNoWarn("machdep.cpu.cache.L1_associativity", 0);
810         int l2associativity = sysctlIntNoWarn("machdep.cpu.cache.L2_associativity", 0);
811         Set<ProcessorCache> caches = new HashSet<>();
812         for (int i = 0; i < perflevels; i++) {
813             int size = sysctlIntNoWarn("hw.perflevel" + i + ".l1icachesize", 0);
814             if (size > 0) {
815                 caches.add(new ProcessorCache(1, l1associativity, linesize, size, ProcessorCache.Type.INSTRUCTION));
816             }
817             size = sysctlIntNoWarn("hw.perflevel" + i + ".l1dcachesize", 0);
818             if (size > 0) {
819                 caches.add(new ProcessorCache(1, l1associativity, linesize, size, ProcessorCache.Type.DATA));
820             }
821             size = sysctlIntNoWarn("hw.perflevel" + i + ".l2cachesize", 0);
822             if (size > 0) {
823                 caches.add(new ProcessorCache(2, l2associativity, linesize, size, ProcessorCache.Type.UNIFIED));
824             }
825             size = sysctlIntNoWarn("hw.perflevel" + i + ".l3cachesize", 0);
826             if (size > 0) {
827                 caches.add(new ProcessorCache(3, 0, linesize, size, ProcessorCache.Type.UNIFIED));
828             }
829         }
830         return caches;
831     }

846     @Override
847     public long[] queryCurrentFreq() {
848         if (isArmCpu) {
849             long[] byClass = nominalFrequencies.get();
850             long topFreq = byClass[byClass.length - 1];
851             Map<Integer, Long> physFreqMap = new HashMap<>();
852             // A class outside the array cannot happen, since the array was sized from the classes the cores report,
853             // but clamping keeps a future mismatch from becoming an exception on a path callers poll
854             getPhysicalProcessors().forEach(p -> physFreqMap.put(p.getPhysicalProcessorNumber(),
855                     byClass[Math.min(Math.max(p.getEfficiency(), 0), byClass.length - 1)]));
856             applyLiveFrequencies(physFreqMap);
857             return getLogicalProcessors().stream().map(LogicalProcessor::getPhysicalProcessorNumber)
858                     .map(p -> physFreqMap.getOrDefault(p, topFreq)).mapToLong(f -> f).toArray();
859         }
860         return new long[] { getProcessorIdentifier().getVendorFreq() };
861     }

863     /**
864      * Replaces nominal core frequencies with the frequency each core actually ran at since the previous call, where
865      * IOReport can supply it. A core whose residency cannot be paired with a frequency table keeps its nominal
866      * frequency, as do all of them if no sample is available or if the channels cannot be matched to the cores.
867      *
868      * @param physFreqMap the frequency in Hz of each physical processor, updated in place
869      */
870     private void applyLiveFrequencies(Map<Integer, Long> physFreqMap) {
871         IOReportCpuSampler sampler = cpuSampler.get();
872         if (sampler == null) {
873             return;
874         }
875         CpuResidencySample sample = sampler.sampleResidencyDelta();
876         if (sample == null) {
877             // No previous sample to subtract from, as on the first call
878             return;
879         }
880         Map<String, Map<String, Long>> residency = sample.getCoreStates();
881         if (residency.isEmpty()) {
882             return;
883         }
884         // Both lists run in ascending performance order, so they line up only if they describe the same chip. Any
885         // disagreement leaves the nominal frequencies in place rather than risk charging a core with another cluster's
886         // frequency table, which would report a plausible but wrong number.
887         Map<Integer, List<String>> channelGroups = groupChannelsByCoreType(residency.keySet());
888         Map<Integer, List<PhysicalProcessor>> coreGroups = groupCoresByEfficiencyClass();
889         if (channelGroups.containsKey(CpuFrequencyResidency.UNKNOWN_RANK)) {
890             // A core type this release does not know. Where it belongs among the ones it does know is a guess, and
891             // guessing wrong charges a core with another cluster's frequency table.
892             LOG.debug("IOReport reports a CPU core type this release does not recognize: {}",
893                     channelGroups.get(CpuFrequencyResidency.UNKNOWN_RANK));
894             return;
895         }
896         if (channelGroups.size() != coreGroups.size()) {
897             LOG.debug("IOReport reports {} core types but this processor has {} efficiency classes.",
898                     channelGroups.size(), coreGroups.size());
899             return;
900         }
901         Iterator<List<String>> channelIterator = channelGroups.values().iterator();
902         for (Map.Entry<Integer, List<PhysicalProcessor>> cores : coreGroups.entrySet()) {
903             List<String> channels = channelIterator.next();
904             if (channels.size() != cores.getValue().size()) {
905                 LOG.debug("IOReport reports {} cores of one type but efficiency class {} has {}.", channels.size(),
906                         cores.getKey(), cores.getValue().size());
907                 return;
908             }
909         }
910         // The state each cluster's hardware actually ran at, which is what any core in it ran at while it was running.
911         // Only usable if the clusters reported are the core types reported, since otherwise which table weights which
912         // cluster is a guess.
913         Map<Integer, Map<String, Long>> complexByRank = CpuFrequencyResidency
914                 .realizedComplexStates(sample.getComplexStates());
915         boolean complexMatchesCores = complexByRank.keySet().equals(channelGroups.keySet());
916         long[][] tables = nominalFrequencyTables.get();
917         Iterator<Map.Entry<Integer, List<String>>> channelEntries = channelGroups.entrySet().iterator();
918         for (Map.Entry<Integer, List<PhysicalProcessor>> cores : coreGroups.entrySet()) {
919             Map.Entry<Integer, List<String>> channelEntry = channelEntries.next();
920             List<String> channels = channelEntry.getValue();
921             long[] table = tables[Math.min(Math.max(cores.getKey(), 0), tables.length - 1)];
922             Map<String, Long> cluster = complexMatchesCores ? complexByRank.get(channelEntry.getKey()) : null;
923             long realized = cluster == null ? 0L : CpuFrequencyResidency.activeWeightedFrequency(cluster, table);
924             for (int i = 0; i < channels.size(); i++) {
925                 Map<String, Long> states = residency.get(channels.get(i));
926                 long frequency = states == null ? 0L : CpuFrequencyResidency.activeWeightedFrequency(states, table);
927                 if (frequency == 0L) {
928                     // Nothing was observed, or the core's states are not a list this table can be paired with
929                     continue;
930                 }
931                 if (realized > 0 && frequency != table[0]) {
932                     // A core's own residency names the state it asked for, which is the fastest one whenever it has
933                     // work, so under a power or thermal cap it reads high by as much as the cap. Where the cluster can
934                     // be read it says what the core got instead, and only a core that did not run at all keeps the
935                     // lowest frequency its cluster can run at rather than being credited with its siblings' work.
936                     frequency = realized;
937                 }
938                 physFreqMap.put(cores.getValue().get(i).getPhysicalProcessorNumber(), frequency);
939             }
940         }
941     }

943     /**
944      * Groups the IOReport channel names by the core type they report, in ascending performance order.
945      *
946      * @param channelNames the channel names sampled
947      * @return the channel names of each core type, keyed by that type's rank, with the names of one type in core order
948      */
949     private static Map<Integer, List<String>> groupChannelsByCoreType(Collection<String> channelNames) {
950         Map<Integer, List<String>> groups = new TreeMap<>();
951         for (String channel : CpuFrequencyResidency.orderChannels(channelNames)) {
952             int rank = CpuFrequencyResidency.prefixRank(channel);
953             groups.computeIfAbsent(rank, k -> new ArrayList<>()).add(channel);
954         }

961      * @return the cores of each class, keyed by class, with the cores of one class in processor number order. A class
962      *         with no cores is absent.

1086         int[] cpuTicks = queryProcessorCpuTicks();
1087         int procCount = cpuTicks.length / CPU_STATE_MAX;
1088         if (procCount > ticks.length) {
1089             LOG.warn("processor_cpu_load_info returned {} CPUs but expected {}; capping iteration", procCount,
1090                     ticks.length);
1091         }
1092         int cpuLimit = Math.min(procCount, ticks.length);
1093         for (int cpu = 0; cpu < cpuLimit; cpu++) {
1094             int offset = cpu * CPU_STATE_MAX;
1095             ticks[cpu][TickType.USER.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_USER]);
1096             ticks[cpu][TickType.NICE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_NICE]);
1097             ticks[cpu][TickType.SYSTEM.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_SYSTEM]);
1098             ticks[cpu][TickType.IDLE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_IDLE]);
1099         }
1100         return ticks;
1101     }

1103     @Override
1104     public double[] getSystemLoadAverage(int nelem) {
1105         if (nelem < 1 || nelem > 3) {
1106             throw new IllegalArgumentException("Must include from one to three elements.");
1107         }
1108         double[] average = new double[nelem];
1109         int retval = getloadavgNative(average, nelem);
1110         if (retval < nelem) {
1111             Arrays.fill(average, -1d);
1112         }
1113         return average;
1114     }
