163     /**
164      * Queries the per-core IORegistry properties used to identify the CPU. Walks the {@code IOPlatformDevice} nodes
165      * named {@code cpuN} once, reading both the {@code compatible} property (which names the core's microarchitecture,
166      * e.g. {@code apple,avalanche arm,v8}) and the {@code cluster-type} property (a single letter, {@code P} for a
167      * performance cluster or {@code E} for an efficiency cluster).
168      *
169      * @return a map of physical processor numbers to a pair of ({@code compatible}, {@code cluster-type}). Either
170      *         element may be {@code null} if that property is absent from the node.
171      */
172     protected Map<Integer, Pair<@Nullable String, @Nullable String>> queryCoreProperties() {
173         Map<Integer, Pair<@Nullable String, @Nullable String>> coreProperties = new HashMap<>();
174         ioKitProvider().forEachMatchingService("IOPlatformDevice", entry -> {
175             String name = entry.getName();
176             if (name != null) {
177                 Matcher m = CPU_N.matcher(name.toLowerCase(Locale.ROOT));
178                 if (m.matches()) {
179                     int procId = ParseUtil.parseIntOrDefault(m.group(1), 0);
180                     coreProperties.put(procId, new Pair<>(ioRegString(entry.getByteArrayProperty("compatible")),
181                             ioRegString(entry.getByteArrayProperty("cluster-type"))));
182                 }
183             }
184         });
185         return coreProperties;
186     }

199     /**
200      * Derives a performance/efficiency class for each physical core, so that hybrid Apple Silicon cores are classified
201      * without a hardcoded table of core codenames. A higher value indicates a higher-performing core, matching
202      * {@link PhysicalProcessor#getEfficiency()}.
203      * <p>
204      * The first of these strategies that classifies every core wins:
205      * <ol>
206      * <li>The {@code cluster-type} property read from each core: {@code P} is a performance cluster, {@code E} an
207      * efficiency cluster. This is chip-independent and exact.</li>
208      * <li>If only some cores report a recognized {@code cluster-type}, the mapping learned from those cores is applied
209      * to the other cores sharing the same codename.</li>
210      * <li>Grouping cores by codename and ordering the groups by their lowest core number, since macOS numbers
211      * efficiency cores first. Where there are exactly two groups and {@code hw.perflevel0.physicalcpu} matches the size
212      * of exactly one of them, that count decides which group is the performance group, overriding the ordering. This
213      * applies only when {@code cluster-type} classified no core at all: a {@code cluster-type} reading is exact, so
214      * where some cores reported one those values are kept and only the remainder falls to the default below.</li>
215      * <li>Failing all of those, every core is class 0, which is the historical behavior for a chip whose cores were not
216      * recognized, and for non-hybrid and Intel processors.</li>
217      * </ol>
218      * Note that {@code hw.nperflevels} is deliberately neither an input nor a gate here: it does not exist before macOS
219      * 12, so requiring it would misclassify an M1 running macOS 11 as non-hybrid.
220      *
221      * @param coreKeys          the physical core keys, which sort in core number order
222      * @param coreProperties    per-core ({@code compatible}, {@code cluster-type}) values from
223      *                          {@link #queryCoreProperties()}; may be empty, and either element of any pair may be null
224      * @param topPerfLevelCores the value of {@code hw.perflevel0.physicalcpu}, or 0 if unknown. Perf level 0 is always
225      *                          the highest-performing level.
226      * @return a map from core key to efficiency class, containing every key in {@code coreKeys}
227      */
228     static Map<Integer, Integer> deriveEfficiencyClasses(List<Integer> coreKeys,
229             Map<Integer, Pair<@Nullable String, @Nullable String>> coreProperties, int topPerfLevelCores) {
230         Map<Integer, Integer> efficiencyMap = new HashMap<>();
231         // Strategy 1 and 2: cluster-type, directly and then propagated to cores sharing a codename
232         Map<String, Integer> classByCodename = new HashMap<>();
233         for (Integer key : coreKeys) {
234             Integer efficiency = clusterTypeClass(coreProperties.get(key));
235             if (efficiency != null) {
236                 efficiencyMap.put(key, efficiency);
237                 String codename = codename(coreProperties.get(key));
238                 if (codename != null) {
239                     classByCodename.put(codename, efficiency);
240                 }
241             }
242         }
243         if (!efficiencyMap.isEmpty() && efficiencyMap.size() < coreKeys.size()) {
244             for (Integer key : coreKeys) {
245                 // Only the cores still unclassified: a core that reported its own cluster-type keeps that exact
246                 // reading rather than one inferred from another core sharing its codename.
247                 if (efficiencyMap.containsKey(key)) {
248                     continue;
249                 }
250                 String codename = codename(coreProperties.get(key));
251                 Integer efficiency = codename == null ? null : classByCodename.get(codename);
252                 if (efficiency != null) {
253                     efficiencyMap.put(key, efficiency);
254                 }
255             }
256         }
257         if (efficiencyMap.size() == coreKeys.size()) {
258             return efficiencyMap;
259         }
260         // Strategy 3: group by codename, ordered by lowest core number, cross-checked against the perf level count.
261         // Only when cluster-type classified nothing at all: where it classified some cores, those readings are exact
262         // and are kept, with strategy 4 below defaulting the remainder, rather than discarded in favor of an
263         // inference over the whole set.
264         Map<String, List<Integer>> groups = new LinkedHashMap<>();
265         if (efficiencyMap.isEmpty()) {
266             for (Integer key : coreKeys) {
267                 String codename = codename(coreProperties.get(key));
268                 if (codename == null) {
269                     // A core with no codename cannot be grouped, so this strategy cannot classify every core
270                     groups.clear();
271                     break;
272                 }
273                 List<Integer> group = groups.get(codename);
274                 if (group == null) {
275                     group = new ArrayList<>();
276                     groups.put(codename, group);
277                 }
278                 group.add(key);
279             }
280         }
281         List<List<Integer>> ordered = new ArrayList<>(groups.values());
282         // Only decisive when the first group alone matches the highest perf level's core count; an even split such as
283         // the base M1's 4+4 matches both groups and leaves the ordering to decide instead.
284         if (ordered.size() == 2 && topPerfLevelCores > 0 && ordered.get(0).size() == topPerfLevelCores
285                 && ordered.get(1).size() != topPerfLevelCores) {
286             Collections.reverse(ordered);
287         }
288         for (int i = 0; i < ordered.size(); i++) {
289             for (Integer key : ordered.get(i)) {
290                 efficiencyMap.put(key, i);
291             }
292         }
293         // Strategy 4: anything still unclassified is class 0
294         for (Integer key : coreKeys) {
295             if (!efficiencyMap.containsKey(key)) {
296                 efficiencyMap.put(key, 0);
297             }
298         }
299         return efficiencyMap;
300     }

302     /**
303      * Maps a core's {@code cluster-type} to an efficiency class.
304      *
305      * @param properties the core's properties, possibly null
306      * @return 1 for a performance cluster, 0 for an efficiency cluster, or null if not recognized
307      */
308     private static @Nullable Integer clusterTypeClass(@Nullable Pair<@Nullable String, @Nullable String> properties) {
309         String clusterType = properties == null ? null : properties.getB();
310         if (clusterType == null || clusterType.isEmpty()) {
311             return null;
312         }
313         char c = Character.toUpperCase(clusterType.charAt(0));
314         if (c == 'P') {
315             return 1;
316         }
317         return c == 'E' ? 0 : null;
318     }

320     /**
321      * Extracts the Apple core codename from a core's {@code compatible} string.
322      *
323      * @param properties the core's properties, possibly null
324      * @return the codename, e.g. {@code avalanche}, or null if absent
325      */
326     private static @Nullable String codename(@Nullable Pair<@Nullable String, @Nullable String> properties) {
327         String compatible = properties == null ? null : properties.getA();
328         if (compatible == null) {
329             return null;
330         }
331         Matcher m = APPLE_CORE.matcher(compatible.toLowerCase(Locale.ROOT));
332         return m.find() ? m.group(1) : null;
333     }

335     /**
336      * Derives a microarchitecture description from the core codenames, highest-performing first, for example
337      * {@code ARM64 SoC: Avalanche + Blizzard}. This lets a chip absent from the architecture table still be described,
338      * rather than reported as unknown.
339      *
340      * @param physicalProcessors the physical processors, whose ID strings hold the IORegistry {@code compatible} value
341      * @return the description, or null if no Apple core codename was found
342      */
343     static @Nullable String deriveMicroarchitecture(List<PhysicalProcessor> physicalProcessors) {
344         // Sorting keys, ordering distinct codenames by descending efficiency class and then by core number
345         Map<String, Pair<Integer, Integer>> codenames = new LinkedHashMap<>();
346         for (PhysicalProcessor processor : physicalProcessors) {
347             Matcher m = APPLE_CORE.matcher(processor.getIdString().toLowerCase(Locale.ROOT));
348             if (m.find() && !codenames.containsKey(m.group(1))) {
349                 codenames.put(m.group(1),
350                         new Pair<>(processor.getEfficiency(), processor.getPhysicalProcessorNumber()));
351             }
352         }
353         if (codenames.isEmpty()) {
354             return null;
355         }
356         List<Map.Entry<String, Pair<Integer, Integer>>> entries = new ArrayList<>(codenames.entrySet());
357         Collections.sort(entries, (a, b) -> {
358             int byEfficiency = b.getValue().getA().compareTo(a.getValue().getA());
359             return byEfficiency == 0 ? a.getValue().getB().compareTo(b.getValue().getB()) : byEfficiency;
360         });
361         StringBuilder sb = new StringBuilder(MICROARCH_PREFIX);
362         for (int i = 0; i < entries.size(); i++) {
363             String codename = entries.get(i).getKey();
364             if (i > 0) {
365                 sb.append(" + ");
366             }
367             sb.append(codename.substring(0, 1).toUpperCase(Locale.ROOT)).append(codename.substring(1));
368         }
369         return sb.toString();
370     }

432     @Override
433     protected ProcessorIdentifier queryProcessorId() {
434         String cpuName = sysctlString("machdep.cpu.brand_string", "");
435         String cpuVendor;
436         String cpuStepping;
437         String cpuModel;
438         String cpuFamily;
439         String processorID;
440         String microarchitecture = null;
441         if (cpuName.startsWith("Apple")) {
442             cpuVendor = vendor.get();
443             cpuStepping = "0";
444             cpuModel = "0";
445             int type;
446             int family;
447             if (isArmCpu) {
448                 type = ARM_CPUTYPE;
449                 // No fallback when this is absent: guessing a family from the brand string reported an unrecognized
450                 // chip as an M1. A zero family simply misses the architecture table, and the derived
451                 // microarchitecture below covers that case instead.
452                 family = sysctlInt("hw.cpufamily", 0);
453                 microarchitecture = deriveMicroarchitecture(getPhysicalProcessors());
454             } else {
455                 type = sysctlInt("hw.cputype", 0);
456                 family = sysctlInt("hw.cpufamily", 0);
457             }
458             cpuFamily = String.format(Locale.ROOT, "0x%08x", family);
459             processorID = String.format(Locale.ROOT, "%08x%08x", type, family);
460         } else {
461             cpuVendor = sysctlString("machdep.cpu.vendor", "");
462             int i = sysctlInt("machdep.cpu.stepping", -1);
463             cpuStepping = i < 0 ? "" : Integer.toString(i);
464             i = sysctlInt("machdep.cpu.model", -1);
465             cpuModel = i < 0 ? "" : Integer.toString(i);
466             i = sysctlInt("machdep.cpu.family", -1);
467             cpuFamily = i < 0 ? "" : Integer.toString(i);
468             long processorIdBits = 0L;
469             processorIdBits |= sysctlInt("machdep.cpu.signature", 0);
470             processorIdBits |= (sysctlLong("machdep.cpu.feature_bits", 0L) & 0xffffffff) << 32;
471             processorID = String.format(Locale.ROOT, "%016x", processorIdBits);
472         }
473         if (isArmCpu) {
474             calculateNominalFrequencies();
475         }
476         long cpuFreq = isArmCpu ? performanceCoreFrequency : sysctlLong("hw.cpufrequency", 0L);
477         boolean cpu64bit = sysctlInt("hw.cpu64bit_capable", 0) != 0;
478 
479         return new ProcessorIdentifier(cpuVendor, cpuName, cpuFamily, cpuModel, cpuStepping, processorID, cpu64bit,
480                 cpuFreq, microarchitecture);
481     }

483     @Override
484     protected Quartet<List<LogicalProcessor>, @Nullable List<PhysicalProcessor>, @Nullable List<ProcessorCache>, List<String>> initProcessorCounts() {
485         int logicalProcessorCount = sysctlInt("hw.logicalcpu", 1);
486         int physicalProcessorCount = sysctlInt("hw.physicalcpu", 1);
487         int physicalPackageCount = sysctlInt("hw.packages", 1);
488         List<LogicalProcessor> logProcs = new ArrayList<>(logicalProcessorCount);
489         Set<Integer> pkgCoreKeys = new HashSet<>();
490         for (int i = 0; i < logicalProcessorCount; i++) {
491             int coreId = i * physicalProcessorCount / logicalProcessorCount;
492             int pkgId = i * physicalPackageCount / logicalProcessorCount;
493             logProcs.add(new LogicalProcessor(i, coreId, pkgId));
494             pkgCoreKeys.add((pkgId << 16) + coreId);
495         }
496         Map<Integer, Pair<@Nullable String, @Nullable String>> coreProps = queryCoreProperties();
497         int perflevels = sysctlIntNoWarn("hw.nperflevels", 1);
498         int topPerfLevelCores = sysctlIntNoWarn("hw.perflevel0.physicalcpu", 0);
499         List<Integer> coreKeys = pkgCoreKeys.stream().sorted().collect(Collectors.toList());
500         Map<Integer, Integer> efficiencyMap = deriveEfficiencyClasses(coreKeys, coreProps, topPerfLevelCores);
501         List<PhysicalProcessor> physProcs = new ArrayList<>(coreKeys.size());
502         for (Integer k : coreKeys) {
503             Pair<@Nullable String, @Nullable String> props = coreProps.get(k);
504             String compat = props == null || props.getA() == null ? "" : props.getA().toLowerCase(Locale.ROOT);
505             physProcs.add(new PhysicalProcessor(k >> 16, k & 0xffff, efficiencyMap.getOrDefault(k, 0), compat));
506         }
507         List<ProcessorCache> caches = orderedProcCaches(getCacheValues(perflevels));
508         List<String> featureFlags = getFeatureFlagsFromSysctl();
509         return new Quartet<>(logProcs, physProcs, caches, featureFlags);
510     }

512     private Set<ProcessorCache> getCacheValues(int perflevels) {
513         int linesize = (int) sysctlLong("hw.cachelinesize", 0L);
514         int l1associativity = sysctlIntNoWarn("machdep.cpu.cache.L1_associativity", 0);
515         int l2associativity = sysctlIntNoWarn("machdep.cpu.cache.L2_associativity", 0);
516         Set<ProcessorCache> caches = new HashSet<>();
517         for (int i = 0; i < perflevels; i++) {
518             int size = sysctlIntNoWarn("hw.perflevel" + i + ".l1icachesize", 0);
519             if (size > 0) {
520                 caches.add(new ProcessorCache(1, l1associativity, linesize, size, ProcessorCache.Type.INSTRUCTION));
521             }
522             size = sysctlIntNoWarn("hw.perflevel" + i + ".l1dcachesize", 0);
523             if (size > 0) {
524                 caches.add(new ProcessorCache(1, l1associativity, linesize, size, ProcessorCache.Type.DATA));
525             }
526             size = sysctlIntNoWarn("hw.perflevel" + i + ".l2cachesize", 0);
527             if (size > 0) {
528                 caches.add(new ProcessorCache(2, l2associativity, linesize, size, ProcessorCache.Type.UNIFIED));
529             }
530             size = sysctlIntNoWarn("hw.perflevel" + i + ".l3cachesize", 0);
531             if (size > 0) {
532                 caches.add(new ProcessorCache(3, 0, linesize, size, ProcessorCache.Type.UNIFIED));
533             }
534         }
535         return caches;
536     }

543     List<String> parseX86FeatureFlags() {
544         return Stream.of("features", "extfeatures", "leaf7_features").map(f -> {
545             String key = "machdep.cpu." + f;
546             String features = sysctlStringNoWarn(key, "");
547             return Util.isBlank(features) ? null : (key + ": " + features);
548         }).filter(Objects::nonNull).collect(Collectors.toList());
549     }

551     @Override
552     public long[] queryCurrentFreq() {
553         if (isArmCpu) {
554             Map<Integer, Long> physFreqMap = new HashMap<>();
555             getPhysicalProcessors().forEach(p -> physFreqMap.put(p.getPhysicalProcessorNumber(),
556                     p.getEfficiency() > 0 ? performanceCoreFrequency : efficiencyCoreFrequency));
557             return getLogicalProcessors().stream().map(LogicalProcessor::getPhysicalProcessorNumber)
558                     .map(p -> physFreqMap.getOrDefault(p, performanceCoreFrequency)).mapToLong(f -> f).toArray();
559         }
560         return new long[] { getProcessorIdentifier().getVendorFreq() };
561     }

585     /**
586      * Extracts the maximum frequency from a byte array property.
587      *
588      * @param data the byte array from IOKit
589      * @return the frequency in Hz, or DEFAULT_FREQUENCY if unavailable
590      */
591     protected long getMaxFreqFromByteArray(byte[] data) {
592         if (data != null && data.length >= 8) {
593             byte[] freqData = Arrays.copyOfRange(data, data.length - 8, data.length - 4);
594             return ParseUtil.byteArrayToLong(freqData, 4, false);
595         }
596         return DEFAULT_FREQUENCY;
597     }

615     /**
616      * Native {@code getloadavg(3)} call, filling {@code loadavg} with up to {@code nelem} samples.
617      *
618      * @param loadavg the array to populate
619      * @param nelem   the number of elements requested
620      * @return the number of samples retrieved, or a negative value on failure
621      */

624     @Override
625     public long[] querySystemCpuLoadTicks() {
626         long[] ticks = new long[TickType.values().length];
627         int[] cpuTicks = queryHostCpuLoadTicks();
628         if (cpuTicks.length < CPU_STATE_MAX) {
629             return ticks;
630         }
631         ticks[TickType.USER.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[CPU_STATE_USER]);
632         ticks[TickType.NICE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[CPU_STATE_NICE]);
633         ticks[TickType.SYSTEM.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[CPU_STATE_SYSTEM]);
634         ticks[TickType.IDLE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[CPU_STATE_IDLE]);
635         return ticks;
636     }
637 
638     @Override
639     public long[][] queryProcessorCpuLoadTicks() {
640         long[][] ticks = new long[getLogicalProcessorCount()][TickType.values().length];
641         int[] cpuTicks = queryProcessorCpuTicks();
642         int procCount = cpuTicks.length / CPU_STATE_MAX;
643         if (procCount > ticks.length) {
644             LOG.warn("processor_cpu_load_info returned {} CPUs but expected {}; capping iteration", procCount,
645                     ticks.length);
646         }
647         int cpuLimit = Math.min(procCount, ticks.length);
648         for (int cpu = 0; cpu < cpuLimit; cpu++) {
649             int offset = cpu * CPU_STATE_MAX;
650             ticks[cpu][TickType.USER.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_USER]);
651             ticks[cpu][TickType.NICE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_NICE]);
652             ticks[cpu][TickType.SYSTEM.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_SYSTEM]);
653             ticks[cpu][TickType.IDLE.getIndex()] = FormatUtil.getUnsignedInt(cpuTicks[offset + CPU_STATE_IDLE]);
654         }
655         return ticks;
656     }

658     @Override
659     public double[] getSystemLoadAverage(int nelem) {
660         if (nelem < 1 || nelem > 3) {
661             throw new IllegalArgumentException("Must include from one to three elements.");
662         }
663         double[] average = new double[nelem];
664         int retval = getloadavgNative(average, nelem);
665         if (retval < nelem) {
666             Arrays.fill(average, -1d);
667         }
668         return average;
669     }
