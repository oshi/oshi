26  /**
27   * Abstract base for the Solaris CPU. The command-based pieces (NUMA mapping, processor-ID feature flags, the {@code
28   * dmesg} physical-processor parsing, the averaged tick rollup, and the {@code kstat -p} context-switch/interrupt
29   * fallbacks) are shared; the {@code kstat}-chain queries are native and implemented by the JNA and FFM subclasses.
30   */
31  @ThreadSafe
32  public abstract class SolarisCentralProcessor extends AbstractCentralProcessor {
33  
34      // SPARC prints decimal (:l2-dcache-size 131072), x86 prints hex (:l2-cache-size 0x80000)
35      private static final Pattern PICL_CACHE_PROPERTY = Pattern
36              .compile("\\s*:l(\\d)-([id]?)cache-(size|line-size|associativity)\\s+(0x\\p{XDigit}+|\\d+)\\s*");
37  
38      @Override
39      protected Quartet<List<LogicalProcessor>, @Nullable List<PhysicalProcessor>, @Nullable List<ProcessorCache>, List<String>> initProcessorCounts() {
40          // Defensive copy: the base appends a fallback entry, so don't assume the subclass returned a mutable list
41          List<LogicalProcessor> logProcs = new ArrayList<>(queryLogicalProcessors());
42          if (logProcs.isEmpty()) {
43              logProcs.add(new LogicalProcessor(0, 0, 0));
44          }
45          List<ProcessorCache> caches = parsePiclCaches(ExecutingCommand.runNative("/usr/sbin/prtpicl -v -c cpu"));
46          // The flags come from isainfo, so they do not depend on the boot messages still being in dmesg
47          List<String> featureFlags = ExecutingCommand.runNative("isainfo -x");
48          Map<Integer, String> dmesg = parseDmesgCpuInfo(ExecutingCommand.runNative("dmesg"));
49          List<PhysicalProcessor> physProcs = dmesg.isEmpty() ? null : createProcListFromDmesg(logProcs, dmesg);
50          return new Quartet<>(logProcs, physProcs, caches, featureFlags);
51      }

53      @Override
54      public long[] querySystemCpuLoadTicks() {
55          long[] ticks = new long[TickType.values().length];
56          // Average processor ticks
57          long[][] procTicks = getProcessorCpuLoadTicks();
58          for (int i = 0; i < ticks.length; i++) {
59              for (long[] procTick : procTicks) {
60                  ticks[i] += procTick[i];
61              }
62              ticks[i] /= procTicks.length;
63          }
64          return ticks;
65      }

67      @Override
68      public long queryContextSwitches() {
69          // Shell-escaped pipe: Java \\\\| -> shell \\| -> kstat regex \| -> matches literal pipe.
70          return sumKstatLong(ExecutingCommand.runNative("kstat -p cpu_stat:::/pswitch\\\\|inv_swtch/"));
71      }

73      @Override
74      public long queryInterrupts() {
75          return sumKstatLong(ExecutingCommand.runNative("kstat -p cpu_stat:::/intr/"));
76      }

107     /**
108      * Parses the output of {@code dmesg} to extract CPU name strings keyed by core id.
109      *
110      * @param dmesg the lines emitted by {@code dmesg}
111      * @return a map of core id to CPU name string
112      */
113     static Map<Integer, String> parseDmesgCpuInfo(List<String> dmesg) {
114         Map<Integer, String> cpuMap = new HashMap<>();
115         // Jan 9 14:04:28 solaris unix: [ID 950921 kern.info] cpu0: Intel(r) Celeron(r)
116         // CPU J3455 @ 1.50GHz
117         // but NOT: Jan 9 14:04:28 solaris unix: [ID 950921 kern.info] cpu0: x86 (chipid
118         // 0x0 GenuineIntel 506C9 family 6 model 92 step 9 clock 1500 MHz)
119         Pattern p = Pattern.compile(".* cpu(\\d+): ((ARM|AMD|Intel).+)");
120         for (String s : dmesg) {
121             Matcher m = p.matcher(s);
122             if (m.matches()) {
123                 int coreId = ParseUtil.parseIntOrDefault(m.group(1), 0);
124                 cpuMap.put(coreId, m.group(2).trim());
125             }
126         }
127         return cpuMap;
128     }

130     /**
131      * Parses the output of {@code prtpicl -v -c cpu} into the set of distinct processor caches. Each {@code cpu} node
132      * carries {@code l<n>-icache-*}, {@code l<n>-dcache-*} or (for a unified cache) {@code l<n>-cache-*} properties
133      * giving the size, line size and associativity of every cache level that CPU sees. SPARC prints the values in
134      * decimal and x86 in hexadecimal.
135      *
136      * @param prtpicl the lines emitted by {@code prtpicl -v -c cpu}
137      * @return the distinct caches, ordered by level and type; empty if the CPU nodes carry no cache properties
138      */
139     static List<ProcessorCache> parsePiclCaches(List<String> prtpicl) {
140         Set<ProcessorCache> caches = new HashSet<>();
141         // Per CPU node, each cache's size, line size and associativity, keyed by its level and type letter
142         Map<String, long[]> nodeCaches = new HashMap<>();
143         for (String line : prtpicl) {
144             Matcher m = PICL_CACHE_PROPERTY.matcher(line);
145             if (m.matches()) {
146                 long[] values = nodeCaches.computeIfAbsent(m.group(1) + m.group(2), k -> new long[3]);
147                 long value = ParseUtil.parseLastLong(m.group(4), 0L);
148                 switch (m.group(3)) {
149                     case "size":
150                         values[0] = value;
151                         break;
152                     case "line-size":
153                         values[1] = value;
154                         break;
155                     default:
156                         values[2] = value;
157                         break;
158                 }
159             } else if (!line.trim().startsWith(":")) {
160                 // A node header line ("cpu (cpu, 298000001c1)") ends the previous node
161                 addPiclCaches(nodeCaches, caches);
162             }
163         }
164         addPiclCaches(nodeCaches, caches);
165         return orderedProcCaches(caches);
166     }

168     /**
169      * Converts one CPU node's accumulated cache properties into {@link ProcessorCache} entries, then clears them.
170      *
171      * @param nodeCaches cache key (level digit plus {@code i}, {@code d} or nothing) to {size, line size,
172      *                   associativity}
173      * @param caches     the set to add the node's caches to
174      */
175     private static void addPiclCaches(Map<String, long[]> nodeCaches, Set<ProcessorCache> caches) {
176         for (Map.Entry<String, long[]> e : nodeCaches.entrySet()) {
177             long[] values = e.getValue();
178             if (values[0] > 0) {
179                 String key = e.getKey();
180                 ProcessorCache.Type type = ProcessorCache.Type.UNIFIED;
181                 if (key.endsWith("i")) {
182                     type = ProcessorCache.Type.INSTRUCTION;
183                 } else if (key.endsWith("d")) {
184                     type = ProcessorCache.Type.DATA;
185                 }
186                 caches.add(new ProcessorCache(key.charAt(0) - '0', (int) values[2], (int) values[1], values[0], type));
187             }
188         }
189         nodeCaches.clear();
190     }

192     /**
193      * Parses the output of {@code lgrpinfo -c leaves} into a map of CPU id to NUMA node.
194      *
195      * @param lgrpinfo the lines emitted by {@code lgrpinfo -c leaves}
196      * @return a map of processor id to NUMA node number
197      */
198     static Map<Integer, Integer> parseNumaNodes(List<String> lgrpinfo) {
199         // Get numa node info from lgrpinfo
200         Map<Integer, Integer> numaNodeMap = new HashMap<>();
201         int lgroup = 0;
202         for (String line : lgrpinfo) {
203             // Format:
204             // lgroup 0 (root):
205             // CPUs: 0 1
206             // CPUs: 0-7
207             // CPUs: 0-3 6 7 12 13
208             // CPU: 0
209             // CPU: 1
210             if (line.startsWith("lgroup")) {
211                 lgroup = ParseUtil.getFirstIntValue(line);
212             } else if (line.contains("CPUs:") || line.contains("CPU:")) {
213                 for (Integer cpu : ParseUtil.parseHyphenatedIntList(line.split(":", -1)[1])) {
214                     numaNodeMap.put(cpu, lgroup);
215                 }
216             }
217         }
218         return numaNodeMap;
219     }

221     /**
222      * Parses the output of {@code isainfo -v} to extract the 64-bit feature flags.
223      *
224      * @param isainfo the lines emitted by {@code isainfo -v}
225      * @return an array of feature flag strings (lowercase)
226      */
227     static String[] parseIsainfoFlags(List<String> isainfo) {
228         StringBuilder flags = new StringBuilder();
229         for (String line : isainfo) {
230             if (line.startsWith("32-bit")) {
231                 break;
232             } else if (!line.startsWith("64-bit")) {
233                 flags.append(' ').append(line.trim());
234             }
235         }
236         return ParseUtil.whitespaces.split(flags.toString().toLowerCase(Locale.ROOT));
237     }

239     /**
240      * Sums the last long value from each line of {@code kstat -p} output.
241      *
242      * @param kstat the lines emitted by a {@code kstat -p} command
243      * @return the sum of the trailing long values across all lines
244      */
245     static long sumKstatLong(List<String> kstat) {
246         long total = 0L;
247         for (String s : kstat) {
248             total += ParseUtil.parseLastLong(s, 0L);
249         }
250         return total;
251     }
