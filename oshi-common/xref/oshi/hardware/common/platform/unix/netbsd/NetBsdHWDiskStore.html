43      // dk1 at ld4: "bce3cb3f-b5ff-4aac-8049-5f748b710db0", 133807967 blocks at 409728, type: ffs
44      private static final Pattern DMESG_WEDGE = Pattern
45              .compile("\\b(dk\\d+) at (\\w+): \"(.*)\", (\\d+) blocks at \\d+, type: (\\S*)");
46      // ld4: 65536 MB, 16383 cyl, 16 head, 63 sec, 512 bytes/sect x 134217728 sectors
47      private static final Pattern DMESG_GEOMETRY = Pattern.compile("\\b(\\w+): .* (\\d+) bytes/sect x \\d+ sectors");

57      /**
58       * Gets the disks on this machine.
59       *
60       * @return a list of {@link HWDiskStore} objects representing the disks
61       */
62      public static List<HWDiskStore> getDisks() {
63          List<HWDiskStore> diskList = new ArrayList<>();
64          // Get list of disks from sysctl
65          // NetBSD: hw.disknames = ld0 fd0 dk0 dk1 cd0 (space-separated, no colon suffix)
66          String[] devices = ParseUtil.whitespaces.split(BsdSysctlUtil.sysctl("hw.disknames", "").trim(), -1);
67          List<String> dmesg = queryDmesg();
68          Map<String, List<HWPartition>> wedges = parseWedges(dmesg, Arrays.asList(devices), queryMounts(),
69                  Disklabel::getMajorMinor);
70          Set<String> wedgeNames = new HashSet<>();
71          for (List<HWPartition> parts : wedges.values()) {
72              for (HWPartition part : parts) {
73                  wedgeNames.add(part.getIdentification());
74              }
75          }
76          Supplier<List<String>> iostatSupplier = memoize(NetBsdHWDiskStore::queryIostat, defaultExpiration());
77  
78          NetBsdHWDiskStore store;
79          for (String diskName : devices) {
80              // A wedge is a partition of the disk it sits on, and is listed there instead, if that disk is present
81              if (wedgeNames.contains(diskName)) {
82                  continue;
83              }
84              // Try disklabel first (needs root)
85              Quartet<String, String, Long, List<HWPartition>> diskdata = Disklabel.getDiskParams(diskName);
86              String model = diskdata.getA();
87              long size = diskdata.getC();
88              if (size <= 1) {
89                  // NetBSD dmesg format for physical disks:
90                  // ld0: 200 GB, 16383 cyl, 16 head, 63 sec, 512 bytes/sect x 419430400 sectors
91                  Pattern diskGeom = Pattern.compile(diskName + ": .* (\\d+) bytes/sect x (\\d+) sectors");
92                  // NetBSD dmesg format for wedge devices (dk*), for one whose parent was not found:
93                  // dk0 at ld0: "uuid", 406845440 blocks at 2048, type: ffs
94                  Pattern wedgeBlocks = Pattern.compile(diskName + " at (\\w+): .*, (\\d+) blocks at .*");
95                  // Model from "device at bus targ N lun N: <Vendor, Model, Rev>"
96                  // but NOT from "features: 0x...<FLAGS>" lines
97                  Pattern diskAt = Pattern.compile(".*" + diskName + " at .* lun \\d+: <(.+)>.*");
98                  for (String line : dmesg) {
99                      Matcher m = diskGeom.matcher(line);
100                     if (m.find()) {
101                         long bytesPerSector = ParseUtil.parseLongOrDefault(m.group(1), 512L);
102                         long sectors = ParseUtil.parseLongOrDefault(m.group(2), 0L);
103                         size = bytesPerSector * sectors;
104                         break;
105                     }
106                     m = wedgeBlocks.matcher(line);
107                     if (m.find()) {
108                         long blocks = ParseUtil.parseLongOrDefault(m.group(2), 0L);
109                         size = blocks * 512L;
110                         break;
111                     }
112                     m = diskAt.matcher(line);
113                     if (m.matches()) {
114                         model = m.group(1);
115                     }
116                 }
117             }
118             store = new NetBsdHWDiskStore(diskName, model, diskdata.getB(), size, iostatSupplier);
119             // On a disk carrying wedges, the wedges are its partitions; any disklabel describes the same space
120             List<HWPartition> diskWedges = wedges.get(diskName);
121             store.setPartitionList(diskWedges == null ? diskdata.getD() : diskWedges);
122             store.updateAttributes();
123 
124             diskList.add(store);
125         }
126         return diskList;
127     }

129     /**
130      * Parses the wedges ({@code dk} devices) announced in the kernel message buffer, grouped by the disk each one sits
131      * on. Package-private for testing.
132      * <p>
133      * A wedge's size is counted in sectors of its parent disk, so the parent's sector size comes from its geometry
134      * line, defaulting to 512 bytes. The quoted wedge name is the partition label if one was set, and otherwise its
135      * GUID.
136      * <p>
137      * The messages are read oldest first, so where a device is announced more than once, as it is in both the boot copy
138      * and the live buffer, or when a wedge number is reused after boot, the last announcement wins. Only wedges that
139      * are still present, on a disk that is still present, are returned: the boot copy remembers wedges deleted since,
140      * and a wedge whose parent is missing has no disk to be listed under.
141      *
142      * @param dmesg            the kernel messages, oldest first
143      * @param devices          the disk devices currently present, wedges included
144      * @param mounts           mount points keyed by device name, such as {@code dk1}
145      * @param majorMinorLookup resolves a device name and suffix to its major and minor numbers
146      * @return the wedges of each disk, keyed by the disk's name
147      */
148     static Map<String, List<HWPartition>> parseWedges(List<String> dmesg, Collection<String> devices,
149             Map<String, String> mounts, BiFunction<String, String, Pair<Integer, Integer>> majorMinorLookup) {
150         Map<String, Long> sectorSizes = new HashMap<>();
151         Map<String, MatchResult> announcements = new LinkedHashMap<>();
152         for (String line : dmesg) {
153             Matcher m = DMESG_GEOMETRY.matcher(line);
154             if (m.find()) {
155                 sectorSizes.put(m.group(1), ParseUtil.parseLongOrDefault(m.group(2), 512L));
156             }
157             m = DMESG_WEDGE.matcher(line);
158             if (m.find()) {
159                 announcements.put(m.group(1), m.toMatchResult());
160             }
161         }
162         Map<String, List<HWPartition>> wedges = new LinkedHashMap<>();
163         for (MatchResult m : announcements.values()) {
164             String name = m.group(1);
165             String parent = m.group(2);
166             if (!devices.contains(name) || !devices.contains(parent)) {
167                 continue;
168             }
169             String wedgeName = m.group(3);
170             String uuid = ParseUtil.parseUuidOrDefault(wedgeName, "");
171             long size = ParseUtil.parseLongOrDefault(m.group(4), 0L) * sectorSizes.getOrDefault(parent, 512L);
172             Pair<Integer, Integer> majorMinor = majorMinorLookup.apply(name, "");
173             wedges.computeIfAbsent(parent, p -> new ArrayList<>())
174                     .add(new HWPartition(name, name, m.group(5), uuid, uuid.isEmpty() ? wedgeName : "", size,
175                             majorMinor.getA(), majorMinor.getB(), mounts.getOrDefault(name, "")));
176         }
177         return wedges;
178     }

180     /**
181      * Parses {@code df} output into mount points keyed by device name. Package-private for testing.
182      *
183      * @param df the output of {@code df}
184      * @return mount points keyed by the device name without its {@code /dev/} prefix
185      */
186     static Map<String, String> parseMounts(List<String> df) {
187         Map<String, String> mounts = new HashMap<>();
188         for (String line : df) {
189             if (line.startsWith("/dev/")) {
190                 // Six columns; the last, the mount point, may contain spaces
191                 String[] split = ParseUtil.whitespaces.split(line, 6);
192                 if (split.length == 6) {
193                     mounts.put(split[0].substring(5), split[5]);
194                 }
195             }
196         }
197         return mounts;
198     }

200     /**
201      * Reads the copy of the kernel messages saved at boot, followed by the live buffer, so the result runs oldest
202      * first. The boot copy still holds the disks once the live buffer has wrapped past them, and the live buffer holds
203      * wedges added since boot.
204      */
205     private static List<String> queryDmesg() {
206         List<String> dmesg = new ArrayList<>(FileUtil.readFile("/var/run/dmesg.boot", false));
207         dmesg.addAll(ExecutingCommand.runNative("dmesg"));
208         return dmesg;
209     }
210 
211     private static Map<String, String> queryMounts() {
212         return parseMounts(ExecutingCommand.runNative("df"));
213     }

215     @Override
216     public synchronized boolean updateAttributes() {
217         long now = System.currentTimeMillis();
218         boolean diskFound = false;
219         for (String line : iostat.get()) {
220             String[] split = ParseUtil.whitespaces.split(line.trim(), -1);
221             // iostat -x -I output (cumulative totals, 9 fields):
222             // device read KB/t xfr time MB write KB/t xfr time MB
223             // ld0 26.36 2087 8.23 53.72 25.42 74725 8.23 1855.30
224             // [0] [1] [2] [3] [4] [5] [6] [7] [8]
225             if (split.length >= 9 && split[0].equals(getName())) {
226                 diskFound = true;
227                 this.reads = ParseUtil.parseLongOrDefault(split[2], 0L);
228                 this.writes = ParseUtil.parseLongOrDefault(split[6], 0L);
229                 // Bytes come from the MB totals, not from KB-per-transfer times transfers: iostat rounds KB/t to two
230                 // decimals, and multiplying that by a transfer count scales the rounding error up with it, enough that
231                 // two readings seconds apart can go backwards. Rounding a total cannot.
232                 this.readBytes = (long) (ParseUtil.parseDoubleOrDefault(split[4], 0d) * 1024 * 1024);
233                 this.writeBytes = (long) (ParseUtil.parseDoubleOrDefault(split[8], 0d) * 1024 * 1024);
234                 // Both time columns carry the same drive-busy total (cur.time[dn] in iostat's drive_statsx), which
235                 // covers reads and writes together, so it is counted once.
236                 this.transferTime = (long) (ParseUtil.parseDoubleOrDefault(split[3], 0d) * 1000);
237                 this.timeStamp = now;
238                 break;
239             }
240         }
241         return diskFound;
242     }
243 
244     private static List<String> queryIostat() {
245         return ExecutingCommand.runNative("iostat -x -I");
246     }
