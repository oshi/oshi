7   import static oshi.util.Memoizer.memoize;
8   
9   import java.util.HashMap;
10  import java.util.List;
11  import java.util.Map;
12  import java.util.function.Supplier;
13  
14  import oshi.annotation.concurrent.Immutable;
15  import oshi.hardware.Baseboard;
16  import oshi.hardware.Firmware;
17  import oshi.hardware.common.platform.unix.BsdComputerSystem;
18  import oshi.hardware.common.platform.unix.UnixBaseboard;
19  import oshi.util.Constants;
20  import oshi.util.ExecutingCommand;
21  import oshi.util.ParseUtil;

23  /**
24   * NetBSD ComputerSystem implementation. Reads the SMBIOS strings the kernel publishes under {@code machdep.dmi}. The
25   * serial numbers and UUID there are readable only by root, and the board values are absent where the firmware has no
26   * baseboard table, in which case the system values stand in for them.
27   */
28  @Immutable
29  public class NetBsdComputerSystem extends BsdComputerSystem {
30  
31      private static final String DMI_PREFIX = "machdep.dmi.";
32  
33      private final Supplier<Map<String, String>> dmi = memoize(NetBsdComputerSystem::queryDmi);
34  
35      @Override
36      protected String queryManufacturer() {
37          return dmi("system-vendor");
38      }
39  
40      @Override
41      protected String queryModel() {
42          return dmi("system-product");
43      }
44  
45      @Override
46      protected String querySerialNumber() {
47          return dmi("system-serial");
48      }
49  
50      @Override
51      protected String queryUUID() {
52          return dmi("system-uuid");
53      }
54  
55      @Override
56      protected Firmware createFirmware() {
57          return new NetBsdFirmware();
58      }
59  
60      @Override
61      protected Baseboard createBaseboard() {
62          return new UnixBaseboard(boardOrSystem("vendor"), boardOrSystem("product"), boardOrSystem("serial"),
63                  boardOrSystem("version"));
64      }
65  
66      private String dmi(String key) {
67          return ParseUtil.getStringValueOrUnknown(dmi.get().get(key));
68      }
69  
70      private String boardOrSystem(String field) {
71          String board = dmi("board-" + field);
72          return Constants.UNKNOWN.equals(board) ? dmi("system-" + field) : board;
73      }
74  
75      /**
76       * Reads the {@code machdep.dmi} sysctls.
77       *
78       * @return the values keyed by the name after {@code machdep.dmi.}, e.g. {@code system-vendor}
79       */
80      static Map<String, String> queryDmi() {
81          return parseDmi(ExecutingCommand.runNative("sysctl machdep.dmi"));
82      }
83  
84      /**
85       * Parses the output of {@code sysctl machdep.dmi}, one {@code machdep.dmi.key = value} line per string.
86       *
87       * @param lines the command output
88       * @return the values keyed by the name after {@code machdep.dmi.}, e.g. {@code system-vendor}
89       */
90      static Map<String, String> parseDmi(List<String> lines) {
91          Map<String, String> values = new HashMap<>();
92          for (Map.Entry<String, String> e : ParseUtil.parseStringListToMap(lines, " = ").entrySet()) {
93              if (e.getKey().startsWith(DMI_PREFIX)) {
94                  values.put(e.getKey().substring(DMI_PREFIX.length()), e.getValue());
95              }
96          }
97          return values;
98      }
99  }
