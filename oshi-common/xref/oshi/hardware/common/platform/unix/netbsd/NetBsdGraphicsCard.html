24  /**
25   * Graphics card info obtained from the NetBSD boot messages.
26   * <p>
27   * Listing PCI devices with their class ({@code pcictl}) requires root, so a display adapter is identified instead as
28   * the PCI device that a {@code wsdisplay} console attaches to, directly ({@code vga}, {@code genfb}) or through a
29   * framebuffer driver ({@code i915drmkms} and similar).
30   */
31  @Immutable
32  final class NetBsdGraphicsCard extends AbstractGraphicsCard {
33  
34      private static final String DMESG_BOOT = "/var/run/dmesg.boot";
35  
36      // wsdisplay0 at vga0 kbdmux 1: console (80x25, vt100 emulation), using wskbd0
37      private static final Pattern ATTACH = Pattern.compile("^(\\w++(?<=\\d)) at (\\w++(?<=\\d))\\b.*+");
38      // vga0 at pci0 dev 2 function 0: vendor 1234 product 1111 (rev. 0x02)
39      // virtio0 at pci0 dev 2 function 0
40      private static final Pattern PCI_ATTACH = Pattern.compile(
41              "^(\\w++(?<=\\d)) at pci\\d++ dev \\d++ function \\d++[^:]*+(?:: (.+?)(?: \\(rev\\. (0x\\p{XDigit}++)\\))?)?$");
42      // virtio0: GPU device (id 16, rev. 0x01), printed after a bare virtio attachment
43      private static final Pattern VIRTIO_TYPE = Pattern
44              .compile("^(\\w++(?<=\\d)): (.+) device \\(id \\d++, rev\\. (0x\\p{XDigit}++)\\)$");
45      private static final Pattern PCI_IDS = Pattern.compile("vendor (\\p{XDigit}{4}) product (\\p{XDigit}{4})");
46  
47      NetBsdGraphicsCard(String name, String deviceId, String vendor, String versionInfo, long vram) {
48          super(name, deviceId, vendor, versionInfo, vram);
49      }
50  
51      /**
52       * Gets the graphics cards that a console display is attached to.
53       *
54       * @return the graphics cards
55       */
56      static List<GraphicsCard> getGraphicsCards() {
57          return parseDmesg(FileUtil.readFile(DMESG_BOOT, false));
58      }
59  
60      /**
61       * Parses boot messages for the PCI devices that a {@code wsdisplay} attaches to. A virtio device attaches to PCI
62       * with no description and reports its type on the following line.
63       *
64       * @param dmesg the lines of {@code /var/run/dmesg.boot}
65       * @return a graphics card for each such device, in attach order
66       */
67      static List<GraphicsCard> parseDmesg(List<String> dmesg) {
68          Map<String, String> parents = new HashMap<>();
69          Map<String, PciDevice> pciDevices = new HashMap<>();
70          List<String> displays = new ArrayList<>();
71          for (String line : dmesg) {
72              Matcher pci = PCI_ATTACH.matcher(line);
73              Matcher virtio = VIRTIO_TYPE.matcher(line);
74              Matcher m = ATTACH.matcher(line);
75              if (pci.matches()) {
76                  pciDevices.put(pci.group(1), new PciDevice(pci.group(2), pci.group(3), false));
77              } else if (virtio.matches()) {
78                  PciDevice bare = pciDevices.get(virtio.group(1));
79                  if (bare != null && bare.description.isEmpty()) {
80                      pciDevices.put(virtio.group(1), new PciDevice(virtio.group(2), virtio.group(3), true));
81                  }
82              } else if (m.matches()) {
83                  parents.putIfAbsent(m.group(1), m.group(2));
84                  if (m.group(1).startsWith("wsdisplay")) {
85                      displays.add(m.group(1));
86                  }
87              }
88          }
89          // Walk each display up to the PCI device it hangs from
90          Map<String, PciDevice> adapters = new LinkedHashMap<>();
91          for (String display : displays) {
92              String device = parents.get(display);
93              for (int depth = 0; device != null && depth < 8; depth++) {
94                  PciDevice pci = pciDevices.get(device);
95                  if (pci != null) {
96                      adapters.putIfAbsent(device, pci);
97                      break;
98                  }
99                  device = parents.get(device);
100             }
101         }
102         List<GraphicsCard> cards = new ArrayList<>();
103         for (PciDevice pci : adapters.values()) {
104             cards.add(toGraphicsCard(pci));
105         }
106         return Collections.unmodifiableList(cards);
107     }
108 
109     private static GraphicsCard toGraphicsCard(PciDevice pci) {
110         String versionInfo = pci.revision == null ? Constants.UNKNOWN : "Revision: " + pci.revision;
111         if (pci.description.isEmpty()) {
112             return new NetBsdGraphicsCard(Constants.UNKNOWN, Constants.UNKNOWN, Constants.UNKNOWN, versionInfo, 0L);
113         }
114         if (pci.virtio) {
115             return new NetBsdGraphicsCard("Virtio " + pci.description + " device", Constants.UNKNOWN, Constants.UNKNOWN,
116                     versionInfo, 0L);
117         }
118         Matcher ids = PCI_IDS.matcher(pci.description);
119         if (ids.find()) {
120             // No name in the kernel's PCI database: only the IDs are known
121             return new NetBsdGraphicsCard(Constants.UNKNOWN, "0x" + ids.group(2), "0x" + ids.group(1), versionInfo, 0L);
122         }
123         // The kernel prints vendor and product names as one string, with no reliable boundary between them
124         return new NetBsdGraphicsCard(pci.description, Constants.UNKNOWN, Constants.UNKNOWN, versionInfo, 0L);
125     }
126 
127     /**
128      * A PCI attachment: its description (empty if none was printed), revision, and whether it is a virtio device.
129      */
130     @Immutable
131     private static final class PciDevice {
132         private final String description;
133         private final @Nullable String revision;
134         private final boolean virtio;
135 
136         PciDevice(@Nullable String description, @Nullable String revision, boolean virtio) {
137             this.description = description == null ? "" : description;
138             this.revision = revision;
139             this.virtio = virtio;
140         }
141     }
142 }
