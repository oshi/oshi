22  /**
23   * Batteries reported by NetBSD's {@code acpibat(4)} through {@code envstat}, with the external power state from
24   * {@code acpiacad(4)}.
25   */
26  @ThreadSafe
27  final class NetBsdPowerSource extends AbstractPowerSource {
28  
29      private static final double MICRO = 1_000_000d;
30  
31      private NetBsdPowerSource(String psName, double psRemainingCapacityPercent, double psTimeRemaining,
32              double psPowerUsageRate, double psVoltage, double psAmperage, boolean psPowerOnLine, boolean psCharging,
33              boolean psDischarging, CapacityUnits psCapacityUnits, int psCurrentCapacity, int psMaxCapacity,
34              int psDesignCapacity) {
35          super(psName, Constants.UNKNOWN, psRemainingCapacityPercent, psTimeRemaining, psTimeRemaining, psPowerUsageRate,
36                  psVoltage, psAmperage, psPowerOnLine, psCharging, psDischarging, psCapacityUnits, psCurrentCapacity,
37                  psMaxCapacity, psDesignCapacity, -1, Constants.UNKNOWN, null, Constants.UNKNOWN, Constants.UNKNOWN, 0d);
38      }
39  
40      @Override
41      protected List<PowerSource> queryPowerSources() {
42          return getPowerSources();
43      }
44  
45      /**
46       * Gets the batteries present in the system.
47       *
48       * @return the power sources
49       */
50      static List<PowerSource> getPowerSources() {
51          return parsePowerSources(Envstat.querySensors());
52      }
53  
54      /**
55       * Builds power sources from {@code envstat} sensors. A battery is a device reporting {@code present} and
56       * {@code charge}; a slot whose {@code present} indicator is false or invalid is skipped.
57       *
58       * @param sensors the parsed sensors
59       * @return a power source for each battery present
60       */
61      static List<PowerSource> parsePowerSources(List<Sensor> sensors) {
62          Map<String, Map<String, Sensor>> byDevice = new LinkedHashMap<>();
63          @Nullable
64          Boolean acConnected = null;
65          for (Sensor s : sensors) {
66              byDevice.computeIfAbsent(s.getDevice(), d -> new LinkedHashMap<>()).put(s.getDescription(), s);
67              if ("connected".equals(s.getDescription()) && Envstat.INDICATOR.equals(s.getType()) && s.isValid()) {
68                  acConnected = (acConnected != null && acConnected) || s.getValue() != 0;
69              }
70          }
71          List<PowerSource> batteries = new ArrayList<>();
72          for (Map.Entry<String, Map<String, Sensor>> e : byDevice.entrySet()) {
73              Map<String, Sensor> bat = e.getValue();
74              Sensor present = bat.get("present");
75              Sensor charge = bat.get("charge");
76              // A failed presence check leaves the previous value in place but marks every sensor invalid
77              if (present == null || charge == null || !present.isValid() || present.getValue() == 0) {
78                  continue;
79              }
80              batteries.add(toPowerSource(e.getKey(), bat, charge, acConnected));
81          }
82          return Collections.unmodifiableList(batteries);
83      }
84  
85      private static PowerSource toPowerSource(String device, Map<String, Sensor> bat, Sensor charge,
86              @Nullable Boolean acConnected) {
87          CapacityUnits units = Envstat.AMPERE_HOUR.equals(charge.getType()) ? CapacityUnits.MAH : CapacityUnits.MWH;
88          // Energy and charge are in micro units; OSHI reports milli units
89          int current = (int) (value(charge) / 1000);
90          int max = (int) (value(bat.get("last full cap")) / 1000);
91          int design = (int) (value(bat.get("design cap")) / 1000);
92          double percent = max > 0 ? Math.min(1d, (double) current / max) : 1d;
93  
94          Sensor voltageSensor = bat.get("voltage");
95          double voltage = voltageSensor != null && voltageSensor.isValid() ? voltageSensor.getValue() / MICRO : -1d;
96          long chargeRate = value(bat.get("charge rate"));
97          long dischargeRate = value(bat.get("discharge rate"));
98          boolean charging = value(bat.get("charging")) != 0;
99          boolean discharging = !charging && dischargeRate > 0;
100         boolean onLine = acConnected != null ? acConnected : !discharging;
101 
102         // The rate sensors are watts for a battery measured in watt-hours and amperes for one in ampere-hours
103         long signedRate = charging ? chargeRate : -dischargeRate;
104         double powerMilliwatts = 0d;
105         double currentMilliamps = 0d;
106         if (units == CapacityUnits.MWH) {
107             powerMilliwatts = signedRate / 1000d;
108             if (voltage > 0) {
109                 currentMilliamps = powerMilliwatts / voltage;
110             }
111         } else {
112             currentMilliamps = signedRate / 1000d;
113             if (voltage > 0) {
114                 powerMilliwatts = currentMilliamps * voltage;
115             }
116         }
117 
118         // -1 is unknown, -2 is unlimited (running on external power)
119         double timeRemaining = -1d;
120         if (discharging) {
121             timeRemaining = 3600d * value(charge) / dischargeRate;
122         } else if (onLine) {
123             timeRemaining = -2d;
124         }
125 
126         String name = device.startsWith("acpi") ? device.substring(4) : device;
127         return new NetBsdPowerSource(name, percent, timeRemaining, powerMilliwatts, voltage, currentMilliamps, onLine,
128                 charging, discharging, units, current, max, design);
129     }
130 
131     private static long value(@Nullable Sensor s) {
132         return s != null && s.isValid() ? s.getValue() : 0L;
133     }
134 }
