27  /**
28   * FreeBSD and DragonFly BSD sound card, read from the device tree the kernel publishes under {@code dev.}.
29   */
30  @Immutable
31  public class FreeBsdSoundCard extends AbstractSoundCard {
32  
33      /** A device-tree sysctl line, {@code dev.pcm.0.%desc: Dummy Audio Device}. */
34      private static final Pattern DEV_SYSCTL = Pattern.compile("dev\\.(\\w+)\\.(\\d+)\\.%(desc|parent): ?(.*)");
35      /** A device name, {@code hdaa0}: the driver, then the unit. */
36      private static final Pattern DEVICE_NAME = Pattern.compile("([a-z_]\\w*?)(\\d+)");
37      /** The buses a sound controller sits on, which end the walk up the tree. */
38      private static final Pattern BUS = Pattern.compile("(nexus|pci|uhub|usbus|acpi|isa|ofwbus|simplebus)\\d*");
39      /** The kernel release, which is the version of the in-kernel sound drivers. */
40      private static final String OSRELEASE = "kern.osrelease";
41      /** Bounds the walk: HDA is pcm, hdaa, hdacc, hdac. */
42      private static final int MAX_DEPTH = 4;
43  
44      /**
45       * Constructor for FreeBsdSoundCard.
46       *
47       * @param kernelVersion The version
48       * @param name          The name
49       * @param codec         The codec
50       */
51      FreeBsdSoundCard(String kernelVersion, String name, String codec) {
52          super(kernelVersion, name, codec);
53      }
54  
55      /**
56       * Gets the sound cards: one for each {@code pcm} device.
57       *
58       * @return a {@link java.util.List} object.
59       */
60      public static List<SoundCard> getSoundCards() {
61          // -i skips a name sysctl does not know: without it DragonFly BSD stops there and prints nothing further
62          return parseSoundCards(oids -> ExecutingCommand.runNative("sysctl -i " + String.join(" ", oids)));
63      }
64  
65      /**
66       * Builds one sound card for each {@code pcm} device. Package-private for testing.
67       * <p>
68       * Each {@code pcm} device is an audio endpoint, whose description names it, as in
69       * {@code Realtek ALC892 (Rear Analog)}; that is the codec. Its card is the controller it hangs from, found by
70       * following each device's {@code %parent} up the tree until the bus: on an HDA system {@code pcm0} is on
71       * {@code hdaa0}, which is on the codec {@code hdacc0}, which is on the controller {@code hdac0} on {@code pci0}. A
72       * device attached directly to a bus, such as {@code snd_dummy} on {@code nexus0}, is its own card. Reading the tree
73       * needs no privilege, where {@code /dev/sndstat} may not be readable, and {@code lshal}, the previous source, is no
74       * longer packaged.
75       *
76       * @param sysctl runs {@code sysctl} on the given names and returns the lines it prints for those it knows, skipping
77       *               the rest
78       * @return the sound cards, in {@code pcm} unit order
79       */
80      static List<SoundCard> parseSoundCards(UnaryOperator<List<String>> sysctl) {
81          List<String> first = sysctl.apply(Arrays.asList(OSRELEASE, "dev.pcm"));
82          Map<String, Map<String, String>> devices = parseDevices(first);
83          Map<String, Map<String, String>> pcms = devices.entrySet().stream()
84                  .filter(e -> e.getKey().startsWith("pcm") && DEVICE_NAME.matcher(e.getKey()).matches())
85                  .sorted(Map.Entry.comparingByKey((a, b) -> Integer.compare(unit(a), unit(b))))
86                  .collect(Collectors.toMap(Map.Entry::getKey, Map.Entry::getValue, (a, b) -> a, LinkedHashMap::new));
87          if (pcms.isEmpty()) {
88              return new ArrayList<>();
89          }
90          String version = ParseUtil.getStringValueOrUnknown(version(first));
91          // Walk every pcm device's ancestors together, one sysctl call for each level of the tree
92          Set<String> tried = new HashSet<>();
93          List<String> pending = parentsToFetch(pcms.values(), tried);
94          for (int depth = 1; depth < MAX_DEPTH && !pending.isEmpty(); depth++) {
95              tried.addAll(pending);
96              List<String> oids = new ArrayList<>();
97              for (String device : pending) {
98                  Matcher m = DEVICE_NAME.matcher(device);
99                  if (m.matches()) {
100                     oids.add("dev." + m.group(1) + "." + m.group(2) + ".%desc");
101                     oids.add("dev." + m.group(1) + "." + m.group(2) + ".%parent");
102                 }
103             }
104             devices.putAll(parseDevices(sysctl.apply(oids)));
105             pending = parentsToFetch(devices.values(), tried);
106         }
107         List<SoundCard> soundCards = new ArrayList<>();
108         for (Map<String, String> pcm : pcms.values()) {
109             String codec = ParseUtil.getStringValueOrUnknown(pcm.get("desc"));
110             String card = codec;
111             Map<String, String> device = pcm;
112             for (int depth = 0; depth < MAX_DEPTH; depth++) {
113                 Map<String, String> parent = devices.get(device.getOrDefault("parent", ""));
114                 if (parent == null) {
115                     break;
116                 }
117                 card = ParseUtil.getStringValueOrDefault(parent.get("desc"), card);
118                 device = parent;
119             }
120             soundCards.add(new FreeBsdSoundCard(version, card, codec));
121         }
122         return soundCards;
123     }

125     /**
126      * Parses the {@code %desc} and {@code %parent} lines of {@code sysctl dev.<driver>} output, keyed by device name.
127      *
128      * @param lines the lines emitted by {@code sysctl}
129      * @return each device's {@code desc} and {@code parent}
130      */
131     private static Map<String, Map<String, String>> parseDevices(List<String> lines) {
132         Map<String, Map<String, String>> devices = new HashMap<>();
133         for (String line : lines) {
134             Matcher m = DEV_SYSCTL.matcher(line.trim());
135             if (m.matches()) {
136                 devices.computeIfAbsent(m.group(1) + m.group(2), k -> new HashMap<>()).put(m.group(3),
137                         m.group(4).trim());
138             }
139         }
140         return devices;
141     }
142 
143     /**
144      * Finds the parents not yet read that are devices rather than the bus that ends the walk.
145      *
146      * @param children the devices whose parents to look up
147      * @param tried    the devices already asked for, whether or not {@code sysctl} knew them
148      * @return the device names still to read
149      */
150     private static List<String> parentsToFetch(Iterable<Map<String, String>> children, Set<String> tried) {
151         List<String> pending = new ArrayList<>();
152         for (Map<String, String> child : children) {
153             String parent = child.getOrDefault("parent", "");
154             if (!parent.isEmpty() && !tried.contains(parent) && !pending.contains(parent)
155                     && !BUS.matcher(parent).matches() && DEVICE_NAME.matcher(parent).matches()) {
156                 pending.add(parent);
157             }
158         }
159         return pending;
160     }
161 
162     private static int unit(String device) {
163         Matcher m = DEVICE_NAME.matcher(device);
164         return m.matches() ? ParseUtil.parseIntOrDefault(m.group(2), 0) : 0;
165     }
166 
167     private static String version(List<String> lines) {
168         for (String line : lines) {
169             if (line.startsWith(OSRELEASE + ":")) {
170                 return line.substring(OSRELEASE.length() + 1).trim();
171             }
172         }
173         return Constants.UNKNOWN;
174     }
