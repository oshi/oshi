7   import java.util.List;
8   
9   import org.jspecify.annotations.Nullable;
10  
11  import oshi.annotation.concurrent.Immutable;
12  import oshi.hardware.common.platform.unix.BsdFirmware;
13  import oshi.util.ExecutingCommand;
14  import oshi.util.ParseUtil;
15  import oshi.util.tuples.Triplet;
16  
17  /**
18   * Firmware information from dmidecode
19   */
20  @Immutable
21  public class FreeBsdFirmware extends BsdFirmware {
22  
23      @Override
24      protected Triplet<@Nullable String, @Nullable String, @Nullable String> readFirmware() {
25          // Only works with root permissions but it's all we've got
26          return parseDmiDecode(ExecutingCommand.runNative("dmidecode -t bios"));
27      }
28  
29      /**
30       * Parses the output of {@code dmidecode -t bios} into its manufacturer, version, and release date fields. Any field
31       * not present in the output is returned as {@code null} (or an empty date); the caller applies fallbacks.
32       *
33       * @param dmidecode the lines emitted by {@code dmidecode -t bios}
34       * @return a {@link Triplet} of manufacturer, version, and release date
35       */
36      static Triplet<@Nullable String, @Nullable String, @Nullable String> parseDmiDecode(List<String> dmidecode) {
37          String manufacturer = null;
38          String version = null;
39          String releaseDate = "";
40  
41          // $ sudo dmidecode -t bios
42          // # dmidecode 3.0
43          // Scanning /dev/mem for entry point.
44          // SMBIOS 2.7 present.
45          //
46          // Handle 0x0000, DMI type 0, 24 bytes
47          // BIOS Information
48          // Vendor: Parallels Software International Inc.
49          // Version: 11.2.1 (32626)
50          // Release Date: 07/15/2016
51          // ... <snip> ...
52          // BIOS Revision: 11.2
53          // Firmware Revision: 11.2
54  
55          final String manufacturerMarker = "Vendor:";
56          final String versionMarker = "Version:";
57          final String releaseDateMarker = "Release Date:";
58  
59          for (final String checkLine : dmidecode) {
60              if (checkLine.contains(manufacturerMarker)) {
61                  manufacturer = ParseUtil.getTextAfterString(checkLine, manufacturerMarker).trim();
62              } else if (checkLine.contains(versionMarker)) {
63                  version = ParseUtil.getTextAfterString(checkLine, versionMarker).trim();
64              } else if (checkLine.contains(releaseDateMarker)) {
65                  releaseDate = ParseUtil.getTextAfterString(checkLine, releaseDateMarker).trim();
66              }
67          }
68          releaseDate = ParseUtil.parseMmDdYyyyToYyyyMmDD(releaseDate);
69          return new Triplet<>(manufacturer, version, releaseDate);
70      }
71  }
