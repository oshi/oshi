37      // Capture the CSV of hex values as group(1), clients should split on ','
38      // The Origin line of dmesg.boot, and each name=value pair on it, with or without spaces around the =
39      private static final Pattern ORIGIN = Pattern.compile("Origin\\s*=.*");
40      // The lookbehind stops find() from retrying inside a name, so the scan stays linear
41      private static final String DMESG_NAME = "(?<!\\w)(\\w++)\\s*+=\\s*+";
42      private static final String DMESG_VALUE = "(\"[^\"]*+\"|\\S++)";
43      private static final Pattern DMESG_FIELD = Pattern.compile(DMESG_NAME + DMESG_VALUE); // NOSONAR java:S8786 - linear

45      private static final Pattern CPUMASK = Pattern.compile("mask=\"([\\p{XDigit},]+)\"");

50      private static final int CP_SYS = 2;
51      private static final int CP_INTR = 3;
52      private static final int CP_IDLE = 4;
53      private static final int CPUSTATES = 5;

100     /**
101      * Maps the five FreeBSD CPU-state values starting at {@code offset} in {@code cpTimeArray} to the corresponding
102      * {@link TickType} slots of {@code ticks}.
103      *
104      * @param ticks       the destination tick array
105      * @param cpTimeArray the source uint64 values
106      * @param offset      the starting index of this CPU's states within {@code cpTimeArray}
107      */
108     private static void fillTicks(long[] ticks, long[] cpTimeArray, int offset) {
109         ticks[TickType.USER.getIndex()] = cpTimeArray[offset + CP_USER];
110         ticks[TickType.NICE.getIndex()] = cpTimeArray[offset + CP_NICE];
111         ticks[TickType.SYSTEM.getIndex()] = cpTimeArray[offset + CP_SYS];
112         ticks[TickType.IRQ.getIndex()] = cpTimeArray[offset + CP_INTR];
113         ticks[TickType.IDLE.getIndex()] = cpTimeArray[offset + CP_IDLE];
114     }

117     public long[] querySystemCpuLoadTicks() {
118         long[] ticks = new long[TickType.values().length];
119         long[] cpTime = queryCpTimes("kern.cp_time");
120         if (cpTime == null || cpTime.length < CPUSTATES) {
121             return ticks;
122         }
123         fillTicks(ticks, cpTime, 0);
124         return ticks;

132     /**
133      * Reads per-CPU ticks from DragonFly BSD's {@code kern.cputime}, an array of one {@code struct kinfo_cputime} per
134      * CPU. The struct begins with the five CPU-state values and is followed by fields whose count may vary by release,
135      * so the per-CPU stride is derived from the array length.
136      *
137      * @return per-CPU ticks, zero-filled if the sysctl is unavailable or too short
138      */
139     protected long[][] queryKinfoCputimeTicks() {
140         int cpus = getLogicalProcessorCount();
141         long[] cpTimes = queryCpTimes("kern.cputime");
142         return parseProcessorTicks(cpTimes, cpus, cpTimes == null ? 0 : cpTimes.length / cpus);
143     }

145     /**
146      * Maps an array of per-CPU records, each beginning with the five CPU-state values, to per-CPU tick arrays.
147      *
148      * @param cpTimes the uint64 values read from the sysctl, or {@code null} if the read failed
149      * @param cpus    the number of logical processors
150      * @param stride  the number of values in each CPU's record
151      * @return per-CPU ticks; CPUs without a complete record in {@code cpTimes} are zero-filled
152      */
153     static long[][] parseProcessorTicks(long @Nullable [] cpTimes, int cpus, int stride) {
154         long[][] ticks = new long[cpus][TickType.values().length];
155         if (cpTimes == null || stride < CPUSTATES) {
156             return ticks;
157         }
158         int records = Math.min(cpus, cpTimes.length / stride);
159         for (int cpu = 0; cpu < records; cpu++) {
160             fillTicks(ticks[cpu], cpTimes, cpu * stride);
161         }
162         return ticks;
163     }

167         if (nelem < 1 || nelem > 3) {
168             throw new IllegalArgumentException("Must include from one to three elements.");

170         double[] average = new double[nelem];
171         int retval = getloadavgNative(average, nelem);

175         }
176         return average;

191         // Parsing dmesg.boot is apparently the only reliable source for processor identification in FreeBSD. Read

199         return new ProcessorIdentifier(id.vendor(), id.name(), id.family(), id.model(), id.stepping(), processorID,

247     /**
248      * Parses the processor identifier fields from {@code dmesg.boot}. Uses {@code initialName} (from
249      * {@code sysctl hw.model}) unless the file supplies a {@code CPU:} line and the initial name is empty.
250      * Package-private for testing.
251      * <p>
252      * FreeBSD writes {@code Origin="GenuineIntel"  Id=0x306f2  Family=0x6  Model=0x3f  Stepping=2}. DragonFly BSD puts
253      * spaces around each {@code =} and omits the family and model, which are then decoded from the {@code Id}.
254      *
255      * @param dmesg       the lines of {@code dmesg.boot}
256      * @param initialName the CPU name from {@code sysctl hw.model}
257      * @return the parsed identifier fields
258      */
259     static DmesgProcessorId parseProcessorIdFromDmesg(List<String> dmesg, String initialName) {
260         final Pattern featuresPattern = Pattern.compile("Features=(\\S+)<.*");
261 
262         String cpuVendor = "";
263         String cpuName = initialName;
264         String cpuFamily = "";
265         String cpuModel = "";
266         String cpuStepping = "";
267         long processorIdBits = 0L;
268         for (String line : dmesg) {
269             line = line.trim();
270             // Prefer hw.model to this one
271             if (line.startsWith("CPU:") && cpuName.isEmpty()) {
272                 cpuName = line.replace("CPU:", "").trim();
273             } else if (ORIGIN.matcher(line).matches()) {
274                 Map<String, String> fields = new HashMap<>();
275                 Matcher m = DMESG_FIELD.matcher(line);
276                 while (m.find()) {
277                     fields.put(m.group(1), m.group(2));
278                 }
279                 cpuVendor = ParseUtil.getStringBetween(fields.getOrDefault("Origin", ""), '"');
280                 int id = ParseUtil.decodeIntOrDefault(fields.getOrDefault("Id", ""), 0);
281                 processorIdBits |= ParseUtil.unsignedIntToLong(id);
282                 Triplet<Integer, Integer, Integer> fms = cpuidToFamilyModelStepping(id);
283                 cpuFamily = String.valueOf(ParseUtil.decodeIntOrDefault(fields.get("Family"), fms.getA()));
284                 cpuModel = String.valueOf(ParseUtil.decodeIntOrDefault(fields.get("Model"), fms.getB()));
285                 cpuStepping = String.valueOf(ParseUtil.decodeIntOrDefault(fields.get("Stepping"), fms.getC()));
286             } else if (line.startsWith("Features=")) {
287                 Matcher m = featuresPattern.matcher(line);
288                 if (m.matches()) {
289                     processorIdBits |= ParseUtil.decodeLongOrDefault(m.group(1), 0L) << 32;
290                 }
291                 // No further interest in this file
292                 break;
293             }
294         }
295         return new DmesgProcessorId(cpuVendor, cpuName, cpuFamily, cpuModel, cpuStepping, processorIdBits);
296     }

299     protected Quartet<List<LogicalProcessor>, @Nullable List<PhysicalProcessor>, @Nullable List<ProcessorCache>, List<String>> initProcessorCounts() {
300         List<LogicalProcessor> logProcs = parseTopology();
301         // Force at least one processor
302         if (logProcs.isEmpty()) {
303             logProcs.add(new LogicalProcessor(0, 0, 0));
304         }
305         Pair<Map<Integer, String>, List<String>> modelsAndFlags = parseProcModelsAndFlags(
306                 FileUtil.readFile("/var/run/dmesg.boot", false));
307         Map<Integer, String> dmesg = modelsAndFlags.getA();
308         List<String> featureFlags = modelsAndFlags.getB();
309         List<PhysicalProcessor> physProcs = dmesg.isEmpty() ? null : createProcListFromDmesg(logProcs, dmesg);
310         List<ProcessorCache> caches = getCacheInfoFromLscpu();
311         return new Quartet<>(logProcs, physProcs, caches, featureFlags);
312     }

314     /**
315      * Parses the per-core processor model strings and feature flags from {@code dmesg.boot}. Package-private for
316      * testing.
317      *
318      * @param dmesg the lines of {@code dmesg.boot}
319      * @return a pair of (core id to model-string map, feature-flag lines)
320      */
321     static Pair<Map<Integer, String>, List<String>> parseProcModelsAndFlags(List<String> dmesg) {
322         Map<Integer, String> models = new HashMap<>();
323         // cpu0: <Open Firmware CPU> on cpulist0
324         Pattern normal = Pattern.compile("cpu(\\d+): (.+) on .*");
325         // CPU 0: ARM Cortex-A53 r0p4 affinity: 0 0
326         Pattern hybrid = Pattern.compile("CPU\\s*(\\d+): (.+) affinity:.*");
327         List<String> featureFlags = new ArrayList<>();
328         boolean readingFlags = false;
329         for (String s : dmesg) {
330             Matcher h = hybrid.matcher(s);
331             if (h.matches()) {
332                 int coreId = ParseUtil.parseIntOrDefault(h.group(1), 0);
333                 // This always takes priority, overwrite if needed
334                 models.put(coreId, h.group(2).trim());
335             } else {
336                 Matcher n = normal.matcher(s);
337                 if (n.matches()) {
338                     int coreId = ParseUtil.parseIntOrDefault(n.group(1), 0);
339                     // Don't overwrite if h matched earlier
340                     models.putIfAbsent(coreId, n.group(2).trim());
341                 }
342             }
343             if (ORIGIN.matcher(s.trim()).matches()) {
344                 readingFlags = true;
345             } else if (readingFlags) {
346                 if (s.startsWith("  ")) {
347                     featureFlags.add(s.trim());
348                 } else {
349                     readingFlags = false;
350                 }
351             }
352         }
353         return new Pair<>(models, featureFlags);
354     }

360     /**
361      * Parses the CPU cache descriptions from {@code lscpu} output. Package-private for testing.
362      *
363      * @param lscpu the lines of {@code lscpu} output
364      * @return the ordered list of processor caches
365      */
366     static List<ProcessorCache> parseCachesFromLscpu(List<String> lscpu) {
367         Set<ProcessorCache> caches = new HashSet<>();
368         for (String checkLine : lscpu) {
369             if (checkLine.contains("L1d cache:")) {
370                 caches.add(new ProcessorCache(1, 0, 0,
371                         ParseUtil.parseDecimalMemorySizeToBinary(checkLine.split(":", -1)[1].trim()),
372                         ProcessorCache.Type.DATA));
373             } else if (checkLine.contains("L1i cache:")) {
374                 caches.add(new ProcessorCache(1, 0, 0,
375                         ParseUtil.parseDecimalMemorySizeToBinary(checkLine.split(":", -1)[1].trim()),
376                         ProcessorCache.Type.INSTRUCTION));
377             } else if (checkLine.contains("L2 cache:")) {
378                 caches.add(new ProcessorCache(2, 0, 0,
379                         ParseUtil.parseDecimalMemorySizeToBinary(checkLine.split(":", -1)[1].trim()),
380                         ProcessorCache.Type.UNIFIED));
381             } else if (checkLine.contains("L3 cache:")) {
382                 caches.add(new ProcessorCache(3, 0, 0,
383                         ParseUtil.parseDecimalMemorySizeToBinary(checkLine.split(":", -1)[1].trim()),
384                         ProcessorCache.Type.UNIFIED));
385             }
386         }
387         return orderedProcCaches(caches);
388     }

390     private List<LogicalProcessor> parseTopology() {
391         String spec = sysctlString("kern.sched.topology_spec", "");
392         if (spec.isEmpty()) {
393             // DragonFly BSD has no topology_spec, and publishes each CPU's place under hw.cpu_topology instead
394             int ncpu = sysctlInt("hw.ncpu", 0);
395             int[] physicalIds = new int[ncpu];
396             int[] coreIds = new int[ncpu];
397             for (int i = 0; i < ncpu; i++) {
398                 physicalIds[i] = sysctlInt("hw.cpu_topology.cpu" + i + ".physical_id", 0);
399                 coreIds[i] = sysctlInt("hw.cpu_topology.cpu" + i + ".core_id", -1);
400             }
401             return buildTopologyFromIds(physicalIds, coreIds);
402         }
403         return parseTopologySpec(spec);
404     }

406     /**
407      * Builds the logical processors from {@code kern.sched.topology_spec}. Package-private for testing.
408      * <p>
409      * The spec nests groups of CPUs that share a cache. A group flagged {@code THREAD}, {@code SMT} or {@code HTT}
410      * holds the hardware threads of one core, and a CPU in no such group is a core of its own; virtual machines often
411      * publish a single flat group, so every CPU is its own core. The groups at level 2 that are not thread groups are
412      * taken as packages.
413      *
414      * <pre>{@code
415      * <groups>
416      *  <group level="1" cache-level="0">
417      *   <cpu count="24" mask="ffffff">0, 1, 2, ..., 23</cpu>
418      *   <children>
419      *    <group level="2" cache-level="2">
420      *     <cpu count="12" mask="fff">0, 1, ..., 11</cpu>
421      *     <children>
422      *      <group level="3" cache-level="1">
423      *       <cpu count="2" mask="3">0, 1</cpu>
424      *       <flags><flag name="THREAD">THREAD group</flag><flag name="SMT">SMT group</flag></flags>
425      *      </group>
426      * }</pre>
427      *
428      * @param spec the value of {@code kern.sched.topology_spec}
429      * @return the logical processors, in CPU order
430      */
431     static List<LogicalProcessor> parseTopologySpec(String spec) {
432         List<TopologyGroup> groups = new ArrayList<>();
433         Deque<TopologyGroup> open = new ArrayDeque<>();
434         for (String line : spec.split("[\\n\\r]", -1)) {
435             if (line.contains("<group level=")) {
436                 TopologyGroup group = new TopologyGroup(open.size() + 1);
437                 groups.add(group);
438                 open.push(group);
439             } else if (line.contains("</group>")) {
440                 open.poll();
441             } else if (!open.isEmpty()) {
442                 TopologyGroup group = open.element();
443                 if (line.contains("<cpu")) {
444                     group.cpus.addAll(parseCpuList(line));
445                 } else if (line.contains("<flag name=\"THREAD\"") || line.contains("<flag name=\"SMT\"")
446                         || line.contains("<flag name=\"HTT\"")) {
447                     // The kernel prints THREAD alongside either of the others, but any of them marks a core
448                     group.thread = true;
449                 }
450             }
451         }
452         if (groups.isEmpty()) {
453             return new ArrayList<>();
454         }
455         // The CPUs of each core, keyed by the core's lowest CPU, and each CPU's package
456         Map<Integer, Integer> coreOf = new HashMap<>();
457         Map<Integer, Integer> packageOf = new HashMap<>();
458         int packageNumber = 0;
459         for (TopologyGroup group : groups) {
460             if (group.thread && !group.cpus.isEmpty()) {
461                 int first = Collections.min(group.cpus);
462                 for (int cpu : group.cpus) {
463                     coreOf.put(cpu, first);
464                 }
465             } else if (group.level == 2) {
466                 for (int cpu : group.cpus) {
467                     packageOf.put(cpu, packageNumber);
468                 }
469                 packageNumber++;
470             }
471         }
472         List<LogicalProcessor> logProcs = new ArrayList<>();
473         Map<Integer, Integer> coreNumbers = new HashMap<>();
474         for (int cpu : new TreeSet<>(groups.get(0).cpus)) {
475             int core = coreNumbers.computeIfAbsent(coreOf.getOrDefault(cpu, cpu), k -> coreNumbers.size());
476             logProcs.add(new LogicalProcessor(cpu, core, packageOf.getOrDefault(cpu, 0), 0));
477         }
478         return logProcs;
479     }

481     /**
482      * Reads the CPU numbers of a {@code <cpu>} line, from the list it prints or, failing that, from its mask. The mask
483      * is a comma-separated list of 64-bit hex words, lowest CPUs first.
484      */
485     private static List<Integer> parseCpuList(String line) {
486         List<Integer> cpus = new ArrayList<>();
487         int open = line.indexOf('>');
488         int close = line.indexOf("</cpu>");
489         if (open >= 0 && close > open) {
490             for (String cpu : line.substring(open + 1, close).split(",", -1)) {
491                 int n = ParseUtil.parseIntOrDefault(cpu.trim(), -1);
492                 if (n >= 0) {
493                     cpus.add(n);
494                 }
495             }
496         }
497         if (cpus.isEmpty()) {
498             Matcher m = CPUMASK.matcher(line);
499             if (m.find()) {
500                 String[] words = m.group(1).split(",", -1);
501                 for (int w = 0; w < words.length; w++) {
502                     long bits = ParseUtil.hexStringToLong(words[w], 0L);
503                     for (int b = 0; b < Long.SIZE; b++) {
504                         if ((bits & (1L << b)) != 0) {
505                             cpus.add(w * Long.SIZE + b);
506                         }
507                     }
508                 }
509             }
510         }
511         return cpus;
512     }

525     /**
526      * Builds the logical processors from each CPU's package and core IDs, as DragonFly BSD publishes them in
527      * {@code hw.cpu_topology.cpu<n>.physical_id} and {@code core_id}. Package-private for testing.
528      * <p>
529      * A core ID is only unique within its package, so cores are numbered across the system in order of first
530      * appearance. DragonFly BSD omits {@code core_id} when each chip has one core, in which case every CPU is its own
531      * core.
532      *
533      * @param physicalIds each CPU's package ID
534      * @param coreIds     each CPU's core ID within its package, or -1 where it has none
535      * @return the logical processors, in CPU order
536      */
537     static List<LogicalProcessor> buildTopologyFromIds(int[] physicalIds, int[] coreIds) {
538         List<LogicalProcessor> logProcs = new ArrayList<>();
539         Map<String, Integer> coreNumbers = new HashMap<>();
540         for (int i = 0; i < physicalIds.length; i++) {
541             String core = physicalIds[i] + ":" + (coreIds[i] < 0 ? "cpu" + i : coreIds[i]);
542             Integer coreNumber = coreNumbers.computeIfAbsent(core, k -> coreNumbers.size());
543             logProcs.add(new LogicalProcessor(i, coreNumber, physicalIds[i], 0));
544         }
545         return logProcs;
546     }

548     @Override
549     public long[] queryCurrentFreq() {
550         long[] freq = new long[1];
551         freq[0] = sysctlLong("dev.cpu.0.freq", -1L);
552         if (freq[0] > 0) {
553             // If success, value is in MHz
554             freq[0] *= 1_000_000L;
555         } else {
556             freq[0] = sysctlLong("machdep.tsc_freq", -1L);
557         }
558         return freq;
559     }

561     @Override
562     public long queryMaxFreq() {
563         long max = -1L;
564         String freqLevels = sysctlString("dev.cpu.0.freq_levels", "");
565         // MHz/Watts pairs like: 2501/32000 2187/27125 2000/24000
566         for (String s : ParseUtil.whitespaces.split(freqLevels, -1)) {
567             long freq = ParseUtil.parseLongOrDefault(s.split("/", -1)[0], -1L);
568             if (max < freq) {
569                 max = freq;
570             }
571         }
572         if (max > 0) {
573             // If success, value is in MHz
574             max *= 1_000_000;
575         } else {
576             max = sysctlLong("machdep.tsc_freq", -1L);
577         }
578         return max;
579     }

581     /**
582      * Fetches the ProcessorID from dmidecode (if possible with root permissions), otherwise uses the values from
583      * /var/run/dmesg.boot
584      *
585      * @param processorID The processorID as a long
586      * @return The ProcessorID string
587      */
588     protected static String getProcessorIDfromDmiDecode(long processorID) {
589         boolean procInfo = false;
590         String marker = "Processor Information";
591         for (String checkLine : ExecutingCommand.runNative("dmidecode -t processor")) {
592             if (!procInfo && checkLine.contains(marker)) {
593                 marker = "ID:";
594                 procInfo = true;
595             } else if (procInfo && checkLine.contains(marker)) {
596                 return ParseUtil.getTextAfterString(checkLine, marker).trim();
597             }
598         }
599         // If we've gotten this far, dmidecode failed. Used the passed-in values
600         return String.format(Locale.ROOT, "%016X", processorID);
601     }
