35      // A devinfo -v line: its indent, driver instance, optional description, and then its plug-and-play and location
36      // fields, after "pnpinfo" and "at" respectively
37      private static final Pattern DEVINFO_LINE = Pattern.compile("( *)(\\S+)(?: <(.*?)>)?(.*)");
38      // A plug-and-play or location field, its value quoted where it may hold spaces: sernum="1234BBBK5678" ugen=ugen0.2
39      // The lookbehind stops find() from retrying inside a name, so the scan stays linear
40      private static final String PNP_NAME = "(?<!\\w)(\\w++)=";
41      private static final String PNP_VALUE = "(\"[^\"]*+\"|\\S++)";
42      private static final Pattern PNP_FIELD = Pattern.compile(PNP_NAME + PNP_VALUE); // NOSONAR java:S8786 - linear

69      /**
70       * Parses {@code devinfo -v} output into the USB controller device tree. Package-private for testing.
71       * <p>
72       * {@code devinfo} prints the device tree by indentation, each line a driver instance with its description in angle
73       * brackets and its plug-and-play fields after {@code pnpinfo}. A controller is the parent of a {@code usbus} node,
74       * whose children, the root hub and what hangs from it, become the controller's own. A USB device is listed once per
75       * interface driver attached to it, each with the same {@code ugen} name among its location fields, so that name
76       * identifies the device. A driver attached beneath a USB device without a {@code ugen} name of its own, such as a
77       * HID bus, is not a USB device and is left out. Only devices with a driver attached appear. DragonFly, which shares
78       * this class, prints no descriptions, so there a device is named by its driver instance.
79       *
80       * @param devinfo the lines emitted by {@code devinfo -v}
81       * @return a list of USB controllers, each with its connected-device tree.
82       */
83      static List<UsbDevice> parseUsbDevices(List<String> devinfo) {
84          Map<String, String> nameMap = new HashMap<>();
85          Map<String, String> vendorIdMap = new HashMap<>();
86          Map<String, String> productIdMap = new HashMap<>();
87          Map<String, String> serialMap = new HashMap<>();
88          Map<String, String> parentMap = new HashMap<>();
89          Map<String, List<String>> hubMap = new HashMap<>();
90          List<String> usBuses = new ArrayList<>();
91          // The open ancestors of the current line, innermost first, as indent and key
92          Deque<Pair<Integer, String>> ancestors = new ArrayDeque<>();
93          // The root hubs and USB devices seen, and the drivers beneath them that are neither, such as a HID bus
94          Set<String> usbNodes = new HashSet<>();
95          Set<String> notUsb = new HashSet<>();
96          for (String line : devinfo) {
97              Matcher m = DEVINFO_LINE.matcher(line);
98              if (!m.matches()) {
99                  continue;
100             }
101             int indent = m.group(1).length();
102             Pair<Integer, String> ancestor = ancestors.peek();
103             while (ancestor != null && ancestor.getA() >= indent) {
104                 ancestors.pop();
105                 ancestor = ancestors.peek();
106             }
107             String parent = ancestor == null ? null : ancestor.getB();
108             // The ugen name is among the location fields, the IDs among the plug-and-play ones
109             Map<String, String> pnp = parseFields(m.group(4));
110             String ugen = pnp.get("ugen");
111             String key = ugen == null ? m.group(2) : ugen;
112             if (ugen == null && nameMap.containsKey(key)) {
113                 // A name devinfo repeats, such as "unknown" for each device without a driver
114                 key = key + "#" + nameMap.size();
115             }
116             ancestors.push(new Pair<>(indent, key));
117             if (parent != null && ugen == null && (usbNodes.contains(parent) || notUsb.contains(parent))) {
118                 // A child driver of a USB device that is not itself one, such as the HID bus beneath a tablet
119                 notUsb.add(key);
120                 continue;
121             }
122             if (ugen != null || (parent != null && parent.startsWith("usbus"))) {
123                 usbNodes.add(key);
124             }
125             if (nameMap.containsKey(key)) {
126                 // Another interface driver of a device already listed
127                 continue;
128             }
129             if (key.startsWith("usbus")) {
130                 usBuses.add(key);
131             }
132             String description = ParseUtil.getStringValueOrEmpty(m.group(3));
133             int usbClass = description.indexOf(", class ");
134             if (description.isEmpty()) {
135                 // DragonFly's devinfo prints no descriptions, so the driver instance (uhci0, uhub0) names the device
136                 nameMap.put(key, m.group(2));
137             } else {
138                 nameMap.put(key, usbClass < 0 ? description : description.substring(0, usbClass));
139             }
140             putHexId(vendorIdMap, key, pnp.get("vendor"));
141             putHexId(productIdMap, key, pnp.get(pnp.containsKey("product") ? "product" : "device"));
142             String serial = pnp.get("sernum");
143             if (serial != null && !serial.isEmpty()) {
144                 serialMap.put(key, serial);
145             }
146             if (parent != null) {
147                 parentMap.put(key, parent);
148                 hubMap.computeIfAbsent(parent, x -> new ArrayList<>()).add(key);
149             }
150         }
151 
152         // Skip the usbuses: make their parents the controllers and replace parents' children with the buses' children
153         List<UsbDevice> controllerDevices = new ArrayList<>();
154         for (String usbus : usBuses) {
155             String parent = parentMap.get(usbus);
156             if (parent == null) {
157                 continue;
158             }
159             hubMap.put(parent, new ArrayList<>(hubMap.getOrDefault(usbus, Collections.emptyList())));
160             controllerDevices.add(buildDeviceTree(parent, "0000", "0000", nameMap, Collections.emptyMap(), vendorIdMap,
161                     productIdMap, serialMap, hubMap, FreeBsdUsbDevice::new));
162         }
163         return Collections.unmodifiableList(controllerDevices);
164     }

166     private static Map<String, String> parseFields(String fields) {
167         Map<String, String> pnp = new HashMap<>();
168         Matcher m = PNP_FIELD.matcher(fields);
169         while (m.find()) {
170             String value = m.group(2);
171             if (value.length() >= 2 && value.charAt(0) == '"' && value.charAt(value.length() - 1) == '"') {
172                 value = value.substring(1, value.length() - 1);
173             }
174             pnp.put(m.group(1), value);
175         }
176         return pnp;
177     }
178 
179     private static void putHexId(Map<String, String> map, String key, @Nullable String hex) {
180         if (hex != null && hex.startsWith("0x")) {
181             map.put(key, String.format(Locale.ROOT, "%04x", ParseUtil.hexStringToInt(hex.substring(2), 0)));
182         }
183     }
