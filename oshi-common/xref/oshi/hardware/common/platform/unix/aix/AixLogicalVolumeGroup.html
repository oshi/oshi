21  /**
22   * An AIX Logical Volume Manager volume group, read with {@code lsvg} and {@code lspv}.
23   */
24  @Immutable
25  public final class AixLogicalVolumeGroup extends AbstractLogicalVolumeGroup {
26  
27      private AixLogicalVolumeGroup(String name, Map<String, Set<String>> lvMap, Set<String> pvSet) {
28          super(name, lvMap, pvSet);
29      }
30  
31      /**
32       * Gets the active volume groups. None of the commands needs privilege.
33       *
34       * @return the volume groups that are varied on
35       */
36      public static List<LogicalVolumeGroup> getLogicalVolumeGroups() {
37          return queryLogicalVolumeGroups(ExecutingCommand::runNative);
38      }
39  
40      /**
41       * Reads the active volume groups through the given command runner. Package-private for testing.
42       * <p>
43       * {@code lsvg -o} names the volume groups that are varied on, and {@code lsvg -p} and {@code lsvg -l}, given all of
44       * them at once, list each one's physical and logical volumes. {@code -L} keeps those from waiting on a volume
45       * group's lock, which another LVM command may hold for as long as it runs, at the cost of a listing that may be
46       * incomplete while that volume group is being changed. Neither says which physical volumes hold a logical volume,
47       * so in a volume group of more than one, {@code lspv -l} lists the logical volumes on each physical volume; in a
48       * volume group of one, every logical volume is on it. Physical volumes are named as {@code getDiskStores()} names
49       * them, as in {@code hdisk0}.
50       *
51       * @param run runs a command and returns the lines it prints
52       * @return the volume groups that are varied on
53       */
54      static List<LogicalVolumeGroup> queryLogicalVolumeGroups(Function<String, List<String>> run) {
55          List<String> vgs = new ArrayList<>();
56          for (String line : run.apply("lsvg -o")) {
57              String vg = line.trim();
58              if (!vg.isEmpty()) {
59                  vgs.add(vg);
60              }
61          }
62          if (vgs.isEmpty()) {
63              return new ArrayList<>();
64          }
65          String vgList = String.join(" ", vgs);
66          Map<String, Set<String>> pvsByVg = parseSections(run.apply("lsvg -L -p " + vgList), "PV_NAME");
67          Map<String, Set<String>> lvsByVg = parseSections(run.apply("lsvg -L -l " + vgList), "LV NAME");
68          List<LogicalVolumeGroup> groups = new ArrayList<>();
69          for (String vg : vgs) {
70              Set<String> pvs = pvsByVg.getOrDefault(vg, new LinkedHashSet<>());
71              Map<String, Set<String>> lvMap = new LinkedHashMap<>();
72              for (String lv : lvsByVg.getOrDefault(vg, new LinkedHashSet<>())) {
73                  lvMap.put(lv, pvs.size() == 1 ? new LinkedHashSet<>(pvs) : new LinkedHashSet<>());
74              }
75              if (pvs.size() > 1) {
76                  for (String pv : pvs) {
77                      for (Set<String> lvs : parseSections(run.apply("lspv -l " + pv), "LV NAME").values()) {
78                          for (String lv : lvs) {
79                              Set<String> onPvs = lvMap.get(lv);
80                              if (onPvs != null) {
81                                  onPvs.add(pv);
82                              }
83                          }
84                      }
85                  }
86              }
87              groups.add(new AixLogicalVolumeGroup(vg, lvMap, pvs));
88          }
89          return groups;
90      }

92      /**
93       * Parses the output of {@code lsvg -p}, {@code lsvg -l} or {@code lspv -l}, which print a {@code name:} line for
94       * each volume group or physical volume asked about, then a heading row, then one row per volume whose first column
95       * is its name.
96       *
97       * @param lines   the lines the command printed
98       * @param heading the start of the heading row, which is skipped
99       * @return the names in the first column, keyed by the section they appear under
100      */
101     static Map<String, Set<String>> parseSections(List<String> lines, String heading) {
102         Map<String, Set<String>> sections = new LinkedHashMap<>();
103         Set<String> current = null;
104         for (String line : lines) {
105             String trimmed = line.trim();
106             if (trimmed.isEmpty()) {
107                 continue;
108             }
109             if (trimmed.endsWith(":") && !trimmed.contains(" ")) {
110                 current = sections.computeIfAbsent(trimmed.substring(0, trimmed.length() - 1),
111                         k -> new LinkedHashSet<>());
112             } else if (current != null && !isHeading(trimmed, heading)) {
113                 current.add(ParseUtil.whitespaces.split(trimmed, 2)[0]);
114             }
115         }
116         return sections;
117     }
118 
119     /**
120      * Whether a row is the column heading: one that starts with the heading's words followed by whitespace, so a volume
121      * whose name merely begins with them, such as {@code PV_NAMEdata}, is not mistaken for it.
122      *
123      * @param row     a trimmed row
124      * @param heading the heading's first words
125      * @return whether the row is the heading
126      */
127     private static boolean isHeading(String row, String heading) {
128         return row.startsWith(heading)
129                 && (row.length() == heading.length() || Character.isWhitespace(row.charAt(heading.length())));
130     }
