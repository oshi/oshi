20  /**
21   * CUPS-based printer implementation with shared lpstat fallback logic. Subclasses provide native CUPS access via JNA or
22   * FFM.
23   */
24  @Immutable
25  public abstract class CupsPrinter extends AbstractPrinter {
26  
27      /**
28       * Creates a CupsPrinter with the given parameters.
29       *
30       * @param name         the printer name
31       * @param driverName   the driver name
32       * @param description  the description
33       * @param status       the printer status
34       * @param statusReason the status reason
35       * @param isDefault    whether this is the default printer
36       * @param isLocal      whether this is a local printer
37       * @param portName     the port name
38       */
39      protected CupsPrinter(String name, String driverName, String description, PrinterStatus status, String statusReason,
40              boolean isDefault, boolean isLocal, String portName) {
41          super(name, driverName, description, status, statusReason, isDefault, isLocal, portName);
42      }
43  
44      /** CUPS {@code ipp_pstate_t} value for an idle printer. */
45      private static final int IPP_PRINTER_IDLE = 3;
46      /** CUPS {@code ipp_pstate_t} value for a printer processing a job. */
47      private static final int IPP_PRINTER_PROCESSING = 4;
48      /** CUPS {@code ipp_pstate_t} value for a stopped printer. */
49      private static final int IPP_PRINTER_STOPPED = 5;
50  
51      /**
52       * Maps a CUPS printer state and its state reasons to a {@link PrinterStatus}. An error or fault in the reasons
53       * takes precedence over the numeric state, matching the CUPS {@code ipp_pstate_t} values.
54       *
55       * @param state        the CUPS {@code printer-state} value as a string, or empty if unavailable
56       * @param stateReasons the CUPS {@code printer-state-reasons} value, or empty/{@code "none"} if there are none
57       * @return the mapped {@link PrinterStatus}, or {@link PrinterStatus#UNKNOWN} if the state is empty or unrecognized
58       */
59      protected static PrinterStatus parseStateFromCups(String state, String stateReasons) {
60          if (!stateReasons.isEmpty() && !"none".equals(stateReasons)) {
61              String lower = stateReasons.toLowerCase(Locale.ROOT);
62              if (lower.contains("error") || lower.contains("fault")) {
63                  return PrinterStatus.ERROR;
64              }
65          }
66          if (state.isEmpty()) {
67              return PrinterStatus.UNKNOWN;
68          }
69          switch (ParseUtil.parseIntOrDefault(state, -1)) {
70              case IPP_PRINTER_IDLE:
71                  return PrinterStatus.IDLE;
72              case IPP_PRINTER_PROCESSING:
73                  return PrinterStatus.PRINTING;
74              case IPP_PRINTER_STOPPED:
75                  return PrinterStatus.OFFLINE;
76              default:
77                  return PrinterStatus.UNKNOWN;
78          }
79      }
80  
81      /**
82       * Gets printers by parsing lpstat command output. Used as a fallback when libcups is unavailable.
83       *
84       * @param factory function to create concrete printer instances
85       * @return list of printers
86       */
87      protected static List<Printer> getPrintersFromLpstat(PrinterFactory factory) {
88          return getPrintersFromLpstat(ExecutingCommand.runNative(new String[] { "lpstat", "-p" }), factory);
89      }
90  
91      /**
92       * Parse lpstat -p output to build a list of printers.
93       *
94       * @param lpstatLines output of {@code lpstat -p}
95       * @param factory     function to create concrete printer instances
96       * @return list of printers
97       */
98      static List<Printer> getPrintersFromLpstat(List<String> lpstatLines, PrinterFactory factory) {
99          return getPrintersFromLpstat(lpstatLines, Lpstat.queryDefaultPrinter(), Lpstat.queryPortMap(),
100                 Lpstat.queryDescriptionMap(), Lpstat::queryDriver, factory);
101     }
102 
103     /**
104      * Parse lpstat -p output to build a list of printers with pre-fetched data.
105      *
106      * @param lpstatLines    output of {@code lpstat -p}
107      * @param defaultPrinter the default printer name
108      * @param portMap        map of printer name to device URI
109      * @param descriptionMap map of printer name to description
110      * @param driverLookup   function to look up driver name for a printer
111      * @param factory        function to create concrete printer instances
112      * @return list of printers
113      */
114     static List<Printer> getPrintersFromLpstat(List<String> lpstatLines, String defaultPrinter,
115             Map<String, String> portMap, Map<String, String> descriptionMap, UnaryOperator<String> driverLookup,
116             PrinterFactory factory) {
117         List<Printer> printers = new ArrayList<>();
118 
119         for (String line : lpstatLines) {
120             if (line.startsWith("printer ")) {
121                 String[] parts = ParseUtil.whitespaces.split(line, -1);
122                 if (parts.length >= 3) {
123                     String name = parts[1];
124                     PrinterStatus status = Lpstat.parseStatus(line);
125                     boolean isDefault = name.equals(defaultPrinter);
126                     String portName = portMap.getOrDefault(name, "");
127                     boolean isLocal = Lpstat.isLocalUri(portName);
128                     String driverName = driverLookup.apply(name);
129                     String description = descriptionMap.getOrDefault(name, "");
130                     String statusReason = Lpstat.parseStatusReason(line);
131 
132                     printers.add(factory.create(name, driverName, description, status, statusReason, isDefault, isLocal,
133                             portName));
134                 }
135             }
136         }
137         return printers;
138     }

140     /**
141      * Factory interface for creating concrete printer instances.
142      */
143     @FunctionalInterface
144     public interface PrinterFactory {
145         /**
146          * Create a printer instance.
147          *
148          * @param name         printer name
149          * @param driverName   driver name
150          * @param description  description
151          * @param status       status
152          * @param statusReason status reason
153          * @param isDefault    whether this is the default printer
154          * @param isLocal      whether this is a local printer
155          * @param portName     port/URI
156          * @return a new Printer instance
157          */
158         Printer create(String name, String driverName, String description, PrinterStatus status, String statusReason,
159                 boolean isDefault, boolean isLocal, String portName);
160     }
