9   import java.util.List;
10  import java.util.function.Supplier;
11  import java.util.regex.Pattern;
12  
13  import org.jspecify.annotations.Nullable;
14  
15  import oshi.annotation.concurrent.Immutable;
16  import oshi.hardware.common.AbstractFirmware;
17  import oshi.util.Constants;
18  import oshi.util.ExecutingCommand;
19  import oshi.util.ParseUtil;
20  import oshi.util.tuples.Triplet;
21  
22  /**
23   * Common BSD Firmware implementation. The BSDs report the same three firmware attributes, memoized from a single
24   * command, and differ only in which command supplies them: most read the boot-time {@code dmesg} banner, while FreeBSD
25   * uses {@code dmidecode}. Name and description are not available from any of those sources, so both keep the
26   * {@link AbstractFirmware} defaults.
27   */
28  @Immutable
29  public abstract class BsdFirmware extends AbstractFirmware {
30      private static final Pattern VENDOR = Pattern.compile("vendor");
31  
32      private final Supplier<Triplet<@Nullable String, @Nullable String, @Nullable String>> manufVersRelease = memoize(
33              this::queryFirmware);
34  
35      /**
36       * Default constructor.
37       */
38      protected BsdFirmware() {
39      }
40  
41      @Override
42      public String getManufacturer() {
43          return ParseUtil.getStringValueOrUnknown(manufVersRelease.get().getA());
44      }
45  
46      @Override
47      public String getVersion() {
48          return ParseUtil.getStringValueOrUnknown(manufVersRelease.get().getB());
49      }
50  
51      @Override
52      public String getReleaseDate() {
53          return ParseUtil.getStringValueOrUnknown(manufVersRelease.get().getC());
54      }
55  
56      /**
57       * Reads the platform's firmware attributes. Any attribute the platform cannot supply may be returned as null or
58       * blank; the caller substitutes {@link Constants#UNKNOWN}.
59       * <p>
60       * Defaults to the boot-time {@code dmesg} banner, which is what most of the BSDs report firmware through. FreeBSD
61       * overrides this because it reads {@code dmidecode} instead.
62       *
63       * @return a {@link Triplet} of manufacturer, version, and release date
64       */
65      protected Triplet<@Nullable String, @Nullable String, @Nullable String> readFirmware() {
66          return parseDmesg(ExecutingCommand.runNative("dmesg"));
67      }
68  
69      private Triplet<@Nullable String, @Nullable String, @Nullable String> queryFirmware() {
70          Triplet<@Nullable String, @Nullable String, @Nullable String> dmi = readFirmware();
71          return new Triplet<>(unknownIfBlank(dmi.getA()), unknownIfBlank(dmi.getB()), unknownIfBlank(dmi.getC()));
72      }
73  
74      private static String unknownIfBlank(@Nullable String value) {
75          return ParseUtil.getStringValueOrUnknown(value);
76      }
77  
78      /**
79       * Parses the output of {@code dmesg} for BIOS firmware information. Looks for lines starting with
80       * {@code "bios0: vendor"} to extract manufacturer, version, and release date. Any field not present in the output
81       * is returned as {@code null} (or an empty date); the caller applies fallbacks.
82       *
83       * @param dmesg the lines emitted by {@code dmesg}
84       * @return a {@link Triplet} of vendor, version, and release date
85       */
86      static Triplet<@Nullable String, @Nullable String, @Nullable String> parseDmesg(List<String> dmesg) {
87          String version = null;
88          String vendor = null;
89          String releaseDate = "";
90  
91          // bios0 at mainbus0: SMBIOS rev. 2.7 @ 0xdcc0e000 (67 entries)
92          // bios0: vendor LENOVO version "GLET90WW (2.44 )" date 09/13/2017
93          // bios0: LENOVO 20AWA08J00
94          for (String line : dmesg) {
95              if (line.startsWith("bios0: vendor")) {
96                  version = ParseUtil.getStringBetween(line, '"');
97                  releaseDate = ParseUtil.parseMmDdYyyyToYyyyMmDD(ParseUtil.parseLastString(line));
98                  String afterVendor = VENDOR.split(line, -1)[1].trim();
99                  int versionIdx = afterVendor.indexOf(" version ");
100                 vendor = versionIdx > 0 ? afterVendor.substring(0, versionIdx)
101                         : ParseUtil.whitespaces.split(afterVendor, -1)[0];
102             }
103         }
104         return new Triplet<>(vendor, version, releaseDate);
105     }
106 }
