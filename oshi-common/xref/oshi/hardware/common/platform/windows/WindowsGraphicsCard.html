27  /**
28   * Common logic for a Windows {@link GraphicsCard}, including the WMI fallback enumeration used when the display device
29   * registry keys yield nothing.
30   * <p>
31   * Both bindings enumerate cards identically once their driver has returned the WMI rows; only which driver is called
32   * differs, so {@link #buildFromWmi} takes the query result and a factory rather than performing the query itself.
33   */
34  @ThreadSafe
35  public abstract class WindowsGraphicsCard extends AbstractGraphicsCard {
36  
37      /** PDH instance prefix for this adapter's LUID, e.g. {@code luid_0x00000000_0x0001234_phys_0}. */
38      private final String luidPrefix;
39  
40      /** LHM hardware identifier for this GPU, e.g. {@code /gpu-nvidia/0}. Empty if LHM is not available. */
41      private final String lhmParent;
42  
43      /** PCI bus number from DXGI, used to correlate with ADL. -1 if unknown. */
44      private final int pciBusNumber;
45  
46      /** PCI bus ID string for NVML correlation, e.g. {@code 0000:01:00.0}. Empty if unknown. */
47      private final String pciBusId;
48  
49      /**
50       * Creates a Windows graphics card.
51       *
52       * @param name         the card name
53       * @param deviceId     the device ID
54       * @param vendor       the vendor
55       * @param versionInfo  the version info
56       * @param vram         the VRAM in bytes
57       * @param luidPrefix   PDH LUID instance prefix for this adapter, or an empty string if unknown
58       * @param lhmParent    LHM hardware identifier for this GPU, or an empty string if unavailable
59       * @param pciBusNumber PCI bus number for ADL correlation, or -1 if unknown
60       * @param pciBusId     PCI bus ID string for NVML correlation, or an empty string if unknown
61       */
62      protected WindowsGraphicsCard(String name, String deviceId, String vendor, String versionInfo, long vram,
63              String luidPrefix, String lhmParent, int pciBusNumber, String pciBusId) {
64          super(name, deviceId, vendor, versionInfo, vram);
65          this.luidPrefix = luidPrefix;
66          this.lhmParent = lhmParent;
67          this.pciBusNumber = pciBusNumber;
68          this.pciBusId = pciBusId;
69      }
70  
71      /**
72       * Returns this adapter's PDH instance prefix, used to filter GPU Engine and GPU Adapter Memory counters.
73       *
74       * @return the PDH LUID instance prefix, or an empty string if unknown
75       */
76      protected String getLuidPrefix() {
77          return luidPrefix;
78      }
79  
80      /**
81       * Returns this GPU's LibreHardwareMonitor hardware identifier, used to select its LHM sensors.
82       *
83       * @return the LHM hardware identifier, or an empty string if unavailable
84       */
85      protected String getLhmParent() {
86          return lhmParent;
87      }
88  
89      /**
90       * Returns this adapter's PCI bus number, used to correlate with ADL.
91       *
92       * @return the PCI bus number, or -1 if unknown
93       */
94      protected int getPciBusNumber() {
95          return pciBusNumber;
96      }
97  
98      /**
99       * Returns this adapter's PCI bus ID, used to correlate with NVML.
100      *
101      * @return the PCI bus ID string, or an empty string if unknown
102      */
103     protected String getPciBusId() {
104         return pciBusId;
105     }

107     /**
108      * Creates a backend's concrete card type.
109      */
110     @FunctionalInterface
111     public interface CardFactory {
112         /**
113          * @param name         the card name
114          * @param deviceId     the device ID
115          * @param vendor       the vendor
116          * @param versionInfo  the version info
117          * @param vram         the VRAM in bytes
118          * @param luidPrefix   PDH LUID instance prefix, or an empty string if unknown
119          * @param lhmParent    LHM hardware identifier, or an empty string if unavailable
120          * @param pciBusNumber PCI bus number, or -1 if unknown
121          * @param pciBusId     PCI bus ID string, or an empty string if unknown
122          * @return the card
123          */
124         GraphicsCard create(String name, String deviceId, String vendor, String versionInfo, long vram,
125                 String luidPrefix, String lhmParent, int pciBusNumber, String pciBusId);
126     }

128     /**
129      * Builds the card list from {@code Win32_VideoController} rows, used as a fallback when the display device registry
130      * keys yield nothing.
131      * <p>
132      * The returned list is ordered to match DXGI enumeration order, which places the primary desktop adapter first,
133      * with any unmatched cards appended.
134      *
135      * @param dxgiAdapters      all DXGI adapters, retained unmutated as the stable reference for ordering lookups
136      * @param lhmParentMap      normalized GPU name to LHM parent identifier
137      * @param cards             the {@code Win32_VideoController} query result
138      * @param luidPrefixBuilder derives the PDH LUID prefix for a matched adapter, falling back to a PDH instance lookup
139      * @param factory           creates the backend's card type
140      * @return the cards, never null
141      */
142     protected static List<GraphicsCard> buildFromWmi(List<DxgiAdapterInfo> dxgiAdapters,
143             Map<String, String> lhmParentMap, WmiResult<VideoControllerProperty> cards,
144             Function<DxgiAdapterInfo, String> luidPrefixBuilder, CardFactory factory) {
145         boolean dxgiAvailable = !dxgiAdapters.isEmpty();
146         // dxgiAdapters is not mutated; remainingDxgi is the working copy consumed during matching, and dxgiAdapters is
147         // retained as the stable reference for indexOf ordering lookups.
148         List<DxgiAdapterInfo> remainingDxgi = new ArrayList<>(dxgiAdapters);
149         TreeMap<Integer, GraphicsCard> dxgiOrdered = new TreeMap<>();
150         List<GraphicsCard> cardList = new ArrayList<>();
151 
152         for (int index = 0; index < cards.getResultCount(); index++) {
153             // ConfigManagerErrorCode 0 = working properly; non-zero = disabled/error (ghost device).
154             // When DXGI is unavailable, keep all entries for maximum compatibility.
155             if (dxgiAvailable && WmiUtil.getUint32(cards, VideoControllerProperty.CONFIGMANAGERERRORCODE, index) != 0) {
156                 continue;
157             }
158             String name = WmiUtil.getString(cards, VideoControllerProperty.NAME, index);
159             Triplet<String, String, String> idPair = ParseUtil.parseDeviceIdToVendorProductSerial(
160                     WmiUtil.getString(cards, VideoControllerProperty.PNPDEVICEID, index));
161             String deviceId = idPair == null ? Constants.UNKNOWN : idPair.getB();
162             String vendor = WmiUtil.getString(cards, VideoControllerProperty.ADAPTERCOMPATIBILITY, index);
163             if (idPair != null) {
164                 if (Util.isBlank(vendor)) {
165                     deviceId = idPair.getA();
166                 } else {
167                     vendor = vendor + " (" + idPair.getA() + ")";
168                 }
169             }
170             String versionInfo = WmiUtil.getString(cards, VideoControllerProperty.DRIVERVERSION, index);
171             versionInfo = Util.isBlank(versionInfo) ? Constants.UNKNOWN : "DriverVersion=" + versionInfo;
172 
173             // Prefer DXGI DedicatedVideoMemory when a match can be found via the PCI IDs extracted from PNPDEVICEID.
174             // Fall back to WMI AdapterRAM (32-bit capped) only when no DXGI match is available.
175             Pair<Integer, Integer> pciIds = ParseUtil.parseDeviceIdToVendorProductIds(
176                     WmiUtil.getString(cards, VideoControllerProperty.PNPDEVICEID, index));
177             int pciVendorId = pciIds == null ? 0 : pciIds.getA();
178             int pciDeviceId = pciIds == null ? 0 : pciIds.getB();
179             DxgiAdapterInfo dxgiMatch = DxgiUtil.findMatch(remainingDxgi, pciVendorId, pciDeviceId, name);
180 
181             long vram;
182             int dxgiIndex = -1;
183             String luidPrefix = "";
184             if (dxgiMatch == null) {
185                 vram = WmiUtil.getUint32asLong(cards, VideoControllerProperty.ADAPTERRAM, index);
186             } else {
187                 vram = dxgiMatch.getDedicatedVideoMemory();
188                 dxgiIndex = dxgiAdapters.indexOf(dxgiMatch);
189                 luidPrefix = luidPrefixBuilder.apply(dxgiMatch);
190             }
191             String lhmParent = lhmParentMap.getOrDefault(DxgiUtil.normalizeName(Util.isBlank(name) ? "" : name), "");
192 
193             // pciBusNumber and pciBusId are not available via WMI; ADL and NVML correlation are skipped for cards
194             // enumerated through this path.
195             GraphicsCard card = factory.create(ParseUtil.getStringValueOrUnknown(name), deviceId,
196                     ParseUtil.getStringValueOrUnknown(vendor), versionInfo, vram, luidPrefix, lhmParent, -1, "");
197 
198             // Remove dxgiMatch only after the card is successfully constructed, matching the registry path's defensive
199             // pattern so that a failure during construction leaves the match available for subsequent entries.
200             if (dxgiMatch != null) {
201                 remainingDxgi.remove(dxgiMatch);
202             }
203             if (dxgiIndex >= 0) {
204                 dxgiOrdered.put(dxgiIndex, card);
205             } else {
206                 cardList.add(card);
207             }
208         }
209         List<GraphicsCard> result = new ArrayList<>(dxgiOrdered.values());
210         result.addAll(cardList);
211         return result;
212     }
