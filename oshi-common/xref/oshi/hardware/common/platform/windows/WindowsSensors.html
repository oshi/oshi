35  /**
36   * Abstract base shared by the Windows Sensors implementations (JNA and FFM). Holds the orchestration of the sensor
37   * sources, which are tried in order: the Open Hardware Monitor and Libre Hardware Monitor WMI namespaces, published by
38   * either application while it runs (Libre Hardware Monitor only up to version 0.9.4); the web server of Libre Hardware
39   * Monitor 0.9.5 and later, if {@link GlobalConfig#OSHI_OS_WINDOWS_LHM_URL} names one; and finally plain WMI. The source
40   * order and result processing are backend-agnostic; the native driver queries are provided by the subclasses.
41   */
42  @ThreadSafe
43  public abstract class WindowsSensors extends AbstractSensors {
44  
45      private static final Logger LOG = LoggerFactory.getLogger(WindowsSensors.class);

49      /** Open Hardware Monitor's {@code HardwareType} value for a processor. */
50      private static final String OHM_CPU = "CPU";
51  
52      /** Libre Hardware Monitor's {@code HardwareType} value for a processor, which OHM spells {@code CPU}. */
53      private static final String LHM_CPU = "Cpu";
54  
55      /**
56       * Both monitors' {@code HardwareType} value for the motherboard's Super I/O chip, which carries the fan sensors; a
57       * processor carries none.
58       */
59      private static final String SUPER_IO = "SuperIO";

61      /** {@code SensorType} values. Both monitors publish the same names in WMI. */
62      private static final String TEMPERATURE = "Temperature";
63      private static final String FAN = "Fan";
64      private static final String VOLTAGE = "Voltage";

66      // One request to the Libre Hardware Monitor web server serves all three sensor reads of a refresh. The cache
67      // lifetime starts when the request completes rather than when it starts, as Memoizer's does, so that a request
68      // slower than the lifetime is not repeated by the next read.
69      // Null until the first request
70      @GuardedBy("this")
71      private @Nullable List<Reading> lhmReadings;
72      @GuardedBy("this")
73      private long lhmExpiration;
74  
75      private synchronized List<Reading> getLhmMetrics() {
76          long ttl = defaultExpiration();
77          List<Reading> readings = lhmReadings;
78          if (readings == null || (ttl >= 0 && System.nanoTime() - lhmExpiration >= 0)) {
79              readings = queryLhmMetrics();
80              lhmReadings = readings;
81              lhmExpiration = System.nanoTime() + ttl;
82          }
83          return readings;
84      }

86      /**
87       * Queries the Libre Hardware Monitor web server named by {@link GlobalConfig#OSHI_OS_WINDOWS_LHM_URL}.
88       *
89       * @return the readings, empty if no server is configured or it could not be reached
90       */
91      protected List<Reading> queryLhmMetrics() {
92          String url = GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_LHM_URL, "");
93          return url.trim().isEmpty() ? Collections.emptyList() : LhmMetrics.query(url);
94      }

96      /**
97       * Averages the temperature readings of the first processor, using the same rule as the WMI namespaces.
98       *
99       * @param readings the web server's readings
100      * @return the average temperature in degrees Celsius, or 0 if there is none
101      */
102     static double getCpuTemperature(List<Reading> readings) {
103         String cpu = null;
104         double sum = 0;
105         int count = 0;
106         for (Reading r : readings) {
107             if (LhmMetrics.CPU_TEMPERATURE.equals(r.getMetric())) {
108                 if (cpu == null) {
109                     cpu = r.getHardwareId();
110                 }
111                 if (cpu.equals(r.getHardwareId()) && isCpuTemperatureReading(r.getSensorName(), r.getValue())) {
112                     sum += r.getValue();
113                     count++;
114                 }
115             }
116         }
117         return count > 0 ? sum / count : 0;
118     }

120     /**
121      * Collects the motherboard's fan readings, from every Super I/O chip, as the WMI namespaces do.
122      *
123      * @param readings the web server's readings
124      * @return the fan speeds in RPM, empty if there are none
125      */
126     static int[] getFanSpeeds(List<Reading> readings) {
127         return readings.stream().filter(r -> LhmMetrics.MOTHERBOARD_FAN.equals(r.getMetric()))
128                 .mapToInt(r -> (int) r.getValue()).toArray();
129     }

131     /**
132      * Finds the processor's voltage: its own first voltage sensor if it has one, as the WMI namespaces read, otherwise
133      * the Super I/O chip's {@code Vcore}.
134      *
135      * @param readings the web server's readings
136      * @return the voltage in volts, or 0 if there is none
137      */
138     static double getCpuVoltage(List<Reading> readings) {
139         for (Reading r : readings) {
140             if (LhmMetrics.CPU_VOLTAGE.equals(r.getMetric()) && r.getValue() > 0) {
141                 return r.getValue();
142             }
143         }
144         for (Reading r : readings) {
145             if (LhmMetrics.MOTHERBOARD_VOLTAGE.equals(r.getMetric()) && r.getValue() > 0
146                     && r.getSensorName().toLowerCase(Locale.ROOT).contains("vcore")) {
147                 return r.getValue();
148             }
149         }
150         return 0d;
151     }

153     @Override
154     public double queryCpuTemperature() {
155         // Attempt to fetch value from Open Hardware Monitor if it is running, as it will give the most accurate results
156         // and the time to query (or attempt) is trivial
157         double tempC = getTempFromMonitorWmi(OhmHardware.OHM_NAMESPACE, OHM_CPU);
158         if (tempC > 0d) {
159             return tempC;
160         }
161         // Then Libre Hardware Monitor, the maintained successor, which publishes the same schema
162         tempC = getTempFromMonitorWmi(LhmSensor.LHM_NAMESPACE, LHM_CPU);
163         if (tempC > 0d) {
164             return tempC;
165         }
166         // Then Libre Hardware Monitor 0.9.5 and later, which replaced WMI with a web server
167         tempC = getCpuTemperature(getLhmMetrics());
168         if (tempC > 0d) {
169             return tempC;
170         }
171         // If we get this far, OHM is not running. Try from WMI. Other fallbacks to WMI are unreliable so we omit them.
172         return getTempFromWMI();
173     }

175     private double getTempFromMonitorWmi(String namespace, String cpuTypeName) {
176         if (HardwareMonitorDisabled.isWmiDisabled(namespace)) {
177             return 0;
178         }
179         WmiResult<ValueProperty> ohmSensors = queryHardwareMonitorSensor(namespace, OhmHardware.HARDWARE, cpuTypeName,
180                 TEMPERATURE);
181         if (ohmSensors != null && ohmSensors.getResultCount() > 0) {
182             double sum = 0;
183             int count = 0;
184             for (int i = 0; i < ohmSensors.getResultCount(); i++) {
185                 double value = WmiUtil.getFloat(ohmSensors, ValueProperty.VALUE, i);
186                 if (isCpuTemperatureReading(WmiUtil.getString(ohmSensors, ValueProperty.NAME, i), value)) {
187                     sum += value;
188                     count++;
189                 }
190             }
191             return count > 0 ? sum / count : 0;
192         }
193         return 0;
194     }

196     /**
197      * Whether a processor temperature sensor is a reading to average. Both monitors also publish summaries of the
198      * per-core readings ({@code Core Max}, {@code Core Average}) and, on Intel processors, each core's
199      * {@code Distance to TjMax}, which is a margin below the throttling point rather than a temperature.
200      *
201      * @param name  the sensor name
202      * @param value the sensor value, in degrees Celsius
203      * @return true for a positive temperature that is not a summary or a margin
204      */
205     static boolean isCpuTemperatureReading(String name, double value) {
206         return value > 0 && !name.contains("Max") && !name.contains("Average") && !name.contains("Distance");
207     }

209     private double getTempFromWMI() {
210         double tempC = 0d;
211         long tempK = 0L;
212         WmiResult<TemperatureProperty> result = queryWmiTemperature();
213         if (result.getResultCount() > 0) {
214             LOG.debug("Found Temperature data in WMI");
215             tempK = WmiUtil.getUint32asLong(result, TemperatureProperty.CURRENTTEMPERATURE, 0);
216         }
217         if (tempK > 2732L) {
218             tempC = tempK / 10d - 273.15;
219         } else if (tempK > 274L) {
220             tempC = tempK - 273d;
221         }
222         return Math.max(tempC, +0.0);
223     }

225     @Override
226     public int[] queryFanSpeeds() {
227         // Attempt to fetch value from Open Hardware Monitor if it is running
228         int[] fanSpeeds = getFansFromMonitorWmi(OhmHardware.OHM_NAMESPACE);
229         if (fanSpeeds.length > 0) {
230             return fanSpeeds;
231         }
232         // Then Libre Hardware Monitor, the maintained successor, which publishes the same schema
233         fanSpeeds = getFansFromMonitorWmi(LhmSensor.LHM_NAMESPACE);
234         if (fanSpeeds.length > 0) {
235             return fanSpeeds;
236         }
237         // Then Libre Hardware Monitor 0.9.5 and later, which replaced WMI with a web server
238         fanSpeeds = getFanSpeeds(getLhmMetrics());
239         if (fanSpeeds.length > 0) {
240             return fanSpeeds;
241         }
242         // If we get this far, OHM is not running. Try to get from conventional WMI
243         fanSpeeds = getFansFromWMI();
244         if (fanSpeeds.length > 0) {
245             return fanSpeeds;
246         }
247         // Default
248         return new int[0];
249     }

251     private int[] getFansFromMonitorWmi(String namespace) {
252         if (HardwareMonitorDisabled.isWmiDisabled(namespace)) {
253             return new int[0];
254         }
255         WmiResult<ValueProperty> ohmSensors = queryHardwareMonitorSensor(namespace, OhmHardware.HARDWARE, SUPER_IO,
256                 FAN);
257         if (ohmSensors != null && ohmSensors.getResultCount() > 0) {
258             int[] fanSpeeds = new int[ohmSensors.getResultCount()];
259             for (int i = 0; i < ohmSensors.getResultCount(); i++) {
260                 fanSpeeds[i] = (int) WmiUtil.getFloat(ohmSensors, ValueProperty.VALUE, i);
261             }
262             return fanSpeeds;
263         }
264         return new int[0];
265     }

267     private int[] getFansFromWMI() {
268         WmiResult<SpeedProperty> fan = queryWmiFanSpeed();
269         if (fan.getResultCount() > 0) {
270             LOG.debug("Found Fan data in WMI");
271             int[] fanSpeeds = new int[fan.getResultCount()];
272             for (int i = 0; i < fan.getResultCount(); i++) {
273                 fanSpeeds[i] = (int) WmiUtil.getUint64(fan, SpeedProperty.DESIREDSPEED, i);
274             }
275             return fanSpeeds;
276         }
277         return new int[0];
278     }

283         double volts = getVoltsFromMonitorWmi(OhmHardware.OHM_NAMESPACE, OHM_CPU);

290             return volts;
291         }
292         // Then Libre Hardware Monitor 0.9.5 and later, which replaced WMI with a web server
293         volts = getCpuVoltage(getLhmMetrics());
294         if (volts > 0d) {
295             return volts;
296         }
297         // If we get this far, OHM is not running. Try to get from conventional WMI
298         return getVoltsFromWMI();

301     private double getVoltsFromMonitorWmi(String namespace, String cpuTypeName) {
302         if (HardwareMonitorDisabled.isWmiDisabled(namespace)) {
303             return 0d;
304         }
305         // The processor's voltage sensors, reached the same way as its temperature and fan sensors. This asked the
306         // Hardware class for SensorType="Voltage" until 2026-09; that class has no SensorType property, so the query
307         // could never match and this tier always fell through to plain WMI.
308         WmiResult<ValueProperty> ohmSensors = queryHardwareMonitorSensor(namespace, OhmHardware.HARDWARE, cpuTypeName,
309                 VOLTAGE);
310         if (ohmSensors != null && ohmSensors.getResultCount() > 0) {
311             return WmiUtil.getFloat(ohmSensors, ValueProperty.VALUE, 0);
312         }
313         return 0d;
314     }

316     private double getVoltsFromWMI() {
317         WmiResult<VoltProperty> voltage = queryWmiVoltage();
318         if (voltage.getResultCount() > 0) {
319             LOG.debug("Found Voltage data in WMI");
320             int decivolts = WmiUtil.getUint16(voltage, VoltProperty.CURRENTVOLTAGE, 0);
321             // If the eighth bit is set, bits 0-6 contain the voltage multiplied by 10. If the eighth bit is not set,
322             // then the bit setting in VoltageCaps represents the voltage value.
323             if (decivolts > 0) {
324                 if ((decivolts & 0x80) == 0) {
325                     decivolts = WmiUtil.getUint32(voltage, VoltProperty.VOLTAGECAPS, 0);
326                     // This value is really a bit setting, not decivolts
327                     if ((decivolts & 0x1) > 0) {
328                         return 5.0;
329                     } else if ((decivolts & 0x2) > 0) {
330                         return 3.3;
331                     } else if ((decivolts & 0x4) > 0) {
332                         return 2.9;
333                     }
334                 } else {
335                     // Value from bits 0-6, divided by 10
336                     return (decivolts & 0x7F) / 10d;
337                 }
338             }
339         }
340         return 0d;
341     }

343     /**
344      * Selects the hardware identifiers to read sensors from. The query which produced the result already filtered on
345      * {@code HardwareType}. A processor's sensors are read from the first identifier; fans are read from every Super
346      * I/O chip, since a board may carry more than one and the first need not be the one wired to the fan headers.
347      *
348      * @param hardware the hardware identifier result
349      * @param typeName the {@code HardwareType} the result was filtered on
350      * @return the identifiers to query, empty if there are none
351      */
352     protected static List<String> selectHardwareIdentifiers(WmiResult<IdentifierProperty> hardware, String typeName) {
353         List<String> identifiers = new ArrayList<>();
354         int count = SUPER_IO.equals(typeName) ? hardware.getResultCount() : Math.min(1, hardware.getResultCount());
355         for (int i = 0; i < count; i++) {
356             String identifier = WmiUtil.getString(hardware, IdentifierProperty.IDENTIFIER, i);
357             if (!identifier.isEmpty()) {
358                 identifiers.add(identifier);
359             }
360         }
361         return identifiers;
362     }
