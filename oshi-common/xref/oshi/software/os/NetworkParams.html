6   
7   import java.net.InetAddress;
8   import java.net.UnknownHostException;
9   import java.util.Arrays;
10  import java.util.Collections;
11  import java.util.List;
12  
13  import oshi.annotation.PublicApi;
14  import oshi.annotation.concurrent.Immutable;
15  import oshi.annotation.concurrent.ThreadSafe;
16  import oshi.hardware.NetworkIF;
17  
18  /**
19   * Provides network parameters of the running operating system, including the hostname, domain name, DNS server
20   * addresses, default gateways for IPv4 and IPv6, and the routing table.
21   * <p>
22   * The hostname ({@link #getHostName()}) is the local machine name. The domain name ({@link #getDomainName()}) is the
23   * DNS domain suffix. DNS servers ({@link #getDnsServers()}) are the configured name resolution servers. The default
24   * gateways ({@link #getIpv4DefaultGateway()}, {@link #getIpv6DefaultGateway()}) are the routing destinations for
25   * {@code 0.0.0.0/0} and {@code ::/0} respectively, and return an empty string if not defined.
26   * <p>
27   * {@link #getRoutes()} returns the whole routing table those gateways are drawn from:
28   *
29   * <pre>{@code
30   * NetworkParams params = new SystemInfo().getOperatingSystem().getNetworkParams();
31   * for (NetworkParams.IPRoute route : params.getRoutes()) {
32   *     if (route.getPrefixLength() == 0) {
33   *         System.out.println("Default route via " + route.getInterfaceName());
34   *     }
35   * }
36   * }</pre>
37   *
38   * @see IPRoute
39   */
40  @PublicApi
41  @ThreadSafe
42  public interface NetworkParams {
43  
44      /**
45       * Gets the HostName of the machine executing OSHI.
46       *
47       * @return the hostname
48       */
49      String getHostName();
50  
51      /**
52       * Gets the Domain Name of the machine executing OSHI.
53       *
54       * @return the domain name
55       */
56      String getDomainName();
57  
58      /**
59       * Gets the DNS Servers configured for this machine.
60       *
61       * @return the DNS servers
62       */
63      String[] getDnsServers();
64  
65      /**
66       * Gets the default gateway(routing destination for 0.0.0.0/0) for IPv4 connections.
67       *
68       * @return default gateway for IPv4, or empty string if not defined.
69       */
70      String getIpv4DefaultGateway();
71  
72      /**
73       * Gets default gateway(routing destination for ::/0) for IPv6 connections.
74       *
75       * @return default gateway for IPv6, or empty string if not defined.
76       */
77      String getIpv6DefaultGateway();
78  
79      /**
80       * Gets the operating system's routing table, containing both IPv4 and IPv6 routes. Each {@link IPRoute} carries its
81       * own address family in the length of its destination address.
82       * <p>
83       * The routing table is read fresh on each call and is not cached.
84       * <p>
85       * <b>Platform notes:</b> Windows reads the table from the IP Helper API and Linux from {@code /proc/net/route} and
86       * {@code /proc/net/ipv6_route}; the remaining platforms parse {@code netstat}. No platform requires elevated
87       * privileges. Not every platform publishes every field; see {@link IPRoute} for the sentinel each unavailable field
88       * reports.
89       *
90       * @return A list of {@link IPRoute} objects, or an empty list if the routing table could not be read.
91       */
92      default List<IPRoute> getRoutes() {
93          return Collections.emptyList();
94      }
95  
96      /**
97       * A single entry in the operating system's routing table.
98       * <p>
99       * The address family of the entry is given by the length of {@link #getDestination()}: four bytes for IPv4, sixteen
100      * for IPv6. A route whose destination cannot be interpreted is omitted from {@link #getRoutes()}, so that length is
101      * always 4 or 16.
102      * <p>
103      * <b>Platform notes:</b> {@link #getMetric()} is only reported on Linux, Windows and OpenBSD; macOS, AIX, Solaris,
104      * FreeBSD, DragonFly BSD and NetBSD do not publish a route metric. {@link #getPrefixLength()} is {@code -1} on the
105      * rare route whose netmask the operating system does not publish. {@link #getInterfaceIndex()} is {@code -1} when
106      * the interface name cannot be resolved to an index.
107      */
108     @PublicApi
109     @Immutable
110     final class IPRoute {
111         private final byte[] destination;
112         private final int prefixLength;
113         private final byte[] gateway;
114         private final String interfaceName;
115         private final int interfaceIndex;
116         private final long metric;
117         private final boolean isGateway;
118         private final boolean isHost;
119 
120         /**
121          * Constructs a new IPRoute instance.
122          *
123          * @param destination    the destination network address bytes, four for IPv4 or sixteen for IPv6
124          * @param prefixLength   the number of leading bits in the destination network mask, or -1 if not published
125          * @param gateway        the next hop address bytes, or an empty array for a directly attached route
126          * @param interfaceName  the name of the outgoing interface, or an empty string if not published
127          * @param interfaceIndex the index of the outgoing interface, or -1 if not published
128          * @param metric         the route metric or priority, or -1 if not published
129          * @param isGateway      whether the route forwards through the gateway rather than being directly attached
130          * @param isHost         whether the route is to a single host rather than to a network
131          */
132         public IPRoute(byte[] destination, int prefixLength, byte[] gateway, String interfaceName, int interfaceIndex,
133                 long metric, boolean isGateway, boolean isHost) {
134             this.destination = Arrays.copyOf(destination, destination.length);
135             this.prefixLength = prefixLength;
136             this.gateway = Arrays.copyOf(gateway, gateway.length);
137             this.interfaceName = interfaceName;
138             this.interfaceIndex = interfaceIndex;
139             this.metric = metric;
140             this.isGateway = isGateway;
141             this.isHost = isHost;
142         }
143 
144         /**
145          * Gets the destination network address. For IPv4 routes this is a 4-byte array, for IPv6 routes a 16-byte
146          * array, so the length identifies the address family. A default route has an all-zero destination and a
147          * {@link #getPrefixLength()} of 0.
148          *
149          * @return The destination address, never empty.
150          */
151         public byte[] getDestination() {
152             return Arrays.copyOf(destination, destination.length);
153         }
154 
155         /**
156          * Gets the number of leading bits in the destination network mask, as in CIDR notation.
157          *
158          * @return The prefix length, from 0 to 32 for IPv4 or 0 to 128 for IPv6, or -1 if the operating system does not
159          *         publish the mask for this route.
160          */
161         public int getPrefixLength() {
162             return prefixLength;
163         }
164 
165         /**
166          * Gets the next hop address that traffic matching this route is forwarded to. Only gateway routes have one. For
167          * a directly attached route the operating system may report the outgoing interface's own address, its
168          * link-layer address, or a symbolic name, none of which is a next hop, so all of them are reported as empty.
169          *
170          * @return The gateway address, 4 bytes for IPv4 or 16 for IPv6, or an empty array when {@link #isGateway()} is
171          *         {@code false}.
172          */
173         public byte[] getGateway() {
174             return Arrays.copyOf(gateway, gateway.length);
175         }
176 
177         /**
178          * Gets the name of the interface traffic matching this route leaves by.
179          *
180          * @return The interface name, or an empty string if the operating system does not publish one.
181          */
182         public String getInterfaceName() {
183             return interfaceName;
184         }
185 
186         /**
187          * Gets the index of the interface traffic matching this route leaves by, matching {@link NetworkIF#getIndex()}.
188          *
189          * @return The interface index, or -1 if it could not be determined.
190          */
191         public int getInterfaceIndex() {
192             return interfaceIndex;
193         }
194 
195         /**
196          * Gets the cost of this route, used to select between routes to the same destination. Lower is preferred.
197          *
198          * @return The metric, or -1 on the platforms which do not publish one: macOS, AIX, Solaris, FreeBSD, DragonFly
199          *         BSD and NetBSD.
200          */
201         public long getMetric() {
202             return metric;
203         }
204 
205         /**
206          * Tests whether traffic matching this route is forwarded through {@link #getGateway()} rather than delivered
207          * directly on the attached link.
208          *
209          * @return {@code true} for a gateway route.
210          */
211         public boolean isGateway() {
212             return isGateway;
213         }
214 
215         /**
216          * Tests whether this route is to a single host rather than to a network, that is, whether
217          * {@link #getPrefixLength()} covers the whole address.
218          *
219          * @return {@code true} for a host route.
220          */
221         public boolean isHost() {
222             return isHost;
223         }
224 
225         @Override
226         public String toString() {
227             return "IPRoute [destination=" + addressToString(destination) + "/" + prefixLength + ", gateway="
228                     + addressToString(gateway) + ", interfaceName=" + interfaceName + ", interfaceIndex="
229                     + interfaceIndex + ", metric=" + metric + ", isGateway=" + isGateway + ", isHost=" + isHost + "]";
230         }
231 
232         private static String addressToString(byte[] address) {
233             if (address.length > 0) {
234                 try {
235                     return InetAddress.getByAddress(address).getHostAddress();
236                 } catch (UnknownHostException e) { // NOSONAR java:S108
237                     // Cannot happen for a length of 4 or 16; fall through to the placeholder
238                 }
239             }
240             return "*";
241         }
242     }
