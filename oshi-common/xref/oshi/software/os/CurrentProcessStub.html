7   import java.util.Collections;
8   import java.util.List;
9   import java.util.Map;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.software.common.AbstractOSProcess;
13  import oshi.util.Constants;

15  /**
16   * Minimal stand-in for the current process, returned by {@link OperatingSystem#getCurrentProcess()} only when the
17   * platform query fails to produce it.
18   * <p>
19   * The current process exists by definition, so a failed lookup is a failed read — a truncated native process list, a
20   * denied {@code /proc} entry, a race with the query — rather than a missing process. {@code getProcess(int)} is
21   * documented to return null and {@code getCurrentProcess()} is not, so this carries the facts still known to be true
22   * (the process ID, and that it is running) and reports every other value as the unknown/zero/empty default that callers
23   * already handle from a live process.
24   */
25  @ThreadSafe
26  final class CurrentProcessStub extends AbstractOSProcess {
27  
28      CurrentProcessStub(int pid) {
29          super(pid);
30          this.name = "";
31          this.state = State.RUNNING;
32      }
33  
34      @Override
35      public List<OSThread> getThreadDetails() {
36          // Empty rather than absent: every platform's getCurrentThread() already falls back to constructing a thread
37          // from the process ID and thread ID when the list holds no match.
38          return Collections.emptyList();
39      }
40  
41      @Override
42      public String getUser() {
43          return Constants.UNKNOWN;
44      }
45  
46      @Override
47      public String getUserID() {
48          return Constants.UNKNOWN;
49      }
50  
51      @Override
52      public String getGroup() {
53          return Constants.UNKNOWN;
54      }
55  
56      @Override
57      public String getGroupID() {
58          return Constants.UNKNOWN;
59      }
60  
61      @Override
62      public String getCommandLine() {
63          return "";
64      }
65  
66      @Override
67      public List<String> getArguments() {
68          return Collections.emptyList();
69      }
70  
71      @Override
72      public Map<String, String> getEnvironmentVariables() {
73          return Collections.emptyMap();
74      }
75  
76      @Override
77      public String getCurrentWorkingDirectory() {
78          return "";
79      }
80  
81      @Override
82      public long getAffinityMask() {
83          return 0L;
84      }
85  
86      @Override
87      public int getBitness() {
88          return 0;
89      }
90  
91      @Override
92      public long getResidentMemory() {
93          return 0L;
94      }
95  
96      @Override
97      public long getOpenFiles() {
98          return 0L;
99      }
100 
101     @Override
102     public long getSoftOpenFileLimit() {
103         return -1L;
104     }
105 
106     @Override
107     public long getHardOpenFileLimit() {
108         return -1L;
109     }
110 
111     @Override
112     public boolean updateAttributes() {
113         // Nothing to refresh: this instance exists precisely because the platform query did not return this process.
114         return false;
115     }
116 }
