37      @Override
38      public TcpStats getTCPv4Stats() {
39          return parseTcp4Stats(snmpStats(TCP));
40      }

42      /**
43       * Builds TCPv4 statistics from the {@code Tcp} counters of {@code /proc/net/snmp}.
44       *
45       * @param tcp a map of counter names to values
46       * @return the statistics, with zeros for any counter not present
47       */
48      static TcpStats parseTcp4Stats(Map<String, Long> tcp) {
49          return new TcpStats(tcp.getOrDefault("CurrEstab", 0L), tcp.getOrDefault("ActiveOpens", 0L),
50                  tcp.getOrDefault("PassiveOpens", 0L), tcp.getOrDefault("AttemptFails", 0L),
51                  tcp.getOrDefault("EstabResets", 0L), tcp.getOrDefault("OutSegs", 0L), tcp.getOrDefault("InSegs", 0L),
52                  tcp.getOrDefault("RetransSegs", 0L), tcp.getOrDefault("InErrs", 0L), tcp.getOrDefault("OutRsts", 0L));
53      }
54  
55      @Override
56      public UdpStats getUDPv4Stats() {
57          return parseUdp4Stats(snmpStats(UDP));
58      }
59  
60      /**
61       * Builds UDPv4 statistics from the {@code Udp} counters of {@code /proc/net/snmp}.
62       *
63       * @param udp a map of counter names to values
64       * @return the statistics, with zeros for any counter not present
65       */
66      static UdpStats parseUdp4Stats(Map<String, Long> udp) {
67          return new UdpStats(udp.getOrDefault("OutDatagrams", 0L), udp.getOrDefault("InDatagrams", 0L),
68                  udp.getOrDefault("NoPorts", 0L), udp.getOrDefault("InErrors", 0L));
69      }

71      @Override
72      public UdpStats getUDPv6Stats() {
73          return parseUdp6Stats(ProcUtil.parseStatistics(ProcPath.SNMP6));
74      }

76      /**
77       * Builds UDPv6 statistics from the counters of {@code /proc/net/snmp6}.
78       *
79       * @param snmp6 a map of counter names to values, as read from the whole file
80       * @return the statistics, with zeros for any counter not present
81       */
82      static UdpStats parseUdp6Stats(Map<String, Long> snmp6) {
83          return new UdpStats(snmp6.getOrDefault("Udp6OutDatagrams", 0L), snmp6.getOrDefault("Udp6InDatagrams", 0L),
84                  snmp6.getOrDefault("Udp6NoPorts", 0L), snmp6.getOrDefault("Udp6InErrors", 0L));
85      }

87      private static Map<String, Long> snmpStats(String key) {
88          return ProcUtil.parseNestedStatistics(ProcPath.SNMP, key).getOrDefault(key, Collections.emptyMap());
89      }

91      @Override
92      public List<IPConnection> getConnections() {
93          List<IPConnection> conns = new ArrayList<>();
94          Map<Long, Integer> pidMap = ProcessStat.querySocketToPidMap();
95          conns.addAll(queryConnections("tcp", 4, pidMap));
96          conns.addAll(queryConnections("tcp", 6, pidMap));
97          conns.addAll(queryConnections("udp", 4, pidMap));
98          conns.addAll(queryConnections("udp", 6, pidMap));
99          return conns;
100     }
101 
102     private static List<IPConnection> queryConnections(String protocol, int ipver, Map<Long, Integer> pidMap) {
103         return parseConnections(protocol + ipver,
104                 FileUtil.readFile(ProcPath.NET + "/" + protocol + (ipver == 6 ? "6" : "")), pidMap,
105                 ByteOrder.nativeOrder());
106     }
107 
108     /**
109      * Parses the lines of one of {@code /proc/net/tcp}, {@code tcp6}, {@code udp} or {@code udp6}.
110      *
111      * @param type   the connection type to report, such as {@code tcp4}
112      * @param lines  the lines of the file
113      * @param pidMap a map of socket inode to owning process ID
114      * @param order  the byte order of the host that wrote the lines
115      * @return the parsed connections
116      */
117     static List<IPConnection> parseConnections(String type, List<String> lines, Map<Long, Integer> pidMap,
118             ByteOrder order) {
119         List<IPConnection> conns = new ArrayList<>();
120         for (String s : lines) {
121             if (s.indexOf(':') >= 0) {
122                 String[] split = ParseUtil.whitespaces.split(s.trim(), -1);
123                 if (split.length > 9) {
124                     Pair<byte[], Integer> lAddr = parseIpAddr(split[1], order);
125                     Pair<byte[], Integer> fAddr = parseIpAddr(split[2], order);
126                     TcpState state = TcpState.fromLinuxState(ParseUtil.hexStringToInt(split[3], 0));
127                     Pair<Integer, Integer> txQrxQ = parseHexColonHex(split[4]);
128                     long inode = ParseUtil.parseLongOrDefault(split[9], 0);
129                     conns.add(new IPConnection(type, lAddr.getA(), lAddr.getB(), fAddr.getA(), fAddr.getB(), state,
130                             txQrxQ.getA(), txQrxQ.getB(), pidMap.getOrDefault(inode, -1)));
131                 }
132             }
133         }
134         return conns;
135     }
136 
137     private static Pair<byte[], Integer> parseIpAddr(String s, ByteOrder order) {
138         int colon = s.indexOf(':');
139         if (colon > 0 && colon < s.length()) {
140             // Each 32-bit word of the __be32 address is printed with %08X, so in the host's byte order
141             byte[] first = ParseUtil.hexWordsToByteArray(s.substring(0, colon), order);
142             int second = ParseUtil.hexStringToInt(s.substring(colon + 1), 0);
143             return new Pair<>(first, second);
144         }
145         return new Pair<>(new byte[0], 0);
146     }
147 
148     private static Pair<Integer, Integer> parseHexColonHex(String s) {
149         int colon = s.indexOf(':');
150         if (colon > 0 && colon < s.length()) {
151             int first = ParseUtil.hexStringToInt(s.substring(0, colon), 0);
152             int second = ParseUtil.hexStringToInt(s.substring(colon + 1), 0);
153             return new Pair<>(first, second);
154         }
155         return new Pair<>(0, 0);
156     }
