5   package oshi.software.common.os.unix.netbsd;
6   
7   import java.util.List;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.software.common.AbstractNetworkParams;
11  import oshi.software.os.NetworkParams;
12  import oshi.util.ExecutingCommand;
13  import oshi.util.driver.unix.NetstatRoute;
14  import oshi.util.driver.unix.RouteTableDump;
15  
16  /**
17   * NetBsdNetworkParams class.
18   */
19  @ThreadSafe
20  public class NetBsdNetworkParams extends AbstractNetworkParams {
21  
22      /** NetBSD prints the interface in the seventh column: Destination Gateway Flags Refs Use Mtu Interface. */
23      private static final int IF_NAME_INDEX = 6;
24  
25      @Override
26      public String getIpv4DefaultGateway() {
27          return searchGateway(ExecutingCommand.runNative("route -n get default"));
28      }
29  
30      @Override
31      public String getIpv6DefaultGateway() {
32          return searchGateway(ExecutingCommand.runNative("route -n get -inet6 default"));
33      }
34  
35      @Override
36      public List<NetworkParams.IPRoute> getRoutes() {
37          byte[] dump = queryRouteDump();
38          if (dump.length > 0) {
39              List<NetworkParams.IPRoute> routes = RouteTableDump.parse(dump, RouteTableDump.Layout.NETBSD,
40                      queryInterfaceNameByIndex());
41              if (!routes.isEmpty()) {
42                  return routes;
43              }
44          }
45          return NetstatRoute.queryRoutes("netstat -rn -f inet", "netstat -rn -f inet6", IF_NAME_INDEX,
46                  queryInterfaceIndexByName());
47      }
48  
49      /**
50       * Fetches the kernel's routing table dump.
51       *
52       * @return The bytes of a {@code NET_RT_DUMP} sysctl, or an empty array if this build cannot make the call, in which
53       *         case the routing table is read by running a command instead
54       */
55      protected byte[] queryRouteDump() {
56          return new byte[0];
57      }
58  }
