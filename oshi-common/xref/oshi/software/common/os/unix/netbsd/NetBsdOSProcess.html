56      private static final Pattern AFFINITY = Pattern.compile("Affinity:");
57      private static final Pattern COMMA_OR_WHITESPACE = Pattern.compile("[,\\s]+");
58      private static final int ARGS_READ_ATTEMPTS_SELF = 15;
59      private static final int ARGS_READ_ATTEMPTS_OTHER = 2;
60      private static final long ARGS_READ_RETRY_MILLIS = 20L;

62      /**
63       * Ordered {@code ps} columns queried for each process. Shared by NetBsdOSProcess and the NetBSD OperatingSystem so
64       * the column list and parsing stay in lockstep. {@code ARGS} must remain last.
65       */
66      public static final List<BsdPsKeyword> PS_KEYWORDS = Collections.unmodifiableList(Arrays.asList(STATE, PID, PPID,
67              USER, UID, GROUP, GID, PRI, VSZ, RSS, ETIME, CPUTIME, MAJFLT, MINFLT, NVCSW, NIVCSW, ARGS));
68  
69      public static final String PS_COMMAND_ARGS = PS_KEYWORDS.stream().map(Enum::name)
70              .map(name -> name.toLowerCase(Locale.ROOT)).collect(Collectors.joining(","));

97      @Override
98      protected String queryPath(Map<BsdPsKeyword, String> psMap) {
99          // NetBSD's ps prints argv[0] in the comm column, the same value the args column begins with, but clips it to
100         // the width of its "COMMAND" header - seven characters - because only the last column is left unbounded. So the
101         // path is taken from the args column, which is queried last and is therefore printed in full.
102         String argv = parseArgv0(psMap.get(BsdPsKeyword.ARGS));
103         // The placeholder says the arguments were unavailable, so no path is known
104         return isPlaceholder(argv) ? "" : argv;
105     }

107     @Override
108     protected String queryCommandLineBackup(Map<BsdPsKeyword, String> psMap) {
109         return parseCommandLine(psMap.get(BsdPsKeyword.ARGS));
110     }

112     /**
113      * Extracts the command line from a {@code ps} args column.
114      *
115      * @param args the args column, the process's arguments separated by spaces
116      * @return the arguments, or an empty string where {@code ps} substituted the parenthesized command because the
117      *         kernel would not release them, which is a placeholder rather than a command line
118      */
119     static String parseCommandLine(@Nullable String args) {
120         String argv = ParseUtil.getStringValueOrEmpty(args);
121         return isPlaceholder(parseArgv0(argv)) ? "" : argv;
122     }
123 
124     @Override
125     protected String queryName(Map<BsdPsKeyword, String> psMap, String path) {
126         String argv = parseArgv0(psMap.get(BsdPsKeyword.ARGS));
127         // The placeholder wraps the kernel's own name for the process, which is the name whether or not the arguments
128         // could be read; unwrap it rather than reporting ps's punctuation as the name
129         return isPlaceholder(argv) ? argv.substring(1, argv.length() - 1) : path.substring(path.lastIndexOf('/') + 1);
130     }
131 
132     /**
133      * Extracts {@code argv[0]} from a {@code ps} args column.
134      *
135      * @param args the args column, the process's arguments separated by spaces
136      * @return {@code argv[0]}, which is a path when the process was started by one, or the command in parentheses where
137      *         the kernel would not release the arguments
138      */
139     static String parseArgv0(@Nullable String args) {
140         String argv = ParseUtil.getStringValueOrEmpty(args).trim();
141         int space = argv.indexOf(' ');
142         return space < 0 ? argv : argv.substring(0, space);
143     }
144 
145     /**
146      * Whether a {@code ps} args column holds the parenthesized command that {@code ps} substitutes when the kernel
147      * refuses a process's arguments, which it does while that process is inside {@code posix_spawn}.
148      *
149      * @param argv the value from {@link #parseArgv0(String)}
150      * @return {@code true} if it is that placeholder rather than a real {@code argv[0]}
151      */
152     static boolean isPlaceholder(String argv) {
153         return argv.length() > 2 && argv.charAt(0) == '(' && argv.charAt(argv.length() - 1) == ')';
154     }

156     @Override
157     protected List<String> queryArguments() {
158         // NetBSD provides command line via /proc filesystem
159         String cmdline = "/proc/" + getProcessID() + "/cmdline";
160         // The kernel refuses to read a process's arguments while that process is inside posix_spawn, which a JVM is
161         // whenever it runs a command, so an empty read of a file that exists is retried. This process is the case OSHI
162         // provokes itself, by running ps while another thread spawns, and so is given the longer budget; for any other
163         // process an empty read is usually genuine. The result is memoized for the life of this object, so a single
164         // refusal would otherwise stick.
165         int attempts = getProcessID() == this.os.getProcessId() ? ARGS_READ_ATTEMPTS_SELF : ARGS_READ_ATTEMPTS_OTHER;
166         for (int attempt = 1;; attempt++) {
167             byte[] cmdBytes = FileUtil.readAllBytes(cmdline, false);
168             if (cmdBytes.length > 0) {
169                 return Collections.unmodifiableList(ParseUtil.parseByteArrayToStrings(cmdBytes));
170             }
171             if (attempt >= attempts || !new File(cmdline).exists()) {
172                 return Collections.emptyList();
173             }
174             Util.sleep(ARGS_READ_RETRY_MILLIS);
175         }
176     }

178     @Override
179     protected Map<String, String> queryEnvironmentVariables() {
180         // For the current process, use Java's System.getenv()
181         if (getProcessID() == this.os.getProcessId()) {
182             return System.getenv();
183         }
184         return parseEnvironment(queryEnvironmentBytes());
185     }
186 
187     /**
188      * Queries the environment of this process from the kernel, the {@code kern.proc_args} sysctl with
189      * {@code KERN_PROC_ENV}.
190      *
191      * @return The NUL-delimited {@code KEY=VALUE} entries, or an empty array if this build cannot make the call; NetBSD
192      *         offers no command that reads another process's environment
193      */
194     protected byte[] queryEnvironmentBytes() {
195         return new byte[0];
196     }
197 
198     /**
199      * Parses the environment returned by the {@code kern.proc_args} sysctl.
200      *
201      * @param envBytes NUL-delimited {@code KEY=VALUE} entries
202      * @return the environment, empty if there were no entries
203      */
204     static Map<String, String> parseEnvironment(byte[] envBytes) {
205         // Parse by raw bytes rather than by String offsets, which would misalign on a multibyte charset
206         Map<String, String> env = new LinkedHashMap<>();
207         for (String envStr : ParseUtil.parseByteArrayToStrings(envBytes)) {
208             int idx = envStr.indexOf('=');
209             if (idx > 0) {
210                 env.put(envStr.substring(0, idx), envStr.substring(idx + 1));
211             }
212         }
213         return env.isEmpty() ? Collections.emptyMap() : Collections.unmodifiableMap(env);
214     }

226     @Override
227     public long getSoftOpenFileLimit() {
228         if (getProcessID() == this.os.getProcessId()) {
229             // This process's own limit where a backend can call getrlimit, the system-wide maximum otherwise
230             long limit = queryRlimitNofile(true);
231             if (limit >= 0) {
232                 return limit;
233             }
234             limit = BsdSysctlUtil.sysctl("kern.maxfilesperproc", 0L);
235             if (limit <= 0) {
236                 limit = BsdSysctlUtil.sysctl("kern.maxfiles", 0L);
237             }
238             return limit > 0 ? limit : -1L;
239         }
240         return -1L;
241     }
242 
243     @Override
244     public long getHardOpenFileLimit() {
245         if (getProcessID() == this.os.getProcessId()) {
246             long limit = queryRlimitNofile(false);
247             if (limit >= 0) {
248                 return limit;
249             }
250             limit = BsdSysctlUtil.sysctl("kern.maxfiles", 0L);
251             return limit > 0 ? limit : -1L;
252         }
253         return -1L;
254     }

256     @Override
257     protected int queryBitness() {
258         // NetBSD does not maintain a compatibility layer.
259         // Process bitness is OS bitness
260         return System.getProperty("os.arch", "").contains("64") ? 64 : 32;
261     }

263     @Override
264     public long getAffinityMask() {
265         long bitMask = 0L;
266         // schedctl -p <pid> shows affinity; -A cpus would set it
267         List<String> schedctl = ExecutingCommand.runNative("schedctl -p " + getProcessID());
268         for (String line : schedctl) {
269             // Output includes "Affinity: <list>" when bound
270             if (line.contains("Affinity:")) {
271                 String[] parts = COMMA_OR_WHITESPACE.split(AFFINITY.split(line, -1)[1].trim(), -1);
272                 for (String part : parts) {
273                     int bitToSet = ParseUtil.parseIntOrDefault(part.trim(), -1);
274                     if (bitToSet >= 0) {
275                         bitMask |= 1L << bitToSet;
276                     }
277                 }
278                 return bitMask;
279             }
280         }
281         // Not bound — return all CPUs
282         int ncpu = BsdSysctlUtil.sysctl("hw.ncpuonline", 1);
283         return ncpu < 64 ? (1L << ncpu) - 1 : -1L;
284     }

286     @Override
287     public List<OSThread> getThreadDetails() {
288         // NetBSD ps shows per-LWP rows when lid is in the format specifier
289         String psCommand = "ps -awwxo " + NetBsdOSThread.PS_THREAD_COLUMNS;
290         if (getProcessID() >= 0) {
291             psCommand += " -p " + getProcessID();
292         }
293         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsArgs = threadMap -> threadMap
294                 .containsKey(BsdPsThreadKeyword.ARGS);
295         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
296                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class, NetBsdOSThread.PS_THREAD_KEYWORDS,
297                         thread.trim(), ' '))
298                 .filter(hasColumnsArgs).map(threadMap -> new NetBsdOSThread(getProcessID(), threadMap))
299                 .filter(VALID_THREAD).collect(Collectors.toList());
300     }
301 
302     @Override
303     protected void updateThreadCount() {
304         // Use nlwp keyword to get LWP count for this process
305         List<String> nlwpList = ExecutingCommand.runNative("ps -o nlwp -p " + getProcessID());
306         if (nlwpList.size() > 1) {
307             this.threadCount = ParseUtil.parseIntOrDefault(nlwpList.get(1).trim(), 1);
308         } else {
309             this.threadCount = 1;
310         }
311     }
