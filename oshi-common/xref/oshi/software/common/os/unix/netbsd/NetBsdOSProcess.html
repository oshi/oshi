104     @Override
105     protected synchronized boolean updateAttributes(Map<BsdPsKeyword, String> psMap) {
106         Map<BsdPsKeyword, String> row = psMap;
107         int attempts = getProcessID() == this.os.getProcessId() ? SPAWN_RETRY_ATTEMPTS_SELF
108                 : SPAWN_RETRY_ATTEMPTS_OTHER;
109         for (int attempt = 1; attempt < attempts && isTornRow(row); attempt++) {
110             Util.sleep(SPAWN_RETRY_MILLIS);
111             Map<BsdPsKeyword, String> retry = queryPsRow();
112             if (retry.isEmpty()) {
113                 break;
114             }
115             row = retry;
116         }
117         return super.updateAttributes(row);
118     }

120     /**
121      * Whether a {@code ps} row is missing the counters the kernel withholds while the process is inside
122      * {@code posix_spawn}. Package-private for testing.
123      * <p>
124      * {@code fill_kproc2()} fills the CPU time, page faults and context switches only when it can take the process's
125      * {@code p_reflock}; {@code sysctl_doeproc} reports the process with all of them zero otherwise. A JVM holds that
126      * lock whenever one of its threads runs a command, as OSHI does. A live process that has run at all has nonzero
127      * minor faults and context switches, so all four zero marks such a row.
128      *
129      * @param psMap the parsed {@code ps} columns
130      * @return {@code true} if the row's CPU time, minor faults and both context-switch counts are all zero
131      */
132     static boolean isTornRow(Map<BsdPsKeyword, String> psMap) {
133         return !ParseUtil.getStringValueOrEmpty(psMap.get(STATE)).startsWith("Z")
134                 && ParseUtil.parseDHMSOrDefault(psMap.get(CPUTIME), 0L) == 0L
135                 && ParseUtil.parseLongOrDefault(psMap.get(MINFLT), 0L) == 0L
136                 && ParseUtil.parseLongOrDefault(psMap.get(NVCSW), 0L) == 0L
137                 && ParseUtil.parseLongOrDefault(psMap.get(NIVCSW), 0L) == 0L;
138     }

150     @Override
151     protected String queryPath(Map<BsdPsKeyword, String> psMap) {
152         String path = queryExecutablePath();
153         if (!path.isEmpty()) {
154             return path;
155         }
156         // Without the kernel's answer, the path is argv[0]. NetBSD's ps prints argv[0] in the comm column, the same
157         // value the args column begins with, but clips it to the width of its "COMMAND" header - seven characters -
158         // because only the last column is left unbounded. So argv[0] is taken from the args column, which is queried
159         // last and is therefore printed in full.
160         String argv = parseArgv0(psMap.get(BsdPsKeyword.ARGS));
161         // The placeholder says the arguments were unavailable, so no path is known
162         return isPlaceholder(argv) ? "" : argv;
163     }

186     /**
187      * Extracts the command line from a {@code ps} args column.
188      *
189      * @param args the args column, the process's arguments separated by spaces
190      * @return the arguments, or an empty string where {@code ps} substituted the parenthesized command because the
191      *         kernel would not release them, which is a placeholder rather than a command line
192      */
193     static String parseCommandLine(@Nullable String args) {
194         String argv = ParseUtil.getStringValueOrEmpty(args);
195         return isPlaceholder(parseArgv0(argv)) ? "" : argv;
196     }

198     @Override
199     protected String queryName(Map<BsdPsKeyword, String> psMap, String path) {
200         // From argv[0] rather than the path, so that it matches the name each of the process's threads is given
201         return parseName(psMap.get(BsdPsKeyword.ARGS));
202     }

204     /**
205      * Extracts the process name from a {@code ps} args column: the last path component of {@code argv[0]}.
206      * <p>
207      * The placeholder {@code ps} substitutes when the kernel refuses the arguments wraps the kernel's own name for the
208      * process, which is the name whether or not the arguments could be read, so it is unwrapped rather than reporting
209      * {@code ps}'s punctuation as the name. Either way two readings of one process give the same name.
210      *
211      * @param args the args column, the process's arguments separated by spaces
212      * @return the process name
213      */
214     static String parseName(@Nullable String args) {
215         String argv = parseArgv0(args);
216         return isPlaceholder(argv) ? argv.substring(1, argv.length() - 1) : argv.substring(argv.lastIndexOf('/') + 1);
217     }
218 
219     /**
220      * Extracts {@code argv[0]} from a {@code ps} args column.
221      *
222      * @param args the args column, the process's arguments separated by spaces
223      * @return {@code argv[0]}, which is a path when the process was started by one, or the command in parentheses where
224      *         the kernel would not release the arguments
225      */
226     static String parseArgv0(@Nullable String args) {
227         String argv = ParseUtil.getStringValueOrEmpty(args).trim();
228         int space = argv.indexOf(' ');
229         return space < 0 ? argv : argv.substring(0, space);
230     }

232     /**
233      * Whether a {@code ps} args column holds the parenthesized command that {@code ps} substitutes when the kernel
234      * refuses a process's arguments, which it does while that process is inside {@code posix_spawn}.
235      *
236      * @param argv the value from {@link #parseArgv0(String)}
237      * @return {@code true} if it is that placeholder rather than a real {@code argv[0]}
238      */
239     static boolean isPlaceholder(String argv) {
240         return argv.length() > 2 && argv.charAt(0) == '(' && argv.charAt(argv.length() - 1) == ')';
241     }

243     @Override
244     protected List<String> queryArguments() {
245         // NetBSD provides command line via /proc filesystem
246         String cmdline = "/proc/" + getProcessID() + "/cmdline";
247         // The kernel refuses to read a process's arguments while that process is inside posix_spawn, which a JVM is
248         // whenever it runs a command, so an empty read of a file that exists is retried. This process is the case OSHI
249         // provokes itself, by running ps while another thread spawns, and so is given the longer budget; for any other
250         // process an empty read is usually genuine. The result is memoized for the life of this object, so a single
251         // refusal would otherwise stick.
252         int attempts = getProcessID() == this.os.getProcessId() ? SPAWN_RETRY_ATTEMPTS_SELF
253                 : SPAWN_RETRY_ATTEMPTS_OTHER;
254         for (int attempt = 1;; attempt++) {
255             byte[] cmdBytes = FileUtil.readAllBytes(cmdline, false);
256             if (cmdBytes.length > 0) {
257                 return Collections.unmodifiableList(ParseUtil.parseByteArrayToStrings(cmdBytes));
258             }
259             if (attempt >= attempts || !new File(cmdline).exists()) {
260                 return Collections.emptyList();
261             }
262             Util.sleep(SPAWN_RETRY_MILLIS);
263         }
264     }

286     /**
287      * Parses the environment returned by the {@code kern.proc_args} sysctl.
288      *
289      * @param envBytes NUL-delimited {@code KEY=VALUE} entries
290      * @return the environment, empty if there were no entries
291      */
292     static Map<String, String> parseEnvironment(byte[] envBytes) {
293         // Parse by raw bytes rather than by String offsets, which would misalign on a multibyte charset
294         Map<String, String> env = new LinkedHashMap<>();
295         for (String envStr : ParseUtil.parseByteArrayToStrings(envBytes)) {
296             int idx = envStr.indexOf('=');
297             if (idx > 0) {
298                 env.put(envStr.substring(0, idx), envStr.substring(idx + 1));
299             }
300         }
301         return env.isEmpty() ? Collections.emptyMap() : Collections.unmodifiableMap(env);
302     }

325     private long queryOwnOpenFileLimit(boolean soft) {
326         if (getProcessID() == this.os.getProcessId()) {
327             // getrlimit where a backend can call it; otherwise the sysctl a child process inherits, then the system max
328             long limit = queryRlimitNofile(soft);
329             if (limit >= 0) {
330                 return limit;
331             }
332             limit = BsdSysctlUtil.sysctl("proc.curproc.rlimit.descriptors." + (soft ? "soft" : "hard"), 0L);
333             if (limit <= 0) {
334                 limit = BsdSysctlUtil.sysctl("kern.maxfiles", 0L);
335             }
336             return limit > 0 ? limit : -1L;
337         }
338         return -1L;
339     }

348     @Override
349     public long getAffinityMask() {
350         long bitMask = 0L;
351         // schedctl -p <pid> shows affinity; -A cpus would set it
352         List<String> schedctl = ExecutingCommand.runNative("schedctl -p " + getProcessID());
353         for (String line : schedctl) {
354             // Output includes "Affinity: <list>" when bound
355             if (line.contains("Affinity:")) {
356                 String[] parts = COMMA_OR_WHITESPACE.split(AFFINITY.split(line, -1)[1].trim(), -1);
357                 for (String part : parts) {
358                     int bitToSet = ParseUtil.parseIntOrDefault(part.trim(), -1);
359                     if (bitToSet >= 0) {
360                         bitMask |= 1L << bitToSet;
361                     }
362                 }
363                 return bitMask;
364             }
365         }
366         // Not bound — return all CPUs
367         int ncpu = BsdSysctlUtil.sysctl("hw.ncpuonline", 1);
368         return ncpu < 64 ? (1L << ncpu) - 1 : -1L;
369     }

371     @Override
372     public List<OSThread> getThreadDetails() {
373         String psCommand = NetBsdOSThread.PS_THREAD_COMMAND;
374         if (getProcessID() >= 0) {
375             psCommand += " -p " + getProcessID();
376         }
377         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsArgs = threadMap -> threadMap
378                 .containsKey(BsdPsThreadKeyword.ARGS);
379         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
380                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class, NetBsdOSThread.PS_THREAD_KEYWORDS,
381                         thread.trim(), ' '))
382                 .filter(hasColumnsArgs).map(threadMap -> new NetBsdOSThread(getProcessID(), threadMap))
383                 .filter(VALID_THREAD).collect(Collectors.toList());
384     }

386     @Override
387     protected void updateThreadCount() {
388         // Use nlwp keyword to get LWP count for this process
389         List<String> nlwpList = ExecutingCommand.runNative("ps -o nlwp -p " + getProcessID());
390         if (nlwpList.size() > 1) {
391             this.threadCount = ParseUtil.parseIntOrDefault(nlwpList.get(1).trim(), 1);
392         } else {
393             this.threadCount = 1;
394         }
395     }

402     /**
403      * Parses the kernel and user time from NetBSD's Linux-format {@code /proc/<pid>/stat}, which ps does not separate.
404      * Package-private for testing.
405      * <p>
406      * Fields 14 and 15 are the {@code p_uutime} and {@code p_ustime} of the process's {@code kinfo_proc2}, in a fixed
407      * 100 ticks per second whatever the kernel's {@code hz} ({@code UTIME2TICKS} in {@code procfs_linux.c}). The
408      * command name is parenthesized and may itself contain spaces or parentheses, so fields are counted from the last
409      * closing parenthesis, after which field 3, the state, comes first.
410      *
411      * @param stat the contents of {@code /proc/<pid>/stat}, empty if procfs is not mounted or the process has exited
412      * @return the kernel and user time in milliseconds, in that order, or an empty array if they could not be read
413      */
414     static long[] parseKernelUserTime(String stat) {
415         int nameEnd = stat.lastIndexOf(')');
416         if (nameEnd < 0) {
417             return new long[0];
418         }
419         String[] split = ParseUtil.whitespaces.split(stat.substring(nameEnd + 1).trim(), -1);
420         // Field n of the Linux format is at index n - 3
421         if (split.length < 13) {
422             return new long[0];
423         }
424         long utime = ParseUtil.parseLongOrDefault(split[11], -1L);
425         long stime = ParseUtil.parseLongOrDefault(split[12], -1L);
426         if (utime < 0L || stime < 0L) {
427             return new long[0];
428         }
429         return new long[] { stime * 10L, utime * 10L };
430     }
