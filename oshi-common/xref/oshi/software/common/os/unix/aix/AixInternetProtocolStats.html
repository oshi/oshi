5   package oshi.software.common.os.unix.aix;
6   
7   import org.jspecify.annotations.Nullable;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.software.common.AbstractInternetProtocolStats;
11  
12  /**
13   * Abstract base for AIX InternetProtocolStats. Both backends iterate {@code perfstat_protocol_t} and surface the TCP
14   * and UDP variants; the shared decoding lives here, the per-backend data fetch in concrete subclasses.
15   */
16  @ThreadSafe
17  public abstract class AixInternetProtocolStats extends AbstractInternetProtocolStats {
18  
19      @Override
20      public TcpStats getTCPv4Stats() {
21          TcpStats stats = queryTcpStats();
22          return stats == null ? new TcpStats(0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L) : stats;
23      }
24  
25      @Override
26      public UdpStats getUDPv4Stats() {
27          UdpStats stats = queryUdpStats();
28          return stats == null ? new UdpStats(0L, 0L, 0L, 0L) : stats;
29      }
30  
31      /**
32       * Returns the TCP stats from the subclass's perfstat data source.
33       *
34       * @return populated {@link TcpStats}, or {@code null} if the {@code tcp} entry was not found
35       */
36      protected abstract @Nullable TcpStats queryTcpStats();
37  
38      /**
39       * Returns the UDP stats from the subclass's perfstat data source.
40       *
41       * @return populated {@link UdpStats}, or {@code null} if the {@code udp} entry was not found
42       */
43      protected abstract @Nullable UdpStats queryUdpStats();
44  
45      /**
46       * Maps the TCP counters of {@code perfstat_protocol_t} onto {@link TcpStats}.
47       * <p>
48       * {@code perfstat} has no count of connections that failed before being established, nor of retransmitted segments,
49       * so both are 0. Its {@code dropped}, the established connections dropped, is the reset count, which
50       * {@code netstat -s} prints as the drops among connections closed. Its packets sent exclude retransmissions.
51       *
52       * @param established connections established
53       * @param initiated   connections initiated, the active opens
54       * @param accepted    connections accepted, the passive opens
55       * @param dropped     established connections dropped
56       * @param opackets    packets sent
57       * @param ipackets    packets received
58       * @param ierrors     input errors
59       * @return the TCP statistics
60       */
61      protected static TcpStats tcpStats(long established, long initiated, long accepted, long dropped, long opackets,
62              long ipackets, long ierrors) {
63          return new TcpStats(established, initiated, accepted, 0L, dropped, opackets, ipackets, 0L, ierrors, 0L);
64      }
65  
66      /**
67       * Maps the UDP counters of {@code perfstat_protocol_t} onto {@link UdpStats}.
68       * <p>
69       * {@code perfstat}'s {@code ierrors} counts every datagram dropped on input, including those with no socket on
70       * their port, and it keeps no count of the broadcast and multicast ones among them that would let the rest be
71       * separated out. So the receive errors, which exclude datagrams with no application at their port, are 0. Its
72       * {@code no_socket} counts only unicast datagrams with no socket.
73       *
74       * @param opackets packets sent
75       * @param ipackets packets received
76       * @param noSocket unicast datagrams with no socket on their port
77       * @return the UDP statistics
78       */
79      protected static UdpStats udpStats(long opackets, long ipackets, long noSocket) {
80          return new UdpStats(opackets, ipackets, noSocket, 0L);
81      }
82  }
