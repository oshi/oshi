8   import static oshi.util.Memoizer.memoize;

10  import java.nio.ByteBuffer;
11  import java.nio.ByteOrder;
12  import java.util.function.Supplier;
13  
14  import org.jspecify.annotations.Nullable;
15  
16  import oshi.annotation.concurrent.ThreadSafe;
17  import oshi.software.common.AbstractInternetProtocolStats;
18  import oshi.util.driver.unix.NetStat;
19  import oshi.util.tuples.Pair;

21  /**
22   * Abstract base for the FreeBSD InternetProtocolStats. Parses the {@code net.inet.tcp.stats} and
23   * {@code net.inet.udp.stats} sysctls, falling back to {@code netstat -s} when the structure layout is not recognized.
24   * The JNA and FFM subclasses supply the raw sysctl reads and the per-connection reads.
25   * <p>
26   * Since FreeBSD 10.0 every counter in {@code struct tcpstat} and {@code struct udpstat} is a {@code uint64_t}. FreeBSD
27   * 14.0 inserted {@code tcps_progdrops} into the middle of {@code struct tcpstat}, moving every counter from
28   * {@code tcps_sndtotal} onward by 8 bytes. There is no version field, but the structure size identifies the layout: 880
29   * bytes (10.x and 11.x) or 904 bytes (12.x and 13.x) before the insertion, and 984 bytes or more after it.
30   */
31  @ThreadSafe
32  public abstract class FreeBsdInternetProtocolStats extends AbstractInternetProtocolStats {
33  
34      /** The TCP statistics sysctl. */
35      protected static final String TCPSTAT_SYSCTL = "net.inet.tcp.stats";
36      /** The UDP statistics sysctl. */
37      protected static final String UDPSTAT_SYSCTL = "net.inet.udp.stats";
38  
39      // struct tcpstat sizes bounding each known layout
40      private static final int TCPSTAT_MIN_SIZE = 880;
41      private static final int TCPSTAT_MAX_SIZE_BEFORE_PROGDROPS = 904;
42      private static final int TCPSTAT_MIN_SIZE_WITH_PROGDROPS = 984;
43      // struct tcpstat offsets ahead of tcps_progdrops, the same in every layout
44      private static final int TCPS_CONNATTEMPT = 0;
45      private static final int TCPS_ACCEPTS = 8;
46      private static final int TCPS_DROPS = 24;
47      private static final int TCPS_CONNDROPS = 32;
48      // struct tcpstat offsets after tcps_progdrops, as laid out before FreeBSD 14.0
49      private static final int TCPS_SNDPACK = 136;
50      private static final int TCPS_SNDREXMITPACK = 152;
51      private static final int TCPS_RCVPACK = 224;
52      private static final int TCPS_RCVBADSUM = 240;
53      private static final int TCPS_RCVBADOFF = 248;
54      private static final int TCPS_RCVSHORT = 264;
55      // The shift FreeBSD 14.0's tcps_progdrops applies to the offsets above
56      private static final int PROGDROPS_SHIFT = 8;
57  
58      // struct udpstat, unchanged since FreeBSD 10.0
59      private static final int UDPSTAT_SIZE = 112;
60      private static final int UDPS_IPACKETS = 0;
61      private static final int UDPS_HDROPS = 8;
62      private static final int UDPS_BADSUM = 16;
63      private static final int UDPS_BADLEN = 32;
64      private static final int UDPS_NOPORT = 40;
65      private static final int UDPS_NOPORTBCAST = 48;
66      private static final int UDPS_OPACKETS = 80;
67      private static final int UDPS_NOPORTMCAST = 96;
68  
69      private final Supplier<Pair<Long, Long>> establishedv4v6 = memoize(NetStat::queryTcpnetstat, defaultExpiration());
70      private final Supplier<TcpStats> tcpv4 = memoize(this::queryTcpv4Stats, defaultExpiration());
71      private final Supplier<UdpStats> udpv4 = memoize(this::queryUdpv4Stats, defaultExpiration());
72  
73      /**
74       * Reads a sysctl as raw bytes.
75       *
76       * @param name the sysctl name
77       * @return the sysctl value, or an empty array if the query failed
78       */
79      protected abstract byte[] querySysctl(String name);
80  
81      @Override
82      public TcpStats getTCPv4Stats() {
83          return tcpv4.get();
84      }
85  
86      @Override
87      public UdpStats getUDPv4Stats() {
88          return udpv4.get();
89      }
90  
91      private TcpStats queryTcpv4Stats() {
92          long established = establishedv4v6.get().getA();
93          TcpStats stats = parseTcpstat(querySysctl(TCPSTAT_SYSCTL), established);
94          if (stats != null) {
95              return stats;
96          }
97          return NetStat.queryTcpStats("netstat -s -p tcp", established);
98      }
99  
100     private UdpStats queryUdpv4Stats() {
101         UdpStats stats = parseUdpstat(querySysctl(UDPSTAT_SYSCTL));
102         return stats != null ? stats : NetStat.queryUdpStats("netstat -s -p udp");
103     }
104 
105     /**
106      * Parses a {@code struct tcpstat} read from the {@code net.inet.tcp.stats} sysctl.
107      *
108      * @param tcpstat     the raw structure
109      * @param established the number of connections currently established, which the structure does not hold
110      * @return the statistics, or {@code null} if the structure size matches no known layout
111      */
112     static @Nullable TcpStats parseTcpstat(byte[] tcpstat, long established) {
113         int shift;
114         if (tcpstat.length >= TCPSTAT_MIN_SIZE_WITH_PROGDROPS) {
115             shift = PROGDROPS_SHIFT;
116         } else if (tcpstat.length >= TCPSTAT_MIN_SIZE && tcpstat.length <= TCPSTAT_MAX_SIZE_BEFORE_PROGDROPS) {
117             shift = 0;
118         } else {
119             return null;
120         }
121         ByteBuffer bb = ByteBuffer.wrap(tcpstat).order(ByteOrder.nativeOrder());
122         return new TcpStats(established, bb.getLong(TCPS_CONNATTEMPT), bb.getLong(TCPS_ACCEPTS),
123                 bb.getLong(TCPS_CONNDROPS), bb.getLong(TCPS_DROPS), bb.getLong(TCPS_SNDPACK + shift),
124                 bb.getLong(TCPS_RCVPACK + shift), bb.getLong(TCPS_SNDREXMITPACK + shift),
125                 bb.getLong(TCPS_RCVBADSUM + shift) + bb.getLong(TCPS_RCVBADOFF + shift)
126                         + bb.getLong(TCPS_RCVSHORT + shift),
127                 0L);
128     }
129 
130     /**
131      * Parses a {@code struct udpstat} read from the {@code net.inet.udp.stats} sysctl.
132      *
133      * @param udpstat the raw structure
134      * @return the statistics, or {@code null} if the structure size matches no known layout
135      */
136     static @Nullable UdpStats parseUdpstat(byte[] udpstat) {
137         if (udpstat.length < UDPSTAT_SIZE) {
138             return null;
139         }
140         ByteBuffer bb = ByteBuffer.wrap(udpstat).order(ByteOrder.nativeOrder());
141         return new UdpStats(bb.getLong(UDPS_OPACKETS), bb.getLong(UDPS_IPACKETS),
142                 bb.getLong(UDPS_NOPORT) + bb.getLong(UDPS_NOPORTBCAST) + bb.getLong(UDPS_NOPORTMCAST),
143                 bb.getLong(UDPS_HDROPS) + bb.getLong(UDPS_BADSUM) + bb.getLong(UDPS_BADLEN));
144     }
145 }
