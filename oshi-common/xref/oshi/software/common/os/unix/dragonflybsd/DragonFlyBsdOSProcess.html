44  /**
45   * Abstract base for a DragonFly BSD OSProcess. Inherits the shared BSD {@code ps} parsing and adds the fields whose
46   * column set differs on DragonFly.
47   */
48  public abstract class DragonFlyBsdOSProcess extends BsdOSProcess {
49  
50      /**
51       * Ordered {@code ps} columns queried for each process. Shared by the DragonFlyBsdOSProcess subclasses (JNA/FFM) and
52       * the DragonFly OperatingSystem classes so the column list and parsing stay in lockstep. {@code COMMAND} must
53       * remain last.
54       */
55      public static final List<BsdPsKeyword> PS_KEYWORDS = Collections.unmodifiableList(Arrays.asList(STATE, PID, PPID,
56              USER, UID, RGID, NLWP, PRI, VSZ, RSS, TIME, MAJFLT, MINFLT, NVCSW, NIVCSW, UCOMM, COMMAND));
57  
58      public static final String PS_COMMAND_ARGS = PS_KEYWORDS.stream().map(Enum::name)
59              .map(name -> name.toLowerCase(Locale.ROOT)).collect(Collectors.joining(","));
60  
61      /**
62       * The {@code KERN_PROC} sysctl subtype that returns a process's executable path (12 on FreeBSD).
63       */
64      protected static final int KERN_PROC_PATHNAME = 9;
65  
66      /**
67       * The size of the buffer for an executable path, the platform's {@code PATH_MAX}.
68       */
69      protected static final int PATH_MAX = 1024;
70  
71      /**
72       * The longest command name the kernel keeps, {@code MAXCOMLEN}; {@code ucomm} is cut to this length.
73       */
74      private static final int MAXCOMLEN = 16;
75  
76      private final OperatingSystem os;
77  
78      /**
79       * Constructs a new {@code DragonFlyBsdOSProcess}.
80       *
81       * @param pid the process ID
82       * @param os  the operating system this process runs on
83       */
84      protected DragonFlyBsdOSProcess(int pid, OperatingSystem os) {
85          super(pid);
86          this.os = os;
87      }
88  
89      @Override
90      protected List<BsdPsKeyword> psKeywords() {
91          return PS_KEYWORDS;
92      }
93  
94      @Override
95      protected String psCommandArgs() {
96          return PS_COMMAND_ARGS;
97      }

99      @Override
100     public String getCurrentWorkingDirectory() {
101         return ProcstatUtil.getCwd(getProcessID());
102     }
103 
104     @Override
105     public long getOpenFiles() {
106         return ProcstatUtil.getOpenFiles(getProcessID());
107     }
108 
109     @Override
110     protected int queryOwnProcessId() {
111         return this.os.getProcessId();
112     }
113 
114     /**
115      * {@inheritDoc}
116      * <p>
117      * On DragonFly BSD only the current process is asked; every other process falls back to the {@code ps} column. The
118      * kernel's {@code KERN_PROC_PATHNAME} handler dereferences the target's file-descriptor table without checking it,
119      * and an exiting process frees that table before it drops its executable's name, so querying another process at the
120      * wrong moment panics the kernel.
121      */
122     @Override
123     protected final String queryExecutablePath() {
124         return getProcessID() == queryOwnProcessId() ? queryOwnExecutablePath() : "";
125     }
126 
127     /**
128      * Returns the full path of the current process's executable through the {@code KERN_PROC_PATHNAME} sysctl. Only
129      * called when this process is the current one; see {@link #queryExecutablePath()}.
130      *
131      * @return the path, or an empty string if it is unavailable
132      */
133     protected abstract String queryOwnExecutablePath();
134 
135     /**
136      * {@inheritDoc}
137      * <p>
138      * Every process but the current one falls back to {@code argv[0]} from {@code /proc/<pid>/cmdline}, which holds the
139      * path when the process was started by one and is not cut to {@code MAXCOMLEN} as {@code ucomm} is. A process can
140      * rewrite {@code argv[0]}, so it is used only when it names the same program as {@code ucomm}; otherwise, or
141      * without procfs, the result is {@code ucomm}.
142      */
143     @Override
144     protected String queryPath(Map<BsdPsKeyword, String> psMap) {
145         String path = queryExecutablePath();
146         if (!path.isEmpty()) {
147             return path;
148         }
149         String ucomm = ParseUtil.getStringValueOrEmpty(psMap.get(UCOMM));
150         // The NUL-separated arguments keep a path with spaces whole, which ps's space-separated column does not
151         List<String> args = queryArguments();
152         String argv0 = args.isEmpty() ? "" : args.get(0);
153         return !ucomm.isEmpty() && namesProgram(argv0.substring(argv0.lastIndexOf('/') + 1), ucomm) ? argv0 : ucomm;
154     }
155 
156     /**
157      * Whether the last element of {@code argv[0]} names the program whose kernel command name is {@code ucomm}.
158      *
159      * @param argv0Name the last element of {@code argv[0]}
160      * @param ucomm     the kernel's command name, cut to {@code MAXCOMLEN}
161      * @return {@code true} if it is the same name, or extends a truncated {@code ucomm} with more of a file name
162      */
163     static boolean namesProgram(String argv0Name, String ucomm) {
164         if (argv0Name.equals(ucomm)) {
165             return true;
166         }
167         // A setproctitle() title, "name: status", also extends ucomm, but a file name rarely holds a colon or a space
168         return ucomm.length() == MAXCOMLEN && argv0Name.startsWith(ucomm)
169                 && argv0Name.substring(MAXCOMLEN).chars().noneMatch(c -> c == ':' || Character.isWhitespace(c));
170     }
171 
172     @Override
173     protected int queryBitness() {
174         // DragonFly BSD has no KERN_PROC_SV_NAME (its KERN_PROC subtype 9 is the executable's path) and no ABI
175         // compatibility layer: an x86_64 kernel runs only 64-bit programs (kern.elf64 exists, kern.elf32 does not), and
176         // the i386 kernel of 3.8 and earlier only 32-bit ones. So every process has the operating system's bitness.
177         return this.os.getBitness();
178     }
179 
180     @Override
181     protected long queryStartTimeMillis() {
182         return queryStartTimeFromProc(getProcessID());
183     }
184 
185     private static long queryStartTimeFromProc(int pid) {
186         List<String> status = FileUtil.readFile("/proc/" + pid + "/status", false);
187         if (!status.isEmpty()) {
188             // Format: name pid ... startSec,startUsec ...
189             String[] split = ParseUtil.whitespaces.split(status.get(0).trim(), -1);
190             if (split.length >= 8) {
191                 String[] timeParts = split[7].split(",", -1);
192                 long seconds = ParseUtil.parseLongOrDefault(timeParts[0], 0L);
193                 if (seconds > 0) {
194                     return seconds * 1000L;
195                 }
196             }
197         }
198         // Unknown start time: return the sentinel so updateAttributes() falls back to elapsed time
199         return -1L;
200     }

202     @Override
203     public List<OSThread> getThreadDetails() {
204         String psCommand = "ps -awwxo " + DragonFlyBsdOSThread.PS_THREAD_COLUMNS + " -H";
205         if (getProcessID() >= 0) {
206             psCommand += " -p " + getProcessID();
207         }
208         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsPri = threadMap -> threadMap
209                 .containsKey(BsdPsThreadKeyword.PRI);
210         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
211                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class,
212                         DragonFlyBsdOSThread.PS_THREAD_KEYWORDS, thread.trim(), ' '))
213                 .filter(hasColumnsPri).map(threadMap -> new DragonFlyBsdOSThread(getProcessID(), threadMap))
214                 .filter(VALID_THREAD).collect(Collectors.toList());
215     }
216 
217     @Override
218     public long getAffinityMask() {
219         // DragonFly has no cpuset command, which the other BSDs read affinity from
220         return queryAffinityMask();
221     }
222 
223     /**
224      * Reads this process's CPU affinity mask through {@code sched_getaffinity}, which reports the mask of its first LWP
225      * limited to the active CPUs. The kernel allows it for any process.
226      *
227      * @return the mask of CPUs 0 to 63, or 0 if the process has exited
228      */
229     protected abstract long queryAffinityMask();
