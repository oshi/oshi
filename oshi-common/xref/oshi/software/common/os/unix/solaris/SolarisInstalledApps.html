7   import java.time.OffsetDateTime;
8   import java.time.format.DateTimeFormatter;
9   import java.time.format.DateTimeParseException;
10  import java.util.ArrayList;
11  import java.util.LinkedHashMap;
12  import java.util.List;
13  import java.util.Locale;
14  import java.util.Map;
15  import java.util.regex.Matcher;
16  import java.util.regex.Pattern;
17  
18  import oshi.annotation.concurrent.ThreadSafe;
19  import oshi.software.os.ApplicationInfo;
20  import oshi.util.ExecutingCommand;
21  
22  /**
23   * Queries the packages installed on Solaris and illumos from the Image Packaging System, {@code pkg}.
24   */
25  @ThreadSafe
26  public final class SolarisInstalledApps {
27  
28      /**
29       * A {@code pkg list -Hv} line: the FMRI {@code pkg://publisher/name@version:timestamp}, then the state flags, as in
30       * {@code pkg://solaris/archiver/gnu-tar@1.35-11.4.95.0.1.227.0:20260810T173115Z       i--}.
31       */
32      private static final Pattern FMRI = Pattern
33              .compile("pkg://([^/]++)/([^@\\s]++)@([^:\\s]++):(\\d{8}T\\d{6}Z)\\s++(\\S++)");
34      /** The FMRI's timestamp, in ISO 8601 basic format and UTC. */
35      private static final DateTimeFormatter FMRI_TIME = DateTimeFormatter.ofPattern("yyyyMMdd'T'HHmmssX", Locale.ROOT);
36  
37      private SolarisInstalledApps() {
38      }
39  
40      /**
41       * Queries {@code pkg list} for the installed packages, which needs no privilege.
42       *
43       * @return the installed applications
44       */
45      public static List<ApplicationInfo> queryInstalledApps() {
46          return parsePkgList(ExecutingCommand.runNative("pkg list -Hv"));
47      }
48  
49      /**
50       * Parses the output of {@code pkg list -Hv}. Package-private for testing.
51       * <p>
52       * The publisher, such as {@code solaris} or {@code openindiana.org}, is the vendor. {@code pkg list} does not show
53       * when a package was installed, and {@code pkg info}, which does, would cost a further command for every package,
54       * so the timestamp is the one in the FMRI, which is when that version of the package was published.
55       *
56       * @param lines the lines emitted by {@code pkg list -Hv}
57       * @return the installed applications
58       */
59      static List<ApplicationInfo> parsePkgList(List<String> lines) {
60          List<ApplicationInfo> apps = new ArrayList<>();
61          for (String line : lines) {
62              Matcher m = FMRI.matcher(line.trim());
63              if (!m.matches()) {
64                  continue;
65              }
66              Map<String, String> additionalInfo = new LinkedHashMap<>();
67              additionalInfo.put("fmri", line.trim().split("\\s+", 2)[0]);
68              additionalInfo.put("state", m.group(5));
69              apps.add(
70                      new ApplicationInfo(m.group(2), m.group(3), m.group(1), parseFmriTime(m.group(4)), additionalInfo));
71          }
72          return apps;
73      }
74  
75      private static long parseFmriTime(String timestamp) {
76          try {
77              return OffsetDateTime.parse(timestamp, FMRI_TIME).toInstant().toEpochMilli();
78          } catch (DateTimeParseException e) {
79              return 0L;
80          }
81      }
82  }
