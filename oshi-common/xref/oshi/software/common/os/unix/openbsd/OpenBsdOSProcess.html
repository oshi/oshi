46  /**
47   * Abstract base for an OpenBSD OSProcess. Inherits the shared BSD {@code ps} parsing and adds the fields whose column
48   * set differs on OpenBSD.
49   */
50  public abstract class OpenBsdOSProcess extends BsdOSProcess {
51  
52      /**
53       * Ordered {@code ps} columns queried for each process. Shared by the OpenBsdOSProcess subclasses (JNA/FFM) and the
54       * OpenBSD OperatingSystem classes so the column list and parsing stay in lockstep. {@code ARGS} must remain last.
55       */
56      public static final List<BsdPsKeyword> PS_KEYWORDS = Collections.unmodifiableList(Arrays.asList(STATE, PID, PPID,
57              USER, UID, GROUP, GID, PRI, VSZ, RSS, ETIME, CPUTIME, COMM, MAJFLT, MINFLT, NVCSW, NIVCSW, ARGS));
58  
59      /** The {@code cs_flags} bit of {@code struct cpustats} marking a processor the scheduler may use. */
60      private static final long CPUSTATS_ONLINE = 0x1L;
61  
62      /** The {@code sysctl} node of process entries, {@code KERN_PROC} in {@code <sys/sysctl.h>}. */
63      protected static final int KERN_PROC = 66;
64      /** The {@code KERN_PROC} operation selecting one process by id, {@code KERN_PROC_PID}. */
65      protected static final int KERN_PROC_PID = 1;
66  
67      // struct kinfo_proc in <sys/sysctl.h>: fixed-width fields, laid out the same on every architecture up to the time
68      // fields. The kernel copies as many bytes as the caller asks for, so only this leading part is requested.
69      private static final int KINFO_PROC_PID = 108;
70      private static final int KINFO_PROC_UVALID = 400;
71      private static final int KINFO_PROC_UUTIME_SEC = 420;
72      private static final int KINFO_PROC_UUTIME_USEC = 424;
73      private static final int KINFO_PROC_USTIME_SEC = 428;
74      private static final int KINFO_PROC_USTIME_USEC = 432;
75      /** The number of leading bytes of {@code struct kinfo_proc} to request, through {@code p_ustime_usec}. */
76      protected static final int KINFO_PROC_PREFIX_SIZE = 436;
77  
78      public static final String PS_COMMAND_ARGS = PS_KEYWORDS.stream().map(Enum::name)
79              .map(name -> name.toLowerCase(Locale.ROOT)).collect(Collectors.joining(","));

100     @Override
101     public long getAffinityMask() {
102         // OpenBSD has no interface to bind a process to processors, so every online processor is eligible. Processors
103         // are taken offline as a set (SMT siblings when hw.smt=0), so the online ones need not be contiguous.
104         return onlineCpuMask(queryCpuCount(), this::queryCpuStats);
105     }
106 
107     /**
108      * Builds a mask of the online processors from each processor's {@code kern.cpustats} record.
109      *
110      * @param cpuCount the number of processors the kernel found ({@code hw.ncpufound})
111      * @param cpuStats the {@code kern.cpustats} record for a processor: its {@code cs_time} array followed by
112      *                 {@code cs_flags}, or an empty array if it could not be read
113      * @return a mask with a bit set for each online processor, or 0 if none could be read
114      */
115     static long onlineCpuMask(int cpuCount, IntFunction<long[]> cpuStats) {
116         long mask = 0L;
117         for (int cpu = 0; cpu < cpuCount && cpu < Long.SIZE; cpu++) {
118             long[] stats = cpuStats.apply(cpu);
119             // cs_flags follows the cs_time array, whose length depends on the release
120             if (stats.length > 0 && (stats[stats.length - 1] & CPUSTATS_ONLINE) != 0) {
121                 mask |= 1L << cpu;
122             }
123         }
124         return mask;
125     }

127     @Override
128     protected long[] queryKernelUserTime() {
129         return parseKernelUserTime(queryKinfoProcPrefix(), getProcessID());
130     }
131 
132     /**
133      * Queries the leading {@link #KINFO_PROC_PREFIX_SIZE} bytes of this process's {@code struct kinfo_proc}, through
134      * {@code sysctl} with the name {@code CTL_KERN, KERN_PROC, KERN_PROC_PID, pid, KINFO_PROC_PREFIX_SIZE, 1}.
135      *
136      * @return the bytes, or an empty array if the process is not listed or the call failed
137      */
138     protected abstract byte[] queryKinfoProcPrefix();
139 
140     /**
141      * Parses the user and system time from the leading bytes of a {@code struct kinfo_proc}, which OpenBSD's ps does
142      * not separate. Package-private for testing.
143      * <p>
144      * {@code fill_kproc()} fills {@code p_uutime} and {@code p_ustime} from the whole process's accumulated usage, and
145      * leaves them zero for a process that is exiting; {@code p_uvalid} is clear only once it is a zombie. So an entry
146      * without {@code p_uvalid}, or with both times zero, is not used. The entry's {@code p_pid} is checked against the
147      * process asked for, which also confirms the offsets.
148      *
149      * @param kinfo the leading bytes of the entry, in the platform's byte order
150      * @param pid   the process ID the entry was requested for
151      * @return the kernel and user time in milliseconds, in that order, or an empty array if they could not be read or
152      *         are both zero
153      */
154     static long[] parseKernelUserTime(byte[] kinfo, int pid) {
155         if (kinfo.length < KINFO_PROC_PREFIX_SIZE) {
156             return new long[0];
157         }
158         ByteBuffer entry = ByteBuffer.wrap(kinfo).order(ByteOrder.nativeOrder());
159         if (entry.getInt(KINFO_PROC_PID) != pid || entry.getLong(KINFO_PROC_UVALID) == 0L) {
160             return new long[0];
161         }
162         long system = timevalMillis(entry, KINFO_PROC_USTIME_SEC, KINFO_PROC_USTIME_USEC);
163         long user = timevalMillis(entry, KINFO_PROC_UUTIME_SEC, KINFO_PROC_UUTIME_USEC);
164         // An exiting process keeps p_uvalid until it is a zombie, but its times are left zero. Its PS_EXITING flag is
165         // in
166         // p_psflags, past fields whose offsets depend on the architecture's alignment, so zero is treated as unknown:
167         // the ps total stands in, which for a live process that has used no CPU time is zero too
168         if (system == 0L && user == 0L) {
169             return new long[0];
170         }
171         return new long[] { system, user };
172     }
173 
174     private static long timevalMillis(ByteBuffer entry, int secOffset, int usecOffset) {
175         // Both halves are u_int32_t
176         return Integer.toUnsignedLong(entry.getInt(secOffset)) * 1000L
177                 + Integer.toUnsignedLong(entry.getInt(usecOffset)) / 1000L;
178     }

211     @Override
212     protected void updateThreadCount() {
213         List<String> threadList = ExecutingCommand.runNative("ps -xHo tid -p " + getProcessID());
214         int count = this.threadCount;
215         if (!threadList.isEmpty()) {
216             // Subtract 1 for header
217             count = threadList.size() - 1;
218         }
219         // A live process always has at least one thread
220         this.threadCount = Math.max(count, 1);
221     }
222 
223     @Override
224     public List<OSThread> getThreadDetails() {
225         String psCommand = "ps -Hwwxo " + OpenBsdOSThread.PS_THREAD_COLUMNS;
226         if (getProcessID() >= 0) {
227             psCommand += " -p " + getProcessID();
228         }
229         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsArgs = threadMap -> threadMap
230                 .containsKey(BsdPsThreadKeyword.ARGS);
231         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
232                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class, OpenBsdOSThread.PS_THREAD_KEYWORDS,
233                         thread.trim(), ' '))
234                 .filter(hasColumnsArgs).map(threadMap -> new OpenBsdOSThread(getProcessID(), threadMap))
235                 .filter(VALID_THREAD).collect(Collectors.toList());
236     }
