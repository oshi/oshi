22  /**
23   * Common Windows NetworkParams logic shared between the JNA and FFM implementations. The default-gateway lookups parse
24   * {@code route print} output and live here, as does all of the routing table logic; domain/host name, DNS-server
25   * resolution and the IP Helper API routing table read are native and provided by the subclasses.
26   */
27  @ThreadSafe
28  public abstract class WindowsNetworkParams extends AbstractNetworkParams {
29  
30      private static final Logger LOG = LoggerFactory.getLogger(WindowsNetworkParams.class);
31  
32      // Microsoft's recommended starting size for GetAdaptersAddresses, which avoids a second call on most systems
33      private static final int ADAPTER_BUFFER_SIZE = 15_000;
34      private static final int MAX_ADAPTER_ATTEMPTS = 3;
35      private static final int ERROR_SUCCESS = 0;
36      private static final int ERROR_BUFFER_OVERFLOW = 111;
37      private static final int ERROR_NO_DATA = 232;
38  
39      /** Default constructor. */
40      protected WindowsNetworkParams() {
41      }
42  
43      /**
44       * One row of the routing table as read from the IP Helper API by a backend.
45       * <p>
46       * This is a plain carrier so that the interpretation of the rows can live here rather than being written twice,
47       * once against JNA structures and once against FFM memory segments.
48       */
49      public static final class RouteRow {
50          /** The destination prefix address bytes, four for IPv4 or sixteen for IPv6. */
51          public byte[] destination = new byte[0];
52          /** The destination prefix length. */
53          public int prefixLength = -1;
54          /** The next hop address bytes, all zero for a directly attached route. */
55          public byte[] nextHop = new byte[0];
56          /** The index of the outgoing interface. */
57          public int interfaceIndex = -1;
58          /** The route metric. */
59          public long metric = -1L;
60      }
61  
62      /**
63       * Queries the routing table from the native IP Helper API.
64       *
65       * @return the rows, or an empty list if the query failed
66       */
67      protected abstract List<RouteRow> queryRouteRows();
68  
69      /**
70       * Makes one {@code GetAdaptersAddresses} call with a buffer of the given size and, if it succeeds, collects the DNS
71       * server addresses of every adapter that is up.
72       *
73       * @param bufferSize   the size of the buffer to allocate and pass
74       * @param requiredSize receives, in its only element, the size the call reports when the buffer was too small
75       * @param addresses    receives the address bytes in network order, four for IPv4 or sixteen for IPv6
76       * @return the call's return code, or -1 if the call itself could not be made
77       */
78      protected abstract int queryAdapterDnsServers(int bufferSize, int[] requiredSize, List<byte[]> addresses);
79  
80      /**
81       * Queries the DNS server addresses of every adapter that is up, growing the buffer when the adapter list outgrows
82       * it between the call that reports the size and the one that fills it.
83       *
84       * @return the address bytes in network order, or an empty list if the query failed
85       */
86      List<byte[]> queryDnsServerAddresses() {
87          int size = ADAPTER_BUFFER_SIZE;
88          int[] requiredSize = new int[1];
89          for (int attempt = 0; attempt < MAX_ADAPTER_ATTEMPTS; attempt++) {
90              List<byte[]> addresses = new ArrayList<>();
91              int ret = queryAdapterDnsServers(size, requiredSize, addresses);
92              if (ret == ERROR_BUFFER_OVERFLOW) {
93                  size = requiredSize[0];
94                  continue;
95              }
96              if (ret == ERROR_SUCCESS) {
97                  return addresses;
98              }
99              if (ret != ERROR_NO_DATA) {
100                 LOG.error("Failed to get adapter addresses. Error code: {}", ret);
101             }
102             return new ArrayList<>();
103         }
104         LOG.warn("The adapter list outgrew the buffer {} times in a row.", MAX_ADAPTER_ATTEMPTS);
105         return new ArrayList<>();
106     }
107 
108     @Override
109     public String[] getDnsServers() {
110         Set<String> servers = new LinkedHashSet<>();
111         for (byte[] address : queryDnsServerAddresses()) {
112             if ((address.length == 4 || address.length == 16) && !isUnconfiguredIpv6Dns(address)) {
113                 servers.add(ParseUtil.formatIpAddress(address));
114             }
115         }
116         return servers.toArray(new String[0]);
117     }
118 
119     /**
120      * Whether an address is one of the deprecated site-local DNS addresses {@code fec0:0:0:ffff::1} to {@code ::3},
121      * which Windows reports for an adapter with no IPv6 DNS server configured.
122      *
123      * @param address the address bytes
124      * @return true for one of the three placeholder addresses
125      */
126     static boolean isUnconfiguredIpv6Dns(byte[] address) {
127         if (address.length != 16 || address[15] < 1 || address[15] > 3) {
128             return false;
129         }
130         byte[] prefix = { (byte) 0xfe, (byte) 0xc0, 0, 0, 0, 0, (byte) 0xff, (byte) 0xff };
131         for (int i = 0; i < 15; i++) {
132             if (address[i] != (i < prefix.length ? prefix[i] : 0)) {
133                 return false;
134             }
135         }
136         return true;
137     }

139     @Override
140     public List<NetworkParams.IPRoute> getRoutes() {
141         List<RouteRow> rows = queryRouteRows();
142         List<NetworkParams.IPRoute> routes = new ArrayList<>(rows.size());
143         if (rows.isEmpty()) {
144             // Skip the interface enumeration, which is the expensive part, when there is nothing to name
145             return routes;
146         }
147         // Windows reports only an interface index, so the name has to be looked up. Do it once for the whole table.
148         Map<Integer, String> namesByIndex = queryInterfaceNameByIndex();
149         for (RouteRow row : rows) {
150             int addressBits = row.destination.length * 8;
151             if (addressBits != 32 && addressBits != 128) {
152                 // A family we did not recognize. Dropping the row keeps the promise that a destination's length
153                 // always identifies the address family.
154                 continue;
155             }
156             // Windows publishes no flags, so a gateway route is one whose next hop is not the unspecified address.
157             boolean isGateway = !isUnspecified(row.nextHop);
158             routes.add(
159                     new NetworkParams.IPRoute(row.destination, row.prefixLength, isGateway ? row.nextHop : new byte[0],
160                             ParseUtil.getStringValueOrEmpty(namesByIndex.get(row.interfaceIndex)), row.interfaceIndex,
161                             row.metric, isGateway, row.prefixLength == addressBits));
162         }
163         return routes;
164     }
165 
166     private static boolean isUnspecified(byte[] address) {
167         for (byte b : address) {
168             if (b != 0) {
169                 return false;
170             }
171         }
172         return true;
173     }

175     @Override
176     public String getIpv4DefaultGateway() {
177         return parseIpv4Route();
178     }
179 
180     @Override
181     public String getIpv6DefaultGateway() {
182         return parseIpv6Route();
183     }
184 
185     private static String parseIpv4Route() {
186         List<String> lines = ExecutingCommand.runNative("route print -4 0.0.0.0");
187         for (String line : lines) {
188             String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
189             if (fields.length > 2 && "0.0.0.0".equals(fields[0])) {
190                 return fields[2];
191             }
192         }
193         return "";
194     }
195 
196     private static String parseIpv6Route() {
197         List<String> lines = ExecutingCommand.runNative("route print -6 ::/0");
198         for (String line : lines) {
199             String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
200             if (fields.length > 3 && "::/0".equals(fields[2])) {
201                 return fields[3];
202             }
203         }
204         return "";
205     }
