52      /*
53       * Cache full process stats queries. The second query only populates if the first one returns nothing.
54       */
55      private final Supplier<@Nullable Map<Integer, ProcessPerfCounterBlock>> processMapFromRegistry = memoize(
56              () -> buildProcessMapFromRegistry(null), defaultExpiration());
57      private final Supplier<@Nullable Map<Integer, ProcessPerfCounterBlock>> processMapFromPerfCounters = memoize(
58              () -> buildProcessMapFromPerfCounters(null), defaultExpiration());
59      /*
60       * Cache full thread stats queries. Only used if USE_PROCSTATE_SUSPENDED is set true.
61       */
62      private final Supplier<@Nullable Map<Integer, ThreadPerfCounterBlock>> threadMapFromRegistry = memoize(
63              () -> buildThreadMapFromRegistry(null), defaultExpiration());
64      private final Supplier<@Nullable Map<Integer, ThreadPerfCounterBlock>> threadMapFromPerfCounters = memoize(
65              () -> buildThreadMapFromPerfCounters(null), defaultExpiration());

157     /**
158      * Fetches the cached performance data for all processes, from the registry if it is available and from performance
159      * counters otherwise.
160      *
161      * @return A map of process ID to performance counter block, empty if neither source returned data
162      */
163     protected Map<Integer, ProcessPerfCounterBlock> getCachedProcessMap() {
164         return resolveProcessMap(null);
165     }
166 
167     private Map<Integer, ProcessPerfCounterBlock> resolveProcessMap(@Nullable Collection<Integer> pids) {
168         // Get data from the registry if possible
169         Map<Integer, ProcessPerfCounterBlock> processMap = processMapFromRegistry.get();
170         // otherwise performance counters with WMI backup
171         if (processMap == null || processMap.isEmpty()) {
172             processMap = (pids == null) ? processMapFromPerfCounters.get() : buildProcessMapFromPerfCounters(pids);
173         }
174         return processMap == null ? Collections.emptyMap() : processMap;
175     }
176 
177     private List<OSProcess> processMapToList(@Nullable Collection<Integer> pids) {
178         Map<Integer, ProcessPerfCounterBlock> processMap = resolveProcessMap(pids);
179         Map<Integer, ThreadPerfCounterBlock> threadMap = null;
180         if (USE_PROCSTATE_SUSPENDED) {
181             // Get data from the registry if possible
182             threadMap = threadMapFromRegistry.get();
183             // otherwise performance counters with WMI backup
184             if (threadMap == null || threadMap.isEmpty()) {
185                 threadMap = (pids == null) ? threadMapFromPerfCounters.get() : buildThreadMapFromPerfCounters(pids);
186             }
187         }
188 
189         Map<Integer, WtsInfo> processWtsMap = queryProcessWtsMap(pids);
190 
191         // Intersect with the WTS keys whether or not pids were requested. The perf counter map is memoized, so it can
192         // name a process that has since exited; WTS is queried fresh and will not. Keying off the WTS map keeps
193         // getProcess(pid) from returning a process that getProcesses() would not list.
194         Set<Integer> mapKeys = new HashSet<>(processWtsMap.keySet());
195         mapKeys.retainAll(processMap.keySet());
196 
197         final Map<Integer, ProcessPerfCounterBlock> finalProcessMap = processMap;
198         final Map<Integer, ThreadPerfCounterBlock> finalThreadMap = threadMap;
199         return mapKeys.stream().parallel()
200                 .map(pid -> createOSProcess(pid, finalProcessMap, processWtsMap, finalThreadMap)).filter(VALID_PROCESS)
201                 .collect(Collectors.toList());
202     }

216     @Override
217     protected Pair<String, OSVersionInfo> queryFamilyVersionInfo() {
218         String servicePack = "";
219         int suiteMask = 0;
220         String buildNumber = "";
221         WmiResult<OSVersionProperty> versionInfo = queryOsVersion();
222         if (versionInfo.getResultCount() > 0) {
223             servicePack = WmiUtil.getString(versionInfo, OSVersionProperty.CSDVERSION, 0);
224             suiteMask = WmiUtil.getUint32(versionInfo, OSVersionProperty.SUITEMASK, 0);
225             buildNumber = WmiUtil.getString(versionInfo, OSVersionProperty.BUILDNUMBER, 0);
226         }
227         return parseVersionInfo(System.getProperty("os.name"), servicePack, suiteMask, buildNumber);
228     }

230     /**
231      * Assembles the family and version information from the raw values reported by the JDK and by WMI.
232      *
233      * @param osName      The value of the {@code os.name} system property
234      * @param servicePack The service pack name, empty or {@link Constants#UNKNOWN} if none
235      * @param suiteMask   The suite mask bitmask
236      * @param buildNumber The build number reported by WMI
237      * @return A pair of the family name and the version information
238      */
239     static Pair<String, OSVersionInfo> parseVersionInfo(String osName, String servicePack, int suiteMask,
240             String buildNumber) {
241         String version = osName.startsWith("Windows ") ? osName.substring(8) : osName;
242         if (!servicePack.isEmpty() && !Constants.UNKNOWN.equals(servicePack)) {
243             version = version + " " + servicePack.replace("Service Pack ", "SP");
244         }
245         return new Pair<>("Windows",
246                 new OSVersionInfo(resolveVersionAlias(version, buildNumber), parseCodeName(suiteMask), buildNumber));
247     }
248 
249     /**
250      * Maps the version name reported by the JDK to the name of the release that actually shipped with the given build
251      * number. Older JDKs predate Windows 11 and the Server releases after 2016, so {@code os.name} reports them under
252      * the name of the last release the JDK knew about.
253      *
254      * @param version     The version name derived from {@code os.name}
255      * @param buildNumber The build number reported by WMI
256      * @return The version name of the release matching {@code buildNumber}
257      */
258     static String resolveVersionAlias(String version, String buildNumber) {
259         if ("10".equals(version) && buildNumber.compareTo("22000") >= 0) {
260             return "11";
261         }
262         if ("Server 2016".equals(version) && buildNumber.compareTo("17762") > 0) {
263             version = "Server 2019";
264         }
265         if ("Server 2019".equals(version) && buildNumber.compareTo("20347") > 0) {
266             version = "Server 2022";
267         }
268         if ("Server 2022".equals(version) && buildNumber.compareTo("26039") > 0) {
269             version = "Server 2025";
270         }
271         return version;
272     }

274     /**
275      * Gets suites available on the system and return as a codename.
276      *
277      * @param suiteMask The suite mask bitmask
278      * @return Suites
279      */
280     protected static String parseCodeName(int suiteMask) {
281         List<String> suites = new ArrayList<>();
282         if ((suiteMask & 0x00000002) != 0) {
283             suites.add("Enterprise");
284         }
285         if ((suiteMask & 0x00000004) != 0) {
286             suites.add("BackOffice");
287         }
288         if ((suiteMask & 0x00000008) != 0) {
289             suites.add("Communications Server");
290         }
291         if ((suiteMask & 0x00000080) != 0) {
292             suites.add("Datacenter");
293         }
294         if ((suiteMask & 0x00000200) != 0) {
295             suites.add("Home");
296         }
297         if ((suiteMask & 0x00000400) != 0) {
298             suites.add("Web Server");
299         }
300         if ((suiteMask & 0x00002000) != 0) {
301             suites.add("Storage Server");
302         }
303         if ((suiteMask & 0x00004000) != 0) {
304             suites.add("Compute Cluster");
305         }
306         if ((suiteMask & 0x00008000) != 0) {
307             suites.add("Home Server");
308         }
309         return String.join(",", suites);
310     }
