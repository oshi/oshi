29  /**
30   * Common NetworkParams implementation.
31   */
32  @ThreadSafe
33  public abstract class AbstractNetworkParams implements NetworkParams {
34  
35      private static final Logger LOG = LoggerFactory.getLogger(AbstractNetworkParams.class);
36  
37      private static final Pattern SERVER_VALUE_DELIM = Pattern.compile("[ \t#;]");
38  
39      /**
40       * Default constructor.
41       */
42      protected AbstractNetworkParams() {
43      }
44  
45      private static final String NAMESERVER = "nameserver";
46  
47      @Override
48      public String getDomainName() {
49          InetAddress localHost;
50          try {
51              localHost = InetAddress.getLocalHost();
52          } catch (UnknownHostException e) {
53              localHost = InetAddress.getLoopbackAddress();
54          }
55          return localHost.getCanonicalHostName();
56      }
57  
58      @Override
59      public String getHostName() {
60          InetAddress localHost;
61          try {
62              localHost = InetAddress.getLocalHost();
63          } catch (UnknownHostException e) {
64              localHost = InetAddress.getLoopbackAddress();
65          }
66          String hn = localHost.getHostName();
67          int dot = hn.indexOf('.');
68          if (dot == -1) {
69              return hn;
70          }
71          return hn.substring(0, dot);
72      }
73  
74      @Override
75      public String[] getDnsServers() {
76          List<String> resolv = FileUtil.readFile("/etc/resolv.conf");
77          String key = NAMESERVER;
78          int maxNameServer = 3;
79          List<String> servers = new ArrayList<>();
80          for (int i = 0; i < resolv.size() && servers.size() < maxNameServer; i++) {
81              String line = resolv.get(i);
82              if (line.startsWith(key)) {
83                  String value = line.substring(key.length()).replaceFirst("^[ \t]+", "");
84                  if (!value.isEmpty() && value.charAt(0) != '#' && value.charAt(0) != ';') {
85                      String val = SERVER_VALUE_DELIM.split(value, 2)[0];
86                      servers.add(val);
87                  }
88              }
89          }
90          return servers.toArray(new String[0]);
91      }
92  
93      /**
94       * Convenience method to parse the output of the `route` command. While the command arguments vary between OS's the
95       * output is consistently parsable.
96       *
97       * @param lines output of OS-specific route command
98       * @return default gateway
99       */
100     protected static String searchGateway(List<String> lines) {
101         for (String line : lines) {
102             String leftTrimmed = line.replaceFirst("^\\s+", "");
103             if (leftTrimmed.startsWith("gateway:")) {
104                 String[] split = ParseUtil.whitespaces.split(leftTrimmed, -1);
105                 if (split.length < 2) {
106                     return "";
107                 }
108                 return split[1].split("%", -1)[0];
109             }
110         }
111         return "";
112     }

114     /**
115      * Maps interface names to their indices, for the platforms whose routing table publishes only a name.
116      * <p>
117      * Enumerating the interfaces is not free, so callers should invoke this once per routing table read rather than
118      * once per route.
119      *
120      * @return a map of interface name to index, empty if the interfaces could not be enumerated
121      */
122     protected static Map<String, Integer> queryInterfaceIndexByName() {
123         Map<String, Integer> map = new HashMap<>();
124         for (NetworkInterface netIf : queryNetworkInterfaces()) {
125             map.put(netIf.getName(), netIf.getIndex());
126         }
127         return map;
128     }
129 
130     /**
131      * Maps interface indices to their names, for the platforms whose routing table publishes only an index.
132      * <p>
133      * Enumerating the interfaces is not free, so callers should invoke this once per routing table read rather than
134      * once per route.
135      *
136      * @return a map of interface index to name, empty if the interfaces could not be enumerated
137      */
138     protected static Map<Integer, String> queryInterfaceNameByIndex() {
139         Map<Integer, String> map = new HashMap<>();
140         for (NetworkInterface netIf : queryNetworkInterfaces()) {
141             map.put(netIf.getIndex(), netIf.getName());
142         }
143         return map;
144     }
145 
146     private static List<NetworkInterface> queryNetworkInterfaces() {
147         return ExceptionUtil.getOrDefault(() -> {
148             Enumeration<NetworkInterface> interfaces = NetworkInterface.getNetworkInterfaces();
149             return interfaces == null ? Collections.<NetworkInterface>emptyList() : Collections.list(interfaces);
150         }, Collections.<NetworkInterface>emptyList(), LOG, "Socket exception when retrieving interfaces");
151     }
152 
153     @Override
154     public String toString() {
155         return String.format(Locale.ROOT,
156                 "Host name: %s, Domain name: %s, DNS servers: %s, IPv4 Gateway: %s, IPv6 Gateway: %s",
157                 this.getHostName(), this.getDomainName(), Arrays.toString(this.getDnsServers()),
158                 this.getIpv4DefaultGateway(), this.getIpv6DefaultGateway());
159 
160     }
