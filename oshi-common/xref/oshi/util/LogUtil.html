5   package oshi.util;
6   
7   import org.slf4j.Logger;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  
11  /**
12   * Utility methods for logging at a {@link LogLevel} chosen at runtime.
13   */
14  @ThreadSafe
15  public final class LogUtil {
16  
17      private LogUtil() {
18      }
19  
20      /**
21       * Tests whether a level is enabled for a logger, using only SLF4J 1.x-compatible {@link Logger} methods. This is
22       * the equivalent of slf4j 2.x's {@code Logger.isEnabledForLevel}, whose parameter type OSHI cannot reference (see
23       * {@link LogLevel}).
24       * <p>
25       * {@link #logAtLevel(Logger, LogLevel, String, Object...)} does not need this guard: SLF4J checks the level itself
26       * before substituting {@code {}} placeholders. Use this only to skip work the caller must perform <em>before</em>
27       * the logging call, such as formatting an argument.
28       *
29       * @param log   the logger to query
30       * @param level the level to test
31       * @return true if the level is enabled for the logger
32       */
33      public static boolean isEnabled(Logger log, LogLevel level) {
34          switch (level) {
35              case ERROR:
36                  return log.isErrorEnabled();
37              case WARN:
38                  return log.isWarnEnabled();
39              case INFO:
40                  return log.isInfoEnabled();
41              case TRACE:
42                  return log.isTraceEnabled();
43              case DEBUG:
44              default:
45                  return log.isDebugEnabled();
46          }
47      }
48  
49      /**
50       * Logs a message at the given level using only SLF4J 1.x-compatible {@link Logger} methods (no {@code atLevel}
51       * fluent API).
52       * <p>
53       * The message is formatted by SLF4J only if the level is enabled, so callers do not need to guard this call with
54       * {@code isDebugEnabled()} and friends. Use {@code {}} placeholders rather than concatenation so that argument
55       * substitution remains deferred. Guard with {@link #isEnabled(Logger, LogLevel)} only when computing an argument is
56       * itself expensive.
57       * <p>
58       * SLF4J's implicit-cause handling applies: when the last argument is a {@link Throwable} with no matching
59       * placeholder, it is attached as the log event's cause rather than substituted into the format string.
60       *
61       * @param log   the logger to use
62       * @param level the level at which to log
63       * @param msg   the log message, with {@code {}} placeholders for the arguments
64       * @param args  the arguments to substitute into the message
65       */
66      // Forwarding the caller's message is the entire point of this method, so the format string cannot be a constant
67      // here; the callers are the ones that must pass one.
68      @SuppressWarnings("Slf4jFormatShouldBeConst")
69      public static void logAtLevel(Logger log, LogLevel level, String msg, Object... args) {
70          switch (level) {
71              case ERROR:
72                  log.error(msg, args);
73                  break;
74              case WARN:
75                  log.warn(msg, args);
76                  break;
77              case INFO:
78                  log.info(msg, args);
79                  break;
80              case TRACE:
81                  log.trace(msg, args);
82                  break;
83              case DEBUG:
84              default:
85                  log.debug(msg, args);
86                  break;
87          }
88      }
89  }
