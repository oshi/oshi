5   package oshi.util.common.platform.unix.freebsd;
6   
7   import java.util.HashMap;
8   import java.util.List;
9   import java.util.Map;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.util.ExecutingCommand;
13  import oshi.util.ParseUtil;
14  
15  /**
16   * Reads the kernel environment through {@code kenv}. On FreeBSD and DragonFly BSD the loader copies the SMBIOS strings
17   * into it as {@code smbios.*} variables, which any user can read, unlike {@code dmidecode}, which needs root.
18   */
19  @ThreadSafe
20  public final class KenvUtil {
21  
22      private KenvUtil() {
23      }
24  
25      /**
26       * Reads every kernel environment variable.
27       *
28       * @return a map of variable names to their unquoted values, empty if {@code kenv} fails
29       */
30      public static Map<String, String> queryKenv() {
31          return parseKenv(ExecutingCommand.runNative("kenv"));
32      }
33  
34      /**
35       * Parses the output of {@code kenv}, one {@code name="value"} pair per line.
36       *
37       * @param kenv the lines emitted by {@code kenv}
38       * @return a map of variable names to their values, with the enclosing quotes removed
39       */
40      public static Map<String, String> parseKenv(List<String> kenv) {
41          Map<String, String> map = new HashMap<>();
42          for (Map.Entry<String, String> e : ParseUtil.parseStringListToMap(kenv, "=").entrySet()) {
43              map.put(e.getKey(), ParseUtil.getDoubleQuoteStringValue(e.getValue()));
44          }
45          return map;
46      }
47  }
