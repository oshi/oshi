5   package oshi.util.common.platform.mac;
6   
7   import java.util.Locale;
8   import java.util.concurrent.atomic.AtomicBoolean;
9   
10  import org.slf4j.Logger;
11  
12  import oshi.annotation.concurrent.ThreadSafe;
13  import oshi.util.LogLevel;
14  import oshi.util.LogUtil;
15  
16  /**
17   * Reports why a connection to the SMC could not be opened, at most once per condition.
18   * <p>
19   * Sensor readings poll, and each one opens its own connection: a single pass over CPU temperature, fan speeds, and CPU
20   * voltage opens three. When the service is permanently unavailable an unlatched message therefore repeats for the life
21   * of the process, several times per sampling interval. That is the normal state on a virtual machine, which does not
22   * virtualize the SMC, so callers on a VM saw the failure reported continuously.
23   * <p>
24   * Each condition is instead reported once at a level that reflects how much it should alarm a reader, and at
25   * {@link LogLevel#DEBUG} on every later occurrence. An absent service is a degradation rather than an error: OSHI
26   * reports no sensors and carries on. Failing to open a service that is present is an error, because the SMC is there
27   * and should have answered.
28   * <p>
29   * One instance per backend, holding that backend's logger, so the message is attributed to the class that could not
30   * open the connection rather than to this one.
31   */
32  @ThreadSafe
33  public final class SmcOpenFailure {
34  
35      private final Logger log;
36  
37      private final AtomicBoolean serviceNotFoundLogged = new AtomicBoolean();
38      private final AtomicBoolean openFailedLogged = new AtomicBoolean();
39      private final AtomicBoolean nullConnectionLogged = new AtomicBoolean();
40  
41      /**
42       * Creates a reporter.
43       *
44       * @param log the logger of the class opening the connection, under whose name the messages are reported
45       */
46      public SmcOpenFailure(Logger log) {
47          this.log = log;
48      }
49  
50      /**
51       * Reports that the {@code AppleSMC} service could not be found, so no sensor can be read.
52       */
53      public void serviceNotFound() {
54          LogUtil.logAtLevel(log, levelFor(serviceNotFoundLogged, LogLevel.WARN),
55                  "Unable to locate the AppleSMC service; hardware sensors are unavailable."
56                          + " This is expected on a virtual machine, which does not virtualize the SMC.");
57      }
58  
59      /**
60       * Reports that the {@code AppleSMC} service was found but would not open.
61       *
62       * @param result the nonzero {@code kern_return_t} from {@code IOServiceOpen}
63       */
64      public void openFailed(int result) {
65          LogLevel level = levelFor(openFailedLogged, LogLevel.ERROR);
66          if (LogUtil.isEnabled(log, level)) {
67              LogUtil.logAtLevel(log, level, "Unable to open a connection to the AppleSMC service. Error: 0x{}",
68                      String.format(Locale.ROOT, "%08x", result));
69          }
70      }
71  
72      /**
73       * Reports that {@code IOServiceOpen} succeeded but handed back no connection to use.
74       */
75      public void nullConnection() {
76          LogUtil.logAtLevel(log, levelFor(nullConnectionLogged, LogLevel.ERROR),
77                  "IOServiceOpen reported success but returned a null AppleSMC connection handle.");
78      }
79  
80      /**
81       * Returns the level for this occurrence, at the same time latching the condition as reported.
82       *
83       * @param latch the condition's latch
84       * @param first the level for the first occurrence
85       * @return {@code first} if this is the first occurrence, otherwise {@link LogLevel#DEBUG}
86       */
87      private static LogLevel levelFor(AtomicBoolean latch, LogLevel first) {
88          return latch.compareAndSet(false, true) ? first : LogLevel.DEBUG;
89      }
90  }
