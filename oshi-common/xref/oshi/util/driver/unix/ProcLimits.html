5   package oshi.util.driver.unix;
6   
7   import java.nio.file.Files;
8   import java.nio.file.Paths;
9   import java.util.List;
10  import java.util.Locale;
11  import java.util.Optional;
12  
13  import oshi.annotation.concurrent.ThreadSafe;
14  import oshi.util.FileUtil;
15  import oshi.util.ParseUtil;
16  
17  /**
18   * Utility to read process resource limits from {@code /proc/<pid>/limits}, which Linux and the BSDs with a mounted
19   * procfs both provide in the same format.
20   */
21  @ThreadSafe
22  public final class ProcLimits {
23  
24      private ProcLimits() {
25      }
26  
27      /**
28       * Queries {@code /proc/<pid>/limits} for a process's open file limit.
29       *
30       * @param processId the process ID
31       * @param index     {@code 1} for the soft limit, {@code 2} for the hard limit
32       * @return the limit, or {@code -1} if procfs is not mounted, the row is absent, or that field is unlimited
33       */
34      public static long queryOpenFileLimit(long processId, int index) {
35          final String limitsPath = String.format(Locale.ROOT, "/proc/%d/limits", processId);
36          if (!Files.exists(Paths.get(limitsPath))) {
37              return -1; // not supported
38          }
39          return parseOpenFileLimit(FileUtil.readFile(limitsPath), index);
40      }
41  
42      /**
43       * Parses the open file limit out of {@code /proc/<pid>/limits} content. Split from the file read so it can be
44       * tested against fixture data on any operating system.
45       * <p>
46       * A field reading {@code unlimited} contributes no digits, so it is absent from the split and yields {@code -1} for
47       * that index alone. The other field is still returned: a process whose hard limit is unlimited normally still has a
48       * real soft limit.
49       *
50       * @param lines the lines of {@code /proc/<pid>/limits}
51       * @param index {@code 1} for the soft limit, {@code 2} for the hard limit
52       * @return the limit, or {@code -1} if there is no {@code Max open files} row or that field is unlimited
53       */
54      public static long parseOpenFileLimit(List<String> lines, int index) {
55          final Optional<String> maxOpenFilesLine = lines.stream().filter(line -> line.startsWith("Max open files"))
56                  .findFirst();
57          if (!maxOpenFilesLine.isPresent()) {
58              return -1;
59          }
60          // Split all non-digits away -> ["", "{soft-limit}", "{hard-limit}", ""]
61          final String[] split = ParseUtil.notDigits.split(maxOpenFilesLine.get(), -1);
62          // Element 0 is the empty string left of the label, so a usable index is 1 or 2
63          if (index < 1 || split.length <= index) {
64              return -1;
65          }
66          return ParseUtil.parseLongOrDefault(split[index], -1);
67      }
68  }
