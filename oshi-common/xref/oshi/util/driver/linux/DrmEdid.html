5   package oshi.util.driver.linux;
6   
7   import java.io.File;
8   import java.util.ArrayList;
9   import java.util.Collections;
10  import java.util.List;
11  
12  import oshi.annotation.concurrent.ThreadSafe;
13  import oshi.util.FileUtil;
14  import oshi.util.ParseUtil;
15  import oshi.util.linux.SysPath;
16  import oshi.util.tuples.Triplet;
17  
18  /**
19   * Utility to read EDID data from the Linux DRM (Direct Rendering Manager) subsystem. The kernel exposes raw EDID bytes
20   * for each connected display at {@code /sys/class/drm/card<N>-<connector>/edid}, which works regardless of whether X11
21   * or Wayland is in use.
22   */
23  @ThreadSafe
24  public final class DrmEdid {
25  
26      private DrmEdid() {
27      }
28  
29      /**
30       * Read EDID byte arrays from /sys/class/drm for all connected displays.
31       *
32       * @return a list of EDID byte arrays (at least 128 bytes each), or empty if none found
33       */
34      public static List<byte[]> getEdidArrays() {
35          return getEdidArrays(new File(SysPath.DRM));
36      }
37  
38      /**
39       * Read EDID byte arrays from the given DRM directory.
40       *
41       * @param drmDir the directory containing card*-* subdirectories
42       * @return a list of EDID byte arrays (at least 128 bytes each), or empty if none found
43       */
44      static List<byte[]> getEdidArrays(File drmDir) {
45          List<Triplet<String, Integer, byte[]>> data = getDisplayData(drmDir);
46          List<byte[]> edids = new ArrayList<>(data.size());
47          for (Triplet<String, Integer, byte[]> t : data) {
48              edids.add(t.getC());
49          }
50          return Collections.unmodifiableList(edids);
51      }
52  
53      /**
54       * Read display data from /sys/class/drm for all connected displays.
55       *
56       * @return a list of {@link Triplet} of kernel connector name (e.g. {@code HDMI-A-1}), DRM connector ID ({@code -1}
57       *         if not available), and EDID byte array (at least 128 bytes each), or empty if none found
58       */
59      public static List<Triplet<String, Integer, byte[]>> getDisplayData() {
60          return getDisplayData(new File(SysPath.DRM));
61      }
62  
63      /**
64       * Read display data from the given DRM directory.
65       *
66       * @param drmDir the directory containing card*-* subdirectories
67       * @return a list of {@link Triplet} of kernel connector name, DRM connector ID ({@code -1} if not available), and
68       *         EDID byte array (at least 128 bytes each), or empty if none found
69       */
70      static List<Triplet<String, Integer, byte[]>> getDisplayData(File drmDir) {
71          if (!drmDir.isDirectory()) {
72              return Collections.emptyList();
73          }
74          File[] connectors = drmDir.listFiles(f -> f.isDirectory() && f.getName().matches("card\\d+-.+"));
75          if (connectors == null || connectors.length == 0) {
76              return Collections.emptyList();
77          }
78          List<Triplet<String, Integer, byte[]>> results = new ArrayList<>();
79          for (File connector : connectors) {
80              File statusFile = new File(connector, "status");
81              if (statusFile.exists()) {
82                  String status = FileUtil.getStringFromFile(statusFile.getPath()).trim();
83                  if (!"connected".equals(status)) {
84                      continue;
85                  }
86              }
87              File edidFile = new File(connector, "edid");
88              if (edidFile.exists()) {
89                  byte[] edid = FileUtil.readAllBytes(edidFile.getPath(), false);
90                  if (edid.length >= 128) {
91                      // Extract connector name from directory name: "card0-HDMI-A-1" -> "HDMI-A-1"
92                      String dirName = connector.getName();
93                      String connectorName = dirName.substring(dirName.indexOf('-') + 1);
94                      int connectorId = -1;
95                      File connectorIdFile = new File(connector, "connector_id");
96                      if (connectorIdFile.exists()) {
97                          connectorId = ParseUtil
98                                  .parseIntOrDefault(FileUtil.getStringFromFile(connectorIdFile.getPath()).trim(), -1);
99                      }
100                     results.add(new Triplet<>(connectorName, connectorId, edid));
101                 }
102             }
103         }
104         return Collections.unmodifiableList(results);
105     }
