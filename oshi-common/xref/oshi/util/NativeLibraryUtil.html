20  /**
21   * Decides whether an optional native library may be loaded.
22   * <p>
23   * On Solaris and illumos, AIX, NetBSD, DragonFly BSD and OpenBSD, {@code dlerror()} state is shared by the whole
24   * process rather than kept per thread. Another thread's failed {@code dlopen} or {@code dlsym} can rewrite the message
25   * a thread is about to read, and another thread's {@code dlerror()} call clears it. JNA reads that message with an
26   * unguarded {@code strlen} and {@code strcpy}, so a failed load that overlaps another thread's failed lookup can crash
27   * the JVM on a null message or, where the message lives in a reused buffer, corrupt the native heap. The only defense
28   * available from Java is not to attempt a load that will fail, so on those platforms this class checks that the library
29   * file exists before the JNA or FFM backend loads it. On Linux, macOS, FreeBSD and Windows the state is per thread, a
30   * failed load is harmless, and loading is always allowed. NetBSD shares the state too, but OSHI loads no optional
31   * library there, so it is not checked.
32   * <p>
33   * The check covers the runtime linker's default directories and its library path environment variables, not directories
34   * an administrator has added to the linker configuration. A library found only there is reported absent, and the caller
35   * uses its fallback.
36   */
37  @ThreadSafe
38  public final class NativeLibraryUtil {
39  
40      private static final Logger LOG = LoggerFactory.getLogger(NativeLibraryUtil.class);
41  
42      private static final PlatformEnum PLATFORM = PlatformEnum.getCurrentPlatform();
43  
44      private static final Pattern VERSION = Pattern.compile("\\d+(\\.\\d+)?");
45  
46      private NativeLibraryUtil() {
47      }
48  
49      /**
50       * Returns whether an optional native library may be loaded. On a platform whose {@code dlerror()} state is shared
51       * by the process, this is {@code true} only when the library file exists on the runtime linker's search path, so
52       * that a load which would fail is never attempted; on every other platform it is always {@code true}.
53       * <p>
54       * Pass the file name the loader will open: {@code System.mapLibraryName("cups")} for a JNA
55       * {@code Native.load("cups", ...)} or an FFM lookup by simple name, or the literal name for an FFM lookup by file
56       * name such as {@code "libkstat2.so.1"}. On AIX, JNA opens the {@code .a} archive in place of a {@code .so} name,
57       * so a JNA caller passes that. On OpenBSD, whose runtime linker resolves an unversioned name to a versioned file, a
58       * versioned file also counts. On AIX, an archive name such as {@code "libcups.a"} never counts, because an archive
59       * can be loaded only by naming a member inside it.
60       *
61       * @param fileName the library file name, without a directory
62       * @return {@code false} if loading the library on this platform should be skipped
63       */
64      public static boolean isSafeToLoad(String fileName) {
65          return isSafeToLoad(fileName, null);
66      }
67  
68      /**
69       * Returns whether an optional native library may be loaded, as {@link #isSafeToLoad(String)} does, also searching
70       * directories the loader adds ahead of the runtime linker's own, such as JNA's {@code jna.library.path}. The loader
71       * opens a file there by its full path, so only the exact name counts in those directories.
72       *
73       * @param fileName  the library file name, without a directory
74       * @param extraPath further directories to search, separated by {@code :}, or null
75       * @return {@code false} if loading the library on this platform should be skipped
76       */
77      public static boolean isSafeToLoad(String fileName, @Nullable String extraPath) {
78          List<String> dirs = searchDirs(PLATFORM, System::getenv);
79          if (dirs.isEmpty()) {
80              return true;
81          }
82          List<String> extraDirs = new ArrayList<>();
83          addEntries(extraDirs, extraPath);
84          boolean present = isPresent(PLATFORM, fileName, dirs, extraDirs);
85          if (!present) {
86              LOG.debug("{} not found on the library search path; not loading it.", fileName);
87          }
88          return present;
89      }
90  
91      /**
92       * Returns whether the runtime linker on this platform would find a library by this name in any of the directories.
93       *
94       * @param platform the platform whose naming rules apply
95       * @param fileName the library file name
96       * @param dirs     the directories to search
97       * @return whether a matching file exists in one of them
98       */
99      static boolean isPresent(PlatformEnum platform, String fileName, List<String> dirs) {
100         if (isArchive(platform, fileName)) {
101             return false;
102         }
103         return isPresent(fileName, dirs) || (platform == PlatformEnum.OPENBSD && isVersionPresent(fileName, dirs));
104     }
105 
106     /**
107      * Returns whether a library by this name would be found in the runtime linker's directories or, by its exact name,
108      * in directories the loader searches itself. The loader opens a file there by its full path, and only under the
109      * name it was asked for, so a versioned file in those directories is never reached.
110      *
111      * @param platform   the platform whose naming rules apply
112      * @param fileName   the library file name
113      * @param linkerDirs the directories the runtime linker searches
114      * @param loaderDirs the directories the loader searches before handing the name to the runtime linker
115      * @return whether a matching file exists in one of them
116      */
117     static boolean isPresent(PlatformEnum platform, String fileName, List<String> linkerDirs, List<String> loaderDirs) {
118         return isPresent(platform, fileName, linkerDirs)
119                 || (!isArchive(platform, fileName) && isPresent(fileName, loaderDirs));
120     }
121 
122     /**
123      * Returns whether a file with this name exists in any of the directories. A dangling symbolic link does not count.
124      *
125      * @param fileName the library file name
126      * @param dirs     the directories to search
127      * @return whether the file exists in one of them
128      */
129     static boolean isPresent(String fileName, List<String> dirs) {
130         for (String dir : dirs) {
131             if (new File(dir, fileName).isFile()) {
132                 return true;
133             }
134         }
135         return false;
136     }
137 
138     /**
139      * Returns whether a versioned file for this name, such as {@code libcups.so.2} or {@code libcups.so.7.3} for
140      * {@code libcups.so}, exists in any of the directories. The suffix must be a major number with an optional minor
141      * number, the form OpenBSD's runtime linker accepts, so a different library sharing the prefix, such as
142      * {@code libcupsfilters.so}, does not match.
143      *
144      * @param fileName the unversioned library file name
145      * @param dirs     the directories to search
146      * @return whether a versioned file exists in one of them
147      */
148     static boolean isVersionPresent(String fileName, List<String> dirs) {
149         for (String dir : dirs) {
150             String[] names = new File(dir).list();
151             if (names == null) {
152                 continue;
153             }
154             for (String name : names) {
155                 if (isVersionOf(fileName, name) && new File(dir, name).isFile()) {
156                     return true;
157                 }
158             }
159         }
160         return false;
161     }
162 
163     /**
164      * Returns whether a file name is a versioned name for a library: the library name, a dot, and a major number with
165      * an optional minor number.
166      *
167      * @param fileName the unversioned library file name
168      * @param name     the file name to test
169      * @return whether {@code name} is a versioned name for {@code fileName}
170      */
171     static boolean isVersionOf(String fileName, String name) {
172         String prefix = fileName + ".";
173         return name.startsWith(prefix) && VERSION.matcher(name.substring(prefix.length())).matches();
174     }
175 
176     /**
177      * Builds the directories the runtime linker searches by default on a platform whose {@code dlerror()} state is
178      * shared by the process: the entries of its library path environment variables, followed by its default
179      * directories.
180      * <p>
181      * On Solaris and illumos, every JDK is 64-bit, so {@code LD_LIBRARY_PATH_64} replaces {@code LD_LIBRARY_PATH} when
182      * it is set, even to an empty value. On AIX, {@code LIBPATH} likewise replaces {@code LD_LIBRARY_PATH}.
183      *
184      * @param platform the platform
185      * @param getenv   reads an environment variable, returning null if it is unset
186      * @return the directories, without empty entries; empty if the platform's {@code dlerror()} state is per thread
187      */
188     static List<String> searchDirs(PlatformEnum platform, Function<String, @Nullable String> getenv) {
189         List<String> dirs = new ArrayList<>();
190         switch (platform) {
191             case SOLARIS:
192                 @Nullable
193                 String path64 = getenv.apply("LD_LIBRARY_PATH_64");
194                 addEntries(dirs, path64 != null ? path64 : getenv.apply("LD_LIBRARY_PATH"));
195                 dirs.addAll(Arrays.asList("/lib/64", "/usr/lib/64"));
196                 break;
197             case AIX:
198                 @Nullable
199                 String libpath = getenv.apply("LIBPATH");
200                 addEntries(dirs, libpath != null ? libpath : getenv.apply("LD_LIBRARY_PATH"));
201                 dirs.addAll(Arrays.asList("/usr/lib", "/lib"));
202                 break;
203             case OPENBSD:
204                 addEntries(dirs, getenv.apply("LD_LIBRARY_PATH"));
205                 dirs.addAll(Arrays.asList("/usr/lib", "/usr/X11R6/lib", "/usr/local/lib"));
206                 break;
207             case DRAGONFLYBSD:
208                 addEntries(dirs, getenv.apply("LD_LIBRARY_PATH"));
209                 dirs.addAll(Arrays.asList("/lib", "/usr/lib", "/usr/local/lib", "/usr/local/lib/compat/pkg"));
210                 break;
211             default:
212                 break;
213         }
214         return dirs;
215     }
216 
217     // An AIX archive loads only by member name, as libfoo.a(member), which no caller supplies
218     private static boolean isArchive(PlatformEnum platform, String fileName) {
219         return platform == PlatformEnum.AIX && fileName.endsWith(".a");
220     }
221 
222     private static void addEntries(List<String> dirs, @Nullable String path) {
223         if (path != null) {
224             for (String dir : path.split(":", -1)) {
225                 if (!dir.isEmpty()) {
226                     dirs.add(dir);
227                 }
228             }
229         }
230     }
