19  /**
20   * Parses the BSD mount tables and {@code df -i} into the fields a file store needs.
21   * <p>
22   * Each format is parsed by its fixed parts rather than split on whitespace, so a mount point containing spaces keeps
23   * them. None of these commands escapes a space: each prints the kernel's path as it is.
24   */
25  @ThreadSafe
26  public final class MountTable {
27  
28      // A JSON string, its escapes left in place
29      private static final String JSON_STRING = "\"([^\"\\\\]*+(?:\\\\.[^\"\\\\]*+)*+)\"";
30      // One fstab entry of FreeBSD's mount -p --libxo json, whose fields putfsent emits in this order
31      private static final Pattern LIBXO_ENTRY = Pattern
32              .compile("\\{\\s*\"device\"\\s*:\\s*" + JSON_STRING + "\\s*,\\s*\"mntpoint\"\\s*:\\s*" + JSON_STRING
33                      + "\\s*,\\s*\"fstype\"\\s*:\\s*" + JSON_STRING + "\\s*,\\s*\"opts\"\\s*:\\s*" + JSON_STRING);
34      private static final Pattern TABS = Pattern.compile("\\t+");
35      // NetBSD's mount: /dev/dk1 on / type ffs (noatime, local)
36      private static final Pattern NETBSD_MOUNT = Pattern.compile("(.+?) on (.+) type (\\S+) (\\(.*\\))");
37      // OpenBSD's mount -v, which adds the disk's DUID and partition where it has one:
38      // /dev/sd0a (ed2d8ecd4d5492ee.a) on / type ffs (rw, local, ctime=Wed Oct 7 22:17:15 2026)
39      private static final Pattern OPENBSD_MOUNT = Pattern
40              .compile("(.+?)(?: \\(([0-9a-f]{16}\\.[a-p])\\))? on (.+) type (\\S+) (\\(.*\\))");
41      // A df -i row: the volume, three block counts, the capacity, the inodes used and free, their percentage, and the
42      // mount point, which runs to the end of the line. Avail can be negative once the reserve is in use, and a file
43      // system without inodes prints - for its percentage
44      private static final Pattern DF_INODES = Pattern
45              .compile("(.+?)\\s+\\d+\\s+\\d+\\s+-?\\d+\\s+(?:\\d+%|-)\\s+(\\d+)\\s+(-?\\d+)\\s+(?:\\d+%|-)\\s+(.+)");
46  
47      private MountTable() {
48      }

51       * One mounted file system.
52       */
53      @Immutable
54      public static final class Entry {
55          private final String volume;
56          private final String uuid;
57          private final String path;
58          private final String type;
59          private final String options;

61          Entry(String volume, String uuid, String path, String type, String options) {
62              this.volume = volume;
63              this.uuid = uuid;
64              this.path = path;
65              this.type = type;
66              this.options = options;
67          }

115     /**
116      * Parses FreeBSD's {@code mount -p --libxo json}.
117      * <p>
118      * {@code mount -p} pads its fields with tabs, but after a field of 24 characters or more it separates it from the
119      * next with a single space, which a space inside the field cannot be told apart from. The JSON form carries the
120      * same fields unambiguously.
121      *
122      * @param json the lines of {@code mount -p --libxo json}
123      * @return the mounted file systems, or an empty list where {@code mount} does not support libxo
124      */
125     public static List<Entry> parseFreeBsdLibxo(List<String> json) {
126         List<Entry> entries = new ArrayList<>();
127         Matcher m = LIBXO_ENTRY.matcher(String.join("\n", json));
128         while (m.find()) {
129             entries.add(new Entry(unescapeJson(m.group(1)), "", unescapeJson(m.group(2)), unescapeJson(m.group(3)),
130                     unescapeJson(m.group(4))));
131         }
132         return entries;
133     }

135     /**
136      * Parses {@code mount -p} on DragonFly, or on a FreeBSD whose {@code mount} has no libxo.
137      * <p>
138      * DragonFly prints {@code device<TAB>mount point<TAB>type options<TAB>dump pass}, one tab between each, so a mount
139      * point keeps its spaces. FreeBSD pads every field with tabs instead, which leaves the type and options apart,
140      * except that a field of 24 characters or more is followed by a single space: a long mount point then shares its
141      * field with the type, which is split off at the last space. Any other line is split on whitespace, which cannot
142      * keep a space in a mount point.
143      *
144      * @param lines the lines of {@code mount -p}
145      * @return the mounted file systems
146      */
147     public static List<Entry> parseFstab(List<String> lines) {
148         List<Entry> entries = new ArrayList<>();
149         for (String line : lines) {
150             String[] tabs = TABS.split(line.trim(), -1);
151             // FreeBSD follows options of eight characters or more with a space, which is not a separator inside the
152             // field
153             int typeSpace = tabs.length == 4 ? tabs[2].trim().indexOf(' ') : -1;
154             int pathSpace = tabs.length == 4 ? tabs[1].lastIndexOf(' ') : -1;
155             if (typeSpace > 0) {
156                 // DragonFly: the type and options share a field
157                 entries.add(new Entry(tabs[0], "", tabs[1], tabs[2].substring(0, typeSpace),
158                         tabs[2].substring(typeSpace + 1).trim()));
159             } else if (pathSpace > 0) {
160                 // FreeBSD, a mount point of 24 characters or more: it shares a field with the type, which has no space
161                 entries.add(new Entry(tabs[0], "", tabs[1].substring(0, pathSpace), tabs[1].substring(pathSpace + 1),
162                         tabs[2].trim()));
163             } else if (tabs.length >= 5) {
164                 entries.add(new Entry(tabs[0], "", tabs[1], tabs[2], tabs[3].trim()));
165             } else {
166                 String[] split = ParseUtil.whitespaces.split(line.trim(), -1);
167                 if (split.length >= 5) {
168                     entries.add(new Entry(split[0], "", split[1], split[2], split[3]));
169                 }
170             }
171         }
172         return entries;
173     }

175     /**
176      * Parses NetBSD's {@code mount}.
177      *
178      * @param lines the lines of {@code mount}
179      * @return the mounted file systems
180      */
181     public static List<Entry> parseNetBsd(List<String> lines) {
182         List<Entry> entries = new ArrayList<>();
183         for (String line : lines) {
184             Matcher m = NETBSD_MOUNT.matcher(line.trim());
185             if (m.matches()) {
186                 entries.add(new Entry(m.group(1), "", m.group(2), m.group(3), m.group(4)));
187             }
188         }
189         return entries;
190     }

192     /**
193      * Parses OpenBSD's {@code mount -v}. A file system on a disk is followed by its DUID and partition in parentheses,
194      * which becomes the identifier without them; one with no disk, such as {@code tmpfs} or NFS, has none.
195      *
196      * @param lines the lines of {@code mount -v}
197      * @return the mounted file systems
198      */
199     public static List<Entry> parseOpenBsdVerbose(List<String> lines) {
200         List<Entry> entries = new ArrayList<>();
201         for (String line : lines) {
202             Matcher m = OPENBSD_MOUNT.matcher(line.trim());
203             if (m.matches()) {
204                 String duid = m.group(2);
205                 entries.add(new Entry(m.group(1), duid == null ? "" : duid, m.group(3), m.group(4), m.group(5)));
206             }
207         }
208         return entries;
209     }

211     /**
212      * Parses {@code df -i} output into free and total inode counts keyed by mount point. The layout is the same on
213      * FreeBSD, DragonFly, NetBSD and OpenBSD.
214      * <p>
215      * The mount point is the key because it is unique where the volume is not: two {@code tmpfs} mounts both name
216      * {@code tmpfs} as their volume.
217      *
218      * @param df the lines of {@code df -i}
219      * @return a pair of the free and the total inodes, each keyed by mount point
220      */
221     public static Pair<Map<String, Long>, Map<String, Long>> parseDfInodes(List<String> df) {
222         Map<String, Long> free = new HashMap<>();
223         Map<String, Long> total = new HashMap<>();
224         for (String line : df) {
225             Matcher m = DF_INODES.matcher(line.trim());
226             if (m.matches()) {
227                 long used = ParseUtil.parseLongOrDefault(m.group(2), 0L);
228                 long ifree = ParseUtil.parseLongOrDefault(m.group(3), 0L);
229                 free.put(m.group(4), ifree);
230                 total.put(m.group(4), used + ifree);
231             }
232         }
233         return new Pair<>(free, total);
234     }

236     private static String unescapeJson(String s) {
237         if (s.indexOf('\\') < 0) {
238             return s;
239         }
240         StringBuilder sb = new StringBuilder(s.length());
241         int i = 0;
242         while (i < s.length()) {
243             char c = s.charAt(i);
244             if (c != '\\' || i + 1 == s.length()) {
245                 sb.append(c);
246                 i++;
247                 continue;
248             }
249             // The escape and the character after it, plus four hex digits for \\u
250             char e = s.charAt(i + 1);
251             i += 2;
252             switch (e) {
253                 case 'n':
254                     sb.append('\n');
255                     break;
256                 case 't':
257                     sb.append('\t');
258                     break;
259                 case 'r':
260                     sb.append('\r');
261                     break;
262                 case 'b':
263                     sb.append('\b');
264                     break;
265                 case 'f':
266                     sb.append('\f');
267                     break;
268                 case 'u':
269                     if (i + 4 <= s.length()) {
270                         sb.append((char) ParseUtil.hexStringToInt(s.substring(i, i + 4), '?'));
271                         i += 4;
272                     } else {
273                         sb.append(e);
274                     }
275                     break;
276                 default:
277                     // \" \\ and \/ stand for the character itself
278                     sb.append(e);
279                     break;
280             }
281         }
282         return sb.toString();
283     }
