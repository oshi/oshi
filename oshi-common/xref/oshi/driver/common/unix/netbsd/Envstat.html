22  /**
23   * Reads NetBSD's environmental sensors ({@code envsys(4)}) through {@code envstat -x}, which prints the kernel's sensor
24   * dictionary as an XML property list and needs no privileges.
25   * <p>
26   * Values are raw: temperatures in microkelvin, voltages in microvolts, power in microwatts, current in microamperes,
27   * energy in microwatt-hours or microampere-hours, fans in RPM, and indicators as 0 or 1.
28   */
29  @ThreadSafe
30  public final class Envstat {
31  
32      // envstat lives in /usr/sbin, which an unprivileged user's PATH often omits
33      private static final String ENVSTAT_X = "/usr/sbin/envstat -x";
34  
35      private static final Pattern KEY = Pattern.compile("<key>(.*)</key>");
36      private static final Pattern VALUE = Pattern.compile("<(integer|string)>(.*)</\\1>");
37  
38      private static final double MICRO = 1_000_000d;
39      private static final double ZERO_CELSIUS_KELVIN = 273.15;
40  
41      /** The {@code type} of a temperature sensor. */
42      public static final String TEMPERATURE = "Temperature";
43      /** The {@code type} of a fan speed sensor. */
44      public static final String FAN = "Fan";
45      /** The {@code type} of a DC voltage sensor. */
46      public static final String VOLTAGE_DC = "Voltage DC";
47      /** The {@code type} of a power sensor. */
48      public static final String WATTS = "Watts";
49      /** The {@code type} of a current sensor. */
50      public static final String AMPERE = "Ampere";
51      /** The {@code type} of an energy sensor measured in watt-hours. */
52      public static final String WATT_HOUR = "Watt hour";
53      /** The {@code type} of a charge sensor measured in ampere-hours. */
54      public static final String AMPERE_HOUR = "Ampere hour";
55      /** The {@code type} of a true/false sensor. */
56      public static final String INDICATOR = "Indicator";
57      /** The {@code type} of a battery's charging indicator. */
58      public static final String BATTERY_CHARGE = "Battery charge";
59  
60      private Envstat() {
61      }
62  
63      /**
64       * One sensor reading.
65       */
66      @Immutable
67      public static final class Sensor {
68          private final String device;
69          private final String description;
70          private final String type;
71          private final long value;
72          private final boolean valid;
73  
74          Sensor(String device, String description, String type, long value, boolean valid) {
75              this.device = device;
76              this.description = description;
77              this.type = type;
78              this.value = value;
79              this.valid = valid;
80          }
81  
82          /**
83           * Gets the device that reports the sensor.
84           *
85           * @return the device name, such as {@code acpibat0}
86           */
87          public String getDevice() {
88              return device;
89          }
90  
91          /**
92           * Gets the sensor description.
93           *
94           * @return the description, such as {@code cpu0 temperature} or {@code charge}
95           */
96          public String getDescription() {
97              return description;
98          }
99  
100         /**
101          * Gets the sensor type.
102          *
103          * @return the type, such as {@link Envstat#TEMPERATURE}
104          */
105         public String getType() {
106             return type;
107         }
108 
109         /**
110          * Gets the raw value, in the units described for {@link Envstat}.
111          *
112          * @return the value
113          */
114         public long getValue() {
115             return value;
116         }
117 
118         /**
119          * Gets whether the sensor has a current value. Sensors whose state is {@code invalid} or {@code unknown} do
120          * not.
121          *
122          * @return whether the value is usable
123          */
124         public boolean isValid() {
125             return valid;
126         }
127     }
128 
129     /**
130      * Runs {@code envstat -x} and parses every sensor it reports.
131      *
132      * @return the sensors, empty if there are none or {@code envstat} is unavailable
133      */
134     public static List<Sensor> querySensors() {
135         return parseSensors(ExecutingCommand.runNative(ENVSTAT_X));
136     }
137 
138     /**
139      * Parses the XML property list printed by {@code envstat -x}: a dictionary of device names, each holding an array
140      * of sensor dictionaries followed by a {@code device-properties} dictionary.
141      *
142      * @param lines the lines of {@code envstat -x} output
143      * @return the sensors, in the order printed
144      */
145     public static List<Sensor> parseSensors(List<String> lines) {
146         List<Sensor> sensors = new ArrayList<>();
147         String device = "";
148         String key = null;
149         String lastKey = "";
150         Map<String, String> dict = new HashMap<>();
151         for (String raw : lines) {
152             String line = raw.trim();
153             Matcher k = KEY.matcher(line);
154             if (k.matches()) {
155                 key = unescape(k.group(1));
156                 lastKey = key;
157                 continue;
158             }
159             if ("<array>".equals(line)) {
160                 // A device's sensor array follows its name
161                 device = lastKey;
162             } else if ("<dict>".equals(line)) {
163                 dict.clear();
164             } else if ("</dict>".equals(line)) {
165                 String type = dict.get("type");
166                 if (type != null) {
167                     String state = dict.getOrDefault("state", "");
168                     sensors.add(new Sensor(device, dict.getOrDefault("description", ""), type,
169                             ParseUtil.parseLongOrDefault(dict.getOrDefault("cur-value", ""), 0L),
170                             "valid".equals(state) || state.startsWith("warning") || state.startsWith("critical")));
171                 }
172                 dict.clear();
173             } else if (key != null) {
174                 Matcher v = VALUE.matcher(line);
175                 if (v.matches()) {
176                     dict.put(key, unescape(v.group(2)));
177                 }
178             }
179             key = null;
180         }
181         return Collections.unmodifiableList(sensors);
182     }
183 
184     /**
185      * Summarizes sensors as {@code Sensors} reports them.
186      *
187      * @param sensors the parsed sensors
188      * @return the CPU temperature in degrees Celsius (the average of sensors described as {@code cpu...}, or of all
189      *         temperature sensors if none are), the fan speeds in RPM, and the CPU core voltage in volts (from a DC
190      *         voltage sensor described as a core voltage), with 0 for any not reported
191      */
192     public static Triplet<Double, int[], Double> summarize(List<Sensor> sensors) {
193         double cpuTempSum = 0d;
194         int cpuTempCount = 0;
195         double allTempSum = 0d;
196         int allTempCount = 0;
197         List<Integer> fans = new ArrayList<>();
198         double volts = 0d;
199         for (Sensor s : sensors) {
200             if (!s.isValid()) {
201                 continue;
202             }
203             String description = s.getDescription().toLowerCase(Locale.ROOT);
204             switch (s.getType()) {
205                 case TEMPERATURE:
206                     double celsius = s.getValue() / MICRO - ZERO_CELSIUS_KELVIN;
207                     allTempSum += celsius;
208                     allTempCount++;
209                     if (description.startsWith("cpu")) {
210                         cpuTempSum += celsius;
211                         cpuTempCount++;
212                     }
213                     break;
214                 case FAN:
215                     fans.add((int) s.getValue());
216                     break;
217                 case VOLTAGE_DC:
218                     if (volts == 0d && description.contains("core")) {
219                         volts = s.getValue() / MICRO;
220                     }
221                     break;
222                 default:
223                     break;
224             }
225         }
226         // Prefer the CPU sensors; failing those, the average of every temperature sensor
227         double temp = 0d;
228         if (cpuTempCount > 0) {
229             temp = cpuTempSum / cpuTempCount;
230         } else if (allTempCount > 0) {
231             temp = allTempSum / allTempCount;
232         }
233         int[] fanSpeeds = fans.stream().mapToInt(Integer::intValue).toArray();
234         return new Triplet<>(temp, fanSpeeds, volts);
235     }
236 
237     private static String unescape(String s) {
238         return s.replace("&lt;", "<").replace("&gt;", ">").replace("&quot;", "\"").replace("&apos;", "'")
239                 .replace("&amp;", "&");
240     }
