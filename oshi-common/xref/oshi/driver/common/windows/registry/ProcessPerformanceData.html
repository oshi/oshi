28  /**
29   * Common logic for building process performance data maps from registry or performance counter results. Callers
30   * (JNA/FFM variants) supply the platform-specific pre-fetched registry or performance-counter data.
31   */
32  @ThreadSafe
33  public final class ProcessPerformanceData {
34  
35      /**
36       * The performance object name for process counters.
37       */
38      public static final String PROCESS = "Process";
39  
40      private ProcessPerformanceData() {
41      }
42  
43      /**
44       * Builds a process map from registry performance data that has already been read.
45       *
46       * @param pids        An optional collection of process IDs to filter the list to. May be null for no filtering.
47       * @param processData The raw registry data triplet (instance maps, perfTime100nSec, now in ms)
48       * @return A map with Process ID as the key and a {@link ProcessPerfCounterBlock} object populated with performance
49       *         counter information, or null if processData is null.
50       */
51      private static @Nullable Map<Integer, ProcessPerfCounterBlock> mapFromRegistryData(
52              @Nullable Collection<Integer> pids,
53              @Nullable Triplet<List<Map<ProcessPerformanceProperty, Object>>, Long, Long> processData) {
54          if (processData == null) {
55              return null;
56          }
57          List<Map<ProcessPerformanceProperty, Object>> processInstanceMaps = processData.getA();
58          long now = processData.getC(); // 1970 epoch
59  
60          Map<Integer, ProcessPerfCounterBlock> processMap = new HashMap<>();
61          for (Map<ProcessPerformanceProperty, Object> processInstanceMap : processInstanceMaps) {
62              int pid = intValue(processInstanceMap, ProcessPerformanceProperty.IDPROCESS);
63              String name = stringValue(processInstanceMap, ProcessPerformanceProperty.NAME);
64              if ((pids == null || pids.contains(pid)) && !"_Total".equals(name)) {
65                  long ctime = longValue(processInstanceMap, ProcessPerformanceProperty.ELAPSEDTIME);
66                  if (ctime > now) {
67                      ctime = ParseUtil.filetimeToUtcMs(ctime, false);
68                  }
69                  long upTime = now - ctime;
70                  if (upTime < 1L) {
71                      upTime = 1L;
72                  }
73                  processMap.put(pid,
74                          new ProcessPerfCounterBlock(name,
75                                  intValue(processInstanceMap, ProcessPerformanceProperty.CREATINGPROCESSID),
76                                  intValue(processInstanceMap, ProcessPerformanceProperty.PRIORITYBASE),
77                                  longValue(processInstanceMap, ProcessPerformanceProperty.WORKINGSETPRIVATE),
78                                  longValue(processInstanceMap, ProcessPerformanceProperty.WORKINGSET), ctime, upTime,
79                                  longValue(processInstanceMap, ProcessPerformanceProperty.IOREADBYTESPERSEC),
80                                  longValue(processInstanceMap, ProcessPerformanceProperty.IOWRITEBYTESPERSEC),
81                                  Integer.toUnsignedLong(
82                                          intValue(processInstanceMap, ProcessPerformanceProperty.PAGEFAULTSPERSEC))));
83              }
84          }
85          return processMap;
86      }

88      /**
89       * Builds a process map from performance counter query results.
90       *
91       * @param pids           An optional collection of process IDs to filter the list to. May be null for no filtering.
92       * @param instanceValues The query results as a pair of (instances, valueMap)
93       * @return A map with Process ID as the key and a {@link ProcessPerfCounterBlock} object populated with performance
94       *         counter information, or null if instanceValues is null.
95       */
96      private static @Nullable Map<Integer, ProcessPerfCounterBlock> mapFromCounterValues(
97              @Nullable Collection<Integer> pids,
98              @Nullable Pair<List<String>, Map<ProcessPerformanceProperty, List<Long>>> instanceValues) {
99          if (instanceValues == null) {
100             return null;
101         }
102         Map<Integer, ProcessPerfCounterBlock> processMap = new HashMap<>();
103         long now = System.currentTimeMillis(); // 1970 epoch
104         List<String> instances = instanceValues.getA();
105         Map<ProcessPerformanceProperty, List<Long>> valueMap = instanceValues.getB();
106         List<Long> pidList = counterList(valueMap, ProcessPerformanceProperty.IDPROCESS);
107         List<Long> ppidList = counterList(valueMap, ProcessPerformanceProperty.CREATINGPROCESSID);
108         List<Long> priorityList = counterList(valueMap, ProcessPerformanceProperty.PRIORITYBASE);
109         List<Long> ioReadList = counterList(valueMap, ProcessPerformanceProperty.IOREADBYTESPERSEC);
110         List<Long> ioWriteList = counterList(valueMap, ProcessPerformanceProperty.IOWRITEBYTESPERSEC);
111         List<Long> privateWorkingSetList = counterList(valueMap, ProcessPerformanceProperty.WORKINGSETPRIVATE);
112         List<Long> workingSetList = counterList(valueMap, ProcessPerformanceProperty.WORKINGSET);
113         List<Long> elapsedTimeList = counterList(valueMap, ProcessPerformanceProperty.ELAPSEDTIME);
114         List<Long> pageFaultsList = counterList(valueMap, ProcessPerformanceProperty.PAGEFAULTSPERSEC);
115 
116         for (int inst = 0; inst < instances.size(); inst++) {
117             int pid = pidList.get(inst).intValue();
118             // Skip PID 0: PDH reports it as the "ID Process" sentinel for the _Total aggregate, the Idle
119             // pseudo-process, and any instance whose real PID is not yet available (starting/exiting). Keying
120             // the map on 0 would clobber one such entry with another, and no real Windows process has PID 0.
121             if (pid != 0 && (pids == null || pids.contains(pid))) {
122                 long ctime = elapsedTimeList.get(inst);
123                 if (ctime > now) {
124                     ctime = ParseUtil.filetimeToUtcMs(ctime, false);
125                 }
126                 long upTime = now - ctime;
127                 if (upTime < 1L) {
128                     upTime = 1L;
129                 }
130                 processMap.put(pid,
131                         new ProcessPerfCounterBlock(instances.get(inst), ppidList.get(inst).intValue(),
132                                 priorityList.get(inst).intValue(), privateWorkingSetList.get(inst),
133                                 workingSetList.get(inst), ctime, upTime, ioReadList.get(inst), ioWriteList.get(inst),
134                                 pageFaultsList.get(inst).longValue()));
135             }
136         }
137         return processMap;
138     }

140     /**
141      * Reads process performance data from the registry.
142      *
143      * @param executor The backend performing the native queries
144      * @param pids     Process IDs to filter to, or null for all
145      * @return A map of process ID to counter block, or null if the read failed
146      */
147     public static @Nullable Map<Integer, ProcessPerfCounterBlock> buildProcessMapFromRegistry(
148             PerfCounterQueryExecutor executor, @Nullable Collection<Integer> pids) {
149         Triplet<List<Map<ProcessPerformanceProperty, Object>>, Long, Long> processData = null;
150         if (HkeyPerformanceDataUtil.PERFDATA) {
151             processData = executor.readPerfDataFromRegistry(PROCESS, ProcessPerformanceProperty.class);
152         }
153         return mapFromRegistryData(pids, processData);
154     }
155 
156     /**
157      * Reads process performance data from performance counters.
158      *
159      * @param executor The backend performing the native queries
160      * @param pids     Process IDs to filter to, or null for all
161      * @return A map of process ID to counter block, or null if the read failed
162      */
163     public static @Nullable Map<Integer, ProcessPerfCounterBlock> buildProcessMapFromPerfCounters(
164             PerfCounterQueryExecutor executor, @Nullable Collection<Integer> pids) {
165         if (executor.isPerfProcDisabled()) {
166             return Collections.emptyMap();
167         }
168         return mapFromCounterValues(pids, ProcessInformation.queryProcessCounters(executor));
169     }
