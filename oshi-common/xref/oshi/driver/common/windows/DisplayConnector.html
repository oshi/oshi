5   package oshi.driver.common.windows;
6   
7   import java.util.Locale;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.util.Constants;
11  import oshi.util.Util;

13  /**
14   * Shared logic for mapping a Windows display to the physical connector it is attached to, using the Connecting and
15   * Configuring Displays (CCD) APIs in {@code user32.dll}.
16   * <p>
17   * The struct sizes and field offsets the two native backends read are declared here as constants so the JNA and FFM
18   * implementations cannot drift apart, and the pure parsing (connector naming, device-path normalization) is shared and
19   * unit-tested. The native calls themselves - {@code GetDisplayConfigBufferSizes}, {@code QueryDisplayConfig}, and
20   * {@code DisplayConfigGetDeviceInfo} - are made by each backend.
21   * <p>
22   * A {@code LUID} is two {@code DWORD}s, so it is only 4-byte aligned and the {@code adapterId} fields below can land on
23   * an offset that is not a multiple of 8. An 8-byte read there must therefore be unaligned, which the FFM backend
24   * enforces at runtime.
25   */
26  @ThreadSafe
27  public final class DisplayConnector {
28  
29      private DisplayConnector() {
30      }
31  
32      /** {@code QDC_ONLY_ACTIVE_PATHS} flag for {@code GetDisplayConfigBufferSizes}/{@code QueryDisplayConfig}. */
33      public static final int QDC_ONLY_ACTIVE_PATHS = 0x00000002;
34  
35      /** Size in bytes of a {@code DISPLAYCONFIG_PATH_INFO}. */
36      public static final int PATH_INFO_SIZE = 72;
37      /** Size in bytes of a {@code DISPLAYCONFIG_MODE_INFO} (allocated but never read). */
38      public static final int MODE_INFO_SIZE = 64;
39      /** Offset of {@code targetInfo.adapterId} (an 8-byte {@code LUID}) within a {@code DISPLAYCONFIG_PATH_INFO}. */
40      public static final int PATH_TARGET_ADAPTER_ID_OFFSET = 20;
41      /** Offset of {@code targetInfo.id} (a {@code UINT32}) within a {@code DISPLAYCONFIG_PATH_INFO}. */
42      public static final int PATH_TARGET_ID_OFFSET = 28;
43      /** Offset of the path {@code flags} (a {@code UINT32}) within a {@code DISPLAYCONFIG_PATH_INFO}. */
44      public static final int PATH_FLAGS_OFFSET = 68;
45      /** {@code DISPLAYCONFIG_PATH_ACTIVE} bit within the path {@code flags}. */
46      public static final int PATH_ACTIVE = 0x00000001;
47  
48      /** Size in bytes of a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
49      public static final int TARGET_DEVICE_NAME_SIZE = 420;
50      /** {@code DISPLAYCONFIG_DEVICE_INFO_GET_TARGET_NAME} request type. */
51      public static final int DEVICE_INFO_GET_TARGET_NAME = 2;
52      /** Offset of {@code header.size} within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
53      public static final int TDN_HEADER_SIZE_OFFSET = 4;
54      /**
55       * Offset of {@code header.adapterId} (an 8-byte {@code LUID}) within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}.
56       */
57      public static final int TDN_HEADER_ADAPTER_ID_OFFSET = 8;
58      /** Offset of {@code header.id} (a {@code UINT32}) within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
59      public static final int TDN_HEADER_ID_OFFSET = 16;
60      /** Offset of {@code outputTechnology} within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
61      public static final int TDN_OUTPUT_TECHNOLOGY_OFFSET = 24;
62      /** Offset of {@code connectorInstance} within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
63      public static final int TDN_CONNECTOR_INSTANCE_OFFSET = 32;
64      /** Offset of {@code monitorDevicePath} (a {@code WCHAR[128]}) within a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}. */
65      public static final int TDN_MONITOR_DEVICE_PATH_OFFSET = 164;
66  
67      // DISPLAYCONFIG_VIDEO_OUTPUT_TECHNOLOGY values (shared with the D3DKMDT_VIDEO_OUTPUT_TECHNOLOGY enum).
68      private static final int VOT_HD15 = 0;
69      private static final int VOT_SVIDEO = 1;
70      private static final int VOT_COMPOSITE_VIDEO = 2;
71      private static final int VOT_COMPONENT_VIDEO = 3;
72      private static final int VOT_DVI = 4;
73      private static final int VOT_HDMI = 5;
74      private static final int VOT_LVDS = 6;
75      private static final int VOT_SDI = 9;
76      private static final int VOT_DISPLAYPORT_EXTERNAL = 10;
77      private static final int VOT_DISPLAYPORT_EMBEDDED = 11;
78      private static final int VOT_UDI_EXTERNAL = 12;
79      private static final int VOT_UDI_EMBEDDED = 13;
80      private static final int VOT_SDTVDONGLE = 14;
81      private static final int VOT_MIRACAST = 15;
82      private static final int VOT_INTERNAL = 0x80000000;
83  
84      /**
85       * Names the connector a display is attached through, from a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}'s
86       * {@code outputTechnology} and {@code connectorInstance}. The {@code connectorInstance} is Windows' own
87       * disambiguator, zero when the adapter has a single connector of that type and one-based when it has several, so it
88       * is appended verbatim only when non-zero (e.g. {@code HDMI}, {@code DisplayPort-1}, {@code DisplayPort-2}).
89       *
90       * @param outputTechnology  the {@code DISPLAYCONFIG_VIDEO_OUTPUT_TECHNOLOGY} value
91       * @param connectorInstance the connector instance number
92       * @return a connector name such as {@code HDMI} or {@code DisplayPort-1}
93       */
94      public static String connectorName(int outputTechnology, int connectorInstance) {
95          String base = technologyName(outputTechnology);
96          return connectorInstance > 0 ? base + "-" + connectorInstance : base;
97      }
98  
99      private static String technologyName(int outputTechnology) {
100         switch (outputTechnology) {
101             case VOT_HD15:
102                 return "VGA";
103             case VOT_SVIDEO:
104                 return "S-Video";
105             case VOT_COMPOSITE_VIDEO:
106                 return "Composite";
107             case VOT_COMPONENT_VIDEO:
108                 return "Component";
109             case VOT_DVI:
110                 return "DVI";
111             case VOT_HDMI:
112                 return "HDMI";
113             case VOT_LVDS:
114                 return "LVDS";
115             case VOT_SDI:
116                 return "SDI";
117             case VOT_DISPLAYPORT_EXTERNAL:
118                 return "DisplayPort";
119             case VOT_DISPLAYPORT_EMBEDDED:
120                 return "eDP";
121             case VOT_UDI_EXTERNAL:
122             case VOT_UDI_EMBEDDED:
123                 return "UDI";
124             case VOT_SDTVDONGLE:
125                 return "SDTV";
126             case VOT_MIRACAST:
127                 return "Miracast";
128             case VOT_INTERNAL:
129                 return "Internal";
130             default:
131                 return "Other";
132         }
133     }
134 
135     /**
136      * Normalizes a monitor device interface path for case-insensitive matching. The path returned by
137      * {@code SetupDiGetDeviceInterfaceDetail} and the {@code monitorDevicePath} returned by
138      * {@code DisplayConfigGetDeviceInfo} are the same string but may differ in case.
139      *
140      * @param devicePath the device interface path, or {@code null}
141      * @return the lower-cased path, or {@link Constants#UNKNOWN} if the input is null or empty
142      */
143     public static String normalizePath(String devicePath) {
144         if (Util.isBlank(devicePath)) {
145             return Constants.UNKNOWN;
146         }
147         return devicePath.toLowerCase(Locale.ROOT);
148     }
149 }
