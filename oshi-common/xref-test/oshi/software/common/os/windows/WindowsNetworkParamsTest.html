7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.hasSize;
9   import static org.hamcrest.Matchers.is;
10  
11  import java.util.ArrayList;
12  import java.util.Collections;
13  import java.util.List;
14  
15  import org.junit.jupiter.api.Test;
16  
17  import oshi.software.os.NetworkParams.IPRoute;
18  
19  /**
20   * Tests the Windows routing table interpretation, which lives in {@code oshi-common} so that it can be exercised off
21   * Windows. The backends supply only the raw rows, which the stub below fabricates.
22   */
23  class WindowsNetworkParamsTest {
24  
25      /** A params instance whose IP Helper API rows are supplied by the test. */
26      private static final class StubParams extends WindowsNetworkParams {
27          private final List<WindowsNetworkParams.RouteRow> rows;
28  
29          StubParams(List<WindowsNetworkParams.RouteRow> rows) {
30              this.rows = rows;
31          }
32  
33          @Override
34          protected List<WindowsNetworkParams.RouteRow> queryRouteRows() {
35              return this.rows;
36          }
37      }
38  
39      private static WindowsNetworkParams.RouteRow row(byte[] destination, int prefixLength, byte[] nextHop,
40              int interfaceIndex, long metric) {
41          WindowsNetworkParams.RouteRow r = new WindowsNetworkParams.RouteRow();
42          r.destination = destination;
43          r.prefixLength = prefixLength;
44          r.nextHop = nextHop;
45          r.interfaceIndex = interfaceIndex;
46          r.metric = metric;
47          return r;
48      }
49  
50      @Test
51      void testAnUnspecifiedNextHopIsNotAGateway() {
52          // Windows publishes no flags column, so an on-link route is recognized by its all-zero next hop
53          List<IPRoute> routes = new StubParams(
54                  List.of(row(new byte[] { 10, 0, 0, 0 }, 24, new byte[] { 0, 0, 0, 0 }, 5, 256L))).getRoutes();
55          assertThat(routes, hasSize(1));
56          assertThat(routes.get(0).isGateway(), is(false));
57          assertThat("A non-gateway route reports no gateway address", routes.get(0).getGateway(), is(new byte[0]));
58          assertThat(routes.get(0).getMetric(), is(256L));
59      }
60  
61      @Test
62      void testARealNextHopIsAGateway() {
63          List<IPRoute> routes = new StubParams(
64                  List.of(row(new byte[] { 0, 0, 0, 0 }, 0, new byte[] { -64, -88, 1, 1 }, 5, 25L))).getRoutes();
65          assertThat(routes.get(0).isGateway(), is(true));
66          assertThat(routes.get(0).getGateway(), is(new byte[] { -64, -88, 1, 1 }));
67          assertThat(routes.get(0).getPrefixLength(), is(0));
68          assertThat(routes.get(0).isHost(), is(false));
69      }
70  
71      @Test
72      void testHostRoutesAreRecognizedByPrefixWidthInBothFamilies() {
73          byte[] v6 = new byte[16];
74          v6[15] = 1;
75          List<IPRoute> routes = new StubParams(
76                  List.of(row(new byte[] { 127, 0, 0, 1 }, 32, new byte[] { 0, 0, 0, 0 }, 1, 331L),
77                          row(v6, 128, new byte[16], 1, 331L))).getRoutes();
78          assertThat(routes, hasSize(2));
79          assertThat("A /32 IPv4 route is a host route", routes.get(0).isHost(), is(true));
80          assertThat("A /128 IPv6 route is a host route", routes.get(1).isHost(), is(true));
81          assertThat(routes.get(1).getDestination().length, is(16));
82      }
83  
84      @Test
85      void testARowWithAnUnrecognizedFamilyIsDropped() {
86          // The backends report an empty array for a SOCKADDR_INET family that is neither AF_INET nor AF_INET6.
87          // Dropping the row keeps the promise that a destination's length identifies the family.
88          List<WindowsNetworkParams.RouteRow> rows = new ArrayList<>();
89          rows.add(row(new byte[0], 0, new byte[0], 5, 0L));
90          rows.add(row(new byte[] { 10, 0, 0, 0 }, 8, new byte[0], 5, 0L));
91          List<IPRoute> routes = new StubParams(rows).getRoutes();
92          assertThat(routes, hasSize(1));
93          assertThat(routes.get(0).getDestination().length, is(4));
94      }
95  
96      @Test
97      void testAnUnknownInterfaceIndexReportsAnEmptyName() {
98          // Index 999999 will not be present in this machine's interface list
99          List<IPRoute> routes = new StubParams(List.of(row(new byte[] { 10, 0, 0, 0 }, 8, new byte[0], 999999, 0L)))
100                 .getRoutes();
101         assertThat(routes.get(0).getInterfaceName(), is(""));
102         assertThat(routes.get(0).getInterfaceIndex(), is(999999));
103     }
104 
105     @Test
106     void testAnEmptyTableProducesAnEmptyList() {
107         assertThat(new StubParams(Collections.emptyList()).getRoutes(), hasSize(0));
108     }
109 }
