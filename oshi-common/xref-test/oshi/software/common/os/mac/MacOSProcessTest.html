25  /**
26   * Tests the {@code KERN_PROCARGS2} buffer parser.
27   */
28  class MacOSProcessTest {
29  
30      private static final String EXEC = "/usr/bin/foo";
31  
32      /**
33       * Builds a {@code KERN_PROCARGS2} buffer: a little-endian argument count, the null-terminated exec path, then the
34       * null-terminated entries.
35       */
36      private static byte[] procargs(int nargs, String... entries) {
37          return procargsPadded(nargs, 0, entries);
38      }
39  
40      /**
41       * As {@link #procargs}, with {@code padding} extra null bytes between the exec path and the first entry, as the
42       * kernel emits for alignment.
43       */
44      private static byte[] procargsPadded(int nargs, int padding, String... entries) {
45          ByteArrayOutputStream out = new ByteArrayOutputStream();
46          out.write(nargs & 0xff);
47          out.write(nargs >> 8 & 0xff);
48          out.write(nargs >> 16 & 0xff);
49          out.write(nargs >> 24 & 0xff);
50          writeCString(out, EXEC);
51          for (int i = 0; i < padding; i++) {
52              out.write(0);
53          }
54          for (String entry : entries) {
55              writeCString(out, entry);
56          }
57          return out.toByteArray();
58      }
59  
60      private static void writeCString(ByteArrayOutputStream out, String s) {
61          byte[] bytes = s.getBytes(StandardCharsets.UTF_8);
62          out.write(bytes, 0, bytes.length);
63          out.write(0);
64      }

66      @Test
67      void testAsciiArgsAndEnv() {
68          byte[] buf = procargs(3, EXEC, "-c", "value", "HOME=/Users/me", "LANG=en_US.UTF-8");
69          Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
70          assertThat("arguments", result.getA(), contains(EXEC, "-c", "value"));
71          assertThat("HOME", result.getB(), hasEntry("HOME", "/Users/me"));
72          assertThat("LANG", result.getB(), hasEntry("LANG", "en_US.UTF-8"));
73      }
74  
75      /**
76       * The regression this parser exists for: an entry containing multi-byte characters is longer in bytes than in
77       * chars, so advancing the offset by the decoded String's length lands mid-character and corrupts every entry after
78       * it.
79       */
80      @Test
81      void testMultiByteArgsDoNotShiftLaterEntries() {
82          byte[] buf = procargs(4, EXEC, "日本語", "café", "plain-ascii-tail", "TZ=Asia/Tokyo", "GREET=こんにちは");
83          Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
84          assertThat("arguments", result.getA(), contains(EXEC, "日本語", "café", "plain-ascii-tail"));
85          assertThat("TZ", result.getB(), hasEntry("TZ", "Asia/Tokyo"));
86          assertThat("multi-byte value", result.getB(), hasEntry("GREET", "こんにちは"));
87      }
88  
89      @Test
90      void testNullPaddingBetweenSections() {
91          byte[] buf = procargsPadded(2, 3, EXEC, "arg1", "", "", "FOO=bar");
92          Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
93          assertThat("arguments", result.getA(), contains(EXEC, "arg1"));
94          assertThat("FOO", result.getB(), hasEntry("FOO", "bar"));
95      }
96  
97      /**
98       * An empty string is a legal argv entry. Treating its terminator as padding would both drop it and shift an
99       * environment entry into the argument list.
100      */
101     @Test
102     void testEmptyArgumentIsPreserved() {
103         byte[] buf = procargsPadded(4, 2, EXEC, "first", "", "third", "FOO=bar");
104         Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
105         assertThat("arguments", result.getA(), contains(EXEC, "first", "", "third"));
106         assertThat("environment", result.getB(), is(aMapWithSize(1)));
107         assertThat("FOO", result.getB(), hasEntry("FOO", "bar"));
108     }
109 
110     /**
111      * Processes with several thousand arguments are ordinary (a shell glob or a compiler invocation), so the argument
112      * count is bounded only by the buffer that has to hold them.
113      */
114     @Test
115     void testManyArguments() {
116         String[] entries = new String[2000];
117         entries[0] = EXEC;
118         for (int i = 1; i < 1999; i++) {
119             entries[i] = "arg" + i;
120         }
121         entries[1999] = "FOO=bar";
122         byte[] buf = procargs(1999, entries);
123         Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
124         assertThat("argument count", result.getA(), hasSize(1999));
125         assertThat("first argument", result.getA().get(0), is(EXEC));
126         assertThat("last argument", result.getA().get(1998), is("arg1998"));
127         assertThat("FOO", result.getB(), hasEntry("FOO", "bar"));
128     }
129 
130     @Test
131     void testEntryWithoutEqualsIsNotEnvironment() {
132         byte[] buf = procargs(1, EXEC, "=novalue", "notanassignment", "OK=yes");
133         Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length);
134         assertThat("arguments", result.getA(), contains(EXEC));
135         assertThat("environment", result.getB(), is(aMapWithSize(1)));
136         assertThat("environment", result.getB(), hasEntry("OK", "yes"));
137     }
138 
139     @Test
140     void testSizeLimitsParsing() {
141         byte[] buf = procargs(3, EXEC, "kept", "cut");
142         // The sysctl-reported size bounds the parse: entries past it are not read even though the array holds them
143         Pair<List<String>, Map<String, String>> result = MacOSProcess.parseProcArgs(buf, buf.length - 4);
144         assertThat("arguments", result.getA(), contains(EXEC, "kept"));
145     }
146 
147     /**
148      * A size that cuts through the middle of an entry leaves it unterminated. The fragment is not a value the process
149      * ever had, so it must be dropped rather than reported as a short argument or a truncated environment value.
150      */
151     @Test
152     void testTruncatedEntriesAreDropped() {
153         // "/usr/bin/foo\0" twice, then "kept\0"; stop 2 bytes into "kept"
154         byte[] argBuf = procargs(3, EXEC, "kept", "cut");
155         Pair<List<String>, Map<String, String>> argResult = MacOSProcess.parseProcArgs(argBuf, argBuf.length - 7);
156         assertThat("partial argument dropped", argResult.getA(), contains(EXEC));
157 
158         // nargs=1, so the remaining entry is an environment assignment; stop inside its value
159         byte[] envBuf = procargs(1, EXEC, "HOME=/Users/me");
160         Pair<List<String>, Map<String, String>> envResult = MacOSProcess.parseProcArgs(envBuf, envBuf.length - 5);
161         assertThat("arguments", envResult.getA(), contains(EXEC));
162         assertThat("partial environment value dropped", envResult.getB(), is(anEmptyMap()));
163     }
164 
165     @Test
166     void testUnusableBuffers() {
167         byte[] tooShort = { 1, 0, 0 };
168         assertThat("short buffer args", MacOSProcess.parseProcArgs(tooShort, tooShort.length).getA(), is(empty()));
169 
170         byte[] zeroArgs = procargs(0, "ignored");
171         assertThat("zero nargs args", MacOSProcess.parseProcArgs(zeroArgs, zeroArgs.length).getA(), is(empty()));
172         assertThat("zero nargs env", MacOSProcess.parseProcArgs(zeroArgs, zeroArgs.length).getB(), is(anEmptyMap()));
173 
174         // More arguments than the remaining bytes could hold, even at one null terminator each
175         byte[] impossibleArgs = procargs(1025, "ignored");
176         assertThat("impossible nargs", MacOSProcess.parseProcArgs(impossibleArgs, impossibleArgs.length).getA(),
177                 is(empty()));
178 
179         byte[] negativeArgs = procargs(-1, "ignored");
180         assertThat("negative nargs", MacOSProcess.parseProcArgs(negativeArgs, negativeArgs.length).getA(), is(empty()));
181     }
