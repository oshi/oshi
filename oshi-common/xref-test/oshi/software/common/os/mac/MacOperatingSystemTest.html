7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.contains;
9   import static org.hamcrest.Matchers.is;

11  import java.util.List;
12  
13  import org.junit.jupiter.api.Test;
14  
15  import oshi.software.os.OSService;
16  import oshi.software.os.OSService.State;

18  class MacOperatingSystemTest {
19  
20      @Test
21      void testResolveVersionUsesSwVersWhenNonEmpty() {
22          assertThat(MacOperatingSystem.resolveVersion("10.16", "11.2.3"), is("11.2.3"));
23      }
24  
25      @Test
26      void testResolveVersionFallsBackWhenSwVersEmpty() {
27          assertThat(MacOperatingSystem.resolveVersion("10.16", ""), is("10.16"));
28      }
29  
30      @Test
31      void testParseLaunchctlPrint() {
32          List<String> lines = """
33                  system = {
34                  \ttype = system
35                  \tservices = {
36                  \t\t       0      - \tcom.apple.rpmuxd
37                  \t\t   72143      - \tcom.apple.lskdd
38                  \t\t   63915   (pe) \tcom.apple.kernelmanager_helper
39                  \t\t       0     78 \tcom.apple.wifiFirmwareLoader
40                  \t}
41                  \tendpoints = {
42                  \t\t       0      - \tcom.apple.not.a.service
43                  \t}
44                  }
45                  """.lines().toList();
46          List<OSService> services = MacOperatingSystem.parseLaunchctlPrint(lines);
47          assertThat(services.stream().map(OSService::getName).toList(), contains("com.apple.rpmuxd", "com.apple.lskdd",
48                  "com.apple.kernelmanager_helper", "com.apple.wifiFirmwareLoader"));
49          assertThat(services.stream().map(OSService::getProcessID).toList(), contains(0, 72143, 63915, 0));
50          assertThat(services.stream().map(OSService::getState).toList(),
51                  contains(State.STOPPED, State.RUNNING, State.RUNNING, State.STOPPED));
52      }
53  
54      @Test
55      void testParseLaunchctlList() {
56          List<String> lines = """
57                  PID\tStatus\tLabel
58                  -\t0\tcom.apple.SafariHistoryServiceAgent
59                  95678\t-9\tcom.apple.progressd
60                  851\t0\tapplication.com.apple.Preview.1152921500311898404.1152921500311898409
61                  """.lines().toList();
62          List<OSService> services = MacOperatingSystem.parseLaunchctlList(lines);
63          assertThat(services.stream().map(OSService::getName).toList(),
64                  contains("com.apple.SafariHistoryServiceAgent", "com.apple.progressd"));
65          assertThat(services.stream().map(OSService::getProcessID).toList(), contains(0, 95678));
66          assertThat(services.stream().map(OSService::getState).toList(), contains(State.STOPPED, State.RUNNING));
67      }
