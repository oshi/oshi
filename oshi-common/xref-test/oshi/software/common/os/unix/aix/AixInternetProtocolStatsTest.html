5   package oshi.software.common.os.unix.aix;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import oshi.software.os.InternetProtocolStats.TcpStats;
13  import oshi.software.os.InternetProtocolStats.UdpStats;
14  
15  class AixInternetProtocolStatsTest {
16  
17      @Test
18      void testTcpStats() {
19          // perfstat_protocol's tcp counters on AIX 7.3 (cfarm119), read beside netstat -s: 174587 dropped is netstat's
20          // "connections closed (including 174587 drops)", while its 74322 embryonic drops have no perfstat counter
21          TcpStats tcp = AixInternetProtocolStats.tcpStats(2_030_200L, 276_308L, 1_972_702L, 174_587L, 30_915_756L,
22                  57_617_958L, 303_957L);
23          assertThat(tcp.getConnectionsEstablished(), is(2_030_200L));
24          assertThat(tcp.getConnectionsActive(), is(276_308L));
25          assertThat(tcp.getConnectionsPassive(), is(1_972_702L));
26          assertThat(tcp.getConnectionsReset(), is(174_587L));
27          // Not the drops again: perfstat cannot report failures or retransmissions
28          assertThat(tcp.getConnectionFailures(), is(0L));
29          assertThat(tcp.getSegmentsRetransmitted(), is(0L));
30          assertThat(tcp.getSegmentsSent(), is(30_915_756L));
31          assertThat(tcp.getSegmentsReceived(), is(57_617_958L));
32          assertThat(tcp.getInErrors(), is(303_957L));
33      }
34  
35      @Test
36      void testUdpStats() {
37          // The same machine's udp counters. perfstat's ierrors of 12442997 was netstat's 868999 dropped due to no
38          // socket, 11573609 broadcast or multicast dropped due to no socket and 374 bad checksums, near enough
39          UdpStats udp = AixInternetProtocolStats.udpStats(7_035_856L, 18_748_068L, 868_999L);
40          assertThat(udp.getDatagramsSent(), is(7_035_856L));
41          assertThat(udp.getDatagramsReceived(), is(18_748_068L));
42          assertThat(udp.getDatagramsNoPort(), is(868_999L));
43          assertThat(udp.getDatagramsReceivedErrors(), is(0L));
44      }
45  }
