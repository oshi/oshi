25      // Builds a minimal BSD process whose open-file-limit hooks return sentinel values so the base's routing (current
26      // process via getrlimit vs. other process via the fallback) can be observed.
27      private static BsdOSProcess stubProcess(int pid, int ownPid) {
28          return new BsdOSProcess(pid) {
29              @Override
30              protected List<BsdPsKeyword> psKeywords() {
31                  return Collections.emptyList();
32              }
33  
34              @Override
35              protected String psCommandArgs() {
36                  return "";
37              }
38  
39              @Override
40              protected List<String> queryArguments() {
41                  return Collections.emptyList();
42              }
43  
44              @Override
45              protected Map<String, String> queryEnvironmentVariables() {
46                  return Collections.emptyMap();
47              }
48  
49              @Override
50              protected int queryBitness() {
51                  return 0;
52              }
53  
54              @Override
55              protected int queryOwnProcessId() {
56                  return ownPid;
57              }
58  
59              @Override
60              protected long queryRlimitNofile(boolean soft) {
61                  return soft ? 1024L : 4096L;
62              }
63  
64              @Override
65              protected long otherProcessOpenFileLimit(int index) {
66                  return index == 1 ? -11L : -22L;
67              }
68  
69              @Override
70              public String getCurrentWorkingDirectory() {
71                  return "";
72              }
73  
74              @Override
75              public long getOpenFiles() {
76                  return 0L;
77              }
78  
79              @Override
80              public List<OSThread> getThreadDetails() {
81                  return Collections.emptyList();
82              }
83          };
84      }

86      @Test
87      void testOpenFileLimitsForCurrentProcess() {
88          // pid matches the current process id, so getrlimit values are returned
89          BsdOSProcess self = stubProcess(42, 42);
90          assertThat(self.getSoftOpenFileLimit(), is(1024L));
91          assertThat(self.getHardOpenFileLimit(), is(4096L));
92      }

94      @Test
95      void testOpenFileLimitsForOtherProcess() {
96          // pid differs from the current process id, so the non-current fallback is used
97          BsdOSProcess other = stubProcess(42, 7);
98          assertThat(other.getSoftOpenFileLimit(), is(-11L));
99          assertThat(other.getHardOpenFileLimit(), is(-22L));
100     }

109     // Minimal ps column map for the single-TIME-column platforms (DragonFly, OpenBSD, NetBSD).
110     private static Map<BsdPsKeyword, String> psRow(String time) {
111         Map<BsdPsKeyword, String> psMap = new EnumMap<>(BsdPsKeyword.class);
112         psMap.put(BsdPsKeyword.STATE, "S");
113         psMap.put(BsdPsKeyword.PPID, "1");
114         psMap.put(BsdPsKeyword.USER, "root");
115         psMap.put(BsdPsKeyword.UID, "0");
116         psMap.put(BsdPsKeyword.RGID, "0");
117         psMap.put(BsdPsKeyword.PRI, "20");
118         psMap.put(BsdPsKeyword.VSZ, "1024");
119         psMap.put(BsdPsKeyword.RSS, "512");
120         psMap.put(BsdPsKeyword.NLWP, "4");
121         psMap.put(BsdPsKeyword.TIME, time);
122         psMap.put(BsdPsKeyword.ETIMES, "60");
123         psMap.put(BsdPsKeyword.UCOMM, "java");
124         return psMap;
125     }

127     @Test
128     void testUserTimeIsClampedMonotonic() {
129         BsdOSProcess proc = stubProcess(42, 42);
130         // First read establishes the value
131         proc.updateAttributes(psRow("0:09.78"));
132         assertThat(proc.getUserTime(), is(9780L));
133         // DragonFly's ps TIME sums only the currently live LWPs, so an exiting thread can take its accumulated CPU
134         // time out of the process total. Never report a decrease; clamp to the last value instead.
135         proc.updateAttributes(psRow("0:08.25"));
136         assertThat(proc.getUserTime(), is(9780L));
137         // A genuine increase still comes through
138         proc.updateAttributes(psRow("0:10.00"));
139         assertThat(proc.getUserTime(), is(10000L));
140     }
141 
142     @Test
143     void testKernelAndUserTimeClampedWithSystimeColumn() {
144         BsdOSProcess proc = stubProcess(42, 42);
145         // FreeBSD reports systime separately and TIME is user+sys, so user time is a subtraction of two ps columns
146         Map<BsdPsKeyword, String> first = psRow("0:10.00");
147         first.put(BsdPsKeyword.SYSTIME, "0:04.00");
148         proc.updateAttributes(first);
149         assertThat(proc.getKernelTime(), is(4000L));
150         assertThat(proc.getUserTime(), is(6000L));
151         // Both components regress; both are clamped
152         Map<BsdPsKeyword, String> second = psRow("0:09.00");
153         second.put(BsdPsKeyword.SYSTIME, "0:03.50");
154         proc.updateAttributes(second);
155         assertThat(proc.getKernelTime(), is(4000L));
156         assertThat(proc.getUserTime(), is(6000L));
157     }

159     /**
160      * A ps row missing its STATE column used to throw from charAt(0) partway through updateAttributes, leaving the
161      * process half-updated. An absent state now reads as OTHER and the rest of the row is still applied.
162      */
163     @Test
164     void testAbsentStateColumnYieldsOtherRatherThanThrowing() {
165         BsdOSProcess proc = stubProcess(42, 42);
166         Map<BsdPsKeyword, String> psMap = psRow("0:09.78");
167         psMap.remove(BsdPsKeyword.STATE);
168 
169         assertThat(proc.updateAttributes(psMap), is(true));
170         assertThat(proc.getState(), is(OSProcess.State.OTHER));
171         assertThat(proc.getParentProcessID(), is(1));
172         assertThat(proc.getUser(), is("root"));
173     }
174 
175     /**
176      * Absent string columns read as the empty string rather than null, which the OSProcess getters promise.
177      */
178     @Test
179     void testAbsentStringColumnsReadAsEmpty() {
180         BsdOSProcess proc = stubProcess(42, 42);
181         Map<BsdPsKeyword, String> psMap = psRow("0:09.78");
182         psMap.remove(BsdPsKeyword.USER);
183         psMap.remove(BsdPsKeyword.UID);
184 
185         assertThat(proc.updateAttributes(psMap), is(true));
186         assertThat(proc.getUser(), is(""));
187         assertThat(proc.getUserID(), is(""));
188     }
