5   package oshi.software.common.os.unix.openbsd;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  class OpenBsdOSProcessTest {
13  
14      // A kern.cpustats record: six cs_time values followed by cs_flags
15      private static long[] cpuStats(boolean online) {
16          return new long[] { 1L, 2L, 3L, 4L, 5L, 6L, online ? 0x1L : 0x0L };
17      }
18  
19      @Test
20      void testOnlineCpuMask() {
21          // Every processor online
22          assertThat(OpenBsdOSProcess.onlineCpuMask(4, cpu -> cpuStats(true)), is(0b1111L));
23          // hw.smt=0 takes the odd-numbered SMT siblings offline
24          assertThat(OpenBsdOSProcess.onlineCpuMask(4, cpu -> cpuStats(cpu % 2 == 0)), is(0b0101L));
25          // An unreadable record leaves its processor out; none readable is 0
26          assertThat(OpenBsdOSProcess.onlineCpuMask(3, cpu -> cpu == 1 ? new long[0] : cpuStats(true)), is(0b101L));
27          assertThat(OpenBsdOSProcess.onlineCpuMask(4, cpu -> new long[0]), is(0L));
28          assertThat(OpenBsdOSProcess.onlineCpuMask(0, cpu -> cpuStats(true)), is(0L));
29          // More processors than the mask holds
30          assertThat(OpenBsdOSProcess.onlineCpuMask(80, cpu -> cpuStats(true)), is(-1L));
31      }
32  }
