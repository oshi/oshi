26      @Test
27      void testParseEnvironment() {
28          byte[] env = "HOME=/root\0PATH=/bin:/usr/bin\0EMPTY=\0NOEQUALS\0=novalue\0EQ=a=b\0\0".getBytes(UTF_8);
29          Map<String, String> parsed = NetBsdOSProcess.parseEnvironment(env);
30          assertThat(parsed.keySet(), contains("HOME", "PATH", "EMPTY", "EQ"));
31          assertThat(parsed, hasEntry("HOME", "/root"));
32          assertThat(parsed, hasEntry("PATH", "/bin:/usr/bin"));
33          assertThat(parsed, hasEntry("EMPTY", ""));
34          assertThat(parsed, hasEntry("EQ", "a=b"));
35      }

37      @Test
38      void testParseEnvironmentMultibyte() {
39          byte[] env = "GREETING=héllo wörld\0NEXT=ok\0".getBytes(UTF_8);
40          Map<String, String> parsed = NetBsdOSProcess.parseEnvironment(env);
41          assertThat(parsed, hasEntry("GREETING", "héllo wörld"));
42          assertThat(parsed, hasEntry("NEXT", "ok"));

51      @Test
52      void testParseArgv0() {
53          // The comm column would have clipped this to "/opt/bo", the width of ps's COMMAND header
54          assertThat(NetBsdOSProcess.parseArgv0("/opt/bootstrap/bin/java -Xmx64m -cp /tmp/a.jar Main"),
55                  is("/opt/bootstrap/bin/java"));
56          assertThat(NetBsdOSProcess.parseArgv0("/usr/sbin/sshd"), is("/usr/sbin/sshd"));
57          // Started through PATH, so argv[0] is a bare name rather than a path
58          assertThat(NetBsdOSProcess.parseArgv0("java /tmp/S.java"), is("java"));
59          assertThat(NetBsdOSProcess.parseArgv0("  /bin/sh -c true  "), is("/bin/sh"));
60          assertThat(NetBsdOSProcess.parseArgv0(null), is(""));
61          assertThat(NetBsdOSProcess.parseArgv0(""), is(""));
62      }

64      @Test
65      void testParseCommandLine() {
66          assertThat(NetBsdOSProcess.parseCommandLine("/opt/bootstrap/bin/java -Xmx64m Main"),
67                  is("/opt/bootstrap/bin/java -Xmx64m Main"));
68          assertThat(NetBsdOSProcess.parseCommandLine("java /tmp/S.java"), is("java /tmp/S.java"));
69          // ps's placeholder is not a command line, so nothing is known
70          assertThat(NetBsdOSProcess.parseCommandLine("(java)"), is(""));
71          assertThat(NetBsdOSProcess.parseCommandLine("(sshd)"), is(""));
72          assertThat(NetBsdOSProcess.parseCommandLine(null), is(""));
73      }

75      @Test
76      void testIsPlaceholder() {
77          // ps substitutes the command in parentheses where the kernel would not release the arguments
78          assertTrue(NetBsdOSProcess.isPlaceholder("(java)"));
79          assertTrue(NetBsdOSProcess.isPlaceholder("(sshd)"));
80          assertFalse(NetBsdOSProcess.isPlaceholder("/opt/bootstrap/bin/java"));
81          assertFalse(NetBsdOSProcess.isPlaceholder("java"));
82          assertFalse(NetBsdOSProcess.isPlaceholder("()"));
83          assertFalse(NetBsdOSProcess.isPlaceholder(""));
84      }

94      @Test
95      void testThreadNameIgnoresArgsPlaceholder() {
96          // Two readings of one thread, seconds apart on NetBSD 10.1 CI: once the kernel released the arguments, once
97          // the JVM was inside posix_spawn and ps printed its placeholder instead
98          Map<BsdPsThreadKeyword, String> released = new EnumMap<>(BsdPsThreadKeyword.class);
99          released.put(BsdPsThreadKeyword.LID, "1");
100         released.put(BsdPsThreadKeyword.ARGS, "/opt/bootstrap/bin/java -enableassertions -classpath /tmp/a.jar Main");
101         Map<BsdPsThreadKeyword, String> refused = new EnumMap<>(released);
102         refused.put(BsdPsThreadKeyword.ARGS, "(java)");
103         assertThat(new NetBsdOSThread(42, released).getName(), is("java"));
104         assertThat(new NetBsdOSThread(42, refused).getName(), is("java"));
105     }

107     // A process whose kernel answers come from a map keyed by kern.proc_args subtype, absent meaning no answer
108     private static NetBsdOSProcess stubProcess(String args, Map<Integer, String> kernelPaths) {
109         Map<BsdPsKeyword, String> psMap = new EnumMap<>(BsdPsKeyword.class);
110         psMap.put(BsdPsKeyword.PID, "42");
111         psMap.put(BsdPsKeyword.ARGS, args);
112         return new NetBsdOSProcess(42, psMap, new NetBsdOperatingSystem()) {
113             @Override
114             protected String queryProcArgsPath(int subtype) {
115                 return kernelPaths.getOrDefault(subtype, "");
116             }
117         };
118     }

120     @Test
121     void testPathAndCwdFromKernel() {
122         // sshd rewrites its arguments, so argv[0] is no path; the name still comes from argv[0], as its threads' does
123         NetBsdOSProcess proc = stubProcess("sshd: oshi@pts/0", Map.of(NetBsdOSProcess.KERN_PROC_PATHNAME,
124                 "/usr/sbin/sshd", NetBsdOSProcess.KERN_PROC_CWD, "/var/chroot/sshd"));
125         assertThat(proc.getPath(), is("/usr/sbin/sshd"));
126         assertThat(proc.getName(), is("sshd:"));
127         assertThat(proc.getCurrentWorkingDirectory(), is("/var/chroot/sshd"));
128     }

130     @Test
131     void testPathFallsBackToArgv0() {
132         // With no answer from the kernel, as for the kernel itself, the path is argv[0], unless that is the placeholder
133         assertThat(stubProcess("/usr/sbin/cron", Map.of()).getPath(), is("/usr/sbin/cron"));
134         assertThat(stubProcess("(java)", Map.of()).getPath(), is(""));
135     }
