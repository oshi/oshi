5   package oshi.software.common.os.unix.netbsd;

7   import static java.nio.charset.StandardCharsets.UTF_8;
8   import static org.hamcrest.MatcherAssert.assertThat;
9   import static org.hamcrest.Matchers.anEmptyMap;
10  import static org.hamcrest.Matchers.contains;
11  import static org.hamcrest.Matchers.hasEntry;
12  import static org.hamcrest.Matchers.is;
13  import static org.junit.jupiter.api.Assertions.assertFalse;
14  import static org.junit.jupiter.api.Assertions.assertTrue;
15  
16  import java.util.Map;
17  
18  import org.junit.jupiter.api.Test;
19  
20  class NetBsdOSProcessTest {
21  
22      @Test
23      void testParseEnvironment() {
24          byte[] env = "HOME=/root\0PATH=/bin:/usr/bin\0EMPTY=\0NOEQUALS\0=novalue\0EQ=a=b\0\0".getBytes(UTF_8);
25          Map<String, String> parsed = NetBsdOSProcess.parseEnvironment(env);
26          assertThat(parsed.keySet(), contains("HOME", "PATH", "EMPTY", "EQ"));
27          assertThat(parsed, hasEntry("HOME", "/root"));
28          assertThat(parsed, hasEntry("PATH", "/bin:/usr/bin"));
29          assertThat(parsed, hasEntry("EMPTY", ""));
30          assertThat(parsed, hasEntry("EQ", "a=b"));
31      }
32  
33      @Test
34      void testParseEnvironmentMultibyte() {
35          byte[] env = "GREETING=héllo wörld\0NEXT=ok\0".getBytes(UTF_8);
36          Map<String, String> parsed = NetBsdOSProcess.parseEnvironment(env);
37          assertThat(parsed, hasEntry("GREETING", "héllo wörld"));
38          assertThat(parsed, hasEntry("NEXT", "ok"));
39      }
40  
41      @Test
42      void testParseEnvironmentEmpty() {
43          assertThat(NetBsdOSProcess.parseEnvironment(new byte[0]), is(anEmptyMap()));
44          assertThat(NetBsdOSProcess.parseEnvironment(new byte[] { 0, 0 }), is(anEmptyMap()));
45      }
46  
47      @Test
48      void testParseArgv0() {
49          // The comm column would have clipped this to "/opt/bo", the width of ps's COMMAND header
50          assertThat(NetBsdOSProcess.parseArgv0("/opt/bootstrap/bin/java -Xmx64m -cp /tmp/a.jar Main"),
51                  is("/opt/bootstrap/bin/java"));
52          assertThat(NetBsdOSProcess.parseArgv0("/usr/sbin/sshd"), is("/usr/sbin/sshd"));
53          // Started through PATH, so argv[0] is a bare name rather than a path
54          assertThat(NetBsdOSProcess.parseArgv0("java /tmp/S.java"), is("java"));
55          assertThat(NetBsdOSProcess.parseArgv0("  /bin/sh -c true  "), is("/bin/sh"));
56          assertThat(NetBsdOSProcess.parseArgv0(null), is(""));
57          assertThat(NetBsdOSProcess.parseArgv0(""), is(""));
58      }
59  
60      @Test
61      void testParseCommandLine() {
62          assertThat(NetBsdOSProcess.parseCommandLine("/opt/bootstrap/bin/java -Xmx64m Main"),
63                  is("/opt/bootstrap/bin/java -Xmx64m Main"));
64          assertThat(NetBsdOSProcess.parseCommandLine("java /tmp/S.java"), is("java /tmp/S.java"));
65          // ps's placeholder is not a command line, so nothing is known
66          assertThat(NetBsdOSProcess.parseCommandLine("(java)"), is(""));
67          assertThat(NetBsdOSProcess.parseCommandLine("(sshd)"), is(""));
68          assertThat(NetBsdOSProcess.parseCommandLine(null), is(""));
69      }
70  
71      @Test
72      void testIsPlaceholder() {
73          // ps substitutes the command in parentheses where the kernel would not release the arguments
74          assertTrue(NetBsdOSProcess.isPlaceholder("(java)"));
75          assertTrue(NetBsdOSProcess.isPlaceholder("(sshd)"));
76          assertFalse(NetBsdOSProcess.isPlaceholder("/opt/bootstrap/bin/java"));
77          assertFalse(NetBsdOSProcess.isPlaceholder("java"));
78          assertFalse(NetBsdOSProcess.isPlaceholder("()"));
79          assertFalse(NetBsdOSProcess.isPlaceholder(""));
80      }
81  }
