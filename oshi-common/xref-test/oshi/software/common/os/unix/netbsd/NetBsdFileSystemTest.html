5   package oshi.software.common.os.unix.netbsd;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.Arrays;
11  import java.util.Collections;
12  import java.util.List;
13  
14  import org.junit.jupiter.api.Test;
15  
16  class NetBsdFileSystemTest {
17  
18      @Test
19      void testParsePstatTotalFiles() {
20          // Captured from pstat -T on NetBSD 10.1, run as an unprivileged user
21          List<String> lines = Arrays.asList("1100/10000 files", " 747531 vnodes", "no swap devices configured");
22          assertThat(NetBsdFileSystem.parsePstatTotalFiles(lines), is(1100L));
23          // Each count is printed with %3d, so a small one is padded
24          assertThat(NetBsdFileSystem.parsePstatTotalFiles(Arrays.asList(" 42/100 files")), is(42L));
25          // The vnode line has no slash, and the open-files header from pstat -f is a different count
26          assertThat(NetBsdFileSystem.parsePstatTotalFiles(Arrays.asList(" 747531 vnodes", "179/10000 open files")),
27                  is(0L));
28          assertThat(NetBsdFileSystem.parsePstatTotalFiles(Collections.emptyList()), is(0L));
29      }
30  }
