26  /**
27   * Tests that {@link DragonFlyBsdOSProcess} asks the kernel for an executable path only for the current process, and how
28   * it falls back for the others.
29   */
30  class DragonFlyBsdOSProcessTest {
31  
32      private static final int OWN_PID = 42;
33  
34      // Only getProcessId() is reached by the code under test
35      private static final OperatingSystem OS = (OperatingSystem) Proxy.newProxyInstance(
36              OperatingSystem.class.getClassLoader(), new Class<?>[] { OperatingSystem.class }, (proxy, method, args) -> {
37                  if ("getProcessId".equals(method.getName())) {
38                      return OWN_PID;
39                  }
40                  throw new UnsupportedOperationException(method.getName());
41              });
42  
43      private static DragonFlyBsdOSProcess stubProcess(int pid, AtomicInteger kernelQueries) {
44          return stubProcess(pid, kernelQueries, Collections.emptyList());
45      }
46  
47      private static DragonFlyBsdOSProcess stubProcess(int pid, AtomicInteger kernelQueries, List<String> args) {
48          return new DragonFlyBsdOSProcess(pid, OS) {
49              @Override
50              protected String queryOwnExecutablePath() {
51                  kernelQueries.incrementAndGet();
52                  return "/usr/local/bin/java";
53              }
54  
55              @Override
56              protected long queryAffinityMask() {
57                  return 0L;
58              }
59  
60              @Override
61              protected List<String> queryArguments() {
62                  return args;
63              }
64  
65              @Override
66              protected Map<String, String> queryEnvironmentVariables() {
67                  return Collections.emptyMap();
68              }
69          };
70      }
71  
72      @Test
73      void testCurrentProcessPathFromKernel() {
74          AtomicInteger kernelQueries = new AtomicInteger();
75          assertThat(stubProcess(OWN_PID, kernelQueries).queryExecutablePath(), is("/usr/local/bin/java"));
76          assertThat(kernelQueries.get(), is(1));
77      }
78  
79      @Test
80      void testOtherProcessNeverQueriesKernel() {
81          AtomicInteger kernelQueries = new AtomicInteger();
82          assertThat(stubProcess(OWN_PID + 1, kernelQueries).queryExecutablePath(), is(""));
83          assertThat(kernelQueries.get(), is(0));
84      }
85  
86      private static String otherProcessPath(String ucomm, String... args) {
87          Map<BsdPsKeyword, String> psMap = new EnumMap<>(BsdPsKeyword.class);
88          psMap.put(BsdPsKeyword.UCOMM, ucomm);
89          AtomicInteger kernelQueries = new AtomicInteger();
90          String path = stubProcess(OWN_PID + 1, kernelQueries, List.of(args)).queryPath(psMap);
91          assertThat(kernelQueries.get(), is(0));
92          return path;
93      }
94  
95      @Test
96      void testCurrentProcessPathPrefersKernel() {
97          Map<BsdPsKeyword, String> psMap = new EnumMap<>(BsdPsKeyword.class);
98          psMap.put(BsdPsKeyword.UCOMM, "java");
99          assertThat(stubProcess(OWN_PID, new AtomicInteger(), List.of("java", "-jar", "oshi.jar")).queryPath(psMap),
100                 is("/usr/local/bin/java"));
101     }

103     @Test
104     void testOtherProcessPathFromArgv0() {
105         assertThat(otherProcessPath("sshd", "/usr/sbin/sshd", "-D"), is("/usr/sbin/sshd"));
106         // Started through PATH, so argv[0] is a bare name
107         assertThat(otherProcessPath("gmake", "gmake", "-j8", "all"), is("gmake"));
108         // ucomm is cut to MAXCOMLEN; argv[0] keeps the whole name
109         assertThat(otherProcessPath("postgres_exporte", "/usr/local/bin/postgres_exporter", "--log.level=info"),
110                 is("/usr/local/bin/postgres_exporter"));
111         // The arguments are NUL-separated, so a space in the path does not split it
112         assertThat(otherProcessPath("java", "/opt/java tools/java", "-version"), is("/opt/java tools/java"));
113     }
114 
115     @Test
116     void testOtherProcessPathIgnoresRewrittenArgv0() {
117         // Process titles, a login shell and a different program sharing a prefix all fall back to ucomm
118         assertThat(otherProcessPath("sshd", "sshd: oshi@notty (sshd)"), is("sshd"));
119         assertThat(otherProcessPath("postgres_exporte", "postgres_exporter: running"), is("postgres_exporte"));
120         assertThat(otherProcessPath("postgres_exporte", "postgres_exporter:"), is("postgres_exporte"));
121         assertThat(otherProcessPath("sh", "-sh"), is("sh"));
122         assertThat(otherProcessPath("java", "javac", "Main.java"), is("java"));
123         assertThat(otherProcessPath("", "/usr/sbin/cron"), is(""));
124     }

126     @Test
127     void testOtherProcessPathWithoutProcfs() {
128         assertThat(otherProcessPath("dhclient"), is("dhclient"));
129     }

131     private static byte[] cmdline(String... args) {
132         return (String.join("\0", args) + "\0").getBytes(StandardCharsets.UTF_8);
133     }
134 
135     @Test
136     void testArgumentsAgreeingWithPsKeepTheirBoundaries() {
137         assertThat(DragonFlyBsdOSProcess.reconcileArguments(cmdline("/opt/java tools/java", "-version"),
138                 "/opt/java tools/java -version", "java"), is(List.of("/opt/java tools/java", "-version")));
139         assertThat(DragonFlyBsdOSProcess.reconcileArguments(cmdline("sshd", "-D"), "", "sshd"),
140                 is(List.of("sshd", "-D")));
141     }
142 
143     @Test
144     void testCommandNameBufferFallsBackToPs() {
145         // Another process past kern.ps_arg_cache_limit: the file holds the 16-byte p_comm buffer, stale bytes and all
146         byte[] comm = "java\0er\0\0\0\0\0\0\0\0\0".getBytes(StandardCharsets.UTF_8);
147         assertThat(comm.length, is(16));
148         // The kernel drops the cache only past kern.ps_arg_cache_limit (256 bytes), so ps prints a long command line
149         String classpath = IntStream.rangeClosed(1, 12).mapToObj(i -> "/home/oshi/some/long/path/lib" + i + ".jar")
150                 .collect(Collectors.joining(":"));
151         assertThat(DragonFlyBsdOSProcess.reconcileArguments(comm, "/usr/local/bin/java -cp " + classpath + " S arg1",
152                 "java"), is(List.of("/usr/local/bin/java", "-cp", classpath, "S", "arg1")));
153         // The buffer holds ucomm, not argv[0]: started as "worker", ps follows the line with " (java)", not an argument
154         assertThat(DragonFlyBsdOSProcess.reconcileArguments(comm, "worker -cp " + classpath + " S (java)", "java"),
155                 is(List.of("worker", "-cp", classpath, "S")));
156         // When ps cannot read the arguments either, it prints "(ucomm)", and neither source has them
157         assertThat(DragonFlyBsdOSProcess.reconcileArguments(comm, "(java)", "java"), is(empty()));
158     }

160     @Test
161     void testSixteenByteArgumentsThatPsEscapesKeepTheFile() {
162         // A complete argument list can be MAXCOMLEN bytes too; ps prints the tab escaped, so the two disagree
163         byte[] args = cmdline("sh", "-c", "\tread foo");
164         assertThat(args.length, is(16));
165         assertThat(DragonFlyBsdOSProcess.reconcileArguments(args, "sh -c \\tread foo", "sh"),
166                 is(List.of("sh", "-c", "\tread foo")));
167     }

169     @Test
170     void testCurrentProcessTruncatedAtLimitFallsBackToPs() {
171         // The current process reads its own arguments up to the limit, losing the long classpath and what follows
172         assertThat(
173                 DragonFlyBsdOSProcess.reconcileArguments(cmdline("/usr/local/bin/java", "-Dfoo=bar", "-cp"),
174                         "/usr/local/bin/java -Dfoo=bar -cp .:a.jar:b.jar S", "java"),
175                 is(List.of("/usr/local/bin/java", "-Dfoo=bar", "-cp", ".:a.jar:b.jar", "S")));
176     }

178     @Test
179     void testProcessTitleAndKernelThreadKeepTheFile() {
180         // setproctitle(): ps follows the title with "(ucomm)"
181         assertThat(DragonFlyBsdOSProcess.reconcileArguments(cmdline("dhclient: vtnet0"), "dhclient: vtnet0 (dhclient)",
182                 "dhclient"), is(List.of("dhclient: vtnet0")));
183         // Sixteen bytes of real arguments that ps prints exactly, parentheses and all
184         byte[] parenthesized = cmdline("(sd-pam-worker)");
185         assertThat(parenthesized.length, is(16));
186         assertThat(DragonFlyBsdOSProcess.reconcileArguments(parenthesized, "(sd-pam-worker)", "sd-pam-worker"),
187                 is(List.of("(sd-pam-worker)")));
188         // A kernel thread has no arguments, and ps prints its name in parentheses
189         assertThat(DragonFlyBsdOSProcess.reconcileArguments(new byte[0], "(idle_5)", "idle_5"), is(empty()));
190     }

192     @Test
193     void testEmptyFileKeepsNoArguments() {
194         // Without the file a title cannot be told from arguments, so neither is split into them
195         assertThat(DragonFlyBsdOSProcess.reconcileArguments(new byte[0], "dhclient: vtnet0 (dhclient)", "dhclient"),
196                 is(empty()));
197         assertThat(DragonFlyBsdOSProcess.reconcileArguments(new byte[0], "/sbin/init --", "init"), is(empty()));

200     @Test
201     void testCommandNameSuffixIsNotAnArgument() {
202         // ps appends " (ucomm)" to a line that does not start with the command name; it is never an argument
203         assertThat(DragonFlyBsdOSProcess.reconcileArguments(cmdline("worker", "-q"), "worker -q --long (svc)", "svc"),
204                 is(List.of("worker", "-q", "--long")));
205     }

207     @Test
208     void testTrailingCommandNameArgumentIsKept() {
209         // argv[0] names the command, so ps adds no suffix and a trailing "(java)" is a real argument
210         assertThat(
211                 DragonFlyBsdOSProcess.reconcileArguments(cmdline("/usr/local/bin/java", "-cp"),
212                         "/usr/local/bin/java -cp a.jar Main (java)", "java"),
213                 is(List.of("/usr/local/bin/java", "-cp", "a.jar", "Main", "(java)")));
214         // A name longer than MAXCOMLEN still names the command it was cut to
215         assertThat(
216                 DragonFlyBsdOSProcess.reconcileArguments(cmdline("/opt/a_very_long_program_name", "5"),
217                         "/opt/a_very_long_program_name 5", "a_very_long_prog"),
218                 is(List.of("/opt/a_very_long_program_name", "5")));
219     }

221     @Test
222     void testEscapedCharacterIsNotTruncation() {
223         // ps prints a backslash doubled, so the file's line is a prefix of ps's without being cut short
224         assertThat(DragonFlyBsdOSProcess.reconcileArguments(cmdline("echo", "\\"), "echo \\\\", "echo"),
225                 is(List.of("echo", "\\")));

228     @Test
229     void testParseKernelUserTime() {
230         // cfarm152 (DragonFly 6.4): a busy shell loop, all user time, which ps showed as 3.98 s
231         List<String> busy = List
232                 .of("sh 670132 670131 670131 670131 -1,-1 noflags 1795723507,844877 3,977795 0,0 nochan "
233                         + "61873 61873 61873,61873 -");
234         assertThat(DragonFlyBsdOSProcess.parseKernelUserTime(busy), is(new long[] { 0L, 3977L }));
235         // init, mostly system time
236         List<String> init = List.of("init 1 0 1 1 -1,-1 sldr 1793146831,270 0,629556 4,149102 wait 0 0 0,0 -");
237         assertThat(DragonFlyBsdOSProcess.parseKernelUserTime(init), is(new long[] { 4149L, 629L }));
238         // The process exited, or a malformed time field
239         assertThat(DragonFlyBsdOSProcess.parseKernelUserTime(Collections.emptyList()).length, is(0));
240         assertThat(
241                 DragonFlyBsdOSProcess.parseKernelUserTime(List.of("sh 1 0 1 1 -1,-1 noflags 1,0 3 0,0 nochan")).length,
242                 is(0));
243     }
