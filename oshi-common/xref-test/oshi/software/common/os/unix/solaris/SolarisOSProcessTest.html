5   package oshi.software.common.os.unix.solaris;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.Collections;
11  import java.util.List;
12  
13  import org.junit.jupiter.api.Test;
14  
15  class SolarisOSProcessTest {
16  
17      @Test
18      void testParseOpenFileLimitSoft() {
19          List<String> plimit = List.of("1234:", "   coredumpsize      0           unlimited",
20                  "   nofiles(descriptors)  256         65536", "   vmemorysize       unlimited   unlimited");
21          assertThat(SolarisOSProcess.parseOpenFileLimit(plimit, 1), is(256L));
22      }
23  
24      @Test
25      void testParseOpenFileLimitHard() {
26          List<String> plimit = List.of("1234:", "   nofiles(descriptors)  256         65536");
27          assertThat(SolarisOSProcess.parseOpenFileLimit(plimit, 2), is(65536L));
28      }
29  
30      @Test
31      void testParseOpenFileLimitEmpty() {
32          assertThat(SolarisOSProcess.parseOpenFileLimit(Collections.emptyList(), 1), is(-1L));
33      }
34  
35      @Test
36      void testParseOpenFileLimitNoNofilesLine() {
37          List<String> plimit = List.of("1234:", "   coredumpsize      0           unlimited");
38          assertThat(SolarisOSProcess.parseOpenFileLimit(plimit, 1), is(-1L));
39      }
40  
41      @Test
42      void testParseOpenFileLimitIndexOutOfBounds() {
43          List<String> plimit = List.of("   nofiles(descriptors)  256         unlimited");
44          // "unlimited" is non-numeric, so split yields only ["", "256"]
45          assertThat(SolarisOSProcess.parseOpenFileLimit(plimit, 2), is(-1L));
46      }
47  
48      @Test
49      void testParsePbindOutputBound() {
50          String cpuset = "pid 101048 strongly bound to processor(s) 0 1 2 3.";
51          long mask = SolarisOSProcess.parsePbindOutput(cpuset);
52          // bits 0,1,2,3 set = 0b1111 = 15
53          assertThat(mask, is(15L));
54      }
55  
56      @Test
57      void testParsePbindOutputEmpty() {
58          assertThat(SolarisOSProcess.parsePbindOutput(""), is(0L));
59      }
60  
61      @Test
62      void testParsePbindOutputNotBound() {
63          // Some unexpected format that doesn't match the pattern
64          assertThat(SolarisOSProcess.parsePbindOutput("pid 1234 not bound"), is(0L));
65      }
66  
67      @Test
68      void testParsePsrinfo() {
69          List<String> psrinfo = """
70                  0       on-line   since 01/01/2020 00:00:00
71                  1       on-line   since 01/01/2020 00:00:00
72                  4       on-line   since 01/01/2020 00:00:00
73                  """.lines().toList();
74          long mask = SolarisOSProcess.parsePsrinfo(psrinfo);
75          // bits 0, 1, 4 set = 0b10011 = 19
76          assertThat(mask, is(19L));
77      }
78  
79      @Test
80      void testParsePsrinfoEmpty() {
81          assertThat(SolarisOSProcess.parsePsrinfo(Collections.emptyList()), is(0L));
82      }
83  }
