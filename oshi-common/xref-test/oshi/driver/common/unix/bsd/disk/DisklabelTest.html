31  /**
32   * Tests for the shared BSD {@link Disklabel} parser. The {@code parseDiskParams} and {@code parseDfFallback} methods
33   * are exercised with sample command output so they can be covered on any OS; live-system smoke tests are gated to
34   * OpenBSD/NetBSD.
35   */
36  class DisklabelTest {
37  
38      /** Returns deterministic (major,minor) without calling stat(1). */
39      private static final BiFunction<String, String, Pair<Integer, Integer>> ZERO_MAJOR_MINOR = DisklabelTest::zeroMajorMinor;
40  
41      private static Pair<Integer, Integer> zeroMajorMinor(String device, String name) {
42          return new Pair<>(0, 0);
43      }
44  
45      @Test
46      void testParseDiskParamsExtractsHeaderAndPartitions() {
47          // Representative output of `disklabel -n sd0` on OpenBSD/NetBSD.
48          List<String> disklabelOut = """
49                  # /dev/rsd0c:
50                  type: SCSI
51                  disk: SCSI disk
52                  label: Storage Device
53                  duid: 0000000000000000
54                  flags:
55                  bytes/sector: 512
56                  sectors/track: 63
57                  tracks/cylinder: 255
58                  sectors/cylinder: 16065
59                  cylinders: 976
60                  total sectors: 15693824
61                  boundstart: 0
62                  boundend: 15693824
63                  drivedata: 0
64  
65                  16 partitions:
66                  #                size           offset  fstype [fsize bsize   cpg]
67                    a:          2097152             1024  4.2BSD   2048 16384 12958 # /
68                    b:         17023368          2098176    swap                    # none
69                    c:        500118192                0  unused
70                    i:              960               64   MSDOS
71                  """.lines().toList();
72  
73          Quartet<String, String, Long, List<HWPartition>> result = Disklabel.parseDiskParams("sd0", disklabelOut,
74                  ZERO_MAJOR_MINOR);
75  
76          assertThat("label", result.getA(), is("Storage Device"));
77          assertThat("duid", result.getB(), is("0000000000000000"));
78          // total sectors x bytes/sector
79          assertThat("size in bytes", result.getC(), is(15693824L * 512L));
80          // Only `a` and `b` keep enough columns to be considered partitions. The single-letter rows for `c` (unused) and
81          // `i` (MSDOS) have <=4 whitespace fields and are deliberately filtered out by `split.length > 4`.
82          assertThat(result.getD(), hasSize(2));
83  
84          HWPartition root = result.getD().get(0);
85          assertThat(root.getIdentification(), is("sd0a"));
86          assertThat(root.getName(), is("a"));
87          assertThat(root.getType(), is("4.2BSD"));
88          assertThat(root.getUuid(), is("0000000000000000.a"));
89          assertThat(root.getSize(), is(2097152L * 512L));
90          assertThat(root.getMountPoint(), is("/"));
91  
92          HWPartition swap = result.getD().get(1);
93          assertThat(swap.getName(), is("b"));
94          assertThat(swap.getType(), is("swap"));
95          assertThat(swap.getMountPoint(), is("none"));
96      }

98      @Test
99      void testParseDiskParamsLargeDiskWithoutIntOverflow() {
100         // 5 TiB disk at 512-byte sectors = 10737418240 sectors, > Integer.MAX_VALUE.
101         long sectors = 10_737_418_240L;
102         List<String> disklabelOut = List.of("label: Big Disk", "duid: deadbeefdeadbeef", "bytes/sector: 512",
103                 "total sectors: " + sectors);
104 
105         Quartet<String, String, Long, List<HWPartition>> result = Disklabel.parseDiskParams("sd1", disklabelOut,
106                 ZERO_MAJOR_MINOR);
107 
108         assertThat("size must not overflow", result.getC(), is(sectors * 512L));
109     }

111     @Test
112     void testParseDiskParamsEmptyInputReturnsDefaultsAndEmptyPartitions() {
113         Quartet<String, String, Long, List<HWPartition>> result = Disklabel.parseDiskParams("sd0",
114                 Collections.emptyList(), ZERO_MAJOR_MINOR);
115 
116         assertThat(result.getA(), is(""));
117         assertThat(result.getB(), is(""));
118         assertThat(result.getC(), is(1L)); // default 1 sector x 1 byte
119         assertThat(result.getD(), is(empty()));
120     }

122     @Test
123     void testParseDiskParamsUsesMajorMinorLookup() {
124         List<String> disklabelOut = """
125                 label: Test
126                 duid: aaaa
127                 bytes/sector: 512
128                 total sectors: 100
129                   a:  100  0  4.2BSD  2048  16384 12958 # /
130                 """.lines().toList();
131         BiFunction<String, String, Pair<Integer, Integer>> lookup = (disk, name) -> new Pair<>(7, 42);
132 
133         Quartet<String, String, Long, List<HWPartition>> result = Disklabel.parseDiskParams("sd0", disklabelOut,
134                 lookup);
135 
136         assertThat(result.getD(), hasSize(1));
137         assertThat(result.getD().get(0).getMajor(), is(7));
138         assertThat(result.getD().get(0).getMinor(), is(42));
139     }

141     @Test
142     void testParseDfFallbackBuildsUnknownLabeledQuartet() {
143         // `df` columns: device 1k-blocks used avail capacity mounted
144         List<String> dfOut = """
145                 Filesystem  1K-blocks    Used   Avail Capacity  Mounted on
146                 /dev/sd0a     1024000  500000  524000     49%  /
147                 /dev/sd0d     2048000 1000000 1048000     49%  /var
148                 tmpfs           65536       0   65536      0%  /tmp
149                 """.lines().toList();
150 
151         Quartet<String, String, Long, List<HWPartition>> result = Disklabel.parseDfFallback("sd0", dfOut,
152                 ZERO_MAJOR_MINOR);
153 
154         assertThat(result.getA(), is(Constants.UNKNOWN));
155         assertThat(result.getB(), is(Constants.UNKNOWN));
156         assertThat(result.getC(), is(0L));
157         assertThat(result.getD(), hasSize(2));
158         HWPartition rootPart = result.getD().get(0);
159         // df fallback names use the suffix after "/dev/" (e.g. "sd0a"), not just the partition letter.
160         assertThat(rootPart.getName(), is("sd0a"));
161         assertThat(rootPart.getMountPoint(), is("/"));
162         // 1024000 1K blocks x 512 (df fallback assumes 512-byte units)
163         assertThat(rootPart.getSize(), is(1024000L * 512L));
164     }

166     @Test

168         List<String> dfOut = List.of("/dev/sd1a   1000   500   500   50%  /home");

174     @Test
175     void testParseDiskParamsRowWithoutFstypeIsSkipped() {
176         // A partition row with fewer than 5 columns (no fstype) is discarded by the parser.
177         List<String> disklabelOut = """

185                 ZERO_MAJOR_MINOR);
186         assertThat(result.getD(), is(empty()));

191     @Test
192     @EnabledOnOs(OS.OPENBSD)
193     void testGetDiskParamsLiveOpenBsd() {
194         String disknames = ExecutingCommand.getFirstAnswer("sysctl -n hw.disknames");
195         String[] devices = disknames.isEmpty() ? new String[0] : disknames.split(",");
196         boolean elevated = "0".equals(ExecutingCommand.getFirstAnswer("id -u"));
197         for (String device : devices) {
198             String diskName = device.split(":", -1)[0];
199             Quartet<String, String, Long, List<HWPartition>> diskdata = Disklabel.getDiskParams(diskName);
200             if (elevated) {
201                 assertThat("Disk label is not null", diskdata.getA(), is(not(nullValue())));
202                 assertThat("Disk duid is not null", diskdata.getB(), is(not(nullValue())));
203                 assertThat("Disk size is nonnegative", diskdata.getC(), is(greaterThanOrEqualTo(0L)));
204                 for (HWPartition part : diskdata.getD()) {
205                     assertTrue(part.getIdentification().startsWith(diskName), "Partition ID starts with disk");
206                 }
207             }
208         }
209     }

211     @Test
212     @EnabledIfSystemProperty(named = "os.name", matches = "(?i)netbsd")
213     void testGetDiskParamsLiveNetBsd() {
214         String disknames = ExecutingCommand.getFirstAnswer("sysctl -n hw.disknames");
215         String[] devices = disknames.isEmpty() ? new String[0] : disknames.trim().split("\\s+");
216         boolean elevated = "0".equals(ExecutingCommand.getFirstAnswer("id -u"));
217         for (String device : devices) {
218             Quartet<String, String, Long, List<HWPartition>> diskdata = Disklabel.getDiskParams(device);
219             if (elevated) {
220                 assertThat("Disk label is not null", diskdata.getA(), is(not(nullValue())));
221                 assertThat("Disk duid is not null", diskdata.getB(), is(not(nullValue())));
222                 assertThat("Disk size is nonnegative", diskdata.getC(), is(greaterThanOrEqualTo(0L)));
223             }
224         }
