21      @Test
22      void testParsePrtvtocTypicalOutput() {
23          // Typical prtvtoc output for a Solaris disk
24          List<String> prtvtoc = """
25                  * /dev/dsk/c1t0d0s0 partition map
26                  *
27                  * Dimensions:
28                  *     512 bytes/sector
29                  *      63 sectors/track
30                  *      16 tracks/cylinder
31                  *    1008 sectors/cylinder
32                  *   16384 cylinders
33                  *   16382 accessible cylinders
34                  *
35                  * Flags:
36                  *   1: unmountable
37                  *  10: read-only
38                  *
39                  * Volume Name: MyVolume
40                  *
41                  *          First     Sector    Last
42                  * Partition  Tag  Flags  Sector     Count    Sector  Mount Directory
43                         0      2    00       1008   14329440   14330447   /
44                         1      3    01    14330448    2097648   16428095
45                         2      5    00           0   16515072   16515071
46                         3      7    00    16428096      86976   16515071   /var
47                  """.lines().toList();
48  
49          List<HWPartition> result = Prtvtoc.parsePrtvtoc(prtvtoc, "c1t0d0", 102);
50  
51          // Partition 2 (backup/whole disk) is always skipped
52          assertThat(result.size(), is(3));
53  
54          // Partition 0: tag=2 (root), flags=00 (wm)
55          HWPartition p0 = result.get(0);
56          assertThat(p0.getIdentification(), is("c1t0d0s0"));
57          assertThat(p0.getName(), is("root"));
58          assertThat(p0.getType(), is("wm"));
59          assertThat(p0.getSize(), is(512L * 14329440L));
60          assertThat(p0.getMajor(), is(102));
61          assertThat(p0.getMinor(), is(0));
62          assertThat(p0.getMountPoint(), is("/"));
63          assertThat(p0.getLabel(), is("MyVolume"));
64  
65          // Partition 1: tag=3 (swap), flags=01 (wu)
66          HWPartition p1 = result.get(1);
67          assertThat(p1.getIdentification(), is("c1t0d0s1"));
68          assertThat(p1.getName(), is("swap"));
69          assertThat(p1.getType(), is("wu"));
70          assertThat(p1.getSize(), is(512L * 2097648L));
71          assertThat(p1.getMajor(), is(102));
72          assertThat(p1.getMinor(), is(1));
73          assertThat(p1.getMountPoint(), is(""));
74  
75          // Partition 3: tag=7 (var), flags=00 (wm)
76          HWPartition p3 = result.get(2);
77          assertThat(p3.getIdentification(), is("c1t0d0s3"));
78          assertThat(p3.getName(), is("var"));
79          assertThat(p3.getType(), is("wm"));
80          assertThat(p3.getSize(), is(512L * 86976L));
81          assertThat(p3.getMajor(), is(102));
82          assertThat(p3.getMinor(), is(3));
83          assertThat(p3.getMountPoint(), is("/var"));
84      }

86      @Test
87      void testParsePrtvtocTagMappings() {
88          // Test various tag values to verify name mappings
89          List<String> prtvtoc = """
90                  * header line
91                  *     512 bytes/sector
92                         0      1    00       0   1000    999
93                         1     24    00       0   1000    999
94                         3      4    10       0   1000    999
95                         4      6    00       0   1000    999
96                         5      8    00       0   1000    999
97                         6      9    00       0   1000    999
98                         7     10    00       0   1000    999
99                         8     11    00       0   1000    999
100                        9     12    00       0   1000    999
101                       10     14    00       0   1000    999
102                       11     15    00       0   1000    999
103                       12     99    00       0   1000    999
104                 """.lines().toList();
105 
106         List<HWPartition> result = Prtvtoc.parsePrtvtoc(prtvtoc, "c0d0", 10);
107 
108         assertThat(result.size(), is(12));
109         // tag 0x01 -> boot
110         assertThat(result.get(0).getName(), is("boot"));
111         // tag 0x18 (24) -> boot
112         assertThat(result.get(1).getName(), is("boot"));
113         // tag 0x04 -> usr
114         assertThat(result.get(2).getName(), is("usr"));
115         // tag 0x06 -> stand
116         assertThat(result.get(3).getName(), is("stand"));
117         // tag 0x08 -> home
118         assertThat(result.get(4).getName(), is("home"));
119         // tag 0x09 -> altsctr
120         assertThat(result.get(5).getName(), is("altsctr"));
121         // tag 0x0a (10) -> cache
122         assertThat(result.get(6).getName(), is("cache"));
123         // tag 0x0b (11) -> reserved
124         assertThat(result.get(7).getName(), is("reserved"));
125         // tag 0x0c (12) -> system
126         assertThat(result.get(8).getName(), is("system"));
127         // tag 0x0e (14) -> public region
128         assertThat(result.get(9).getName(), is("public region"));
129         // tag 0x0f (15) -> private region
130         assertThat(result.get(10).getName(), is("private region"));
131         // tag 99 -> unknown
132         assertThat(result.get(11).getName(), is(Constants.UNKNOWN));
133     }

135     @Test
136     void testParsePrtvtocFlagMappings() {
137         // Test flag values: 00=wm, 10=rm, 01=wu, other=ru
138         List<String> prtvtoc = """
139                 * header
140                 *     512 bytes/sector
141                        0      2    00       0   1000    999
142                        1      2    10       0   1000    999
143                        3      2    01       0   1000    999
144                        4      2    11       0   1000    999
145                 """.lines().toList();
146 
147         List<HWPartition> result = Prtvtoc.parsePrtvtoc(prtvtoc, "c0d0", 5);
148 
149         assertThat(result.size(), is(4));
150         assertThat(result.get(0).getType(), is("wm"));
151         assertThat(result.get(1).getType(), is("rm"));
152         assertThat(result.get(2).getType(), is("wu"));
153         assertThat(result.get(3).getType(), is("ru"));
154     }

171         // If bytes/sector is never set, partition lines are ignored

177 
178         List<HWPartition> result = Prtvtoc.parsePrtvtoc(prtvtoc, "c0d0", 10);
