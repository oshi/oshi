5   package oshi.driver.common.windows.wmi;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.jspecify.annotations.Nullable;
11  import org.junit.jupiter.api.AfterEach;
12  import org.junit.jupiter.api.Test;
13  import org.junit.jupiter.api.parallel.Isolated;
14  
15  import oshi.driver.common.windows.wmi.LhmSensor.LhmHardwareProperty;
16  import oshi.driver.common.windows.wmi.LhmSensor.LhmSensorProperty;
17  import oshi.util.GlobalConfig;
18  
19  /**
20   * Tests the configuration switches which suppress hardware monitor queries. The switches are global state, so these
21   * tests must not run alongside any other test which reads or writes the configuration.
22   */
23  @Isolated
24  class HardwareMonitorDisabledTest {
25  
26      private static final String OTHER_NAMESPACE = "ROOT\\CIMV2";
27  
28      private final @Nullable String ohmConfig = GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED);
29      private final @Nullable String lhmConfig = GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED);
30  
31      @AfterEach
32      void restoreConfig() {
33          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED, ohmConfig);
34          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, lhmConfig);
35      }
36  
37      @Test
38      void testEnabledByDefault() {
39          assertThat("OHM enabled by default", HardwareMonitorDisabled.isWmiDisabled(OhmHardware.OHM_NAMESPACE),
40                  is(false));
41          assertThat("LHM enabled by default", HardwareMonitorDisabled.isWmiDisabled(LhmSensor.LHM_NAMESPACE), is(false));
42      }
43  
44      @Test
45      void testSwitchesAreIndependent() {
46          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED, true);
47          assertThat("OHM disabled", HardwareMonitorDisabled.isWmiDisabled(OhmHardware.OHM_NAMESPACE), is(true));
48          assertThat("LHM unaffected", HardwareMonitorDisabled.isWmiDisabled(LhmSensor.LHM_NAMESPACE), is(false));
49  
50          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED, false);
51          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, true);
52          assertThat("LHM disabled", HardwareMonitorDisabled.isWmiDisabled(LhmSensor.LHM_NAMESPACE), is(true));
53          assertThat("OHM unaffected", HardwareMonitorDisabled.isWmiDisabled(OhmHardware.OHM_NAMESPACE), is(false));
54      }
55  
56      @Test
57      void testOtherNamespaceUnaffected() {
58          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED, true);
59          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, true);
60          assertThat("Unrelated namespace still queried", HardwareMonitorDisabled.isWmiDisabled(OTHER_NAMESPACE),
61                  is(false));
62      }
63  
64      @Test
65      void testDisabledLhmSkipsQueries() {
66          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, true);
67          CountingExecutor executor = new CountingExecutor();
68  
69          WmiResult<LhmSensorProperty> sensors = LhmSensor.querySensors(executor, "/gpu-nvidia/0", "Load");
70          assertThat("No sensor rows", sensors.getResultCount(), is(0));
71          WmiResult<LhmHardwareProperty> hardware = LhmSensor.queryGpuHardware(executor);
72          assertThat("No hardware rows", hardware.getResultCount(), is(0));
73          assertThat("WMI not queried", executor.queries, is(0));
74      }
75  
76      @Test
77      void testEnabledLhmQueries() {
78          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, false);
79          CountingExecutor executor = new CountingExecutor();
80  
81          LhmSensor.querySensors(executor, "/gpu-nvidia/0", "Load");
82          LhmSensor.queryGpuHardware(executor);
83          assertThat("WMI queried once per call", executor.queries, is(2));
84      }
85  
86      /** A {@link WmiQueryExecutor} which counts queries and returns no rows. */
87      private static final class CountingExecutor implements WmiQueryExecutor {
88          private int queries;
89  
90          @Override
91          public <T extends Enum<T>> WmiResult<T> queryWMI(WmiQuery<T> query) {
92              queries++;
93              return WmiResult.empty();
94          }
95  
96          @Override
97          public <T extends Enum<T>> WmiResult<T> queryWMI(WmiQuery<T> query, boolean initCom) {
98              return queryWMI(query);
99          }
100     }
101 }
