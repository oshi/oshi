63      // One DISPLAYCONFIG_PATH_INFO whose source indexes the given mode, rotated and at the given refresh rate
64      private static ByteBuffer path(int sourceModeIndex, int rotation, int refreshNumerator, int refreshDenominator) {
65          ByteBuffer path = ByteBuffer.allocate(DisplayConnector.PATH_INFO_SIZE).order(ByteOrder.LITTLE_ENDIAN);
66          path.putInt(DisplayConnector.PATH_SOURCE_MODE_IDX_OFFSET, sourceModeIndex);
67          path.putInt(DisplayConnector.PATH_TARGET_ROTATION_OFFSET, rotation);
68          path.putInt(DisplayConnector.PATH_TARGET_REFRESH_NUMERATOR_OFFSET, refreshNumerator);
69          path.putInt(DisplayConnector.PATH_TARGET_REFRESH_DENOMINATOR_OFFSET, refreshDenominator);
70          return path;
71      }
72  
73      // A DISPLAYCONFIG_MODE_INFO array: a target mode at index 0, a source mode at index 1
74      private static ByteBuffer modes() {
75          ByteBuffer modes = ByteBuffer.allocate(2 * DisplayConnector.MODE_INFO_SIZE).order(ByteOrder.LITTLE_ENDIAN);
76          // DISPLAYCONFIG_MODE_INFO_TYPE_TARGET
77          modes.putInt(0, 2);
78          int source = DisplayConnector.MODE_INFO_SIZE;
79          // DISPLAYCONFIG_MODE_INFO_TYPE_SOURCE, then width, height, pixelFormat, position.x, position.y
80          modes.putInt(source, 1);
81          modes.putInt(source + 16, 1080);
82          modes.putInt(source + 20, 1920);
83          modes.putInt(source + 24, 4);
84          modes.putInt(source + 28, -1080);
85          modes.putInt(source + 32, -240);
86          return modes;
87      }
88  
89      private static @Nullable DisplayMode readMode(ByteBuffer path, ByteBuffer modes) {
90          return DisplayConnector.readMode(off -> path.getInt((int) off), off -> modes.getInt((int) off), 2);
91      }
92  
93      @Test
94      void testReadModeFromSourceModeAndTarget() {
95          // DISPLAYCONFIG_ROTATION_ROTATE90, 143.998 Hz as a rational
96          DisplayMode mode = readMode(path(1, 2, 143998, 1000), modes());
97          assertNotNull(mode);
98          assertThat(mode.getX(), is(-1080));
99          assertThat(mode.getY(), is(-240));
100         assertThat(mode.getWidth(), is(1080));
101         assertThat(mode.getHeight(), is(1920));
102         assertThat(mode.getPixelWidth(), is(1080));
103         assertThat(mode.getPixelHeight(), is(1920));
104         assertThat(mode.getRotation(), is(90));
105         assertThat(mode.getRefreshRate(), is(143.998));
106     }

137     @Test
138     void testIsBuiltIn() {
139         // LVDS, embedded DisplayPort, embedded UDI, internal
140         for (int technology : new int[] { 6, 11, 13, 0x80000000 }) {
141             assertThat(DisplayConnector.isBuiltIn(technology), is(true));
142         }
143         // VGA, HDMI, external DisplayPort, external UDI, Miracast, other
144         for (int technology : new int[] { 0, 5, 10, 12, 15, -1 }) {
145             assertThat(DisplayConnector.isBuiltIn(technology), is(false));
146         }
147     }

149     @Test
150     void testConnector() {
151         DisplayMode mode = readMode(path(1, 1, 60, 1), modes());
152         Connector connector = new Connector(10, 2, mode);
153         assertThat(connector.getName(), is("DisplayPort-2"));
154         assertThat(connector.isBuiltIn(), is(false));
155         assertThat(connector.getMode().isPresent(), is(true));
156         assertThat(new Connector(11, 0, null).getMode().isPresent(), is(false));
157     }
