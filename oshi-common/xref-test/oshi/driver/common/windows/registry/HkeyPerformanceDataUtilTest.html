23  /**
24   * Exercises the HKEY_PERFORMANCE_DATA walk against a synthetic block laid out by hand.
25   * <p>
26   * The block is written at the literal field offsets in {@code winperf.h}, chosen independently of the constants the
27   * walk uses, so a wrong constant reads the wrong place and fails here rather than silently returning bad counters on
28   * Windows. The walk is offset arithmetic over a buffer and makes no native call, so this runs on every platform - the
29   * only way this code gets tested at all.
30   */
31  class HkeyPerformanceDataUtilTest {
32  
33      // winperf.h, 64-bit
34      private static final int DATA_HEADER_LENGTH = 24;
35      private static final int DATA_NUM_OBJECT_TYPES = 28;
36      private static final int DATA_PERF_TIME_100N_SEC = 72;
37      private static final int OBJ_TOTAL_BYTE_LENGTH = 0;
38      private static final int OBJ_DEFINITION_LENGTH = 4;
39      private static final int OBJ_HEADER_LENGTH = 8;
40      private static final int OBJ_NAME_TITLE_INDEX = 12;
41      private static final int OBJ_NUM_COUNTERS = 32;
42      private static final int OBJ_NUM_INSTANCES = 40;
43      private static final int DEF_BYTE_LENGTH = 0;
44      private static final int DEF_NAME_TITLE_INDEX = 4;
45      private static final int DEF_COUNTER_SIZE = 32;
46      private static final int DEF_COUNTER_OFFSET = 36;
47      private static final int INST_BYTE_LENGTH = 0;
48      private static final int INST_NAME_OFFSET = 16;
49      private static final int BLOCK_BYTE_LENGTH = 0;
50  
51      private static final int PERF_DATA_BLOCK_SIZE = 88;
52      private static final int OBJECT_HEADER_SIZE = 64;
53      private static final int COUNTER_DEF_SIZE = 40;
54      private static final int INSTANCE_DEF_SIZE = 40;
55      private static final int COUNTER_BLOCK_SIZE = 32;
56  
57      private static final int WANTED_OBJECT = 230;
58      private static final int DECOY_OBJECT = 238;
59      private static final int PROC_COUNTER = 780;
60      private static final int TIME_COUNTER = 784;
61      private static final long PERF_TIME = 133_000_000_000_000_000L;
62      private static final long PERF_TIME_MILLIS = 1_655_526_400_000L;
63  
64      /** Counters to extract; the first constant names the instance rather than a counter. */
65      enum TestProperty implements PdhCounterWildcardProperty {
66          INSTANCE("Name"), PROC_ID("ID Process"), ELAPSED("Elapsed Time");
67  
68          private final String counter;
69  
70          TestProperty(String counter) {
71              this.counter = counter;
72          }
73  
74          @Override
75          public String getCounter() {
76              return this.counter;
77          }
78      }

80      /** A {@link PerfDataBuffer} over a plain byte array, standing in for JNA's Memory or an FFM MemorySegment. */
81      private static final class ArrayBuffer implements PerfDataBuffer {
82          private final ByteBuffer bb;
83  
84          private ArrayBuffer(byte[] bytes) {
85              this.bb = ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN);
86          }
87  
88          @Override
89          public int getInt(long offset) {
90              return this.bb.getInt((int) offset);
91          }
92  
93          @Override
94          public long getLong(long offset) {
95              return this.bb.getLong((int) offset);
96          }
97  
98          @Override
99          public String getWideString(long offset) {
100             int end = (int) offset;
101             while (this.bb.getShort(end) != 0) {
102                 end += 2;
103             }
104             byte[] chars = new byte[end - (int) offset];
105             // ByteBuffer.position(int) returns Buffer before Java 9, so these cannot be chained here
106             this.bb.position((int) offset);
107             this.bb.get(chars);
108             return new String(chars, StandardCharsets.UTF_16LE);
109         }
110     }

112     /**
113      * Builds a block holding a decoy object followed by the wanted one, so the walk has to skip by TotalByteLength to
114      * reach it, and two instances, so it has to step by the counter block length to reach the second.
115      */
116     private static byte[] buildBlock() {
117         return buildBlock(4);
118     }
119 
120     private static byte[] buildBlock(int firstCounterSize) {
121         int decoyAt = PERF_DATA_BLOCK_SIZE;
122         int decoyLen = OBJECT_HEADER_SIZE;
123         int objAt = decoyAt + decoyLen;
124         int defsAt = objAt + OBJECT_HEADER_SIZE;
125         int definitionLength = OBJECT_HEADER_SIZE + 2 * COUNTER_DEF_SIZE;
126         int instancesAt = objAt + definitionLength;
127         int inst1Block = instancesAt + INSTANCE_DEF_SIZE;
128         int inst2At = inst1Block + COUNTER_BLOCK_SIZE;
129         int inst2Block = inst2At + INSTANCE_DEF_SIZE;
130         byte[] bytes = new byte[inst2Block + COUNTER_BLOCK_SIZE];
131         ByteBuffer bb = ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN);
132 
133         bb.putInt(DATA_HEADER_LENGTH, PERF_DATA_BLOCK_SIZE);
134         bb.putInt(DATA_NUM_OBJECT_TYPES, 2);
135         bb.putLong(DATA_PERF_TIME_100N_SEC, PERF_TIME);
136 
137         // Decoy object: only its title index and total length are ever read
138         bb.putInt(decoyAt + OBJ_NAME_TITLE_INDEX, DECOY_OBJECT);
139         bb.putInt(decoyAt + OBJ_TOTAL_BYTE_LENGTH, decoyLen);
140 
141         bb.putInt(objAt + OBJ_NAME_TITLE_INDEX, WANTED_OBJECT);
142         bb.putInt(objAt + OBJ_TOTAL_BYTE_LENGTH, bytes.length - objAt);
143         bb.putInt(objAt + OBJ_HEADER_LENGTH, OBJECT_HEADER_SIZE);
144         bb.putInt(objAt + OBJ_DEFINITION_LENGTH, definitionLength);
145         bb.putInt(objAt + OBJ_NUM_COUNTERS, 2);
146         bb.putInt(objAt + OBJ_NUM_INSTANCES, 2);
147 
148         // A 4-byte counter at block offset 8, and an 8-byte counter at block offset 16
149         bb.putInt(defsAt + DEF_BYTE_LENGTH, COUNTER_DEF_SIZE);
150         bb.putInt(defsAt + DEF_NAME_TITLE_INDEX, PROC_COUNTER);
151         bb.putInt(defsAt + DEF_COUNTER_SIZE, firstCounterSize);
152         bb.putInt(defsAt + DEF_COUNTER_OFFSET, 8);
153         int def2 = defsAt + COUNTER_DEF_SIZE;
154         bb.putInt(def2 + DEF_BYTE_LENGTH, COUNTER_DEF_SIZE);
155         bb.putInt(def2 + DEF_NAME_TITLE_INDEX, TIME_COUNTER);
156         bb.putInt(def2 + DEF_COUNTER_SIZE, 8);
157         bb.putInt(def2 + DEF_COUNTER_OFFSET, 16);
158 
159         putInstance(bb, instancesAt, "firefox", inst1Block, 1234, 5_000_000_000L);
160         putInstance(bb, inst2At, "chrome", inst2Block, 5678, 9_000_000_000L);
161         return bytes;
162     }
163 
164     private static void putInstance(ByteBuffer bb, int at, String name, int blockAt, int pid, long elapsed) {
165         bb.putInt(at + INST_BYTE_LENGTH, INSTANCE_DEF_SIZE);
166         bb.putInt(at + INST_NAME_OFFSET, 24);
167         bb.position(at + 24);
168         bb.put(name.getBytes(StandardCharsets.UTF_16LE));
169         bb.putShort((short) 0);
170         bb.putInt(blockAt + BLOCK_BYTE_LENGTH, COUNTER_BLOCK_SIZE);
171         bb.putInt(blockAt + 8, pid);
172         bb.putLong(blockAt + 16, elapsed);
173     }

175     private static EnumMap<TestProperty, Integer> indexMap() {
176         EnumMap<TestProperty, Integer> map = new EnumMap<>(TestProperty.class);
177         map.put(TestProperty.PROC_ID, PROC_COUNTER);
178         map.put(TestProperty.ELAPSED, TIME_COUNTER);
179         return map;
180     }

182     @Test
183     void testParsePerfData() {
184         Triplet<List<Map<TestProperty, Object>>, Long, Long> result = HkeyPerformanceDataUtil
185                 .parsePerfData(new ArrayBuffer(buildBlock()), WANTED_OBJECT, indexMap(), TestProperty.class);
186         assertNotNull(result);
187         assertEquals(PERF_TIME, result.getB());
188         // 1601 epoch 100nSec units converted to 1970 epoch milliseconds, stated rather than recomputed
189         assertEquals(PERF_TIME_MILLIS, result.getC());
190 
191         List<Map<TestProperty, Object>> instances = result.getA();
192         assertEquals(2, instances.size());
193 
194         assertEquals("firefox", instances.get(0).get(TestProperty.INSTANCE));
195         assertEquals(1234, instances.get(0).get(TestProperty.PROC_ID));
196         assertEquals(5_000_000_000L, instances.get(0).get(TestProperty.ELAPSED));
197 
198         // Reaching the second instance means stepping by the counter block length, not just the definition length
199         assertEquals("chrome", instances.get(1).get(TestProperty.INSTANCE));
200         assertEquals(5678, instances.get(1).get(TestProperty.PROC_ID));
201         assertEquals(9_000_000_000L, instances.get(1).get(TestProperty.ELAPSED));
202     }

210     @Test
211     void testUnsupportedCounterSizeReturnsNull() {
212         // Only 4- and 8-byte counters can be read; anything else abandons the object rather than guessing
213         assertNull(HkeyPerformanceDataUtil.parsePerfData(new ArrayBuffer(buildBlock(2)), WANTED_OBJECT, indexMap(),
214                 TestProperty.class));
215     }

217     @Test
218     void testUnknownCounterReturnsNull() {
219         EnumMap<TestProperty, Integer> map = indexMap();
220         map.put(TestProperty.ELAPSED, 12345);
221         assertNull(HkeyPerformanceDataUtil.parsePerfData(new ArrayBuffer(buildBlock()), WANTED_OBJECT, map,
222                 TestProperty.class));
