5   package oshi.driver.common.mac;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   import static org.hamcrest.Matchers.sameInstance;
10  
11  import java.util.List;
12  import java.util.Optional;
13  
14  import org.junit.jupiter.api.Test;
15  
16  import oshi.hardware.DisplayMode;
17  import oshi.util.ParseUtil;
18  
19  class CoreGraphicsDisplayTest {
20  
21      // An LG monitor's EDID: manufacturer 0x1e6d (GSM), product code 0x5b08, serial number 0x0b0b0b0b
22      private static final byte[] EDID = ParseUtil
23              .hexStringToByteArray("00ffffffffffff001e6d085b0b0b0b0b" + "0c1c0104b53c2278fb2eb5ae4f46a527"
24                      + "0d5054254b80714f81809500a9c0b300" + "d1c001010101565e00a0a0a029503020"
25                      + "35000f282100001a000000fd00384b1e" + "5a1900000a202020202020000000fc00"
26                      + "4c4720554c545241474541520000ff00" + "3630344e54505a4832313337370a0100");
27  
28      private static final CoreGraphicsDisplay BUILT_IN = new CoreGraphicsDisplay(0x610, 0xa050, 0, true, true, null);
29      private static final CoreGraphicsDisplay LG = new CoreGraphicsDisplay(0x1e6d, 0x5b08, 0x0b0b0b0b, false, false,
30              CoreGraphicsDisplay.toMode(1728, 0, 2560, 1440, 2560, 1440, 60, 0));
31      private static final CoreGraphicsDisplay OTHER = new CoreGraphicsDisplay(0x10ac, 0x4114, 0x42543034, false, false,
32              null);
33  
34      @Test
35      void testMatchEdidByVendorModelAndSerial() {
36          Optional<CoreGraphicsDisplay> match = CoreGraphicsDisplay.matchEdid(List.of(BUILT_IN, LG, OTHER), EDID);
37          assertThat(match.isPresent(), is(true));
38          assertThat(match.get(), is(sameInstance(LG)));
39      }
40  
41      @Test
42      void testMatchEdidAmbiguousIsEmpty() {
43          // Two monitors reporting the same identity cannot be told apart
44          CoreGraphicsDisplay twin = new CoreGraphicsDisplay(0x1e6d, 0x5b08, 0x0b0b0b0b, false, false, null);
45          assertThat(CoreGraphicsDisplay.matchEdid(List.of(LG, twin), EDID).isPresent(), is(false));
46      }
47  
48      @Test
49      void testMatchEdidNoMatchOrShortEdid() {
50          assertThat(CoreGraphicsDisplay.matchEdid(List.of(BUILT_IN, OTHER), EDID).isPresent(), is(false));
51          assertThat(CoreGraphicsDisplay.matchEdid(List.of(LG), new byte[15]).isPresent(), is(false));
52          assertThat(CoreGraphicsDisplay.matchEdid(List.of(), EDID).isPresent(), is(false));
53      }
54  
55      @Test
56      void testMatchBuiltIn() {
57          assertThat(CoreGraphicsDisplay.matchBuiltIn(List.of(LG, BUILT_IN)).get(), is(sameInstance(BUILT_IN)));
58          assertThat(CoreGraphicsDisplay.matchBuiltIn(List.of(LG, OTHER)).isPresent(), is(false));
59          assertThat(CoreGraphicsDisplay.matchBuiltIn(List.of(BUILT_IN, BUILT_IN)).isPresent(), is(false));
60      }
61  
62      @Test
63      void testToModeRetina() {
64          // A 3456x2234 panel presented as 1728x1117 points, captured from a MacBook Pro
65          DisplayMode mode = CoreGraphicsDisplay.toMode(0, 0, 1728, 1117, 3456, 2234, 120, 0);
66          assertThat(mode.getWidth(), is(1728));
67          assertThat(mode.getHeight(), is(1117));
68          assertThat(mode.getPixelWidth(), is(3456));
69          assertThat(mode.getPixelHeight(), is(2234));
70          assertThat(mode.getRefreshRate(), is(120d));
71      }
72  
73      @Test
74      void testToModeRotatedTurnsPixelSize() {
75          // CGDisplayBounds is in the rotated orientation; the mode's pixel size is not
76          DisplayMode mode = CoreGraphicsDisplay.toMode(-1080, -300, 1080, 1920, 1920, 1080, 60, 90);
77          assertThat(mode.getWidth(), is(1080));
78          assertThat(mode.getHeight(), is(1920));
79          assertThat(mode.getPixelWidth(), is(1080));
80          assertThat(mode.getPixelHeight(), is(1920));
81          assertThat(mode.getRotation(), is(90));
82          assertThat(mode.getX(), is(-1080));
83          assertThat(mode.getY(), is(-300));
84      }
85  }
