22   * Tests {@link CpuFrequencyResidency} against the performance state residency and voltage state tables of a real M3 Pro

28      private static final long[] E_TABLE = { 744_000_000L, 1_044_000_000L, 1_476_000_000L, 2_004_000_000L,
29              2_268_000_000L, 2_448_000_000L, 2_640_000_000L, 2_748_000_000L };

37      /** How much of a frequency a rounding difference may account for. */
38      private static final long TOLERANCE_HZ = 1_000_000L;

40      // Compares two frequencies in Hz, allowing for the rounding of the expected value computed by hand.
41      private static void assertFrequency(String message, long expectedHz, long actualHz) {
42          assertThat(message + ", " + actualHz + " Hz", Math.abs(actualHz - expectedHz),
43                  is(lessThanOrEqualTo(TOLERANCE_HZ)));
44      }

46      // Builds a residency map in channel state order, the first name being the idle state.
47      private static Map<String, Long> residency(String[] names, long... ticks) {
48          Map<String, Long> states = new LinkedHashMap<>();
49          for (int i = 0; i < names.length; i++) {
50              states.put(names[i], ticks[i]);
51          }
52          return states;
53      }

55      private static final String[] E_STATES = { "IDLE", "V0P7", "V1P6", "V2P5", "V3P4", "V4P3", "V5P2", "V6P1", "V7P0" };

58              "V7P12", "V8P11", "V9P10", "V10P9", "V11P8", "V12P7", "V13P6", "V14P5", "V15P4", "V16P3", "V17P2", "V18P1",
59              "V19P0" };

61      // ECPU0 of the real dump: mostly parked at the lowest frequency, with a little time at the highest
62      private static Map<String, Long> ecpu0() {
63          return residency(E_STATES, 1_567_158L, 890_293L, 8_869L, 34_395L, 95_064L, 0L, 0L, 0L, 15_183L);
64      }

67      private static Map<String, Long> pcpu0() {
68          return residency(P_STATES, 2_553_514L, 656L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 40_607L, 10_414L, 6_130L, 0L, 0L,
69                  0L, 0L, 0L, 0L, 0L, 229L);
70      }

73      private static Map<String, Long> pcpu3() {
74          return residency(P_STATES, 2_611_550L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L,
75                  0L, 0L);

78      @Test
79      void testWeightsOnlyTheTimeTheCoreWasRunning() {
80          // Weighted over the 1043804 active ticks; weighting the 1567158 idle ticks at zero would report 364 MHz, and
81          // the nominal maximum this replaces reports 2748 MHz
82          assertFrequency("efficiency core", 914_573_414L,
83                  CpuFrequencyResidency.activeWeightedFrequency(ecpu0(), E_TABLE));
84          assertFrequency("performance core", 3_025_986_835L,
85                  CpuFrequencyResidency.activeWeightedFrequency(pcpu0(), P_TABLE));
86      }

93      @Test
94      void testAnIntervalWithNoTicksAtAllIsNotAFrequency() {
95          // Two samples taken close enough together that no tick accumulated, which is not the same as an idle core: it
96          // says nothing about the frequency, so the caller keeps the nominal one
97          Map<String, Long> nothing = residency(P_STATES, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L, 0L,
98                  0L, 0L, 0L, 0L, 0L);
99          assertThat("nothing observed", CpuFrequencyResidency.activeWeightedFrequency(nothing, P_TABLE), is(0L));
100         assertThat("idle observed", CpuFrequencyResidency.activeWeightedFrequency(pcpu3(), P_TABLE), is(P_TABLE[0]));
101     }

103     @Test
104     void testNoFrequencyFallsOutsideItsTable() {
105         long efficiency = CpuFrequencyResidency.activeWeightedFrequency(ecpu0(), E_TABLE);
106         assertThat("efficiency core at least the minimum", efficiency >= E_TABLE[0], is(true));
107         assertThat("efficiency core at most the maximum", efficiency <= E_TABLE[E_TABLE.length - 1], is(true));
108         long performance = CpuFrequencyResidency.activeWeightedFrequency(pcpu0(), P_TABLE);
109         assertThat("performance core at least the minimum", performance >= P_TABLE[0], is(true));
110         assertThat("performance core at most the maximum", performance <= P_TABLE[P_TABLE.length - 1], is(true));
111     }

113     @Test
114     void testAPowerGatedStateAheadOfIdleIsAlsoExcluded() {
115         // Some chips report DOWN or OFF for a power-gated core, ahead of IDLE, which adds a state the table cannot
116         // explain. The frequencies still pair correctly from the end of the list.
117         String[] names = new String[E_STATES.length + 1];
118         names[0] = "DOWN";
119         System.arraycopy(E_STATES, 0, names, 1, E_STATES.length);
120         Map<String, Long> gated = residency(names, 42L, 1_567_158L, 890_293L, 8_869L, 34_395L, 95_064L, 0L, 0L, 0L,
121                 15_183L);
122         assertThat("DOWN ahead of IDLE", CpuFrequencyResidency.activeWeightedFrequency(gated, E_TABLE),
123                 is(CpuFrequencyResidency.activeWeightedFrequency(ecpu0(), E_TABLE)));
124     }

126     @Test
127     void testAMissingFrequencyStateIsNotPairedByShiftingTheTable() {
128         // One frequency state short of the table. Pairing what is there against the end of the table would charge the
129         // idle ticks as the lowest frequency and shift every other frequency onto its neighbour's ticks, reporting a
130         // number that looks plausible and is wrong throughout.
131         String[] names = Arrays.copyOf(E_STATES, E_STATES.length - 1);
132         Map<String, Long> incomplete = residency(names, 1_567_158L, 890_293L, 8_869L, 34_395L, 95_064L, 0L, 0L, 0L);
133         assertThat("one active state missing", CpuFrequencyResidency.activeWeightedFrequency(incomplete, E_TABLE),
134                 is(0L));
135     }

137     @Test
138     void testAnUnrecognizedLeadingStateIsStillExcluded() {
139         // A power-gated state named something this release does not know cannot be recognized as an idle state, so the
140         // surplus is resolved from the end of the list instead. Doing so only drops leading states, never a frequency.
141         String[] names = new String[E_STATES.length + 1];
142         names[0] = "POWERGATED";
143         System.arraycopy(E_STATES, 0, names, 1, E_STATES.length);
144         Map<String, Long> gated = residency(names, 42L, 1_567_158L, 890_293L, 8_869L, 34_395L, 95_064L, 0L, 0L, 0L,
145                 15_183L);
146         assertThat("unrecognized state ahead of IDLE", CpuFrequencyResidency.activeWeightedFrequency(gated, E_TABLE),
147                 is(CpuFrequencyResidency.activeWeightedFrequency(ecpu0(), E_TABLE)));

166     // Residency of a CPU complex that spent every active tick at one frequency of its table. A complex reports the same
167     // states a core of it does, its idle one being the time every one of its cores was idle.
168     private static Map<String, Long> complexAt(String[] states, int index) {
169         long[] ticks = new long[states.length];
170         ticks[0] = 500L;
171         ticks[index + 1] = 1_000L;
172         return residency(states, ticks);
173     }

175     private static final int RANK_E = CpuFrequencyResidency.prefixRank("ECPU0");
176     private static final int RANK_P = CpuFrequencyResidency.prefixRank("PCPU0");
177 
178     @Test
179     void testTheClusterStateTheHardwareRanAtIsPreferredToTheOneTheCoresAskedFor() {
180         // Both channels of a complex are published, and only the one reporting what the cores got is weighted: on a
181         // busy M3 Pro the cores ask for the top state throughout while the hardware runs several states below it
182         Map<String, Map<String, Long>> complexes = new LinkedHashMap<>();
183         complexes.put("ECPU", complexAt(E_STATES, 7));
184         complexes.put("ECPM", complexAt(E_STATES, 3));
185         complexes.put("PCPU", complexAt(P_STATES, 19));
186         complexes.put("PCPM", complexAt(P_STATES, 13));
187         Map<Integer, Map<String, Long>> byRank = CpuFrequencyResidency.realizedComplexStates(complexes);
188         assertThat("one entry per core type", byRank.keySet(), is(new TreeSet<>(List.of(RANK_E, RANK_P))));
189         Map<String, Long> efficiency = byRank.get(RANK_E);
190         Map<String, Long> performance = byRank.get(RANK_P);
191         assertNotNull(efficiency, "efficiency complex");
192         assertNotNull(performance, "performance complex");
193         assertThat("efficiency cluster", CpuFrequencyResidency.activeWeightedFrequency(efficiency, E_TABLE),
194                 is(E_TABLE[3]));
195         assertThat("performance cluster", CpuFrequencyResidency.activeWeightedFrequency(performance, P_TABLE),
196                 is(P_TABLE[13]));
197     }

199     @Test
200     void testEveryClusterOfOneCoreTypeIsSummed() {
201         // An Ultra prefixes each die's complex, and a chip with two clusters of one type on a die is expected to number
202         // them. Every cluster of a type runs the same frequencies, so one type ran at the average of its clusters.
203         for (String[] names : List.of(new String[] { "DIE_0_PCPM", "DIE_1_PCPM" }, new String[] { "PCPM0", "PCPM1" })) {
204             Map<String, Map<String, Long>> complexes = new LinkedHashMap<>();
205             complexes.put(names[0], complexAt(P_STATES, 5));
206             complexes.put(names[1], complexAt(P_STATES, 15));
207             Map<String, Long> summed = CpuFrequencyResidency.realizedComplexStates(complexes).get(RANK_P);
208             assertNotNull(summed, names[0] + " and " + names[1] + " are one core type");
209             assertFrequency(names[0] + " with " + names[1], (P_TABLE[5] + P_TABLE[15]) / 2,
210                     CpuFrequencyResidency.activeWeightedFrequency(summed, P_TABLE));
211         }
212     }

214     @Test
215     void testClustersOfOneTypeThatNameTheirStatesDifferentlyAreNotSummed() {
216         // Summing them would add one cluster's ticks at a frequency to another's at a different frequency, so the type
217         // is reported as one the caller cannot weight rather than as a number that is wrong throughout
218         Map<String, Map<String, Long>> complexes = new LinkedHashMap<>();
219         complexes.put("PCPM0", complexAt(P_STATES, 5));
220         complexes.put("PCPM1", complexAt(E_STATES, 3));
221         Map<String, Long> summed = CpuFrequencyResidency.realizedComplexStates(complexes).get(RANK_P);
222         assertNotNull(summed, "the core type is still reported");
223         assertThat("not summable", CpuFrequencyResidency.activeWeightedFrequency(summed, P_TABLE), is(0L));
224     }

236     @Test
237     void testChannelsAreOrderedByCoreTypeThenCore() {
238         assertThat("M3 Pro", CpuFrequencyResidency.orderChannels(List.of("PCPU5", "ECPU0", "PCPU0", "ECPU5", "ECPU1")),
239                 is(List.of("ECPU0", "ECPU1", "ECPU5", "PCPU0", "PCPU5")));
240         // The M5 Pro and Max name their performance cores MCPU and their fastest cores PCPU, with no ECPU at all, so
241         // the letters shift but the order does not
242         assertThat("M5 Max", CpuFrequencyResidency.orderChannels(List.of("PCPU1", "MCPU11", "PCPU0", "MCPU0", "MCPU2")),
243                 is(List.of("MCPU0", "MCPU2", "MCPU11", "PCPU0", "PCPU1")));
244         // An Ultra prefixes its channels with the die, and macOS numbers both dies' cores of one type before the next
245         assertThat("Ultra",
246                 CpuFrequencyResidency
247                         .orderChannels(List.of("DIE_1_PCPU0", "DIE_0_ECPU0", "DIE_1_ECPU0", "DIE_0_PCPU0")),
248                 is(List.of("DIE_0_ECPU0", "DIE_1_ECPU0", "DIE_0_PCPU0", "DIE_1_PCPU0")));
249     }

269     @Test
270     void testCoreTypesRankInAscendingPerformanceOrder() {
271         assertThat("E before P", CpuFrequencyResidency.prefixRank("ECPU0") < CpuFrequencyResidency.prefixRank("PCPU0"),
272                 is(true));
273         assertThat("M before P", CpuFrequencyResidency.prefixRank("MCPU0") < CpuFrequencyResidency.prefixRank("PCPU0"),
274                 is(true));
275         assertThat("P before S", CpuFrequencyResidency.prefixRank("PCPU0") < CpuFrequencyResidency.prefixRank("SCPU0"),
276                 is(true));
277         assertThat("one type per rank regardless of die",
278                 CpuFrequencyResidency.prefixRank("DIE_1_ECPU0") == CpuFrequencyResidency.prefixRank("ECPU0"), is(true));
279     }
