5   package oshi.hardware;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   import static org.hamcrest.Matchers.not;
10  
11  import org.junit.jupiter.api.Test;
12  
13  class DisplayModeImplTest {
14  
15      @Test
16      void testGetters() {
17          DisplayMode mode = new DisplayModeImpl(-1080, 120, 1080, 1920, 2160, 3840, 59.94, 90);
18          assertThat(mode.getX(), is(-1080));
19          assertThat(mode.getY(), is(120));
20          assertThat(mode.getWidth(), is(1080));
21          assertThat(mode.getHeight(), is(1920));
22          assertThat(mode.getPixelWidth(), is(2160));
23          assertThat(mode.getPixelHeight(), is(3840));
24          assertThat(mode.getRefreshRate(), is(59.94));
25          assertThat(mode.getRotation(), is(90));
26      }
27  
28      @Test
29      void testRotationNormalizedToQuarterTurns() {
30          assertThat(DisplayModeImpl.normalizeRotation(0), is(0));
31          assertThat(DisplayModeImpl.normalizeRotation(270), is(270));
32          assertThat(DisplayModeImpl.normalizeRotation(360), is(0));
33          assertThat(DisplayModeImpl.normalizeRotation(450), is(90));
34          assertThat(DisplayModeImpl.normalizeRotation(-90), is(270));
35          assertThat(DisplayModeImpl.normalizeRotation(-180), is(180));
36          assertThat(DisplayModeImpl.normalizeRotation(89), is(90));
37      }
38  
39      @Test
40      void testNegativeOrNaNRefreshRateIsZero() {
41          assertThat(new DisplayModeImpl(0, 0, 1, 1, 1, 1, -1d, 0).getRefreshRate(), is(0d));
42          assertThat(new DisplayModeImpl(0, 0, 1, 1, 1, 1, Double.NaN, 0).getRefreshRate(), is(0d));
43      }
44  
45      @Test
46      void testToString() {
47          assertThat(new DisplayModeImpl(0, 0, 1920, 1080, 1920, 1080, 60d, 0).toString(),
48                  is("1920x1080 @ 60.00 Hz at (0,0)"));
49          assertThat(new DisplayModeImpl(1920, -200, 1728, 1117, 3456, 2234, 0d, 0).toString(),
50                  is("1728x1117 (3456x2234 pixels) at (1920,-200)"));
51          assertThat(new DisplayModeImpl(0, 0, 1080, 1920, 1080, 1920, 59.93, 270).toString(),
52                  is("1080x1920 @ 59.93 Hz, rotated 270° at (0,0)"));
53      }
54  
55      @Test
56      void testValueEquality() {
57          DisplayMode mode = new DisplayModeImpl(1920, 0, 2560, 1440, 2560, 1440, 59.95, 0);
58          DisplayMode same = new DisplayModeImpl(1920, 0, 2560, 1440, 2560, 1440, 59.95, 360);
59          assertThat(mode, is(same));
60          assertThat(mode.hashCode(), is(same.hashCode()));
61          assertThat(mode, is(not(new DisplayModeImpl(1920, 0, 2560, 1440, 2560, 1440, 60d, 0))));
62          assertThat(mode, is(not(new DisplayModeImpl(0, 0, 2560, 1440, 2560, 1440, 59.95, 0))));
63          assertThat(mode, is(not(new DisplayModeImpl(1920, 0, 2560, 1440, 2560, 1440, 59.95, 90))));
64      }
65  }
