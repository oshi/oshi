34      @Test
35      void testWin32MediaTypeCannotIdentifyMediaTechnology() {
36          // Win32_DiskDrive's MediaType is mapped from DISK_GEOMETRY.MediaType and reports only removability.
37          // A fixed or external disk must not be guessed at: either could be an SSD.
38          assertThat("Fixed media is not an HDD", WindowsHWDiskStore.parseWindowsMediaType("Fixed hard disk media"),
39                  is("Unknown"));
40          assertThat("External media is not an HDD", WindowsHWDiskStore.parseWindowsMediaType("External hard disk media"),
41                  is("Unknown"));
42          assertThat("Unknown format", WindowsHWDiskStore.parseWindowsMediaType("Format is unknown"), is("Unknown"));
43          assertThat("Removable media", WindowsHWDiskStore.parseWindowsMediaType("Removable media other than floppy"),
44                  is("Removable"));
45      }

55          assertThat("No seek penalty is solid state", WindowsHWDiskStore.classifyDiskType(

145     /**
146      * Builds a WmiResult standing in for an {@code MSFT_PhysicalDisk} query.
147      *
148      * @param deviceIds    the DeviceId of each row
149      * @param mediaTypes   the MediaType of each row
150      * @param spindleSpeed the SpindleSpeed of each row
151      * @return the result
152      */
153     private static WmiResult<PhysicalDiskMediaTypeProperty> mediaTypeResult(String[] deviceIds, int[] mediaTypes,
154             long[] spindleSpeed) {
155         return new WmiResult<PhysicalDiskMediaTypeProperty>() {
156             @Override
157             public int getResultCount() {
158                 return deviceIds.length;
159             }
160 
161             @Override
162             public @Nullable Object getValue(PhysicalDiskMediaTypeProperty property, int index) {
163                 return switch (property) {
164                     case DEVICEID -> deviceIds[index];
165                     case MEDIATYPE -> mediaTypes[index];
166                     // SpindleSpeed is a uint32, carried in a VT_I4 and read back as unsigned
167                     case SPINDLESPEED -> (int) spindleSpeed[index];
168                 };
169             }
170 
171             @Override
172             public int getVtType(PhysicalDiskMediaTypeProperty property) {
173                 return property == PhysicalDiskMediaTypeProperty.DEVICEID ? WmiConstants.VT_BSTR : WmiConstants.VT_I4;
174             }
175 
176             @Override
177             public int getCIMType(PhysicalDiskMediaTypeProperty property) {
178                 return switch (property) {
179                     case DEVICEID -> WmiConstants.CIM_STRING;
180                     case MEDIATYPE -> WmiConstants.CIM_UINT16;
181                     case SPINDLESPEED -> WmiConstants.CIM_UINT32;
182                 };
183             }
184         };
185     }

187     @Test
188     void testClassifyDiskTypesOverAQueryResult() {
189         WmiResult<DiskDriveProperty> vals = diskDriveResult(
190                 new String[] { "\\\\.\\PHYSICALDRIVE0", "\\\\.\\PHYSICALDRIVE1" },
191                 new String[] { "Fixed hard disk media", "Removable media other than floppy" }, new int[] { 0, 1 });
192         // Resolver answers for drive 0 only, as a driver lacking the seek penalty property would
193         String[] diskTypes = WindowsHWDiskStore.classifyDiskTypes(vals,
194                 name -> name.endsWith("0") ? new StorageProperties(BUS_NVME, false, Boolean.FALSE) : null);
195         assertThat("Driver answered", diskTypes[0], is("SSD"));
196         assertThat("Driver silent, WMI says removable", diskTypes[1], is("Removable"));
197     }

199     @Test
200     void testFillUnknownDiskTypesOnlyFillsUnknowns() {
201         WmiResult<DiskDriveProperty> vals = diskDriveResult(
202                 new String[] { "\\\\.\\PHYSICALDRIVE0", "\\\\.\\PHYSICALDRIVE1", "\\\\.\\PHYSICALDRIVE2" },
203                 new String[] { "Fixed hard disk media", "Fixed hard disk media", "Fixed hard disk media" },
204                 new int[] { 0, 1, 2 });
205         // MSFT_PhysicalDisk claims HDD for all three, but only the Unknown may be overwritten
206         String[] diskTypes = { "Removable", "Unknown", "SSD" };
207         WindowsHWDiskStore.fillUnknownDiskTypes(diskTypes, vals, mediaTypeResult(new String[] { "0", "1", "2" },
208                 new int[] { 3, 3, 3 }, new long[] { 7200L, 7200L, 7200L }));
209         assertThat("Removable is not relabeled by media technology", diskTypes[0], is("Removable"));
210         assertThat("Unknown is filled from MSFT_PhysicalDisk", diskTypes[1], is("HDD"));
211         assertThat("SSD is not relabeled", diskTypes[2], is("SSD"));
212     }

214     @Test
215     void testFillUnknownDiskTypesJoinsOnDriveNumber() {
216         // Win32_DiskDrive's Index and MSFT_PhysicalDisk's DeviceId are both the physical drive number,
217         // and neither query is ordered by it, so the join must not rely on row position
218         WmiResult<DiskDriveProperty> vals = diskDriveResult(
219                 new String[] { "\\\\.\\PHYSICALDRIVE3", "\\\\.\\PHYSICALDRIVE7" },
220                 new String[] { "Fixed hard disk media", "Fixed hard disk media" }, new int[] { 3, 7 });
221         String[] diskTypes = { "Unknown", "Unknown" };
222         WindowsHWDiskStore.fillUnknownDiskTypes(diskTypes, vals,
223                 mediaTypeResult(new String[] { "7", "3" }, new int[] { 3, 4 }, new long[] { 7200L, 0L }));
224         assertThat("Drive 3 matched DeviceId 3", diskTypes[0], is("SSD"));
225         assertThat("Drive 7 matched DeviceId 7", diskTypes[1], is("HDD"));
226     }

249     /**
250      * Builds a WmiResult standing in for a {@code Win32_DiskDrive} query.
251      *
252      * @param names      the Name of each row
253      * @param mediaTypes the MediaType of each row
254      * @param indices    the Index of each row
255      * @return the result
256      */
257     private static WmiResult<DiskDriveProperty> diskDriveResult(String[] names, String[] mediaTypes, int[] indices) {
258         return new WmiResult<DiskDriveProperty>() {
259             @Override
260             public int getResultCount() {
261                 return names.length;
262             }
263 
264             @Override
265             public @Nullable Object getValue(DiskDriveProperty property, int index) {
266                 return switch (property) {
267                     case NAME -> names[index];
268                     case MEDIATYPE -> mediaTypes[index];
269                     case INDEX -> indices[index];
270                     default -> null;
271                 };
272             }
273 
274             @Override
275             public int getVtType(DiskDriveProperty property) {
276                 return property == DiskDriveProperty.INDEX ? WmiConstants.VT_I4 : WmiConstants.VT_BSTR;
277             }
278 
279             @Override
280             public int getCIMType(DiskDriveProperty property) {
281                 return property == DiskDriveProperty.INDEX ? WmiConstants.CIM_UINT32 : WmiConstants.CIM_STRING;
282             }
283         };
284     }
