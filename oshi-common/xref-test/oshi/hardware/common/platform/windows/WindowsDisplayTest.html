5   package oshi.hardware.common.platform.windows;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.Optional;
11  
12  import org.jspecify.annotations.Nullable;
13  import org.junit.jupiter.api.Test;
14  
15  import oshi.driver.common.windows.DisplayConnector.Connector;
16  import oshi.hardware.DisplayMode;
17  import oshi.hardware.DisplayModeImpl;
18  
19  class WindowsDisplayTest {
20  
21      private static final class TestDisplay extends WindowsDisplay {
22          TestDisplay(@Nullable Connector connector) {
23              super(new byte[128], connector);
24          }
25      }
26  
27      @Test
28      void testUnresolvedConnector() {
29          TestDisplay display = new TestDisplay(null);
30          assertThat(display.getDevicePort(), is("unknown"));
31          assertThat(display.getCurrentMode().isPresent(), is(false));
32          assertThat(display.isBuiltIn().isPresent(), is(false));
33          assertThat(display.isPrimary().isPresent(), is(false));
34      }
35  
36      @Test
37      void testResolvedConnector() {
38          DisplayMode mode = new DisplayModeImpl(0, 0, 2560, 1600, 2560, 1600, 165d, 0);
39          // VOT_DISPLAYPORT_EMBEDDED
40          TestDisplay display = new TestDisplay(new Connector(11, 0, mode));
41          assertThat(display.getDevicePort(), is("eDP"));
42          assertThat(display.getCurrentMode(), is(Optional.of(mode)));
43          assertThat(display.isBuiltIn(), is(Optional.of(Boolean.TRUE)));
44          // Its desktop origin is (0, 0)
45          assertThat(display.isPrimary(), is(Optional.of(Boolean.TRUE)));
46      }
47  
48      @Test
49      void testSecondaryDisplayIsNotPrimary() {
50          DisplayMode mode = new DisplayModeImpl(2560, 0, 1920, 1080, 1920, 1080, 60d, 0);
51          TestDisplay display = new TestDisplay(new Connector(5, 0, mode));
52          assertThat(display.isPrimary(), is(Optional.of(Boolean.FALSE)));
53      }
54  
55      @Test
56      void testResolvedConnectorWithoutMode() {
57          // VOT_HDMI
58          TestDisplay display = new TestDisplay(new Connector(5, 1, null));
59          assertThat(display.getDevicePort(), is("HDMI-1"));
60          assertThat(display.getCurrentMode().isPresent(), is(false));
61          assertThat(display.isBuiltIn(), is(Optional.of(Boolean.FALSE)));
62          assertThat(display.isPrimary().isPresent(), is(false));
63      }
64  }
