33   * Tests the shared Windows graphics card enumeration without Windows, by stubbing the WMI result. This path is the

39      /** A concrete card so the factory has something to build. */
40      private static final class TestCard extends WindowsGraphicsCard {
41          private TestCard(String name, String deviceId, String vendor, String versionInfo, long vram, String luidPrefix,
42                  String lhmParent, int pciBusNumber, String pciBusId) {
43              super(name, deviceId, vendor, versionInfo, vram, luidPrefix, lhmParent, pciBusNumber, pciBusId);
44          }

46          @Override
47          public oshi.hardware.GpuStats createStatsSession() {
48              throw new UnsupportedOperationException("not exercised by this test");
49          }

52      /** One Win32_VideoController row. */
53      private static final class Row {
54          private final Map<VideoControllerProperty, Object> values = new EnumMap<>(VideoControllerProperty.class);
55  
56          private Row set(VideoControllerProperty property, Object value) {
57              values.put(property, value);
58              return this;
59          }
60      }
61  
62      private static Row row(String name, String pnpDeviceId, String vendor, String driverVersion, int errorCode,
63              int adapterRam) {
64          return new Row().set(VideoControllerProperty.NAME, name).set(VideoControllerProperty.PNPDEVICEID, pnpDeviceId)
65                  .set(VideoControllerProperty.ADAPTERCOMPATIBILITY, vendor)
66                  .set(VideoControllerProperty.DRIVERVERSION, driverVersion)
67                  .set(VideoControllerProperty.CONFIGMANAGERERRORCODE, errorCode)
68                  .set(VideoControllerProperty.ADAPTERRAM, adapterRam);
69      }
70  
71      private static WmiResult<VideoControllerProperty> result(Row... rows) {
72          List<Row> list = List.of(rows);
73          return new WmiResult<VideoControllerProperty>() {
74              @Override
75              public int getResultCount() {
76                  return list.size();
77              }
78  
79              @Override
80              public @Nullable Object getValue(VideoControllerProperty property, int index) {
81                  return list.get(index).values.get(property);
82              }
83  
84              @Override
85              public int getVtType(VideoControllerProperty property) {
86                  return isNumeric(property) ? VT_I4 : VT_BSTR;
87              }
88  
89              @Override
90              public int getCIMType(VideoControllerProperty property) {
91                  return isNumeric(property) ? CIM_UINT32 : CIM_STRING;
92              }
93  
94              private boolean isNumeric(VideoControllerProperty property) {
95                  return property == VideoControllerProperty.CONFIGMANAGERERRORCODE
96                          || property == VideoControllerProperty.ADAPTERRAM;
97              }
98          };
99      }

101     private static List<GraphicsCard> build(List<DxgiAdapterInfo> adapters, WmiResult<VideoControllerProperty> cards) {
102         return WindowsGraphicsCard.buildFromWmi(adapters, Collections.emptyMap(), cards, adapter -> "luid_test",
103                 TestCard::new);
104     }

106     private static List<String> names(List<GraphicsCard> cards) {
107         return cards.stream().map(GraphicsCard::getName).toList();
108     }

110     @Test
111     void testWithoutDxgiEveryRowIsKeptInOrder() {
112         // With no DXGI adapters there is nothing to match against, so all rows are kept for maximum compatibility,
113         // including ones reporting a non-zero ConfigManagerErrorCode.
114         List<GraphicsCard> cards = build(Collections.emptyList(),
115                 result(row("Card A", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 4096),
116                         row("Card B", "PCI\\VEN_8086&DEV_56A0", "Intel", "4.5.6", 22, 2048)));
117         assertThat("both rows are kept", names(cards), contains("Card A", "Card B"));
118         assertThat("VRAM falls back to AdapterRAM", cards.get(0).getVRam(), is(4096L));
119     }
120 
121     @Test
122     void testGhostDevicesAreSkippedWhenDxgiIsAvailable() {
123         // A non-zero ConfigManagerErrorCode marks a disabled or errored device once DXGI can confirm what is present.
124         List<DxgiAdapterInfo> adapters = Collections
125                 .singletonList(new DxgiAdapterInfo("NVIDIA GeForce", 0x10DE, 0x1C03, 8_000_000_000L, 1, 0));
126         List<GraphicsCard> cards = build(adapters,
127                 result(row("Ghost", "PCI\\VEN_8086&DEV_56A0", "Intel", "4.5.6", 22, 2048),
128                         row("NVIDIA GeForce", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 2048)));
129         assertThat("the ghost row is dropped", names(cards), contains("NVIDIA GeForce"));
130         assertThat("DXGI VRAM beats the 32-bit AdapterRAM", cards.get(0).getVRam(), is(8_000_000_000L));
131         assertThat("the LUID prefix comes from the supplied builder", ((TestCard) cards.get(0)).getLuidPrefix(),
132                 is("luid_test"));
133     }

135     @Test
136     void testMatchedCardsAreOrderedByDxgiIndex() {
137         // DXGI enumerates the primary desktop adapter first, so the result follows DXGI order, not WMI row order.
138         List<DxgiAdapterInfo> adapters = List.of(new DxgiAdapterInfo("Primary", 0x10DE, 0x1C03, 100L, 1, 0),
139                 new DxgiAdapterInfo("Secondary", 0x8086, 0x56A0, 200L, 2, 0));
140         List<GraphicsCard> cards = build(adapters,
141                 result(row("Secondary", "PCI\\VEN_8086&DEV_56A0", "Intel", "4.5.6", 0, 1),
142                         row("Primary", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 1)));
143         assertThat("DXGI order wins over row order", names(cards), contains("Primary", "Secondary"));
144     }

146     @Test
147     void testAnAdapterIsConsumedByTheFirstMatchingRow() {
148         // Two rows for the same PCI IDs must not both claim the single DXGI adapter, or the second would inherit the
149         // first one's VRAM.
150         List<DxgiAdapterInfo> adapters = Collections
151                 .singletonList(new DxgiAdapterInfo("Shared", 0x10DE, 0x1C03, 9999L, 1, 0));
152         List<GraphicsCard> cards = build(adapters,
153                 result(row("Shared", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 111),
154                         row("Shared", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 222)));
155         assertThat("both rows are returned", cards.size(), is(2));
156         List<Long> vram = cards.stream().map(GraphicsCard::getVRam).toList();
157         assertThat("only one row got the DXGI VRAM", vram, contains(9999L, 222L));
158     }

160     @Test
161     void testVendorAndDeviceIdDerivation() {
162         List<GraphicsCard> cards = build(Collections.emptyList(),
163                 result(row("Card", "PCI\\VEN_10DE&DEV_1C03&SUBSYS_0&REV_A1\\4&SERIAL&0", "NVIDIA", "1.2.3", 0, 1)));
164         GraphicsCard card = cards.get(0);
165         assertThat("the vendor keeps its PCI id in parentheses", card.getVendor(), is("NVIDIA (0x10de)"));
166         assertThat("the version is prefixed", card.getVersionInfo(), is("DriverVersion=1.2.3"));
167     }

169     @Test
170     void testBlankFieldsFallBackToUnknown() {
171         List<GraphicsCard> cards = build(Collections.emptyList(), result(row("", "", "", "", 0, 1)));
172         GraphicsCard card = cards.get(0);
173         assertThat("a blank name is unknown", card.getName(), is(Constants.UNKNOWN));
174         assertThat("a blank vendor is unknown", card.getVendor(), is(Constants.UNKNOWN));
175         assertThat("a blank version is unknown", card.getVersionInfo(), is(Constants.UNKNOWN));
176     }

183     @Test
184     void testPciCorrelationFieldsAreUnsetOnThisPath() {
185         // The bus number and bus ID come from the registry LocationInformation value, which WMI does not expose, so
186         // ADL and NVML correlation are deliberately skipped for cards found this way.
187         List<GraphicsCard> cards = build(Collections.emptyList(),
188                 result(row("Card", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 1)));
189         assertThat("bus number is unknown", ((TestCard) cards.get(0)).getPciBusNumber(), is(-1));
190         assertThat("bus id is empty", ((TestCard) cards.get(0)).getPciBusId(), is(""));
191     }

193     @Test
194     void testLhmParentIsMatchedOnNormalizedName() {
195         Map<String, String> lhm = new HashMap<>();
196         lhm.put("nvidia geforce rtx 3080", "/gpu-nvidia/0");
197         List<GraphicsCard> cards = new ArrayList<>(WindowsGraphicsCard.buildFromWmi(Collections.emptyList(), lhm,
198                 result(row("NVIDIA GeForce RTX 3080", "PCI\\VEN_10DE&DEV_1C03", "NVIDIA", "1.2.3", 0, 1)),
199                 adapter -> "", TestCard::new));
200         assertThat("the LHM identifier is attached", ((TestCard) cards.get(0)).getLhmParent(), is("/gpu-nvidia/0"));
