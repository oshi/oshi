28  /**
29   * Tests the sensor values taken from the web server of LibreHardwareMonitor 0.9.5 and later. One test clears the server
30   * address, which is global state, so these tests must not run alongside any other test which reads or writes the
31   * configuration.
32   */
33  @Isolated
34  class WindowsSensorsLhmMetricsTest {
35  
36      private final @Nullable String urlConfig = GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_LHM_URL);
37  
38      @AfterEach
39      void restoreConfig() {
40          GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_URL, urlConfig);
41      }
42  
43      private static final List<Reading> READINGS = List.of(
44              new Reading("lhm_cpu_load_percent", "/intelcpu/0", "CPU Total", 15.2),
45              new Reading(LhmMetrics.CPU_TEMPERATURE, "/intelcpu/0", "CPU Package", 52.0),
46              new Reading(LhmMetrics.CPU_TEMPERATURE, "/intelcpu/0", "Core #1", 48.0),
47              new Reading(LhmMetrics.CPU_TEMPERATURE, "/intelcpu/0", "Core Max", 60.0),
48              new Reading(LhmMetrics.CPU_TEMPERATURE, "/intelcpu/0", "Core #1 Distance to TjMax", 52.0),
49              new Reading(LhmMetrics.CPU_TEMPERATURE, "/intelcpu/1", "CPU Package", 90.0),
50              new Reading("lhm_storage_temperature_celsius", "/nvme/0", "Composite Temperature", 37.5),
51              new Reading(LhmMetrics.MOTHERBOARD_FAN, "/lpc/nct6798d", "Fan 1", 0.0),
52              new Reading(LhmMetrics.MOTHERBOARD_FAN, "/lpc/nct6798d", "Fan 2", 1123.0),
53              new Reading(LhmMetrics.MOTHERBOARD_FAN, "/lpc/it8689e", "Fan 1", 850.0),
54              new Reading("lhm_gpunvidia_fan_rpm", "/gpu-nvidia/0", "GPU Fan 1", 1500.0),
55              new Reading(LhmMetrics.MOTHERBOARD_VOLTAGE, "/lpc/nct6798d", "+3.3V", 3.3),
56              new Reading(LhmMetrics.MOTHERBOARD_VOLTAGE, "/lpc/nct6798d", "Vcore", 1.2));
57  
58      @Test
59      void testCpuTemperatureAveragesTheFirstProcessorsReadings() {
60          assertThat("Summaries, margins and the second socket are left out", WindowsSensors.getCpuTemperature(READINGS),
61                  closeTo(50.0, 1e-9));
62          assertThat(WindowsSensors.getCpuTemperature(List.of()), is(0d));
63      }
64  
65      @Test
66      void testFanSpeedsAreTheMotherboardsFans() {
67          assertThat("Every Super I/O chip's fans, an idle header included; not the GPU's",
68                  WindowsSensors.getFanSpeeds(READINGS), is(new int[] { 0, 1123, 850 }));
69          assertThat(WindowsSensors.getFanSpeeds(List.of()), is(new int[0]));
70      }
71  
72      @Test
73      void testCpuVoltagePrefersTheProcessorsOwnSensor() {
74          assertThat("Without a processor voltage sensor, the Super I/O Vcore", WindowsSensors.getCpuVoltage(READINGS),
75                  closeTo(1.2, 1e-9));
76          List<Reading> withCpu = List.of(new Reading(LhmMetrics.MOTHERBOARD_VOLTAGE, "/lpc/nct6798d", "Vcore", 1.2),
77                  new Reading(LhmMetrics.CPU_VOLTAGE, "/amdcpu/0", "Core (SVI2 TFN)", 1.35));
78          assertThat(WindowsSensors.getCpuVoltage(withCpu), closeTo(1.35, 1e-9));
79          assertThat(WindowsSensors.getCpuVoltage(List.of()), is(0d));
80      }
81  
82      @Test
83      void testWebServerIsReadAfterBothWmiNamespaces() {
84          StubSensors sensors = new StubSensors(READINGS);
85          assertThat(sensors.queryCpuTemperature(), closeTo(50.0, 1e-9));
86          assertThat(sensors.queryFanSpeeds(), is(new int[] { 0, 1123, 850 }));
87          assertThat(sensors.queryCpuVoltage(), closeTo(1.2, 1e-9));
88          assertThat("One request serves the three reads of a refresh", sensors.requests, is(1));
89      }
90  
91      @Test
92      void testASlowRequestIsNotRepeatedByTheNextRead() {
93          long slowerThanTheCache = Memoizer.defaultExpiration() / 1_000_000L + 100L;
94          StubSensors sensors = new StubSensors(READINGS) {
95              @Override
96              protected List<Reading> queryLhmMetrics() {
97                  try {
98                      Thread.sleep(slowerThanTheCache);
99                  } catch (InterruptedException e) {
100                     Thread.currentThread().interrupt();
101                 }
102                 return super.queryLhmMetrics();
103             }
104         };
105         sensors.queryCpuTemperature();
106         sensors.queryFanSpeeds();
107         sensors.queryCpuVoltage();
108         assertThat("The cache lifetime starts when the request completes", sensors.requests, is(1));
109     }
110 
111     @Test
112     void testNoServerConfiguredMakesNoRequest() {
113         GlobalConfig.set(GlobalConfig.OSHI_OS_WINDOWS_LHM_URL, "");
114         StubSensors sensors = new StubSensors(READINGS) {
115             @Override
116             protected List<Reading> queryLhmMetrics() {
117                 return unstubbedQuery();
118             }
119         };
120         assertThat(sensors.queryCpuTemperature(), is(0d));
121         assertThat(sensors.queryFanSpeeds(), is(new int[0]));
122         assertThat(sensors.queryCpuVoltage(), is(0d));
123     }
124 
125     /** Neither WMI namespace answers; the web server returns the given readings, counting requests. */
126     private static class StubSensors extends WindowsSensors {
127         private final List<Reading> readings;
128         private int requests;
129 
130         StubSensors(List<Reading> readings) {
131             this.readings = readings;
132         }
133 
134         @Override
135         protected List<Reading> queryLhmMetrics() {
136             requests++;
137             return readings;
138         }
139 
140         /** The real query, which reads an unset address and so returns nothing. */
141         List<Reading> unstubbedQuery() {
142             return super.queryLhmMetrics();
143         }
144 
145         @Override
146         protected @Nullable WmiResult<ValueProperty> queryHardwareMonitorSensor(String namespace, String typeToQuery,
147                 String typeName, String sensorType) {
148             return null;
149         }
150 
151         @Override
152         protected WmiResult<TemperatureProperty> queryWmiTemperature() {
153             return WmiResult.empty();
154         }
155 
156         @Override
157         protected WmiResult<SpeedProperty> queryWmiFanSpeed() {
158             return WmiResult.empty();
159         }
160 
161         @Override
162         protected WmiResult<VoltProperty> queryWmiVoltage() {
163             return WmiResult.empty();
164         }
165     }
