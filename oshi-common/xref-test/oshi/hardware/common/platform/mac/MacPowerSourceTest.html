5   package oshi.hardware.common.platform.mac;

7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  
12  import oshi.hardware.PowerSource.CapacityUnits;
13  import oshi.util.tuples.Triplet;
14  
15  class MacPowerSourceTest {
16  
17      @Test
18      void testChooseCapacityKeepsPairInSameUnits() {
19          // Apple Silicon: the raw pair is mAh, the unprefixed pair a percentage
20          assertCapacity(MacPowerSource.chooseCapacity(5736, 7514, 80, 100), CapacityUnits.MAH, 5736, 7514);
21          // Either raw key missing: never mix a raw mAh value with a percentage
22          assertCapacity(MacPowerSource.chooseCapacity(null, 7514, 80, 100), CapacityUnits.RELATIVE, 80, 100);
23          assertCapacity(MacPowerSource.chooseCapacity(5736, null, 80, 100), CapacityUnits.RELATIVE, 80, 100);
24          // Intel Mac without the raw keys: the unprefixed pair is mAh
25          assertCapacity(MacPowerSource.chooseCapacity(null, null, 4100, 5103), CapacityUnits.MAH, 4100, 5103);
26          // Nothing usable
27          assertCapacity(MacPowerSource.chooseCapacity(null, 7514, 80, null), CapacityUnits.RELATIVE, 0, 1);
28          // A maximum of zero is not a usable reading in either pair
29          assertCapacity(MacPowerSource.chooseCapacity(5736, 0, 80, 100), CapacityUnits.RELATIVE, 80, 100);
30          assertCapacity(MacPowerSource.chooseCapacity(null, null, 0, 0), CapacityUnits.RELATIVE, 0, 1);
31      }
32  
33      private static void assertCapacity(Triplet<CapacityUnits, Integer, Integer> capacity, CapacityUnits units,
34              int current, int max) {
35          assertThat(capacity.getA(), is(units));
36          assertThat(capacity.getB(), is(current));
37          assertThat(capacity.getC(), is(max));
38      }
39  
40      @Test
41      void testMinutesToSeconds() {
42          assertThat(MacPowerSource.minutesToSeconds(90), is(5400d));
43          assertThat(MacPowerSource.minutesToSeconds(0), is(0d));
44          assertThat(MacPowerSource.minutesToSeconds(65535), is(-1d));
45          assertThat(MacPowerSource.minutesToSeconds(-1), is(-1d));
46          assertThat(MacPowerSource.minutesToSeconds(null), is(-1d));
47      }
48  }
