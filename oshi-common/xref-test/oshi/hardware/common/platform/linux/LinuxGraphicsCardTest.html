81      @Test
82      void testVramTotalFallsBackToParsedValue(@TempDir Path tempDir) throws IOException {
83          Path device = tempDir.resolve("device");
84          Files.createDirectories(device);
85  
86          LinuxGraphicsCard.Attrs amdNoFile = new LinuxGraphicsCard.Attrs("RX 7900", "0x744c", "AMD", "Rev: c1", 20480L,
87                  device.toString(), "amdgpu", "0000:03:00.0");
88          assertThat(LinuxGraphicsCard.vramTotal(amdNoFile), is(20480L));
89  
90          // Only amdgpu publishes this file, so another driver must not be read out of the same path
91          writeFile(device.resolve("mem_info_vram_total"), "25753026560\n");
92          LinuxGraphicsCard.Attrs intel = new LinuxGraphicsCard.Attrs("Arc A770", "0x56a0", "Intel", "Rev: 08", 20480L,
93                  device.toString(), "i915", "0000:03:00.0");
94          assertThat(LinuxGraphicsCard.vramTotal(intel), is(20480L));
95  
96          LinuxGraphicsCard.Attrs noPath = new LinuxGraphicsCard.Attrs("RX 7900", "0x744c", "AMD", "Rev: c1", 20480L, "",
97                  "amdgpu", "0000:03:00.0");
98          assertThat(LinuxGraphicsCard.vramTotal(noPath), is(20480L));
99      }

202     /**
203      * Creates a card directory with a driver symlink and uevent file.
204      *
205      * @param drmDir     the base DRM directory
206      * @param cardName   e.g. "card0"
207      * @param driverName e.g. "amdgpu"
208      * @param slotName   e.g. "0000:01:00.0"
209      * @throws IOException if file creation fails
210      */
211     private static void createCardWithDriver(Path drmDir, String cardName, String driverName, String slotName)
212             throws IOException {
213         Path deviceDir = drmDir.resolve(cardName + "/device");
214         Files.createDirectories(deviceDir);
215         // Create a fake driver target and symlink
216         Path driverTarget = drmDir.resolve("drivers/" + driverName);
217         Files.createDirectories(driverTarget);
218         Files.createSymbolicLink(deviceDir.resolve("driver"), driverTarget);
219         // Write uevent with PCI_SLOT_NAME
220         Files.writeString(deviceDir.resolve("uevent"), ("PCI_SLOT_NAME=" + slotName + "\n"));
221     }

308     @Test
309     void testGetGraphicsCardsFromLspciPassesEachSlotToLookups() {
310         List<String> twoCards = """
311                 Slot:\t00:02.0
312                 Class:\tVGA compatible controller [0300]
313                 Vendor:\tIntel Corporation [8086]
314                 Device:\tUHD Graphics 630 [3E92]
315                 Rev:\t00
316 
317                 Slot:\t01:00.0
318                 Class:\tVGA compatible controller [0300]
319                 Vendor:\tNVIDIA Corporation [10de]
320                 Device:\tGA102 [GeForce RTX 3090] [2204]
321                 Rev:\ta1
322 
323                 """.lines().toList();
324 
325         List<String> drmSlots = new ArrayList<>();
326         List<String> vramSlots = new ArrayList<>();
327         List<GraphicsCard> cards = LinuxGraphicsCard.getGraphicsCardsFromLspci(twoCards, STUB_FACTORY, slot -> {
328             vramSlots.add(slot);
329             return 1024L;
330         }, slot -> {
331             drmSlots.add(slot);
332             return new Triplet<>("/sys/class/drm/card0/device", "driver", slot);
333         });
334 
335         // Each card must be correlated using its own PCI slot, not null and not a shared one
336         assertThat(drmSlots, is(List.of("00:02.0", "01:00.0")));
337         assertThat(vramSlots, is(List.of("00:02.0", "01:00.0")));
338         assertThat(((LinuxGraphicsCard) cards.get(0)).getPciBusId(), is("00:02.0"));
339         assertThat(((LinuxGraphicsCard) cards.get(1)).getPciBusId(), is("01:00.0"));
340         assertThat(cards.get(0).getVRam(), is(1024L));
341     }

358                 Rev:\t31

377     @Test
378     void testIsDisplayClass() {
379         assertThat(LinuxGraphicsCard.isDisplayClass("VGA compatible controller [0300]"), is(true));
380         assertThat(LinuxGraphicsCard.isDisplayClass("XGA compatible controller [0301]"), is(true));
381         assertThat(LinuxGraphicsCard.isDisplayClass("3D controller [0302]"), is(true));
382         assertThat(LinuxGraphicsCard.isDisplayClass("Display controller [0380]"), is(true));
383         // Contains "VGA" but is PCI class 0x0000, not a graphics card
384         assertThat(LinuxGraphicsCard.isDisplayClass("Non-VGA unclassified device [0000]"), is(false));
385         assertThat(LinuxGraphicsCard.isDisplayClass("Signal processing controller [1180]"), is(false));
386         assertThat(LinuxGraphicsCard.isDisplayClass("Host bridge [0600]"), is(false));
387         // Without numeric class codes, fall back to the class name
388         assertThat(LinuxGraphicsCard.isDisplayClass("VGA compatible controller"), is(true));
389         assertThat(LinuxGraphicsCard.isDisplayClass("3D controller"), is(true));
390         assertThat(LinuxGraphicsCard.isDisplayClass("Non-VGA unclassified device"), is(false));
391         // A bracket group too short to hold a base class falls through to the name check
392         assertThat(LinuxGraphicsCard.isDisplayClass("Bogus controller [3]"), is(false));
393     }

524     @Test
525     void testGetGraphicsCardsFromLshwTwoCards() {
526         List<GraphicsCard> cards = LinuxGraphicsCard.getGraphicsCardsFromLshw(LSHW, STUB_FACTORY, NO_DRM);
527         assertThat(cards.size(), is(2));
528 
529         GraphicsCard nvidia = cards.get(0);
530         assertThat(nvidia.getName(), is("GA102 [GeForce RTX 3090]"));
531         assertThat(nvidia.getVendor(), is("NVIDIA Corporation"));
532         assertThat(nvidia.getVersionInfo(), is("version: a1"));
533         // 16 MiB (f6000000-f6ffffff) + 256 MiB (e0000000-efffffff)
534         assertThat(nvidia.getVRam(), is(16L * 1024 * 1024 + 256L * 1024 * 1024));

536         GraphicsCard intel = cards.get(1);
537         assertThat(intel.getName(), is("UHD Graphics 630"));
538         assertThat(intel.getVendor(), is("Intel Corporation"));
539         // No version line -> UNKNOWN version info; single 16 MiB memory range
540         assertThat(intel.getVersionInfo(), is(Constants.UNKNOWN));
541         assertThat(intel.getVRam(), is(16L * 1024 * 1024));
