100         assertThat(LinuxGraphicsCard.vramTotal(noPath), is(20480L));

204     /**
205      * Creates a card directory with a driver symlink and uevent file.
206      *
207      * @param drmDir     the base DRM directory
208      * @param cardName   e.g. "card0"
209      * @param driverName e.g. "amdgpu"
210      * @param slotName   e.g. "0000:01:00.0"
211      * @throws IOException if file creation fails
212      */
213     private static void createCardWithDriver(Path drmDir, String cardName, String driverName, String slotName)
214             throws IOException {
215         Path deviceDir = drmDir.resolve(cardName + "/device");
216         Files.createDirectories(deviceDir);
217         // Create a fake driver target and symlink
218         Path driverTarget = drmDir.resolve("drivers/" + driverName);
219         Files.createDirectories(driverTarget);
220         Files.createSymbolicLink(deviceDir.resolve("driver"), driverTarget);
221         // Write uevent with PCI_SLOT_NAME
222         Files.writeString(deviceDir.resolve("uevent"), ("PCI_SLOT_NAME=" + slotName + "\n"));
223     }

229     @Test
230     // A PCI function's udev record is named +pci:<slot>, and Windows does not allow a colon in a file name
231     @DisabledOnOs(OS.WINDOWS)
232     void testGraphicsCardsFromDrmPci(@TempDir Path tempDir) throws IOException {
233         Path drm = tempDir.resolve("drm");
234         createCardWithDriver(drm, "card1", "amdgpu", "0000:03:00.0");
235         Path device = drm.resolve("card1/device");
236         writeFile(device.resolve("vendor"), "0x1002");
237         writeFile(device.resolve("device"), "0x744c");
238         writeFile(device.resolve("revision"), "0xc8");
239         // A connector is not a card, and the firmware's framebuffer is not a GPU
240         Files.createDirectories(drm.resolve("card1-DP-1"));
241         createCardWithDriver(drm, "card0", "simple-framebuffer", "");
242         Path udev = Files.createDirectories(tempDir.resolve("udev"));
243         Files.writeString(udev.resolve("+pci:0000:03:00.0"), """
244                 E:ID_PCI_CLASS_FROM_DATABASE=Display controller
245                 E:ID_VENDOR_FROM_DATABASE=Advanced Micro Devices, Inc. [AMD/ATI]
246                 E:ID_MODEL_FROM_DATABASE=Navi 31 [Radeon RX 7900 XT/7900 XTX/7900 GRE/7900M]
247                 """);
248 
249         List<GraphicsCard> cards = LinuxGraphicsCard.getGraphicsCardsFromDrm(STUB_FACTORY, drm + "/", udev + "/");
250         assertThat(cards.size(), is(1));
251         GraphicsCard card = cards.get(0);
252         assertThat(card.getName(), is("Navi 31 [Radeon RX 7900 XT/7900 XTX/7900 GRE/7900M]"));
253         assertThat(card.getVendor(), is("Advanced Micro Devices, Inc. [AMD/ATI] (0x1002)"));
254         assertThat(card.getDeviceId(), is("0x744c"));
255         assertThat(card.getVersionInfo(), is("Rev:\tc8"));
256 
257         // Without a udev database the identifiers still come through
258         card = LinuxGraphicsCard.getGraphicsCardsFromDrm(STUB_FACTORY, drm + "/", tempDir.resolve("none") + "/").get(0);
259         assertThat(card.getName(), is(Constants.UNKNOWN));
260         assertThat(card.getVendor(), is("0x1002"));
261     }

367     @Test
368     void testGetGraphicsCardsFromLspciPassesEachSlotToLookups() {
369         List<String> twoCards = """
370                 Slot:\t00:02.0
371                 Class:\tVGA compatible controller [0300]
372                 Vendor:\tIntel Corporation [8086]
373                 Device:\tUHD Graphics 630 [3E92]
374                 Rev:\t00
375 
376                 Slot:\t01:00.0
377                 Class:\tVGA compatible controller [0300]
378                 Vendor:\tNVIDIA Corporation [10de]
379                 Device:\tGA102 [GeForce RTX 3090] [2204]
380                 Rev:\ta1
381 
382                 """.lines().toList();
383 
384         List<String> drmSlots = new ArrayList<>();
385         List<String> vramSlots = new ArrayList<>();
386         List<GraphicsCard> cards = LinuxGraphicsCard.getGraphicsCardsFromLspci(twoCards, STUB_FACTORY, slot -> {
387             vramSlots.add(slot);
388             return 1024L;
389         }, slot -> {
390             drmSlots.add(slot);
391             return new Triplet<>("/sys/class/drm/card0/device", "driver", slot);
392         });
393 
394         // Each card must be correlated using its own PCI slot, not null and not a shared one
395         assertThat(drmSlots, is(List.of("00:02.0", "01:00.0")));
396         assertThat(vramSlots, is(List.of("00:02.0", "01:00.0")));
397         assertThat(((LinuxGraphicsCard) cards.get(0)).getPciBusId(), is("00:02.0"));
398         assertThat(((LinuxGraphicsCard) cards.get(1)).getPciBusId(), is("01:00.0"));
399         assertThat(cards.get(0).getVRam(), is(1024L));
400     }

436     @Test
437     void testIsDisplayClass() {
438         assertThat(LinuxGraphicsCard.isDisplayClass("VGA compatible controller [0300]"), is(true));
439         assertThat(LinuxGraphicsCard.isDisplayClass("XGA compatible controller [0301]"), is(true));
440         assertThat(LinuxGraphicsCard.isDisplayClass("3D controller [0302]"), is(true));
441         assertThat(LinuxGraphicsCard.isDisplayClass("Display controller [0380]"), is(true));
442         // Contains "VGA" but is PCI class 0x0000, not a graphics card
443         assertThat(LinuxGraphicsCard.isDisplayClass("Non-VGA unclassified device [0000]"), is(false));
444         assertThat(LinuxGraphicsCard.isDisplayClass("Signal processing controller [1180]"), is(false));
445         assertThat(LinuxGraphicsCard.isDisplayClass("Host bridge [0600]"), is(false));
446         // Without numeric class codes, fall back to the class name
447         assertThat(LinuxGraphicsCard.isDisplayClass("VGA compatible controller"), is(true));
448         assertThat(LinuxGraphicsCard.isDisplayClass("3D controller"), is(true));
449         assertThat(LinuxGraphicsCard.isDisplayClass("Non-VGA unclassified device"), is(false));
450         // A bracket group too short to hold a base class falls through to the name check
451         assertThat(LinuxGraphicsCard.isDisplayClass("Bogus controller [3]"), is(false));
452     }

579                    physical id: 2

583     @Test
584     void testGetGraphicsCardsFromLshwTwoCards() {
585         List<GraphicsCard> cards = LinuxGraphicsCard.getGraphicsCardsFromLshw(LSHW, STUB_FACTORY, NO_DRM);
586         assertThat(cards.size(), is(2));
587 
588         GraphicsCard nvidia = cards.get(0);
589         assertThat(nvidia.getName(), is("GA102 [GeForce RTX 3090]"));
590         assertThat(nvidia.getVendor(), is("NVIDIA Corporation"));

597         assertThat(intel.getVendor(), is("Intel Corporation"));

599         assertThat(intel.getVersionInfo(), is(Constants.UNKNOWN));

601     }
