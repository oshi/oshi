81      @Test
82      void testCreateMIDRBasic() {
83          // vendor=0x41 (ARM), stepping=r1p3, model=0xd07, family=8
84          String midr = LinuxCentralProcessor.createMIDR("0x41", "r1p3", "0xd07", "8");
85          // Implementer 0x41 << 24 = 0x41000000
86          // Variant 1 << 20 = 0x00100000
87          // Architecture 8 << 16 = 0x00080000
88          // PartNum 0xd07 << 4 = 0x0000D070
89          // Revision 3 = 0x00000003
90          // Total = 0x41180073... let me compute:
91          // 0x41 << 24 = 0x41000000
92          // 1 << 20 = 0x00100000
93          // 8 << 16 = 0x00080000
94          // 0xd07 << 4 = 0x0000D070
95          // 3 = 0x00000003
96          // Sum = 0x4118D073
97          assertThat(midr, is("4118D073"));
98      }

726     @Test
727     void testApplyHybridCoreTypes() {
728         // Core i7-12700: 8 two-thread P-cores (core_id 0-28 step 4, CPUs 0-15) and 4 E-cores (core_id 32-35)
729         List<LogicalProcessor> logProcs = new ArrayList<>();
730         Map<Integer, Integer> efficiency = new HashMap<>();
731         for (int cpu = 0; cpu < 20; cpu++) {
732             int core = cpu < 16 ? (cpu / 2) * 4 : 32 + cpu - 16;
733             logProcs.add(new LogicalProcessor(cpu, core, 0));
734             efficiency.put(core, 0);
735         }
736         LinuxCentralProcessor.applyHybridCoreTypes(efficiency, logProcs, "0-15");
737         assertThat(efficiency.get(0), is(1));
738         assertThat(efficiency.get(28), is(1));
739         assertThat(efficiency.get(32), is(0));
740         assertThat(efficiency.values().stream().filter(e -> e == 1).count(), is(8L));
741 
742         // No hybrid PMU, or capacities the kernel already published, leave the values alone
743         Map<Integer, Integer> plain = new HashMap<>(Map.of(0, 0, 4, 0));
744         LinuxCentralProcessor.applyHybridCoreTypes(plain, logProcs, "");
745         assertThat(plain, is(Map.of(0, 0, 4, 0)));
746         Map<Integer, Integer> arm = new HashMap<>(Map.of(0, 446, 4, 1024));
747         LinuxCentralProcessor.applyHybridCoreTypes(arm, logProcs, "0-3");
748         assertThat(arm, is(Map.of(0, 446, 4, 1024)));
749     }
