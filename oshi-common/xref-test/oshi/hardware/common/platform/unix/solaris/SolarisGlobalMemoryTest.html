21      @Test
22      void testParseSmbiosMemoryDevicesQemu() {
23          // Captured unprivileged on a vmactions OpenIndiana VM
24          List<String> smbios = """
25                  ID    SIZE TYPE
26                  4352  51   SMB_TYPE_MEMDEVICE (type 17) (memory device)
27  
28                    Manufacturer: QEMU
29                    Location Tag: DIMM 0
30  
31                    Physical Memory Array: 4096
32                    Memory Error Data: Not Supported
33                    Total Width: Unknown
34                    Data Width: Unknown
35                    Size: 6442450944 bytes
36                    Form Factor: 9 (DIMM)
37                    Set: None
38                    Rank: Unknown
39                    Memory Type: 7 (RAM)
40                    Flags: 0x2
41                  \tSMB_MDF_OTHER (other)
42                    Speed: Unknown
43                    Configured Speed: Unknown
44                    Device Locator: DIMM 0
45                    Bank Locator:\s
46                    Minimum Voltage: Unknown
47                    Maximum Voltage: Unknown
48                    Configured Voltage: Unknown
49                  """.lines().toList();
50          List<PhysicalMemory> modules = SolarisGlobalMemory.parseSmbiosMemoryDevices(smbios);
51          assertThat(modules, hasSize(1));
52          PhysicalMemory dimm = modules.get(0);
53          assertThat(dimm.getBankLabel(), is("DIMM 0"));
54          assertThat(dimm.getCapacity(), is(6_442_450_944L));
55          assertThat(dimm.getManufacturer(), is("QEMU"));
56          assertThat(dimm.getMemoryType(), is("RAM"));
57          assertThat(dimm.getClockSpeed(), is(-1L));
58          assertThat(dimm.getPartNumber(), is("unknown"));
59          assertThat(dimm.getSerialNumber(), is("unknown"));
60      }

62      @Test
63      void testParseSmbiosMemoryDevicesPopulatedAndEmptySlot() {
64          // Constructed in the same layout, with the fields real modules carry and an empty slot
65          List<String> smbios = """
66                  ID    SIZE TYPE
67                  40    92   SMB_TYPE_MEMDEVICE (type 17) (memory device)
68  
69                    Manufacturer: Samsung
70                    Serial Number: 4A1B2C3D
71                    Part Number: M393A2K43DB3-CWE
72                    Size: 17179869184 bytes
73                    Memory Type: 26 (DDR4)
74                    Speed: 3200 MT/s
75                    Device Locator: P1-DIMMA1
76                    Bank Locator: P0_Node0_Channel0_Dimm0
77  
78                  ID    SIZE TYPE
79                  41    40   SMB_TYPE_MEMDEVICE (type 17) (memory device)
80  
81                    Manufacturer: NO DIMM
82                    Size: Not Installed
83                    Device Locator: P1-DIMMA2
84                  """.lines().toList();
85          List<PhysicalMemory> modules = SolarisGlobalMemory.parseSmbiosMemoryDevices(smbios);
86          assertThat(modules, hasSize(1));
87          PhysicalMemory dimm = modules.get(0);
88          assertThat(dimm.getBankLabel(), is("P0_Node0_Channel0_Dimm0/P1-DIMMA1"));
89          assertThat(dimm.getCapacity(), is(17_179_869_184L));
90          assertThat(dimm.getMemoryType(), is("DDR4"));
91          assertThat(dimm.getClockSpeed(), is(3_200_000_000L));
92          assertThat(dimm.getPartNumber(), is("M393A2K43DB3-CWE"));
93          assertThat(dimm.getSerialNumber(), is("4A1B2C3D"));
94      }
