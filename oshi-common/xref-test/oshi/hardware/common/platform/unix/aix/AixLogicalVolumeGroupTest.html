68      private static String lspvOther(int disk) {
69          return "hdisk" + disk + ":\n" + "LV NAME               LPs     PPs     DISTRIBUTION          MOUNT POINT\n"
70                  + "hd01                  1680    1680    397..397..396..396..94 /home\n";
71      }

73      @Test
74      void testQueryLogicalVolumeGroups() {
75          Map<String, String> outputs = new HashMap<>();
76          outputs.put("lsvg -o", "vgHome\nvgScratch\nrootvg\n");
77          outputs.put("lsvg -L -p vgHome vgScratch rootvg", LSVG_P);
78          outputs.put("lsvg -L -l vgHome vgScratch rootvg", LSVG_L);
79          outputs.put("lspv -l hdisk2", LSPV_HDISK2);
80          for (int disk = 3; disk <= 9; disk++) {
81              outputs.put("lspv -l hdisk" + disk, lspvOther(disk));
82          }
83          List<String> commands = new ArrayList<>();
84          List<LogicalVolumeGroup> groups = AixLogicalVolumeGroup.queryLogicalVolumeGroups(cmd -> {
85              commands.add(cmd);
86              return outputs.getOrDefault(cmd, "").lines().toList();
87          });
88          assertThat(groups.stream().map(LogicalVolumeGroup::getName).toList(),
89                  contains("vgHome", "vgScratch", "rootvg"));
90  
91          LogicalVolumeGroup home = groups.get(0);
92          assertThat(home.getPhysicalVolumes(),
93                  containsInAnyOrder("hdisk2", "hdisk3", "hdisk4", "hdisk5", "hdisk6", "hdisk7", "hdisk8", "hdisk9"));
94          assertThat(home.getLogicalVolumes().get("hd01"),
95                  containsInAnyOrder("hdisk2", "hdisk3", "hdisk4", "hdisk5", "hdisk6", "hdisk7", "hdisk8", "hdisk9"));
96          assertThat(home.getLogicalVolumes().get("log08"), contains("hdisk2"));
97  
98          // A volume group of one physical volume needs no lspv
99          LogicalVolumeGroup scratch = groups.get(1);
100         assertThat(scratch.getPhysicalVolumes(), contains("hdisk1"));
101         assertThat(scratch.getLogicalVolumes().get("lvscratch"), contains("hdisk1"));
102         assertThat(groups.get(2).getLogicalVolumes().keySet(), containsInAnyOrder("hd5", "hd6", "hd4", "livedump"));
103         assertThat(commands.stream().filter(c -> c.startsWith("lspv")).count(), is(8L));
104     }

106     @Test
107     void testParseSectionsKeepsEachGroupsOwnVolumes() {
108         // Constructed: two groups, one with a disk whose name begins with the heading's first word
109         List<String> lsvg = """
110                 vgA:
111                 PV_NAME           PV STATE          TOTAL PPs   FREE PPs    FREE DISTRIBUTION
112                 hdisk0            active            991         629         198..35..00..198..198
113                 PV_NAMEdata       active            100         0           00..00..00..00..00
114                 vgB:
115                 PV_NAME           PV STATE          TOTAL PPs   FREE PPs    FREE DISTRIBUTION
116                 hdisk1            active            1022        0           00..00..00..00..00
117                 """.lines().toList();
118         Map<String, Set<String>> sections = AixLogicalVolumeGroup.parseSections(lsvg, "PV_NAME");
119         assertThat(sections.get("vgA"), contains("hdisk0", "PV_NAMEdata"));
120         assertThat(sections.get("vgB"), contains("hdisk1"));
121 
122         // A volume group whose name begins with the heading's first word is still a section of its own
123         List<String> named = """
124                 rootvg:
125                 PV_NAME           PV STATE          TOTAL PPs   FREE PPs    FREE DISTRIBUTION
126                 hdisk0            active            991         629         198..35..00..198..198
127                 PV_NAMEdata:
128                 PV_NAME           PV STATE          TOTAL PPs   FREE PPs    FREE DISTRIBUTION
129                 hdisk1            active            1022        0           00..00..00..00..00
130                 """.lines().toList();
131         Map<String, Set<String>> namedSections = AixLogicalVolumeGroup.parseSections(named, "PV_NAME");
132         assertThat(namedSections.get("rootvg"), contains("hdisk0"));
133         assertThat(namedSections.get("PV_NAMEdata"), contains("hdisk1"));
134     }
