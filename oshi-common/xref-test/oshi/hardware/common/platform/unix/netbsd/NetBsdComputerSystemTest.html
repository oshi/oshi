5   package oshi.hardware.common.platform.unix.netbsd;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.aMapWithSize;
9   import static org.hamcrest.Matchers.hasEntry;
10  
11  import java.util.Map;
12  
13  import org.junit.jupiter.api.Test;
14  
15  class NetBsdComputerSystemTest {
16  
17      @Test
18      void testParseDmiArmVm() {
19          // NetBSD 10.1 evbarm, a QEMU/KVM guest: no BIOS or baseboard table
20          Map<String, String> dmi = NetBsdComputerSystem.parseDmi("""
21                  machdep.dmi.system-vendor = QEMU
22                  machdep.dmi.system-product = KVM Virtual Machine
23                  machdep.dmi.system-version = virt-9.2
24                  machdep.dmi.chassis-vendor = QEMU
25                  machdep.dmi.chassis-type = QEMU
26                  machdep.dmi.chassis-version = virt-9.2
27                  machdep.dmi.processor-vendor = QEMU
28                  machdep.dmi.processor-version = virt-9.2
29                  machdep.dmi.processor-frequency = 2000 MHz
30                  """.lines().toList());
31          assertThat(dmi, aMapWithSize(9));
32          assertThat(dmi, hasEntry("system-vendor", "QEMU"));
33          assertThat(dmi, hasEntry("system-product", "KVM Virtual Machine"));
34          assertThat(dmi, hasEntry("processor-frequency", "2000 MHz"));
35      }
36  
37      @Test
38      void testParseDmiIgnoresOtherLines() {
39          Map<String, String> dmi = NetBsdComputerSystem.parseDmi("""
40                  machdep.dmi.bios-vendor = SeaBIOS
41                  machdep.dmi.bios-date = 20140401
42                  machdep.dmi.board-product = 440FX - 82441FX PMC [Natoma]
43                  sysctl: unrelated warning
44                  kern.hostname = example
45                  """.lines().toList());
46          assertThat(dmi, aMapWithSize(3));
47          assertThat(dmi, hasEntry("bios-date", "20140401"));
48          assertThat(dmi, hasEntry("board-product", "440FX - 82441FX PMC [Natoma]"));
49      }
50  }
