21      @Test
22      void testParseVmStats() {
23          List<String> vmstat = """
24                         42983 CPU context switches
25                          8301 device interrupts
26                             0 software interrupts
27                         12345 some other stat
28                  """.lines().toList();
29          Pair<Long, Long> result = NetBsdCentralProcessor.parseVmStats(vmstat);
30          assertThat(result.getA(), is(42983L));
31          assertThat(result.getB(), is(8301L));
32      }
33  
34      @Test
35      void testParseVmStatsEmpty() {
36          Pair<Long, Long> result = NetBsdCentralProcessor.parseVmStats(Collections.emptyList());
37          assertThat(result.getA(), is(0L));
38          assertThat(result.getB(), is(0L));
39      }

41      @Test
42      void testParseFamilyModelStepping() {
43          List<String> dmesg = List.of(//
44                  "cpu0: Intel(R) Celeron(R) N4000 CPU @ 1.10GHz, 2491.67 MHz, 06-7a-01", //
45                  "cpu1: Intel(R) Celeron(R) N4000 CPU @ 1.10GHz, 2491.67 MHz, 06-7a-01");
46          String[] fms = NetBsdCentralProcessor.parseFamilyModelStepping(dmesg);
47          assertThat(fms, is(arrayContaining("06", "7a", "01")));
48      }
49  
50      @Test
51      void testParseFamilyModelSteppingAmd() {
52          List<String> dmesg = List.of(//
53                  "cpu0: AMD EPYC 7313P 16-Core Processor, 2994.74 MHz, 19-01-01");
54          String[] fms = NetBsdCentralProcessor.parseFamilyModelStepping(dmesg);
55          assertThat(fms, is(arrayContaining("19", "01", "01")));
56      }
57  
58      @Test
59      void testParseFamilyModelSteppingEmpty() {
60          String[] fms = NetBsdCentralProcessor.parseFamilyModelStepping(Collections.emptyList());
61          assertThat(fms, is(arrayContaining("", "", "")));
62      }
63  
64      @Test
65      void testParseFamilyModelSteppingNoMatch() {
66          List<String> dmesg = List.of(//
67                  "cpu0 at mainbus0 mpidr 0: ARM Cortex-A53 r0p4", //
68                  "cpu0: 32KB 64b/line 2-way L1 VIPT I-cache");
69          String[] fms = NetBsdCentralProcessor.parseFamilyModelStepping(dmesg);
70          assertThat(fms, is(arrayContaining("", "", "")));
71      }

73      @Test
74      void testParseCpTime() {
75          long[] ticks = NetBsdCentralProcessor
76                  .parseCpTime("kern.cp_time: user = 2930, nice = 42, sys = 1334, intr = 1877, idle = 46354");
77          assertThat(ticks[0], is(2930L));
78          assertThat(ticks[1], is(42L));
79          assertThat(ticks[2], is(1334L));
80          assertThat(ticks[3], is(1877L));
81          assertThat(ticks[4], is(46354L));
82      }
83  
84      @Test
85      void testParseCpTimeEmpty() {
86          long[] ticks = NetBsdCentralProcessor.parseCpTime("");
87          assertThat(ticks.length, is(5));
88          for (long tick : ticks) {
89              assertThat(tick, is(0L));
90          }
91      }

102     @Test
103     void testParseCpTimePerCpu() {
104         long[] ticks = NetBsdCentralProcessor
105                 .parseCpTime("kern.cp_time.0: user = 100, nice = 0, sys = 50, intr = 10, idle = 840");
106         assertThat(ticks[0], is(100L));
107         assertThat(ticks[1], is(0L));
108         assertThat(ticks[2], is(50L));
109         assertThat(ticks[3], is(10L));
110         assertThat(ticks[4], is(840L));
111     }
112 
113     @Test
114     void testCpTimeToTicks() {
115         long[] ticks = NetBsdCentralProcessor.cpTimeToTicks(new long[] { 2930, 42, 1334, 1877, 46354 });
116         assertThat(ticks[TickType.USER.getIndex()], is(2930L));
117         assertThat(ticks[TickType.NICE.getIndex()], is(42L));
118         assertThat(ticks[TickType.SYSTEM.getIndex()], is(1334L));
119         assertThat(ticks[TickType.IRQ.getIndex()], is(1877L));
120         assertThat(ticks[TickType.IDLE.getIndex()], is(46354L));
121         assertThat(ticks[TickType.IOWAIT.getIndex()], is(0L));
122         assertThat(ticks[TickType.SOFTIRQ.getIndex()], is(0L));
123         assertThat(ticks[TickType.STEAL.getIndex()], is(0L));
124     }
125 
126     @Test
127     void testCpTimeToTicksShort() {
128         long[] ticks = NetBsdCentralProcessor.cpTimeToTicks(new long[] { 1, 2, 3 });
129         assertThat(ticks.length, is(TickType.values().length));
130         for (long tick : ticks) {
131             assertThat(tick, is(0L));
132         }
133     }
