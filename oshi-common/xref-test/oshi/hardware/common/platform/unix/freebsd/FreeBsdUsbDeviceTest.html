50      @Test
51      void testParseUsbDevices() {
52          List<UsbDevice> controllers = FreeBsdUsbDevice.parseUsbDevices(MORELLO);
53          assertThat(controllers, hasSize(1));
54          UsbDevice xhci = controllers.get(0);
55          assertThat(xhci.getName(), is("XHCI (generic) USB 3.0 controller"));
56          assertThat(xhci.getVendorId(), is("104c"));
57          assertThat(xhci.getProductId(), is("8241"));
58          // The usbus is skipped: the root hub hangs from the controller
59          assertThat(xhci.getConnectedDevices(), hasSize(1));
60          UsbDevice hub = xhci.getConnectedDevices().get(0);
61          assertThat(hub.getName(), is("(0x104c) XHCI root HUB"));
62          // One device, though three interface drivers list it
63          assertThat(hub.getConnectedDevices(), hasSize(1));
64          UsbDevice beagle = hub.getConnectedDevices().get(0);
65          assertThat(beagle.getName(), is("BeagleBoard.org TIAM335xBeagleBoneBlack"));
66          assertThat(beagle.getVendorId(), is("1d6b"));
67          assertThat(beagle.getProductId(), is("0104"));
68          assertThat(beagle.getSerialNumber(), is("1234BBBK5678"));
69          assertThat(beagle.getUniqueDeviceId(), is("ugen0.2"));
70          assertThat(beagle.getConnectedDevices(), is(empty()));
71      }

73      @Test
74      void testParseUsbDevicesRootHubsOnly() {
75          List<UsbDevice> controllers = FreeBsdUsbDevice.parseUsbDevices(VM);
76          assertThat(controllers, hasSize(2));
77          assertThat(controllers.get(0).getName(), is("Intel 82801I (ICH9) USB controller"));
78          assertThat(controllers.get(1).getProductId(), is("2938"));
79          UsbDevice hub = controllers.get(1).getConnectedDevices().get(0);
80          assertThat(hub.getName(), is("Intel UHCI root HUB"));
81          // A root hub has no pnpinfo of its own, so carries its controller's IDs
82          assertThat(hub.getVendorId(), is("8086"));
83          assertThat(hub.getConnectedDevices(), is(empty()));
84      }

86      @Test
87      void testParseUsbDevicesSkipsChildDrivers() {
88          // devinfo -v on cfarm430: QEMU's tablet, whose HID driver carries a HID bus and a mouse driver beneath it. The
89          // mouse driver has a vendor and product of its own, but no ugen name
90          List<String> tablet = """
91                          ehci1 <Intel 82801I (ICH9) USB 2.0 controller> pnpinfo vendor=0x8086 device=0x293a subvendor=0x1af4 subdevice=0x1100 class=0x0c0320 at slot=29 function=7 dbsf=pci0:0:29:7 handle=\\_SB_.PCI0.SEF_
92                            usbus7
93                              uhub3 <Intel EHCI root HUB, class 9/0, rev 2.00/1.00, addr 1>
94                                usbhid0 <QEMU QEMU USB Tablet, class 0/0, rev 2.00/0.00, addr 2> pnpinfo vendor=0x0627 product=0x0001 devclass=0x00 devsubclass=0x00 devproto=0x00 sernum="28754-0000:00:1d.7-1" release=0x0000 mode=host intclass=0x03 intsubclass=0x00 intprotocol=0x00 at bus=7 hubaddr=1 port=1 devaddr=2 interface=0 ugen=ugen7.2
95                                  hidbus0 <HID bus>

98                  .lines().toList();
99          UsbDevice hub = FreeBsdUsbDevice.parseUsbDevices(tablet).get(0).getConnectedDevices().get(0);
100         assertThat(hub.getConnectedDevices(), hasSize(1));

102         assertThat(usbTablet.getName(), is("QEMU QEMU USB Tablet"));
103         assertThat(usbTablet.getVendorId(), is("0627"));
104         assertThat(usbTablet.getSerialNumber(), is("28754-0000:00:1d.7-1"));
105         assertThat(usbTablet.getConnectedDevices(), is(empty()));
106     }

108     @Test
109     void testParseUsbDevicesWithoutDescriptions() {
110         // devinfo -v on DragonFly 6.4.2 in CI, trimmed: no descriptions, so nothing to name the devices by but their
111         // driver instances
112         List<String> dragonfly = """
113                         uhci0 pnpinfo vendor=0x8086 device=0x7020 subvendor=0x1af4 subdevice=0x1100 class=0x0c0300 at slot=1 function=2 handle=\\_SB_.PCI0.S0A_
114                           usbus0
115                             uhub0
116                         unknown pnpinfo vendor=0x8086 device=0x7113 subvendor=0x1af4 subdevice=0x1100 class=0x068000 at slot=1 function=3 handle=\\_SB_.PCI0.S0B_
117                 """
118                 .lines().toList();
119         List<UsbDevice> controllers = FreeBsdUsbDevice.parseUsbDevices(dragonfly);
120         assertThat(controllers, hasSize(1));
121         assertThat(controllers.get(0).getName(), is("uhci0"));
122         assertThat(controllers.get(0).getVendorId(), is("8086"));
123         assertThat(controllers.get(0).getConnectedDevices().get(0).getName(), is("uhub0"));
124     }
