28                  class      = bridge
29                  subclass   = HOST-PCI
30              vgapci0@pci0:0:1:0:\tclass=0x030000 rev=0x02 hdr=0x00 vendor=0x1234 device=0x1111 subvendor=0x1af4 subdevice=0x1100
31                  class      = display
32                  subclass   = VGA
33              uhci0@pci0:0:26:0:\tclass=0x0c0300 rev=0x03 hdr=0x00 vendor=0x8086 device=0x2937 subvendor=0x1af4 subdevice=0x1100

45      // pciconf -lv on DragonFly BSD 6.4, trimmed, which gives the IDs as chip=0xDDDDVVVV
46      private static final String PCICONF_DRAGONFLY = """
47              none0@pci0:0:1:3:\tclass=0x068000 card=0x11001af4 chip=0x71138086 rev=0x03 hdr=0x00
48                  vendor     = 'Intel Corporation'

50                  class      = bridge
51              vgapci0@pci0:0:2:0:\tclass=0x030000 card=0x11001af4 chip=0x10501af4 rev=0x01 hdr=0x00
52                  vendor     = 'Red Hat, Inc.'
53                  device     = 'Virtio GPU'
54                  class      = display
55                  subclass   = VGA
56              virtio_pci0@pci0:0:3:0:\tclass=0x020000 card=0x00011af4 chip=0x10001af4 rev=0x00 hdr=0x00

63      @Test
64      void testParsePciconfFreeBsd() {
65          List<GraphicsCard> cards = FreeBsdGraphicsCard.parsePciconf(PCICONF_FREEBSD.lines().toList());
66          assertThat(cards, hasSize(2));
67  
68          // Nothing carried over from the device before
69          GraphicsCard qemu = cards.get(0);
70          assertThat(qemu.getName(), is(Constants.UNKNOWN));
71          assertThat(qemu.getVendor(), is("0x1234"));
72          assertThat(qemu.getDeviceId(), is("0x1111"));
73          assertThat(qemu.getVersionInfo(), is("rev=0x02"));
74  
75          GraphicsCard intel = cards.get(1);
76          assertThat(intel.getName(), is("WhiskeyLake-U GT2 [UHD Graphics 620]"));
77          assertThat(intel.getVendor(), is("Intel Corporation (0x8086)"));
78          assertThat(intel.getDeviceId(), is("0x3ea0"));
79          assertThat(intel.getVersionInfo(), is("rev=0x00"));
80          assertThat(intel.getVRam(), is(0L));
81      }
82  
83      @Test
84      void testParsePciconfDragonFly() {
85          List<GraphicsCard> cards = FreeBsdGraphicsCard.parsePciconf(PCICONF_DRAGONFLY.lines().toList());
86          assertThat(cards, hasSize(1));
87          GraphicsCard virtio = cards.get(0);
88          assertThat(virtio.getName(), is("Virtio GPU"));
89          assertThat(virtio.getVendor(), is("Red Hat, Inc. (0x1af4)"));
90          assertThat(virtio.getDeviceId(), is("0x1050"));
91          assertThat(virtio.getVersionInfo(), is("rev=0x01"));
92      }
93  
94      @Test
95      void testParsePciconfEmpty() {
96          assertThat(FreeBsdGraphicsCard.parsePciconf(Collections.emptyList()), is(empty()));
97          assertThat(FreeBsdGraphicsCard.parsePciconf(List.of("")), is(empty()));
98      }
