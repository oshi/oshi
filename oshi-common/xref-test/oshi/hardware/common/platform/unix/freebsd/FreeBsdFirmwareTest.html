7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.emptyString;
9   import static org.hamcrest.Matchers.is;
10  import static org.hamcrest.Matchers.nullValue;
11  import static org.junit.jupiter.api.Assertions.assertDoesNotThrow;
12  
13  import java.util.Collections;
14  import java.util.List;
15  
16  import org.jspecify.annotations.Nullable;
17  import org.junit.jupiter.api.Test;
18  
19  import oshi.util.tuples.Triplet;
20  
21  class FreeBsdFirmwareTest {
22  
23      @Test
24      void testParseDmiDecode() {
25          // Representative `dmidecode -t bios` (DMI type 0) block; MM/DD/YYYY date is normalized to ISO.
26          List<String> dmidecode = """
27                  Handle 0x0000, DMI type 0, 24 bytes
28                  BIOS Information
29                  \tVendor: Parallels Software International Inc.
30                  \tVersion: 11.2.1 (32626)
31                  \tRelease Date: 07/15/2016
32                  \tBIOS Revision: 11.2
33                  \tFirmware Revision: 11.2
34                  """.lines().toList();
35          Triplet<@Nullable String, @Nullable String, @Nullable String> fw = FreeBsdFirmware.parseDmiDecode(dmidecode);
36          assertThat(fw.getA(), is("Parallels Software International Inc."));
37          assertThat(fw.getB(), is("11.2.1 (32626)"));
38          assertThat(fw.getC(), is("2016-07-15"));
39      }
40  
41      @Test
42      void testParseDmiDecodeEmpty() {
43          // No output (e.g. dmidecode not available / not root): the parser returns raw absent values (null
44          // manufacturer/version, empty date); the BsdFirmware base applies the Constants.UNKNOWN fallbacks.
45          Triplet<@Nullable String, @Nullable String, @Nullable String> fw = FreeBsdFirmware
46                  .parseDmiDecode(Collections.emptyList());
47          assertThat(fw.getA(), is(nullValue()));
48          assertThat(fw.getB(), is(nullValue()));
49          assertThat(fw.getC(), is(emptyString()));
50      }
51  
52      @Test
53      void testDmiDecodeOverridesTheDmesgDefault() {
54          // BsdFirmware defaults to the dmesg banner; FreeBSD is the one BSD that reads dmidecode instead. Deleting its
55          // override would silently switch it to dmesg while the parseDmiDecode tests above kept passing.
56          assertDoesNotThrow(() -> FreeBsdFirmware.class.getDeclaredMethod("readFirmware"),
57                  "FreeBsdFirmware must declare its own readFirmware, not inherit the dmesg default");
58      }
59  }
