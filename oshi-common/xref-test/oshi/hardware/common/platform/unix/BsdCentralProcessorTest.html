25      // Representative OpenBSD/NetBSD dmesg topology lines
26      private static final List<String> DMESG_TOPOLOGY = """
27              cpu0: smt 0, core 0, package 0
28              cpu1: smt 0, core 1, package 0
29              cpu2: smt 0, core 0, package 1
30              cpu3: smt 0, core 1, package 1
31              """.lines().toList();

33      @Test
34      void testParseTopology() {
35          Pair<Map<Integer, Integer>, Map<Integer, Integer>> result = BsdCentralProcessor.parseTopology(DMESG_TOPOLOGY);
36          Map<Integer, Integer> coreMap = result.getA();
37          Map<Integer, Integer> packageMap = result.getB();
38          assertThat(coreMap.size(), is(4));
39          assertThat(coreMap.get(0), is(0));
40          assertThat(coreMap.get(1), is(1));
41          assertThat(coreMap.get(2), is(0));
42          assertThat(coreMap.get(3), is(1));
43          assertThat(packageMap.get(0), is(0));
44          assertThat(packageMap.get(2), is(1));
45      }

47      @Test
48      void testParseTopologyNetBsd() {
49          // NetBSD's order, captured from NetBSD 10.1: aarch64 (cfarm428), then amd64 with a NUMA node in front
50          List<String> dmesg = """
51                  cpu0: package 0, core 0, smt 0
52                  cpu15: package 0, core 15, smt 0
53                  cpu2: node 0, package 1, core 2, smt 0
54                  """.lines().toList();
55          Pair<Map<Integer, Integer>, Map<Integer, Integer>> result = BsdCentralProcessor.parseTopology(dmesg);
56          assertThat(result.getA(), is(Map.of(0, 0, 15, 15, 2, 2)));
57          assertThat(result.getB(), is(Map.of(0, 0, 15, 0, 2, 1)));
58      }

76      @Test
77      void testParseDmesgModelsAndCachesIntel() {
78          // Representative x86 dmesg — cache entries are split across separate lines but each line ends with "cache"
79          List<String> dmesg = """
80                  cpu0: Intel(R) Celeron(R) N4000 CPU @ 1.10GHz, 2491.67 MHz, 06-7a-01
81                  cpu0: 32KB 64b/line 8-way D-cache, 32KB 64b/line 8-way I-cache
82                  cpu0: 4MB 64b/line 16-way L2 cache
83                  cpu0: FPU,VME,DE,PSE,TSC,MSR,PAE,MCE
84                  """.lines().toList();
85          DmesgStrings result = BsdCentralProcessor.parseDmesgModelsAndCaches(dmesg);
86          assertThat(result.getCpuMap().get(0), is("Intel(R) Celeron(R) N4000 CPU @ 1.10GHz, 2491.67 MHz, 06-7a-01"));
87          assertThat(result.getCaches(), is(not(empty())));
88          // D-cache, I-cache (from first cache line), L2 cache (from second)
89          assertThat(result.getCaches().size(), is(3));
90      }

92      @Test
93      void testParseDmesgModelsAndCachesArm() {
94          // Representative ARM big.LITTLE dmesg
95          List<String> dmesg = """
96                  cpu0 at mainbus0 mpidr 0: ARM Cortex-A53 r0p4
97                  cpu0: 32KB 64b/line 2-way L1 VIPT I-cache, 32KB 64b/line 4-way L1 D-cache
98                  cpu0: 512KB 64b/line 16-way L2 cache
99                  cpu4 at mainbus0 mpidr 100: ARM Cortex-A72 r0p2
100                 cpu4: 48KB 64b/line 3-way L1 PIPT I-cache, 32KB 64b/line 2-way L1 D-cache
101                 cpu4: 1024KB 64b/line 16-way L2 cache
102                 """.lines().toList();
103         DmesgStrings result = BsdCentralProcessor.parseDmesgModelsAndCaches(dmesg);
104         assertThat(result.getCpuMap().get(0), is("ARM Cortex-A53 r0p4"));
105         assertThat(result.getCpuMap().get(4), is("ARM Cortex-A72 r0p2"));
106         // Should have L1 I-cache, L1 D-cache, L2 from each CPU type (unique by cache params)
107         assertThat(result.getCaches().size(), is(6));
108     }

110     @Test
111     void testParseDmesgModelsAndCachesNetBsdArm() {
112         // NetBSD 10.1 on a KVM Neoverse N1 (cfarm428). QEMU reports one-set caches, so the sizes round down to 0KB
113         List<String> dmesg = """
114                 cpu0 at acpi0: Arm Neoverse N1 r3p1 (v8.2-A+), id 0x0
115                 cpu0: L1 0KB/64B 1-way (1 set) PIPT Instruction cache
116                 cpu0: L2 0KB/64B 1-way (1 set) PIPT Unified cache
117                 cpu0: package 0, core 0, smt 0
118                 cpu1 at acpi0: Arm Neoverse N1 r3p1 (v8.2-A+), id 0x1
119                 cpu1: L1 64KB/64B 4-way (256 set) VIPT Data cache
120                 cpu1: L2 1024KB/64B 8-way (2048 set) PIPT Unified cache
121                 """.lines().toList();
122         DmesgStrings result = BsdCentralProcessor.parseDmesgModelsAndCaches(dmesg);
123         // The affinity register differs on every processor, so it must not reach the name, or each core is a type
124         assertThat(result.getCpuMap(),
125                 is(Map.of(0, "Arm Neoverse N1 r3p1 (v8.2-A+)", 1, "Arm Neoverse N1 r3p1 (v8.2-A+)")));
126         assertThat(result.getCaches().size(), is(4));
127         assertThat(result.getCaches().contains(new ProcessorCache(1, 4, 64, 65_536L, ProcessorCache.Type.DATA)),
128                 is(true));
129         assertThat(result.getCaches().contains(new ProcessorCache(2, 8, 64, 1_048_576L, ProcessorCache.Type.UNIFIED)),
130                 is(true));
131         assertThat(result.getCaches().contains(new ProcessorCache(1, 1, 64, 0L, ProcessorCache.Type.INSTRUCTION)),
132                 is(true));
133         assertThat(result.getFeatureFlags(), is(empty()));
134     }

169     @Test
170     void testParseCacheStrDCache() {
171         ProcessorCache cache = BsdCentralProcessor.parseCacheStr("32KB 64b/line 8-way D-cache");
172         assertThat(cache, is(not(nullValue())));
173         assertThat(cache.getType(), is(ProcessorCache.Type.DATA));
174         assertThat(cache.getLevel(), is((byte) 1));
175         assertThat(cache.getAssociativity(), is((byte) 8));
176         assertThat(cache.getLineSize(), is((short) 64));
177         assertThat(cache.getCacheSize(), is(32768));
178     }
