24      @Test
25      void testToString() {
26          AbstractGlobalMemory mem = new AbstractGlobalMemory() {
27              @Override
28              public long getTotal() {
29                  return 8L * 1024 * 1024 * 1024;
30              }
31  
32              @Override
33              public long getAvailable() {
34                  return 4L * 1024 * 1024 * 1024;
35              }
36  
37              @Override
38              public long getPageSize() {
39                  return 4096L;
40              }
41  
42              @Override
43              public VirtualMemory getVirtualMemory() {
44                  throw new UnsupportedOperationException("not exercised by this test");
45              }
46          };
47          String s = mem.toString();
48          assertThat(s, containsString("Available:"));
49          assertThat(s, containsString("4 GiB"));
50          assertThat(s, containsString("8 GiB"));
51      }

116     @Test
117     void testParsePhysicalMemoryTwoDimms() {
118         List<PhysicalMemory> result = AbstractGlobalMemory.getPhysicalMemory(DMI_TYPE_17);
119         assertThat(result, hasSize(2));
120 
121         PhysicalMemory first = result.get(0);
122         assertThat(first.getBankLabel(), is("NODE 0/DIMM_A1"));
123         assertThat(first.getCapacity(), is(16384L * 1024 * 1024));
124         assertThat(first.getClockSpeed(), is(2_666_000_000L));
125         assertThat(first.getManufacturer(), is("Samsung"));
126         assertThat(first.getMemoryType(), is("DDR4"));
127         assertThat(first.getPartNumber(), is("M393A2K43CB2-CTD"));
128         assertThat(first.getSerialNumber(), is("12345ABC"));
129 
130         PhysicalMemory second = result.get(1);
131         assertThat(second.getBankLabel(), is("NODE 1/DIMM_B1"));
132         assertThat(second.getCapacity(), is(8192L * 1024 * 1024));
133         assertThat(second.getClockSpeed(), is(2_400_000_000L));
134         assertThat(second.getManufacturer(), is("Micron"));
135         assertThat(second.getMemoryType(), is("DDR4"));
136         assertThat(second.getPartNumber(), is("MTA18ASF1G72PZ"));
137         assertThat(second.getSerialNumber(), is("67890DEF"));
138     }

165     @Test
166     void testParsePhysicalMemoryNoStateLeak() {
167         // Second DIMM omits Manufacturer and Part Number — should NOT inherit from first
168         List<String> twoEntries = """
169                 Handle 0x0038, DMI type 17, 40 bytes
170                 Memory Device
171                 \tSize: 8192 MB
172                 \tLocator: DIMM_A1
173                 \tBank Locator: Bank0
174                 \tType: DDR4
175                 \tSpeed: 2666 MT/s
176                 \tManufacturer: Samsung
177                 \tPart Number: M393A2K43CB2-CTD
178                 \tSerial Number: SN111
179                 Handle 0x0039, DMI type 17, 40 bytes
180                 Memory Device
181                 \tSize: 4096 MB
182                 \tLocator: DIMM_B1
183                 \tBank Locator: Bank1
184                 \tType: DDR4
185                 \tSpeed: 2400 MT/s
186                 \tSerial Number: SN222
187                 """.lines().toList();
188         List<PhysicalMemory> result = AbstractGlobalMemory.getPhysicalMemory(twoEntries);
189         assertThat(result, hasSize(2));
190 
191         assertThat(result.get(0).getManufacturer(), is("Samsung"));
192         assertThat(result.get(0).getPartNumber(), is("M393A2K43CB2-CTD"));
193 
194         // Second DIMM should have defaults, not Samsung/M393...
195         assertThat(result.get(1).getBankLabel(), is("Bank1/DIMM_B1"));
196         assertThat(result.get(1).getCapacity(), is(4096L * 1024 * 1024));
197         assertThat(result.get(1).getManufacturer(), is(Constants.UNKNOWN));
198         assertThat(result.get(1).getPartNumber(), is(Constants.UNKNOWN));
199     }
