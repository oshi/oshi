5   package oshi.util.driver.linux.proc;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.anEmptyMap;
9   import static org.hamcrest.Matchers.greaterThan;
10  import static org.hamcrest.Matchers.is;
11  
12  import java.nio.ByteBuffer;
13  import java.nio.ByteOrder;
14  import java.util.Map;
15  
16  import org.junit.jupiter.api.Test;
17  import org.junit.jupiter.api.condition.EnabledOnOs;
18  import org.junit.jupiter.api.condition.OS;
19  
20  class AuxvTest {
21  
22      // The leading entries of a real s390x /proc/self/auxv: HWCAP, PAGESZ, CLKTCK, then the terminating AT_NULL
23      private static final long[] VECTOR = { Auxv.AT_HWCAP, 0x5fffffL, Auxv.AT_PAGESZ, 4096L, Auxv.AT_CLKTCK, 100L,
24              Auxv.AT_NULL, 0L };
25  
26      @Test
27      @EnabledOnOs(OS.LINUX)
28      void testQueryAuxv() {
29          Map<Integer, Long> auxv = Auxv.queryAuxv();
30          assertThat("Clock Ticks should be positive", auxv.getOrDefault(Auxv.AT_CLKTCK, 0L), greaterThan(0L));
31          assertThat("Page Size should be positive", auxv.getOrDefault(Auxv.AT_PAGESZ, 0L), greaterThan(0L));
32          assertThat(Auxv.userHz(auxv), is(auxv.get(Auxv.AT_CLKTCK)));
33          assertThat(Auxv.pageSize(auxv), is(auxv.get(Auxv.AT_PAGESZ)));
34      }
35  
36      @Test
37      void testParseAuxv64BitBigEndian() {
38          assertParsesVector(true, ByteOrder.BIG_ENDIAN);
39      }
40  
41      @Test
42      void testParseAuxv64BitLittleEndian() {
43          assertParsesVector(true, ByteOrder.LITTLE_ENDIAN);
44      }
45  
46      @Test
47      void testParseAuxv32BitBigEndian() {
48          assertParsesVector(false, ByteOrder.BIG_ENDIAN);
49      }
50  
51      @Test
52      void testParseAuxv32BitLittleEndian() {
53          assertParsesVector(false, ByteOrder.LITTLE_ENDIAN);
54      }
55  
56      private static void assertParsesVector(boolean is64Bit, ByteOrder order) {
57          ByteBuffer buff = ByteBuffer.allocate(VECTOR.length * (is64Bit ? 8 : 4)).order(order);
58          for (long v : VECTOR) {
59              if (is64Bit) {
60                  buff.putLong(v);
61              } else {
62                  buff.putInt((int) v);
63              }
64          }
65          buff.flip();
66          Map<Integer, Long> auxv = Auxv.parseAuxv(buff, is64Bit);
67          assertThat(auxv, is(Map.of(Auxv.AT_HWCAP, 0x5fffffL, Auxv.AT_PAGESZ, 4096L, Auxv.AT_CLKTCK, 100L)));
68          assertThat(Auxv.userHz(auxv), is(100L));
69          assertThat(Auxv.pageSize(auxv), is(4096L));
70      }
71  
72      @Test
73      void testParseAuxvEmpty() {
74          // An unreadable /proc/self/auxv reads as no bytes, and the parse stops at once
75          assertThat(Auxv.parseAuxv(ByteBuffer.allocate(0), true), is(anEmptyMap()));
76      }
77  
78      @Test
79      void testFallbacksArePositive() {
80          // A missing or nonpositive entry falls back to getconf, and to a default if that gives nothing usable
81          Map<Integer, Long> auxv = Map.of(Auxv.AT_CLKTCK, 0L, Auxv.AT_PAGESZ, -1L);
82          assertThat(Auxv.userHz(auxv), greaterThan(0L));
83          assertThat(Auxv.pageSize(auxv), greaterThan(0L));
84          assertThat(Auxv.userHz(Map.of()), greaterThan(0L));
85          assertThat(Auxv.pageSize(Map.of()), greaterThan(0L));
86      }
87  }
