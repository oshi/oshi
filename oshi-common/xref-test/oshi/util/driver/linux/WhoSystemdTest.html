27  /**
28   * Tests the systemd session-file fallback shared by the bindings. It reads only files, so a temporary directory stands
29   * in for {@code /run/systemd/sessions} and the parse runs without Linux or a running systemd.
30   */
31  class WhoSystemdTest {
32  
33      // Writes one session file: systemd session files are KEY=value lines, named by numeric session id.
34      private static void session(Path dir, String name, String... keyValues) throws IOException {
35          Files.write(dir.resolve(name), List.of(keyValues), StandardCharsets.UTF_8);
36      }
37  
38      private static List<String> users(List<OSSession> sessions) {
39          return sessions.stream().map(OSSession::getUserName).toList();
40      }
41  
42      @Test
43      void testReadsUserTerminalAndHost(@TempDir Path dir) throws IOException {
44          // REALTIME is microseconds since the epoch, so the reported login time is it divided by 1000
45          session(dir, "12", "USER=alice", "TTY=pts/3", "REMOTE_HOST=10.0.0.5", "REALTIME=1600000000000000");
46          List<OSSession> sessions = Who.querySystemdFiles(dir.toFile());
47  
48          assertThat(sessions, hasSize(1));
49          OSSession s = sessions.get(0);
50          assertThat(s.getUserName(), is("alice"));
51          assertThat(s.getTerminalDevice(), is("pts/3"));
52          assertThat(s.getHost(), is("10.0.0.5"));
53          assertThat("REALTIME microseconds become milliseconds", s.getLoginTime(), is(1600000000000L));
54      }
55  
56      @Test
57      void testDefaultsTerminalToTheSessionIdAndHostToEmpty(@TempDir Path dir) throws IOException {
58          session(dir, "7", "USER=bob", "REALTIME=1600000000000000");
59          List<OSSession> sessions = Who.querySystemdFiles(dir.toFile());
60  
61          assertThat(sessions, hasSize(1));
62          assertThat(sessions.get(0).getTerminalDevice(), is("7"));
63          assertThat(sessions.get(0).getHost(), is(""));
64      }
65  
66      @Test
67      void testFallsBackToFileTimeWhenRealtimeIsAbsentOrUnparseable(@TempDir Path dir) throws IOException {
68          session(dir, "1", "USER=carol");
69          session(dir, "2", "USER=dave", "REALTIME=notanumber");
70          List<OSSession> sessions = Who.querySystemdFiles(dir.toFile());
71  
72          assertThat(sessions, hasSize(2));
73          for (OSSession s : sessions) {
74              assertThat("A session with no usable REALTIME still gets the file's timestamp", s.getLoginTime() > 0L,
75                      is(true));
76          }
77      }
78  
79      @Test
80      void testSkipsFilesWithoutAUser(@TempDir Path dir) throws IOException {
81          session(dir, "1", "TTY=pts/0", "REALTIME=1600000000000000");
82          session(dir, "2", "USER=", "REALTIME=1600000000000000");
83          session(dir, "3", "USER=erin", "REALTIME=1600000000000000");
84          assertThat("Only the session naming a user counts", users(Who.querySystemdFiles(dir.toFile())),
85                  contains("erin"));
86      }
87  
88      @Test
89      void testIgnoresNonNumericFileNames(@TempDir Path dir) throws IOException {
90          // systemd names session files by numeric id; the directory also holds other bookkeeping entries
91          session(dir, "sessions.state", "USER=root", "REALTIME=1600000000000000");
92          session(dir, "c1", "USER=root", "REALTIME=1600000000000000");
93          session(dir, "42", "USER=frank", "REALTIME=1600000000000000");
94          assertThat(users(Who.querySystemdFiles(dir.toFile())), contains("frank"));
95      }
96  
97      @Test
98      void testAbsentDirectoryYieldsNoSessions(@TempDir Path dir) {
99          assertThat(Who.querySystemdFiles(dir.resolve("nope").toFile()), is(empty()));
100         assertThat("An empty directory is not an error", Who.querySystemdFiles(dir.toFile()), is(empty()));
101     }
102 
103     @Test
104     void testAFileThatIsADirectoryDoesNotAbortTheScan(@TempDir Path dir) throws IOException {
105         // A numerically named subdirectory would fail to read as a properties file; the rest must still be returned.
106         Files.createDirectory(dir.resolve("5"));
107         session(dir, "6", "USER=grace", "REALTIME=1600000000000000");
108         assertThat(users(Who.querySystemdFiles(dir.toFile())), contains("grace"));
109     }
110 
111     @Test
112     void testDefaultDirectoryIsAlwaysSafeToRead() {
113         // The no-arg form reads /run/systemd/sessions, which is absent off Linux and populated on a Linux CI runner.
114         // Asserting the contents either way would be environment-dependent, so assert only that it never throws and
115         // never returns null, which is what every caller relies on.
116         ThrowingSupplier<List<OSSession>> read = Who::querySystemdFiles;
117         assertThat(assertDoesNotThrow(read), is(notNullValue()));
118     }
