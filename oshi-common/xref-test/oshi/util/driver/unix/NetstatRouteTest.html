104     @Test
105     void testParseMacOsIpv4() {
106         List<IPRoute> routes = NetstatRoute.parseRoutes(MACOS_V4, false, 3, NO_INDICES);
107         assertThat("Banner, blank and header lines should be skipped", routes, hasSize(5));
108 
109         IPRoute def = routes.get(0);
110         assertThat(def.getDestination(), is(new byte[] { 0, 0, 0, 0 }));
111         assertThat(def.getPrefixLength(), is(0));
112         assertThat(def.isGateway(), is(true));
113         assertThat(def.getGateway(), is(new byte[] { 10, 0, 0, 1 }));
114         assertThat(def.getInterfaceName(), is("en0"));
115         assertThat(def.isHost(), is(false));
116         assertThat("macOS publishes no metric column", def.getMetric(), is(-1L));
117 
118         IPRoute abbreviated = routes.get(1);
119         assertThat("10/24 should expand to 10.0.0.0", abbreviated.getDestination(), is(new byte[] { 10, 0, 0, 0 }));
120         assertThat(abbreviated.getPrefixLength(), is(24));
121         assertThat("link#14 is not a gateway", abbreviated.isGateway(), is(false));
122         assertThat(abbreviated.getGateway(), is(new byte[0]));
123         assertThat("The trailing ! should not shift the interface column", abbreviated.getInterfaceName(), is("en0"));
124 
125         assertThat(routes.get(2).getPrefixLength(), is(32));
126         assertThat(routes.get(2).isHost(), is(true));
127 
128         // The regression this rule exists for: an ARP cache row whose gateway column holds a MAC address
129         IPRoute arp = routes.get(3);
130         assertThat(arp.getDestination(), is(new byte[] { 10, 0, 0, 1 }));
131         assertThat("A host route with no prefix takes its width from the H flag", arp.getPrefixLength(), is(32));
132         assertThat(arp.isHost(), is(true));
133         assertThat(arp.isGateway(), is(false));
134         assertThat("A MAC address must never be reported as a gateway", arp.getGateway(), is(new byte[0]));
135         assertThat(routes.get(4).getGateway(), is(new byte[0]));
136     }

216

225 
226         IPRoute loopback = routes.get(1);

228         assertThat("The prefix comes from the Mask column, which -rn does not print", loopback.getPrefixLength(),
229                 is(32));
230         assertThat(loopback.isHost(), is(true));
231         assertThat(loopback.isGateway(), is(false));
232         assertThat(loopback.getGateway(), is(new byte[0]));

234         IPRoute subnet = routes.get(2);
235         assertThat(subnet.getInterfaceName(), is("net0"));
236         assertThat("255.255.255.248 is a /29", subnet.getPrefixLength(), is(29));
237         assertThat(subnet.isHost(), is(false));

283     /**
284      * DragonFly BSD prints Refs and Use columns that macOS does not, putting Netif at index 5 rather than 3, and
285      * abbreviates an octet-aligned network without stating its prefix.
286      */
287     @Test
288     void testParseDragonFlyIpv4() {
289         List<IPRoute> routes = NetstatRoute.parseRoutes(DRAGONFLY_V4, false, 3, NO_INDICES);
290         assertThat(routes, hasSize(4));
291         assertThat("The header should move the interface column from 3 to 5", routes.get(0).getInterfaceName(),
292                 is("vtnet0"));
293         assertThat(routes.get(0).getGateway(), is(new byte[] { -64, -88, 122, 2 }));
294 
295         assertThat("A UH route with a gateway column but no G flag reports no gateway", routes.get(1).getGateway(),
296                 is(new byte[0]));
297         assertThat(routes.get(1).getPrefixLength(), is(32));
298 
299         IPRoute subnet = routes.get(2);
300         assertThat(subnet.getDestination(), is(new byte[] { -64, -88, 122, 0 }));
301         assertThat("192.168.122 abbreviates a /24 and must not report an unknown prefix", subnet.getPrefixLength(),
302                 is(24));
303         assertThat(subnet.isHost(), is(false));
304         assertThat("link#1 is not a gateway", subnet.getGateway(), is(new byte[0]));
305 
306         assertThat("A MAC in the gateway column must not leak out", routes.get(3).getGateway(), is(new byte[0]));
307         assertThat(routes.get(3).isHost(), is(true));
308     }

327     /**
328      * OpenBSD is the only BSD publishing a route metric, in a Prio column, and the only platform whose interface sits
329      * in the eighth column. This layout is taken from the netstat documentation rather than captured, since no OpenBSD
330      * host was available; the header scan means a wrong default index self-corrects.
331      */
332     @Test
333     void testParseOpenBsdPrioMetric() {
334         List<String> openBsd = """
335                 Routing tables
336 
337                 Internet:
338                 Destination        Gateway            Flags   Refs      Use   Mtu  Prio Iface
339                 default            192.168.1.1        UGS        0        8     -     8 em0
340                 224/4              127.0.0.1          URS        0        0 32768     8 lo0
341                 """.lines().toList();
342         List<IPRoute> routes = NetstatRoute.parseRoutes(openBsd, false, 7, NO_INDICES);
343         assertThat(routes, hasSize(2));
344         assertThat(routes.get(0).getDestination(), is(new byte[] { 0, 0, 0, 0 }));
345         assertThat(routes.get(0).isGateway(), is(true));
346         assertThat(routes.get(0).getGateway(), is(new byte[] { -64, -88, 1, 1 }));
347         assertThat("The header should locate Iface in the eighth column", routes.get(0).getInterfaceName(), is("em0"));
348         assertThat("OpenBSD publishes a metric where the other BSDs do not", routes.get(0).getMetric(), is(8L));
349         assertThat(routes.get(1).getDestination(), is(new byte[] { -32, 0, 0, 0 }));
350         assertThat(routes.get(1).getPrefixLength(), is(4));
351         assertThat(routes.get(1).getMetric(), is(8L));
352     }
