5   package oshi.util.driver.unix;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import java.util.List;
11  
12  import org.junit.jupiter.api.Test;
13  
14  import oshi.hardware.NetworkIF.IfOperStatus;
15  
16  class IfconfigTest {
17  
18      // AIX 7.3 (cfarm gcc119): no status line, so RUNNING decides
19      private static final List<String> AIX_EN0 = """
20              en0: flags=1e084863,81cc0<UP,BROADCAST,NOTRAILERS,RUNNING,SIMPLEX,MULTICAST,GROUPRT,64BIT,CHECKSUM_OFFLOAD(ACTIVE),LARGESEND,CHAIN>
21              \tinet 10.1.0.3 netmask 0xfffffe00 broadcast 10.1.1.255
22              \t tcp_sendspace 262144 tcp_recvspace 262144 rfc1323 1
23              """
24              .lines().toList();
25  
26      private static final List<String> AIX_LO0 = """
27              lo0: flags=e08084b,c0<UP,BROADCAST,LOOPBACK,RUNNING,SIMPLEX,MULTICAST,GROUPRT,64BIT,LARGESEND,CHAIN>
28              \tinet 127.0.0.1 netmask 0xff000000 broadcast 127.255.255.255
29              \tinet6 ::1%1/64
30              \t tcp_sendspace 131072 tcp_recvspace 131072 rfc1323 1
31              """.lines().toList();
32  
33      @Test
34      void testParseOperStatusAix() {
35          assertThat(Ifconfig.parseOperStatus(AIX_EN0), is(IfOperStatus.UP));
36          assertThat(Ifconfig.parseOperStatus(AIX_LO0), is(IfOperStatus.UP));
37          // Administratively up, no link
38          assertThat(
39                  Ifconfig.parseOperStatus(List
40                          .of("en2: flags=1e084843,480<UP,BROADCAST,NOTRAILERS,SIMPLEX,MULTICAST,GROUPRT,64BIT,CHAIN>")),
41                  is(IfOperStatus.DOWN));
42          // ifconfig en2 down
43          assertThat(
44                  Ifconfig.parseOperStatus(
45                          List.of("en2: flags=1e084842,480<BROADCAST,NOTRAILERS,SIMPLEX,MULTICAST,GROUPRT,64BIT,CHAIN>")),
46                  is(IfOperStatus.DOWN));
47          // ifconfig on an interface that does not exist writes its error to stderr, leaving no output
48          assertThat(Ifconfig.parseOperStatus(List.of()), is(IfOperStatus.UNKNOWN));
49      }
50  }
