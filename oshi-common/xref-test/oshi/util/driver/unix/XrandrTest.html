31      // Fixture: xrandr --verbose output with one EDID block (128 bytes = 256 hex chars)
32      private static List<String> createXrandrWithEdid() {
33          List<String> lines = new ArrayList<>();
34          lines.add("Screen 0: minimum 8 x 8, current 1920 x 1080, maximum 32767 x 32767");
35          lines.add("HDMI-1 connected primary 1920x1080+0+0");
36          lines.add("\tEDID:");
37          // 8 lines x 32 hex chars = 256 hex chars = 128 bytes (standard EDID block)
38          lines.add("\t\t00ffffffffffff001e6d085b0b0b0b0b");
39          lines.add("\t\t0c1c0104b53c2278fb2eb5ae4f46a527");
40          lines.add("\t\t0d5054254b80714f81809500a9c0b300");
41          lines.add("\t\td1c001010101565e00a0a0a029503020");
42          lines.add("\t\t35000f282100001a000000fd00384b1e");
43          lines.add("\t\t5a1900000a202020202020000000fc00");
44          lines.add("\t\t4c4720554c545241474541520000ff00");
45          lines.add("\t\t3630344e54505a4832313337370a0100");
46          lines.add("  1920x1080 (0x48) 148.500MHz +HSync +VSync *current +preferred");
47          return lines;
48      }

50      @Test
51      void testGetEdidArraysSingleDisplay() {
52          List<byte[]> edids = Xrandr.getEdidArrays(createXrandrWithEdid());
53          assertThat(edids, hasSize(1));
54          byte[] edid = edids.get(0);
55          assertThat(edid.length, is(128));
56          // Verify EDID magic header: 00 FF FF FF FF FF FF 00
57          assertThat(edid[0], is((byte) 0x00));
58          assertThat(edid[1], is((byte) 0xFF));
59          assertThat(edid[2], is((byte) 0xFF));
60          assertThat(edid[7], is((byte) 0x00));
61      }

151     @Test
152     void testGetDisplayDataSingleWithConnectorId() {
153         Map<String, Pair<Integer, byte[]>> data = Xrandr.getDisplayData(createXrandrWithConnectorId());
154         assertThat(data.size(), is(1));
155         assertThat(data.containsKey("DP2"), is(true));
156         Pair<Integer, byte[]> pair = data.get("DP2");
157         assertNotNull(pair);
158         assertThat(pair.getA(), is(96));
159         assertThat(pair.getB().length, is(128));
160         assertThat(pair.getB()[0], is((byte) 0x00));
161         assertThat(pair.getB()[1], is((byte) 0xFF));
162     }

175     @Test
176     void testGetDisplayDataTwoDisplays() {
177         Map<String, Pair<Integer, byte[]>> data = Xrandr.getDisplayData(createXrandrTwoDisplays());
178         assertThat(data.size(), is(2));
179         assertThat(data.containsKey("DP2"), is(true));
180         assertThat(data.containsKey("HDMI1"), is(true));
181         Pair<Integer, byte[]> dp2 = data.get("DP2");
182         Pair<Integer, byte[]> hdmi1 = data.get("HDMI1");
183         assertNotNull(dp2);
184         assertNotNull(hdmi1);
185         // DP2 has CONNECTOR_ID, HDMI1 does not
186         assertThat(dp2.getA(), is(96));
187         assertThat(hdmi1.getA(), is(-1));
188         // Both have valid EDIDs
189         assertThat(dp2.getB().length, is(128));
190         assertThat(hdmi1.getB().length, is(128));
191     }

205     @Test
206     void testGetDisplayDataDisconnectedDoesNotLeakState() {
207         Map<String, Pair<Integer, byte[]>> data = Xrandr.getDisplayData(createXrandrDisconnectedWithConnectorId());
208         assertThat(data.size(), is(1));
209         assertThat(data.containsKey("eDP"), is(true));
210         Pair<Integer, byte[]> edp = data.get("eDP");
211         assertNotNull(edp);
212         // eDP's own CONNECTOR_ID is 51, not 70 or 80 from the disconnected outputs
213         assertThat(edp.getA(), is(51));
214         // Disconnected outputs should not appear
215         assertThat(data.containsKey("DP-1"), is(false));
216         assertThat(data.containsKey("HDMI-A-1"), is(false));
217     }

267     @Test
268     void testFindOutputNameByEdid() {
269         Map<String, Pair<Integer, byte[]>> data = Xrandr.getDisplayData(createXrandrTwoDisplays());
270         Pair<Integer, byte[]> hdmi1 = data.get("HDMI1");
271         assertNotNull(hdmi1);
272         // HDMI1 has no connector ID in xrandr, so only the EDID can identify it
273         assertThat(Xrandr.findOutputName(data, -1, hdmi1.getB()), is(Optional.of("HDMI1")));
