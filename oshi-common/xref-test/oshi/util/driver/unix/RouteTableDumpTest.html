22  /**
23   * Exercises the routing message walk against buffers laid out by hand.
24   * <p>
25   * The walk reads a kernel buffer, so the interesting cases are the ones a live table will not produce: a message
26   * claiming to be shorter than its own header, a sockaddr claiming more bytes than its message holds, and a buffer
27   * ending mid-message. It makes no native call, so this runs on every platform rather than only the BSDs.
28   */
29  class RouteTableDumpTest {
30  
31      private static final int RTM_GET = 4;
32      private static final int RTF_UP = 0x1;
33      private static final int RTF_GATEWAY = 0x2;
34      /** macOS RTF_WASCLONED and RTF_LLINFO, the pair Layout.MACOS names. */
35      private static final int RTF_WASCLONED = 0x20000;
36      private static final int RTF_LLINFO = 0x400;
37  
38      private static final int RTAX_DST = 1 << 0;
39      private static final int RTAX_GATEWAY = 1 << 1;
40      private static final int RTAX_NETMASK = 1 << 2;
41  
42      /** macOS: 92-byte header, addresses padded to 4. */
43      private static final int HEADER = 92;
44      private static final int PAD = 4;
45  
46      /** FreeBSD's header, for the one case that reads with its layout. */
47      private static final int FREEBSD_HEADER = 152;
48  
49      private static final Map<Integer, String> IF_NAMES = Collections.singletonMap(7, "en0");
50  
51      /**
52       * One message: a default route via a gateway, then a /24 with a truncated netmask.
53       *
54       * @param truncateBy      bytes to cut off the end of the buffer, to model a short read
55       * @param maskLenOverride a length to write into the netmask sockaddr instead of its real one, or -1
56       */
57      private static byte[] buildDump(int truncateBy, int maskLenOverride) {
58          // message 1: destination, gateway (sockaddr_in, 16 bytes each), then a netmask of no length, which is how a
59          // default route states a prefix of zero
60          int msg1 = HEADER + 16 + 16 + PAD;
61          // message 2: destination, gateway, then a 7-byte netmask padded to 8
62          int msg2 = HEADER + 16 + 16 + 8;
63          byte[] bytes = new byte[msg1 + msg2];
64          ByteBuffer bb = ByteBuffer.wrap(bytes).order(ByteOrder.nativeOrder());
65  
66          writeHeader(bb, 0, msg1, RTF_UP | RTF_GATEWAY, RTAX_DST | RTAX_GATEWAY | RTAX_NETMASK);
67          writeInet(bb, HEADER, new byte[] { 0, 0, 0, 0 });
68          writeInet(bb, HEADER + 16, new byte[] { 10, 0, 0, 1 });
69          // sa_len 0: the mask has no bytes at all, so every bit of it is zero
70  
71          int m2 = msg1;
72          writeHeader(bb, m2, msg2, RTF_UP, RTAX_DST | RTAX_GATEWAY | RTAX_NETMASK);
73          writeInet(bb, m2 + HEADER, new byte[] { (byte) 192, (byte) 168, 1, 0 });
74          writeInet(bb, m2 + HEADER + 16, new byte[] { 10, 0, 0, 1 });
75          // a netmask carries only the bytes it needs: 4 of header plus 3 of mask for a /24
76          int mask = m2 + HEADER + 32;
77          bytes[mask] = (byte) (maskLenOverride < 0 ? 7 : maskLenOverride);
78          bytes[mask + 1] = (byte) 255;
79          bytes[mask + 4] = (byte) 255;
80          bytes[mask + 5] = (byte) 255;
81          bytes[mask + 6] = (byte) 255;
82  
83          return truncateBy == 0 ? bytes : Arrays.copyOf(bytes, bytes.length - truncateBy);
84      }
85  
86      private static void writeHeader(ByteBuffer bb, int at, int msgLen, int flags, int addrs) {
87          bb.putShort(at, (short) msgLen);
88          bb.put(at + 3, (byte) RTM_GET);
89          bb.putShort(at + 4, (short) 7); // interface index
90          bb.putInt(at + 8, flags);
91          bb.putInt(at + 12, addrs);
92      }
93  
94      private static void writeInet(ByteBuffer bb, int at, byte[] address) {
95          bb.put(at, (byte) 16); // sa_len
96          bb.put(at + 1, (byte) 2); // AF_INET
97          for (int i = 0; i < 4; i++) {
98              bb.put(at + 4 + i, address[i]);
99          }
100     }

102     @Test
103     void testWellFormedDump() {
104         List<IPRoute> routes = RouteTableDump.parse(buildDump(0, -1), Layout.MACOS, IF_NAMES);
105         assertEquals(2, routes.size());
106 
107         IPRoute def = routes.get(0);
108         assertEquals(0, def.getPrefixLength());
109         assertTrue(def.isGateway());
110         assertEquals("en0", def.getInterfaceName());
111         assertEquals(7, def.getInterfaceIndex());
112         assertTrue(Arrays.equals(new byte[] { 10, 0, 0, 1 }, def.getGateway()));
113 
114         IPRoute subnet = routes.get(1);
115         // the mask carries three bytes, and the fourth is implied zero
116         assertEquals(24, subnet.getPrefixLength());
117         assertTrue(Arrays.equals(new byte[] { (byte) 192, (byte) 168, 1, 0 }, subnet.getDestination()));
118         // a route without the gateway flag reports no gateway, whatever the message carries
119         assertEquals(0, subnet.getGateway().length);
120     }

122     @Test
123     void testTrailingPartialHeaderIsRejected() {
124         // the last message loses all but a few bytes, leaving too little for a header
125         byte[] dump = buildDump(HEADER + 32, -1);
126         assertTrue(RouteTableDump.parse(dump, Layout.MACOS, IF_NAMES).isEmpty(),
127                 "a buffer ending mid-message should yield nothing, so the caller falls back to the command");
128     }
129 
130     @Test
131     void testOversizedSockaddrIsRejected() {
132         // the netmask claims far more bytes than its message holds
133         assertTrue(RouteTableDump.parse(buildDump(0, 200), Layout.MACOS, IF_NAMES).isEmpty(),
134                 "a sockaddr longer than its message should yield nothing");
135     }
136 
137     @Test
138     void testMessageShorterThanItsHeaderIsRejected() {
139         byte[] dump = buildDump(0, -1);
140         ByteBuffer.wrap(dump).order(ByteOrder.nativeOrder()).putShort(0, (short) (HEADER - 4));
141         assertTrue(RouteTableDump.parse(dump, Layout.MACOS, IF_NAMES).isEmpty(),
142                 "a message shorter than its own header should yield nothing");
143     }

145     @Test
146     void testEmptyBufferYieldsNoRoutes() {
147         assertTrue(RouteTableDump.parse(new byte[0], Layout.MACOS, IF_NAMES).isEmpty());
148     }

150     @Test
151     void testFlaggedButAbsentSockaddrIsRejected() {
152         // The header names a netmask, but the message ends after the gateway. Every message on every platform
153         // measured ends exactly on its last address, so this is truncation rather than a shorthand for "no mask"
154         int msgLen = HEADER + 16 + 16;
155         byte[] bytes = new byte[msgLen];
156         ByteBuffer bb = ByteBuffer.wrap(bytes).order(ByteOrder.nativeOrder());
157         writeHeader(bb, 0, msgLen, RTF_UP, RTAX_DST | RTAX_GATEWAY | RTAX_NETMASK);
158         writeInet(bb, HEADER, new byte[] { (byte) 192, (byte) 168, 1, 0 });
159         writeInet(bb, HEADER + 16, new byte[] { 10, 0, 0, 1 });
160 
161         assertTrue(RouteTableDump.parse(bytes, Layout.MACOS, IF_NAMES).isEmpty(),
162                 "a message naming an address it does not carry should yield nothing");
163     }

165     /**
166      * Builds one message with the given flags, carrying a destination and a gateway.
167      */
168     private static byte[] oneRoute(int flags, byte[] destination, int headerSize) {
169         int msgLen = headerSize + 16 + 16;
170         byte[] bytes = new byte[msgLen];
171         ByteBuffer bb = ByteBuffer.wrap(bytes).order(ByteOrder.nativeOrder());
172         writeHeader(bb, 0, msgLen, flags, RTAX_DST | RTAX_GATEWAY);
173         writeInet(bb, headerSize, destination);
174         writeInet(bb, headerSize + 16, new byte[] { 10, 0, 0, 1 });
175         return bytes;
176     }

178     @Test
179     void testClonedRouteIsExcluded() {
180         // A route cloned from another is a cache entry for one host, which the routing table does not list
181         byte[] dump = oneRoute(RTF_UP | RTF_GATEWAY | RTF_WASCLONED, new byte[] { 93, (byte) 184, (byte) 216, 1 },
182                 HEADER);
183         assertTrue(RouteTableDump.parse(dump, Layout.MACOS, IF_NAMES).isEmpty(), "a cloned route should be dropped");
184     }
185 
186     @Test
187     void testClonedLinkLayerRouteIsKept() {
188         // Except a link-layer one: that is how the neighbour cache appears, and the table does show those
189         byte[] dump = oneRoute(RTF_UP | RTF_WASCLONED | RTF_LLINFO, new byte[] { 10, 0, 0, 42 }, HEADER);
190         List<IPRoute> routes = RouteTableDump.parse(dump, Layout.MACOS, IF_NAMES);
191         assertEquals(1, routes.size(), "a cloned link-layer route is the neighbour cache and should be kept");
192         assertTrue(Arrays.equals(new byte[] { 10, 0, 0, 42 }, routes.get(0).getDestination()));
193     }
194 
195     @Test
196     void testClonedRouteIsKeptWhereThePlatformDoesNotClone() {
197         // FreeBSD stopped cloning routes and names no flag for it, so the same bits mean nothing there. Its header
198         // is longer than the macOS one the other cases use
199         byte[] dump = oneRoute(RTF_UP | RTF_GATEWAY | RTF_WASCLONED, new byte[] { 93, (byte) 184, (byte) 216, 1 },
200                 FREEBSD_HEADER);
201         assertEquals(1, RouteTableDump.parse(dump, Layout.FREEBSD, IF_NAMES).size(),
202                 "a platform that does not clone should not have routes filtered out from under it");
203     }
