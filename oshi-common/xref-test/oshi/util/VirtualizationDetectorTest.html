37      @Test
38      void testShippedPropertiesLoadWithUnescapedKeys() {
39          // A missing backslash before a space in a key truncates that key silently, so assert the unescaped forms
40          Properties props = shippedVmProps();
41          assertThat("oshi.vm.properties should be on the classpath", props.stringPropertyNames().size(),
42                  is(greaterThan(0)));
43          assertThat(props.getProperty("cpuid.bhyve bhyve"), is("bhyve"));
44          assertThat(props.getProperty("cpuid.Microsoft Hv"), is("Microsoft Hyper-V"));
45          assertThat(props.getProperty("cpuid.lrpepyh vr"), is("Parallels"));
46          assertThat(props.getProperty("model.Linux KVM"), is("KVM"));
47          assertThat(props.getProperty("model.Microsoft Corporation Virtual Machine"), is("Microsoft Hyper-V"));
48      }
49  
50      @Test
51      void testShippedPropertiesAreWellFormed() {
52          Properties props = shippedVmProps();
53          for (String key : props.stringPropertyNames()) {
54              assertThat("Unexpected prefix on key " + key, key.startsWith("cpuid.") || key.startsWith("model."),
55                      is(true));
56              assertFalse(key.contains("\\"), "Key retains a backslash, so it was over-escaped: " + key);
57              String value = props.getProperty(key);
58              assertThat("Blank value for key " + key, value, is(not(emptyString())));
59              assertThat("Untrimmed value for key " + key, value, is(value.trim()));
60          }
61      }

63      @Test
64      void testShippedMacPropertiesAreWellFormed() {
65          Properties props = shippedMacProps();
66          assertThat("oshi.vmmacaddr.properties should be on the classpath", props.stringPropertyNames().size(),
67                  is(greaterThan(0)));
68          for (String key : props.stringPropertyNames()) {
69              assertThat("OUI key is not uppercase colon-delimited hex: " + key, key,
70                      is(VirtualizationDetector.extractOui(key)));
71          }
72      }

74      @Test
75      void testMatchCpuid() {
76          Properties props = shippedVmProps();
77          assertThat(VirtualizationDetector.matchCpuid("VMwareVMware", props), is(Optional.of("VMware")));
78          // The vendor string is trimmed before lookup
79          assertThat(VirtualizationDetector.matchCpuid("  KVMKVMKVM  ", props), is(Optional.of("KVM")));
80          assertThat(VirtualizationDetector.matchCpuid("bhyve bhyve", props), is(Optional.of("bhyve")));
81          assertThat(VirtualizationDetector.matchCpuid("GenuineIntel", props), is(Optional.empty()));
82          assertThat(VirtualizationDetector.matchCpuid("", props), is(Optional.empty()));
83      }

85      @Test
86      void testMatchSystemAgainstRealDmiStrings() {
87          List<Pair<String, String>> table = VirtualizationDetector.buildModelTable(shippedVmProps());
88          // Manufacturer is part of the haystack, so a platform that names itself only there still matches
89          assertThat(VirtualizationDetector.matchSystem("QEMU", "Standard PC (i440FX + PIIX, 1996)", table),
90                  is(Optional.of("QEMU")));
91          assertThat(VirtualizationDetector.matchSystem("innotek GmbH", "VirtualBox", table),
92                  is(Optional.of("VirtualBox")));
93          assertThat(VirtualizationDetector.matchSystem("VMware, Inc.", "VMware Virtual Platform", table),
94                  is(Optional.of("VMware")));
95          assertThat(VirtualizationDetector.matchSystem("Microsoft Corporation", "Virtual Machine", table),
96                  is(Optional.of("Microsoft Hyper-V")));
97          assertThat(VirtualizationDetector.matchSystem("Xen", "HVM domU", table), is(Optional.of("Xen")));
98          // Apple Virtualization guest, as reported by the Apple Silicon GitHub Actions runners
99          assertThat(VirtualizationDetector.matchSystem("Apple Inc.", "VirtualMac2,1", table),
100                 is(Optional.of("Apple Virtualization")));
101         // The Hyper-V model carries a UEFI version suffix on Azure Linux, so this cannot be an equality test
102         assertThat(
103                 VirtualizationDetector.matchSystem("Microsoft Corporation",
104                         "Virtual Machine (version: Hyper-V UEFI Release v4.1)", table),
105                 is(Optional.of("Microsoft Hyper-V")));
106         assertThat(VirtualizationDetector.matchSystem("Parallels Software International Inc.",
107                 "Parallels Virtual" + " Platform", table), is(Optional.of("Parallels")));
108         // Physical hardware matches nothing
109         assertThat(VirtualizationDetector.matchSystem("Dell Inc.", "PowerEdge R740", table), is(Optional.empty()));
110         assertThat(VirtualizationDetector.matchSystem("Apple Inc.", "MacBookPro18,3", table), is(Optional.empty()));
111         assertThat(VirtualizationDetector.matchSystem("Apple Inc.", "Macmini6,2", table), is(Optional.empty()));
112         assertThat(VirtualizationDetector.matchSystem("", "", table), is(Optional.empty()));
113     }

115     @Test
116     void testMatchSystemPrefersTheLongerSignature() {
117         Properties props = new Properties();
118         props.setProperty("model.Xen", "Xen");
119         props.setProperty("model.Xen HVM domU", "Xen HVM");
120         assertThat(VirtualizationDetector.matchSystem("Xen", "HVM domU", VirtualizationDetector.buildModelTable(props)),
121                 is(Optional.of("Xen HVM")));
122 
123         // Insertion order must not matter, since Properties iteration is unordered
124         Properties reversed = new Properties();
125         reversed.setProperty("model.Xen HVM domU", "Xen HVM");
126         reversed.setProperty("model.Xen", "Xen");
127         assertThat(
128                 VirtualizationDetector.matchSystem("Xen", "HVM domU", VirtualizationDetector.buildModelTable(reversed)),
129                 is(Optional.of("Xen HVM")));
130     }

132     @Test
133     void testExtractOui() {
134         assertThat(VirtualizationDetector.extractOui("08:00:27:aa:bb:cc"), is("08:00:27"));
135         assertThat(VirtualizationDetector.extractOui("08-00-27-AA-BB-CC"), is("08:00:27"));
136         assertThat(VirtualizationDetector.extractOui("080027aabbcc"), is("08:00:27"));
137         assertThat(VirtualizationDetector.extractOui("00:50:56"), is("00:50:56"));
138         // Constants.UNKNOWN and other unparseable values yield no OUI rather than throwing
139         assertThat(VirtualizationDetector.extractOui(Constants.UNKNOWN), is(emptyString()));
140         assertThat(VirtualizationDetector.extractOui(""), is(emptyString()));
141         assertThat(VirtualizationDetector.extractOui("00:50"), is(emptyString()));
142         assertThat(VirtualizationDetector.extractOui("zz:zz:zz:zz:zz:zz"), is(emptyString()));
143     }

145     @Test
146     void testMatchMac() {
147         Properties props = shippedMacProps();
148         assertThat(VirtualizationDetector.matchMac(Collections.singletonList("08:00:27:12:34:56"), props),
149                 is(Optional.of("VirtualBox")));
150         // The library formats MAC addresses in lowercase; the table is keyed uppercase
151         assertThat(VirtualizationDetector.matchMac(Collections.singletonList("00:15:5d:aa:bb:cc"), props),
152                 is(Optional.of("Microsoft Hyper-V")));
153         assertThat(VirtualizationDetector.matchMac(Collections.singletonList("42:01:0a:80:00:01"), props),
154                 is(Optional.of("Google Cloud Platform")));
155         // The first matching address wins, and unparseable entries are skipped rather than throwing
156         assertThat(
157                 VirtualizationDetector
158                         .matchMac(Arrays.asList(Constants.UNKNOWN, "3c:22:fb:11:22:33", "08:00:27:12:34:56"), props),
159                 is(Optional.of("VirtualBox")));
160         assertThat(VirtualizationDetector.matchMac(Collections.singletonList("3c:22:fb:11:22:33"), props),
161                 is(Optional.empty()));
162         assertThat(VirtualizationDetector.matchMac(Collections.<String>emptyList(), props), is(Optional.empty()));
163     }
