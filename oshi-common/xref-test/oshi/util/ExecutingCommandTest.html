56      @Test
57      void testRunNative() {
58          List<String> test = ExecutingCommand.runNative(ECHO);
59          assertThat("echo output", test, hasSize(1));
60          assertThat("echo output", test.get(0), is("Test"));
61          assertThat("echo first answer", ExecutingCommand.getFirstAnswer(ECHO), is("Test"));
62  
63          assertThat("bad command", ExecutingCommand.runNative(BAD_COMMAND), is(empty()));
64          assertThat("bad command first answer", ExecutingCommand.getFirstAnswer(BAD_COMMAND), is(emptyString()));
65      }

128     @Test
129     void testFindExecutableSearchesDirectoriesInOrder(@TempDir Path tmp) throws IOException {
130         Path first = Files.createDirectory(tmp.resolve("sbin"));
131         Path second = Files.createDirectory(tmp.resolve("usr-sbin"));
132         Path tool = Files.createFile(second.resolve("sysctl"));
133         assertThat(tool.toFile().setExecutable(true), is(true));
134         List<String> dirs = List.of(first.toString(), second.toString());
135         assertThat(ExecutingCommand.findExecutable(dirs, "sysctl"), is(tool.toString()));
136         assertThat(ExecutingCommand.findExecutable(dirs, BAD_COMMAND), is(nullValue()));
137         // A name with a path is the caller's to resolve, and an empty one names nothing
138         assertThat(ExecutingCommand.findExecutable(dirs, "../usr-sbin/sysctl"), is(nullValue()));
139         assertThat(ExecutingCommand.findExecutable(dirs, ""), is(nullValue()));
140     }

142     @Test
143     @DisabledOnOs(OS.WINDOWS)
144     void testFindExecutableSkipsFilesThatCannotRun(@TempDir Path tmp) throws IOException {
145         Files.createDirectory(tmp.resolve("route"));
146         assertThat(ExecutingCommand.findExecutable(List.of(tmp.toString()), "route"), is(nullValue()));
147         // access(X_OK) lets root execute a file with no execute bit, so only an unprivileged run can see it skipped
148         assumeFalse(UserGroupInfo.isElevated(), "root may execute any file");
149         Path tool = Files.createFile(tmp.resolve("sysctl"));
150         assertThat(tool.toFile().setExecutable(false, false), is(true));
151         assertThat(ExecutingCommand.findExecutable(List.of(tmp.toString()), "sysctl"), is(nullValue()));
152     }

154     private static String firstLine(Process p) throws IOException, InterruptedException {
155         try (BufferedReader reader = new BufferedReader(
156                 new InputStreamReader(p.getInputStream(), StandardCharsets.UTF_8))) {
157             String line = reader.readLine();
158             p.waitFor();
159             return line == null ? "" : line;
160         }
161     }
162 
163     private static boolean isRunnable(Path script) throws InterruptedException {
164         try {
165             return "found".equals(firstLine(Runtime.getRuntime().exec(new String[] { script.toString() })));
166         } catch (IOException e) {
167             return false;
168         }
169     }
170 
171     @Test
172     @DisabledOnOs(OS.WINDOWS)
173     void testExecFindsAndRefindsACommandOffThePath(@TempDir Path tmp) throws IOException, InterruptedException {
174         // A name unique to this test, so that no other test shares its remembered path
175         String name = "oshi-exec-test-" + System.nanoTime();
176         Path first = Files.createDirectory(tmp.resolve("first"));
177         Path second = Files.createDirectory(tmp.resolve("second"));
178         Path tool = first.resolve(name);
179         Files.writeString(tool, "#!/bin/sh\necho found\n");
180         assertThat(tool.toFile().setExecutable(true), is(true));
181         // A temporary directory on a noexec mount cannot run the script at all
182         assumeTrue(isRunnable(tool), "temporary directory does not allow execution");
183         List<String> dirs = List.of(first.toString(), second.toString());
184         assertThat(firstLine(ExecutingCommand.exec(new String[] { name }, null, dirs)), is("found"));
185         // Moved after it was remembered: the stale path is dropped and the command found again
186         Files.move(tool, second.resolve(name), StandardCopyOption.ATOMIC_MOVE);
187         assertThat(firstLine(ExecutingCommand.exec(new String[] { name }, null, dirs)), is("found"));
188         // Gone entirely: starting it fails as it would have without the search
189         Files.delete(second.resolve(name));
190         assertThrows(IOException.class, () -> ExecutingCommand.exec(new String[] { name }, null, dirs));
191     }
