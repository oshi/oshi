25  /**
26   * Tests that each reason an SMC connection could not be opened is reported once and then demoted to debug, so that
27   * polling sensors on a machine with no SMC does not repeat the message on every reading.
28   */
29  class SmcOpenFailureTest {
30  
31      private static final Pattern PLACEHOLDER = Pattern.compile("\\{\\}");
32  
33      /**
34       * Records the name and formatted text of every logging call. The {@code isXxxEnabled} guards answer with whatever
35       * the recorder was told to report, so a test can check what is skipped when a level is off.
36       */
37      private static final class RecordingLogger implements InvocationHandler {
38          private final List<String> calls = new ArrayList<>();
39          private final List<String> messages = new ArrayList<>();
40          private final boolean levelsEnabled;
41  
42          private RecordingLogger(boolean levelsEnabled) {
43              this.levelsEnabled = levelsEnabled;
44          }
45  
46          @Override
47          public @Nullable Object invoke(Object proxy, Method method, Object @Nullable [] args) {
48              String name = method.getName();
49              if (name.startsWith("is") && name.endsWith("Enabled")) {
50                  return levelsEnabled;
51              }
52              calls.add(name);
53              messages.add(format(args));
54              return null;
55          }
56  
57          /**
58           * Substitutes the arguments into the format string the way SLF4J would. The varargs overloads arrive through
59           * the proxy as {@code {format, Object[]}}, so the substitutions have to be unwrapped from the second argument.
60           *
61           * @param args the arguments the proxy was invoked with
62           * @return the formatted message
63           */
64          private static String format(Object @Nullable [] args) {
65              if (args == null || args.length == 0) {
66                  return "";
67              }
68              String message = String.valueOf(args[0]);
69              if (args.length > 1 && args[1] instanceof Object[]) {
70                  for (Object substitution : (Object[]) args[1]) {
71                      message = PLACEHOLDER.matcher(message)
72                              .replaceFirst(Matcher.quoteReplacement(String.valueOf(substitution)));
73                  }
74              }
75              return message;
76          }
77  
78          private Logger logger() {
79              return (Logger) Proxy.newProxyInstance(Logger.class.getClassLoader(), new Class<?>[] { Logger.class },
80                      this);
81          }
82      }
83  
84      private static RecordingLogger recorder() {
85          return new RecordingLogger(true);
86      }
87  
88      @Test
89      void testServiceNotFoundWarnsOnceThenDebugs() {
90          RecordingLogger recorder = recorder();
91          SmcOpenFailure failure = new SmcOpenFailure(recorder.logger());
92          failure.serviceNotFound();
93          failure.serviceNotFound();
94          failure.serviceNotFound();
95          assertThat(recorder.calls, contains("warn", "debug", "debug"));
96      }
97  
98      @Test
99      void testOpenFailedErrorsOnceThenDebugs() {
100         RecordingLogger recorder = recorder();
101         SmcOpenFailure failure = new SmcOpenFailure(recorder.logger());
102         failure.openFailed(0xe00002c2);
103         failure.openFailed(0xe00002c2);
104         assertThat(recorder.calls, contains("error", "debug"));
105     }
106 
107     @Test
108     void testNullConnectionErrorsOnceThenDebugs() {
109         RecordingLogger recorder = recorder();
110         SmcOpenFailure failure = new SmcOpenFailure(recorder.logger());
111         failure.nullConnection();
112         failure.nullConnection();
113         assertThat(recorder.calls, contains("error", "debug"));
114     }
115 
116     @Test
117     void testConditionsLatchIndependently() {
118         RecordingLogger recorder = recorder();
119         SmcOpenFailure failure = new SmcOpenFailure(recorder.logger());
120         failure.serviceNotFound();
121         failure.openFailed(1);
122         failure.nullConnection();
123         assertThat("Each condition gets its own first report", recorder.calls, contains("warn", "error", "error"));
124     }
125 
126     @Test
127     void testEachInstanceLatchesSeparately() {
128         RecordingLogger first = recorder();
129         RecordingLogger second = recorder();
130         new SmcOpenFailure(first.logger()).serviceNotFound();
131         new SmcOpenFailure(second.logger()).serviceNotFound();
132         assertThat(first.calls, contains("warn"));
133         assertThat("A second backend reports under its own logger", second.calls, contains("warn"));
134     }
135 
136     @Test
137     void testServiceNotFoundExplainsTheCommonCause() {
138         RecordingLogger recorder = recorder();
139         new SmcOpenFailure(recorder.logger()).serviceNotFound();
140         assertThat(recorder.messages.get(0), containsString("virtual machine"));
141     }
142 
143     @Test
144     void testOpenFailedReportsTheErrorInHex() {
145         RecordingLogger recorder = recorder();
146         new SmcOpenFailure(recorder.logger()).openFailed(0xe00002c2);
147         assertThat(recorder.messages.get(0), containsString("0xe00002c2"));
148     }
149 
150     @Test
151     void testHexIsNotFormattedWhenTheLevelIsOff() {
152         RecordingLogger recorder = new RecordingLogger(false);
153         new SmcOpenFailure(recorder.logger()).openFailed(1);
154         assertThat("The guard skips the logging call entirely", recorder.calls, is(empty()));
155     }
