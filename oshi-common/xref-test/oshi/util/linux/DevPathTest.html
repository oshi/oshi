5   package oshi.util.linux;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.is;
9   
10  import org.junit.jupiter.api.Test;
11  import org.junit.jupiter.api.condition.DisabledOnOs;
12  import org.junit.jupiter.api.condition.OS;
13  
14  // Loading DevPath validates that the configured /dev path exists, which fails on Windows
15  @DisabledOnOs(OS.WINDOWS)
16  class DevPathTest {
17  
18      private static final String HOST_DEV = "/host/dev/";
19  
20      @Test
21      void testNamesUseTheNodeRoot() {
22          assertThat(DevPath.NODE_ROOT, is("/dev/"));
23          assertThat(DevPath.DM, is("/dev/dm"));
24          assertThat(DevPath.LOOP, is("/dev/loop"));
25          assertThat(DevPath.MAPPER, is("/dev/mapper/"));
26          assertThat(DevPath.RAM, is("/dev/ram"));
27      }
28  
29      @Test
30      void testCustomRootToNodeName() {
31          assertThat(DevPath.reroot("/host/dev/sda1", HOST_DEV, DevPath.NODE_ROOT), is("/dev/sda1"));
32          assertThat(DevPath.reroot("/host/dev/mapper/vg-root", HOST_DEV, DevPath.NODE_ROOT), is("/dev/mapper/vg-root"));
33      }
34  
35      @Test
36      void testNodeNameToCustomRoot() {
37          assertThat(DevPath.reroot("/dev/mapper/vg-root", DevPath.NODE_ROOT, HOST_DEV), is("/host/dev/mapper/vg-root"));
38      }
39  
40      @Test
41      void testPathOutsideTheRootIsUnchanged() {
42          assertThat(DevPath.reroot("tmpfs", HOST_DEV, DevPath.NODE_ROOT), is("tmpfs"));
43          assertThat(DevPath.reroot("/device/sda", DevPath.NODE_ROOT, HOST_DEV), is("/device/sda"));
44          assertThat(DevPath.reroot("/dev/sda1", HOST_DEV, DevPath.NODE_ROOT), is("/dev/sda1"));
45      }
46  
47      @Test
48      void testConfiguredRootRoundTrip() {
49          assertThat(DevPath.toFilePath("/dev/sda1"), is(DevPath.DEV + "sda1"));
50          assertThat(DevPath.toNodeName(DevPath.DEV + "sda1"), is("/dev/sda1"));
51      }
52  }
