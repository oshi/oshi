124      * A number whose decimals are introduced by something other than a decimal point is not parseable, and the
125      * documented -1 is the answer. This used to throw NumberFormatException instead.
126      * <p>
127      * These cases also pin the pattern's deliberately unescaped {@code .}, which matches any separator and so captures
128      * the malformed number whole for the parser to reject. Escaping it would let the match re-anchor on a fragment and
129      * report "2 40GHz" as a 40 GHz processor.
130      */
131     @Test
132     void testParseHertzNonDecimalSeparator() {
133         assertThat("comma decimal separator", ParseUtil.parseHertz("2,40GHz"), is(-1L));
134         assertThat("comma separator in a CPU name", ParseUtil.parseHertz("Intel Core i7 @ 2,40GHz"), is(-1L));
135         assertThat("space separator", ParseUtil.parseHertz("2 40GHz"), is(-1L));
136         assertThat("letter separator", ParseUtil.parseHertz("2X00MHz"), is(-1L));
137         assertThat("still parses a decimal point", ParseUtil.parseHertz("Intel Core i7 @ 2.40GHz"), is(2_400_000_000L));
138         assertThat("still parses without a decimal", ParseUtil.parseHertz("Some CPU @ 3GHz"), is(3_000_000_000L));
