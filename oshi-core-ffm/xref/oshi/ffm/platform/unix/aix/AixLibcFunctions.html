5   package oshi.ffm.platform.unix.aix;
6   
7   import static java.lang.foreign.ValueLayout.ADDRESS;
8   import static java.lang.foreign.ValueLayout.JAVA_INT;
9   import static java.lang.foreign.ValueLayout.JAVA_LONG;
10  
11  import java.lang.foreign.FunctionDescriptor;
12  import java.lang.foreign.MemorySegment;
13  import java.lang.foreign.ValueLayout;
14  import java.lang.invoke.MethodHandle;
15  
16  import oshi.ffm.platform.unix.PosixLibcFunctions;
17  
18  /**
19   * FFM bindings for AIX libc functions used by OSHI.
20   * <p>
21   * Most performance and configuration data on AIX comes from {@code libperfstat} (see {@link PerfstatFunctions}); the
22   * libc surface here is limited to {@code thread_self} and the {@code open}/{@code close}/{@code pread} trio used to
23   * read {@code /proc}. The POSIX bindings ({@code getpid}, {@code getrlimit}, {@code gethostname}) are inherited from
24   * {@link PosixLibcFunctions}.
25   */
26  public final class AixLibcFunctions extends PosixLibcFunctions {
27  
28      private AixLibcFunctions() {
29      }
30  
31      /** Layout of the C {@code size_t} type on 64-bit AIX. */
32      public static final ValueLayout.OfLong SIZE_T = ValueLayout.JAVA_LONG;
33  
34      /** {@code getrlimit} resource: maximum number of open file descriptors. AIX value (7). */
35      public static final int RLIMIT_NOFILE = 7;
36  
37      // tid_t thread_self(void); // AIX-specific. tid_t follows long -- measured at 4 bytes compiling
38      // with -maix32 and 8 with -maix64 -- so it is 8 here, as SIZE_T above is, because FFM needs a
39      // 64-bit JVM. Declaring it 4 read only the low half of the returned register. The JNA twin maps
40      // it to NativeLong instead, which is the same width here and stays correct in either model.
41      private static final MethodHandle thread_self = LINKER.downcallHandle(LIBC.findOrThrow("thread_self"),
42              FunctionDescriptor.of(JAVA_LONG));
43  
44      /**
45       * Calls {@code thread_self()} — returns the kernel thread ID of the calling thread (AIX).
46       *
47       * @return the kernel thread ID of the calling thread
48       * @throws Throwable on FFM invocation error
49       */
50      public static long thread_self() throws Throwable {
51          return (long) thread_self.invokeExact();
52      }
53  
54      // int open(const char *path, int flags);
55      private static final MethodHandle open = LINKER.downcallHandle(LIBC.findOrThrow("open"),
56              FunctionDescriptor.of(JAVA_INT, ADDRESS, JAVA_INT));
57  
58      /**
59       * Calls {@code open(path, flags)}.
60       *
61       * @param path  null-terminated path segment
62       * @param flags open flags (e.g. {@code O_RDONLY})
63       * @return file descriptor on success, -1 on error
64       * @throws Throwable on FFM invocation error
65       */
66      public static int open(MemorySegment path, int flags) throws Throwable {
67          return (int) open.invokeExact(path, flags);
68      }
69  
70      // int close(int fd);
71      private static final MethodHandle close = LINKER.downcallHandle(LIBC.findOrThrow("close"),
72              FunctionDescriptor.of(JAVA_INT, JAVA_INT));
73  
74      /**
75       * Calls {@code close(fd)}.
76       *
77       * @param fd file descriptor
78       * @return 0 on success, -1 on error
79       * @throws Throwable on FFM invocation error
80       */
81      public static int close(int fd) throws Throwable {
82          return (int) close.invokeExact(fd);
83      }
84  
85      // ssize_t pread(int fd, void *buf, size_t count, off_t offset);
86      private static final MethodHandle pread = LINKER.downcallHandle(LIBC.findOrThrow("pread"),
87              FunctionDescriptor.of(SIZE_T, JAVA_INT, ADDRESS, SIZE_T, JAVA_LONG));
88  
89      /**
90       * Calls {@code pread(fd, buf, count, offset)}. {@code ssize_t} is mapped to {@code long}.
91       *
92       * @param fd     file descriptor
93       * @param buf    buffer for the read
94       * @param count  number of bytes to read
95       * @param offset starting byte offset in the file
96       * @return number of bytes actually read, or -1 on error
97       * @throws Throwable on FFM invocation error
98       */
99      public static long pread(int fd, MemorySegment buf, long count, long offset) throws Throwable {
100         return (long) pread.invokeExact(fd, buf, count, offset);
101     }
102 
103 }
