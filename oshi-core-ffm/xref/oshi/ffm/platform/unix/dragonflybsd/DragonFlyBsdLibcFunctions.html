5   package oshi.ffm.platform.unix.dragonflybsd;
6   
7   import static java.lang.foreign.ValueLayout.ADDRESS;
8   import static java.lang.foreign.ValueLayout.JAVA_INT;
9   import static java.lang.foreign.ValueLayout.JAVA_LONG;
10  
11  import java.lang.foreign.FunctionDescriptor;
12  import java.lang.foreign.MemorySegment;
13  import java.lang.foreign.SymbolLookup;
14  import java.lang.invoke.MethodHandle;
15  
16  import oshi.ffm.ForeignFunctions;
17  
18  /**
19   * FFM bindings for DragonFly BSD-specific libc functions. DragonFly shares most of its libc surface with FreeBSD, so
20   * shared functions ({@code sysctlbyname}, {@code sysctl}, {@code getpid}, {@code getrlimit}, etc.) are called via
21   * {@link oshi.ffm.platform.unix.freebsd.FreeBsdLibcFunctions}. This class adds the DragonFly-specific
22   * {@code lwp_gettid()} (FreeBSD uses {@code thr_self} instead) and {@code sched_getaffinity()}.
23   */
24  public final class DragonFlyBsdLibcFunctions extends ForeignFunctions {
25  
26      private DragonFlyBsdLibcFunctions() {
27      }
28  
29      private static final SymbolLookup LIBC = LINKER.defaultLookup();
30  
31      // int lwp_gettid(void);
32      private static final MethodHandle lwp_gettid = LINKER.downcallHandle(LIBC.findOrThrow("lwp_gettid"),
33              FunctionDescriptor.of(JAVA_INT));
34  
35      /**
36       * Calls {@code lwp_gettid()} — returns the kernel thread ID of the calling LWP.
37       *
38       * @return the kernel thread ID, or -1 on error
39       * @throws Throwable on FFM invocation error
40       */
41      public static int lwp_gettid() throws Throwable {
42          return (int) lwp_gettid.invokeExact();
43      }
44  
45      /**
46       * The size of DragonFly's {@code cpu_set_t}, a {@code cpumask_t} of four 64-bit words on x86_64.
47       */
48      public static final long CPU_SET_SIZE = 32L;
49  
50      // int sched_getaffinity(pid_t pid, size_t cpusetsize, cpu_set_t *mask);
51      private static final MethodHandle sched_getaffinity = LINKER.downcallHandle(LIBC.findOrThrow("sched_getaffinity"),
52              FunctionDescriptor.of(JAVA_INT, JAVA_INT, JAVA_LONG, ADDRESS));
53  
54      /**
55       * Calls {@code sched_getaffinity()} — gets the CPU affinity mask of a process, that of its first LWP.
56       *
57       * @param pid        the process ID, or 0 for the calling process
58       * @param cpusetsize the size of {@code mask}, at most {@link #CPU_SET_SIZE} of which is written
59       * @param mask       the segment that receives the mask, CPU 0 in the lowest bit of the first word
60       * @return 0 on success, -1 on error
61       * @throws Throwable on FFM invocation error
62       */
63      public static int sched_getaffinity(int pid, long cpusetsize, MemorySegment mask) throws Throwable {
64          return (int) sched_getaffinity.invokeExact(pid, cpusetsize, mask);
65      }
66  }
