5   package oshi.ffm.platform.unix;
6   
7   import static java.lang.foreign.ValueLayout.ADDRESS;
8   import static java.lang.foreign.ValueLayout.JAVA_INT;
9   import static java.lang.foreign.ValueLayout.JAVA_LONG;
10  
11  import java.lang.foreign.FunctionDescriptor;
12  import java.lang.foreign.MemoryLayout;
13  import java.lang.foreign.MemoryLayout.PathElement;
14  import java.lang.foreign.MemorySegment;
15  import java.lang.foreign.StructLayout;
16  import java.lang.foreign.SymbolLookup;
17  import java.lang.invoke.MethodHandle;
18  import java.lang.invoke.VarHandle;
19  
20  import oshi.ffm.ForeignFunctions;
21  
22  /**
23   * FFM bindings for the libc functions POSIX requires every UNIX to provide, shared by the per-platform
24   * {@code *LibcFunctions} classes.
25   * <p>
26   * <strong>Membership rule: POSIX-mandated functions only.</strong> These handles are resolved with {@code findOrThrow}
27   * in a static initializer, so a symbol missing on any one platform would raise {@code ExceptionInInitializerError}
28   * there and take every other member of this class down with it. Being POSIX-mandated is what guarantees the symbol
29   * exists everywhere, so it is the criterion for belonging here rather than a stylistic preference.
30   * <p>
31   * {@code getloadavg} is the cautionary example: it looks like a peer of these, and four of the six platforms bind it,
32   * but it is a BSD extension rather than POSIX and AIX does not have it (AIX reads load average from
33   * {@code perfstat_cpu_total} instead). It therefore stays in the per-platform classes.
34   * <p>
35   * Values that POSIX names but does not fix stay per-platform too: {@code RLIMIT_NOFILE} is 7 on Linux and AIX, 8 on the
36   * BSDs, and 5 on Solaris/illumos, so each subclass declares its own.
37   */
38  public abstract class PosixLibcFunctions extends ForeignFunctions {
39  
40      /**
41       * Constructs a new {@code PosixLibcFunctions}. Subclasses are utility holders and are never instantiated; this
42       * exists only so they can inherit the members below.
43       */
44      protected PosixLibcFunctions() {
45      }
46  
47      /**
48       * The libc symbol lookup, shared with the per-platform subclasses. libc is already loaded into the JVM process, so
49       * {@code defaultLookup()} finds it without the {@code libc.so} versioning pitfalls that
50       * {@code SymbolLookup.libraryLookup("c")} hits on some platforms.
51       */
52      protected static final SymbolLookup LIBC = LINKER.defaultLookup();
53  
54      /** Layout of {@code struct rlimit}: two {@code rlim_t} (LP64 long) fields. */
55      public static final StructLayout RLIMIT_LAYOUT = MemoryLayout.structLayout(JAVA_LONG.withName("rlim_cur"),
56              JAVA_LONG.withName("rlim_max"));
57  
58      private static final VarHandle RLIMIT_CUR = RLIMIT_LAYOUT.varHandle(PathElement.groupElement("rlim_cur"));
59      private static final VarHandle RLIMIT_MAX = RLIMIT_LAYOUT.varHandle(PathElement.groupElement("rlim_max"));
60  
61      // pid_t getpid(void);
62      private static final MethodHandle getpid = LINKER.downcallHandle(LIBC.findOrThrow("getpid"),
63              FunctionDescriptor.of(JAVA_INT));
64  
65      /**
66       * Calls {@code getpid()}.
67       *
68       * @return the process ID of the calling process
69       * @throws Throwable on FFM invocation error
70       */
71      public static int getpid() throws Throwable {
72          return (int) getpid.invokeExact();
73      }
74  
75      // int getrlimit(int resource, struct rlimit *rlim);
76      private static final MethodHandle getrlimit = LINKER.downcallHandle(LIBC.findOrThrow("getrlimit"),
77              FunctionDescriptor.of(JAVA_INT, JAVA_INT, ADDRESS));
78  
79      /**
80       * Calls {@code getrlimit(resource, rlim)}.
81       *
82       * @param resource resource constant; use the calling class's {@code RLIMIT_NOFILE}, whose value is
83       *                 platform-specific
84       * @param rlim     segment allocated with {@link #RLIMIT_LAYOUT}
85       * @return 0 on success, -1 on error
86       * @throws Throwable on FFM invocation error
87       */
88      public static int getrlimit(int resource, MemorySegment rlim) throws Throwable {
89          return (int) getrlimit.invokeExact(resource, rlim);
90      }
91  
92      /**
93       * Reads {@code rlim_cur} from an rlimit segment populated by {@link #getrlimit(int, MemorySegment)}.
94       *
95       * @param rlim segment allocated with {@link #RLIMIT_LAYOUT}
96       * @return the soft resource limit
97       */
98      public static long rlimitCur(MemorySegment rlim) {
99          return (long) RLIMIT_CUR.get(rlim, 0L);
100     }
101 
102     /**
103      * Reads {@code rlim_max} from an rlimit segment populated by {@link #getrlimit(int, MemorySegment)}.
104      *
105      * @param rlim segment allocated with {@link #RLIMIT_LAYOUT}
106      * @return the hard resource limit
107      */
108     public static long rlimitMax(MemorySegment rlim) {
109         return (long) RLIMIT_MAX.get(rlim, 0L);
110     }
111 
112     // int gethostname(char *name, size_t namelen);
113     private static final MethodHandle gethostname = LINKER.downcallHandle(LIBC.findOrThrow("gethostname"),
114             FunctionDescriptor.of(JAVA_INT, ADDRESS, JAVA_LONG));
115 
116     /**
117      * Calls {@code gethostname(name, namelen)}.
118      *
119      * @param name    buffer to receive the host name
120      * @param namelen size of the buffer in bytes
121      * @return 0 on success, -1 on error
122      * @throws Throwable on FFM invocation error
123      */
124     public static int gethostname(MemorySegment name, long namelen) throws Throwable {
125         return (int) gethostname.invokeExact(name, namelen);
126     }
127 }
