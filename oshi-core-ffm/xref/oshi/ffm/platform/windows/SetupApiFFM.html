74      /**
75       * Enumerates the device interfaces of a single device, scoping the enumeration to the given {@code devInfoData}
76       * rather than the whole device info set.
77       *
78       * @param hDevInfo            the device info set handle
79       * @param devInfoData         the device to scope enumeration to, or {@code MemorySegment.NULL} for the whole set
80       * @param classGuid           the interface class GUID
81       * @param memberIndex         zero-based index of the interface to retrieve
82       * @param deviceInterfaceData receives information about the device interface
83       * @return 0 if no more items, 1 if success, -1 on other error
84       */
85      public static int SetupDiEnumDeviceInterfaces(MemorySegment hDevInfo, MemorySegment devInfoData,
86              MemorySegment classGuid, int memberIndex, MemorySegment deviceInterfaceData) {
87          return getIntOrDefault(() -> {
88              int result = (int) SetupDiEnumDeviceInterfaces.invokeExact(hDevInfo, devInfoData, classGuid, memberIndex,
89                      deviceInterfaceData);
90              if (isSuccess(result)) {
91                  return 1;
92              }
93              int err = Kernel32FFM.GetLastError().orElse(0);
94              return err == ERROR_NO_MORE_ITEMS ? 0 : -1;
95          }, -1, LOG, "SetupApiFFM.SetupDiEnumDeviceInterfaces failed");
96      }

99              "SetupDiGetDeviceInterfaceDetailW", JAVA_INT, ADDRESS, ADDRESS, ADDRESS, JAVA_INT, ADDRESS, ADDRESS);

101     /**
102      * First call with null detail buffer to get required size.
103      *
104      * @param hDevInfo            the device info set handle
105      * @param deviceInterfaceData the device interface data
106      * @param arena               the arena to allocate the size buffer from
107      * @return required buffer size, or 0 on unexpected error
108      */
109     public static int SetupDiGetDeviceInterfaceDetailSize(MemorySegment hDevInfo, MemorySegment deviceInterfaceData,
110             Arena arena) {
111         return getIntOrDefault(() -> {
112             MemorySegment requiredSize = arena.allocate(JAVA_INT);
113             // The sizing call is expected to fail with ERROR_INSUFFICIENT_BUFFER, having written the size it needs, so
114             // the written size is what to read; an unexpected success leaves nothing to size. Reading it rather than
115             // gating on GetLastError() == ERROR_INSUFFICIENT_BUFFER is deliberate. Arena allocations are
116             // zero-initialized, so a failure that writes no size reads back 0 -- the same answer the error check would
117             // give -- whereas a separate GetLastError() downcall is not a dependable read: the runtime may overwrite
118             // the thread's last error first, which is what Linker.Option.captureCallState exists to work around. A
119             // clobbered read would discard a size the API did write and report UNKNOWN for every display. The JNA twin
120             // trusts the written size too. Cast the result even though it is only tested for success: invokeExact() in
121             // an expression statement infers a void return type, and a handle declared to return int then throws
122             // WrongMethodTypeException.
123             if (isSuccess((int) SetupDiGetDeviceInterfaceDetail.invokeExact(hDevInfo, deviceInterfaceData,
124                     MemorySegment.NULL, 0, requiredSize, MemorySegment.NULL))) {
125                 return 0;
126             }
127             return requiredSize.get(JAVA_INT, 0);
128         }, 0, LOG, "SetupApiFFM.SetupDiGetDeviceInterfaceDetailSize failed");
129     }

131     /**
132      * Second call with allocated buffer to get device path. The detail buffer layout is: cbSize (DWORD=4) + DevicePath
133      * (WCHAR[]). On 64-bit, cbSize field value must be 8; on 32-bit it is 5 or 6.
134      *
135      * @param hDevInfo            the device info set handle
136      * @param deviceInterfaceData the device interface data
137      * @param requiredSize        the buffer size returned by {@link #SetupDiGetDeviceInterfaceDetailSize}
138      * @param arena               the arena to allocate buffers from
139      * @return the device path string, or empty if failed
140      */
141     public static Optional<String> SetupDiGetDeviceInterfaceDetail(MemorySegment hDevInfo,
142             MemorySegment deviceInterfaceData, int requiredSize, Arena arena) {
143         return getOptional(() -> {
144             MemorySegment detail = arena.allocate(requiredSize);
145             detail.set(JAVA_INT, 0, is64Bit() ? 8 : 6);
146             MemorySegment reqSize = arena.allocate(JAVA_INT);
147             int result = (int) SetupDiGetDeviceInterfaceDetail.invokeExact(hDevInfo, deviceInterfaceData, detail,
148                     requiredSize, reqSize, MemorySegment.NULL);
149             return isSuccess(result) ? readWideString(detail.asSlice(4)) : null;
150         }, LOG, "SetupApiFFM.SetupDiGetDeviceInterfaceDetail failed");
151     }

192      *
193      * @param hDevInfo    the device info set handle
194      * @param devInfoData the device info data
195      * @param scope       DICS_FLAG_GLOBAL or DICS_FLAG_CONFIGSPECIFIC
196      * @param hwProfile   hardware profile (0 for current)
197      * @param keyType     DIREG_DEV or DIREG_DRV
198      * @param samDesired  registry access mask (e.g. KEY_QUERY_VALUE)
199      * @return registry key handle, or null if failed
200      */
201     public static MemorySegment SetupDiOpenDevRegKey(MemorySegment hDevInfo, MemorySegment devInfoData, int scope,
202             int hwProfile, int keyType, int samDesired) {
203         return getOrDefault(() -> {
204             MemorySegment key = (MemorySegment) SetupDiOpenDevRegKey.invokeExact(hDevInfo, devInfoData, scope,
205                     hwProfile, keyType, samDesired);
206             return Kernel32FFM.isInvalidHandle(key) ? null : key;
207         }, null, LOG, "SetupApiFFM.SetupDiOpenDevRegKey failed");
