32  /**
33   * FFM-based optional runtime binding to the NVIDIA Management Library (NVML). All methods return sentinel values
34   * ({@code -1} or {@code -1L}) when NVML is unavailable or a specific query fails.
35   * <p>
36   * The query skeleton shared with the JNA binding lives in {@link NvmlQuery}; only the native reads below differ.
37   */
38  @ThreadSafe
39  public final class NvmlUtilFFM {
40  
41      private static final Logger LOG = LoggerFactory.getLogger(NvmlUtilFFM.class);
42  
43      private static final long BUS_ID_LEGACY_OFFSET = NvmlFunctions.PCI_INFO_LAYOUT
44              .byteOffset(MemoryLayout.PathElement.groupElement("busIdLegacy"));
45      private static final long BUS_ID_OFFSET = NvmlFunctions.PCI_INFO_LAYOUT
46              .byteOffset(MemoryLayout.PathElement.groupElement("busId"));
47  
48      // Derived from the layout rather than repeated as literals, so the reads cannot drift from the struct definition.
49      private static final int BUS_ID_LEGACY_LENGTH = (int) NvmlFunctions.PCI_INFO_LAYOUT
50              .select(MemoryLayout.PathElement.groupElement("busIdLegacy")).byteSize();
51      private static final int BUS_ID_LENGTH = (int) NvmlFunctions.PCI_INFO_LAYOUT
52              .select(MemoryLayout.PathElement.groupElement("busId")).byteSize();
53  
54      private static final NvmlDeviceCache DEVICE_CACHE = new NvmlDeviceCache("FFM");
55  
56      private static final NvmlScope<Device> SCOPE = new NvmlScope<Device>() {
57          @Override
58          public boolean init() {
59              return nvmlInit();
60          }
61  
62          @Override
63          public void uninit() {
64              nvmlUninit();
65          }
66  
67          @Override
68          public <R> R withDevice(String deviceId, Function<Device, R> body, R sentinel) {
69              // The arena is opened and closed here so no MemorySegment outlives the query or escapes this package.
70              try (Arena arena = Arena.ofConfined()) {
71                  MemorySegment handle = acquireHandleByBusId(deviceId, arena);
72                  return handle == null ? sentinel : body.apply(new Device(handle, arena));
73              }
74          }
75      };

81       * An acquired device handle together with the arena its out-parameters are allocated from. Both are valid only for
82       * the duration of one {@link NvmlScope#withDevice} call.
83       */
84      private static final class Device {
85          private final MemorySegment handle;
86          private final Arena arena;
87  
88          Device(MemorySegment handle, Arena arena) {
89              this.handle = handle;
90              this.arena = arena;
91          }
92      }

94      private static boolean nvmlInit() {
95          if (!NvmlFunctions.isAvailable()) {
96              return false;
97          }
98          int ret = NvmlFunctions.init();
99          if (ret == NvmlFunctions.NVML_SUCCESS) {
100             return true;
101         }
102         LOG.debug("nvmlInit_v2 failed with code {}", ret);
103         return false;
104     }
105 
106     private static void nvmlUninit() {
107         NvmlFunctions.shutdown();
108     }

114     /**
115      * Applies {@code visitor} to each NVML device handle in index order, stopping early if it returns true. Devices
116      * whose handle cannot be acquired are skipped. Must be called while NVML is initialized.
117      *
118      * @param arena   allocates the out-parameters for the walk
119      * @param visitor applied to each handle, returning true to stop the walk
120      * @return false if the device count could not be read, letting callers distinguish an NVML failure from a machine
121      *         with no devices
122      */
123     private static boolean forEachDevice(Arena arena, Predicate<MemorySegment> visitor) {
124         MemorySegment countSeg = arena.allocate(JAVA_INT);
125         if (NvmlFunctions.deviceGetCount(countSeg) != NvmlFunctions.NVML_SUCCESS) {
126             return false;
127         }
128         int count = countSeg.get(JAVA_INT, 0);
129         for (int i = 0; i < count; i++) {
130             MemorySegment handleSeg = arena.allocate(ADDRESS);
131             if (NvmlFunctions.deviceGetHandleByIndex(i, handleSeg) != NvmlFunctions.NVML_SUCCESS) {
132                 continue;
133             }
134             if (visitor.test(handleSeg.get(ADDRESS, 0))) {
135                 return true;
136             }
137         }
138         return true;
139     }

141     /**
142      * Reads a device's two PCI bus ID forms, lowercased.
143      *
144      * @param handle the device handle
145      * @param arena  allocates the PCI info struct
146      * @return the modern and legacy bus IDs, or {@code null} if the PCI info could not be read
147      */
148     private static @Nullable Pair<String, String> readBusIds(MemorySegment handle, Arena arena) {
149         MemorySegment pciSeg = arena.allocate(NvmlFunctions.PCI_INFO_LAYOUT);
150         if (NvmlFunctions.deviceGetPciInfo(handle, pciSeg) != NvmlFunctions.NVML_SUCCESS) {
151             return null;
152         }
153         return new Pair<>(NvmlFunctions.readString(pciSeg, BUS_ID_OFFSET, BUS_ID_LENGTH).toLowerCase(Locale.ROOT),
154                 NvmlFunctions.readString(pciSeg, BUS_ID_LEGACY_OFFSET, BUS_ID_LEGACY_LENGTH).toLowerCase(Locale.ROOT));
155     }
156 
157     /**
158      * Reads a device's name, lowercased.
159      *
160      * @param handle the device handle
161      * @param arena  allocates the name buffer
162      * @return the name, or {@code null} if it could not be read
163      */
164     private static @Nullable String readName(MemorySegment handle, Arena arena) {
165         MemorySegment nameSeg = arena.allocate(NvmlFunctions.NVML_DEVICE_NAME_BUFFER_SIZE);
166         if (NvmlFunctions.deviceGetName(handle, nameSeg,
167                 NvmlFunctions.NVML_DEVICE_NAME_BUFFER_SIZE) != NvmlFunctions.NVML_SUCCESS) {
168             return null;
169         }
170         return nameSeg.getString(0).toLowerCase(Locale.ROOT);
171     }
172 
173     /**
174      * Returns a set of PCI bus ID strings for all NVML devices, or {@code null} on NVML error (so the caller can
175      * distinguish a real failure from a legitimate empty result).
176      *
177      * @return set of PCI bus ID strings, or {@code null} on NVML error
178      */
179     private static @Nullable Set<String> enumerateDeviceBusIds() {
180         try (Arena arena = Arena.ofConfined()) {
181             Set<String> ids = new HashSet<>();
182             boolean enumerated = forEachDevice(arena, handle -> {
183                 Pair<String, String> busIds = readBusIds(handle, arena);
184                 if (busIds != null) {
185                     addIfNotEmpty(ids, busIds.getA());
186                     addIfNotEmpty(ids, busIds.getB());
187                 }
188                 return false;
189             });
190             return enumerated ? Collections.unmodifiableSet(ids) : null;
191         }
192     }
193 
194     private static void addIfNotEmpty(Set<String> ids, String id) {
195         if (!id.isEmpty()) {
196             ids.add(id);
197         }
198     }

200     private static @Nullable MemorySegment acquireHandleByBusId(String pciBusId, Arena arena) {
201         String needle = pciBusId.toLowerCase(Locale.ROOT);
202         MemorySegment[] found = new MemorySegment[1];
203         forEachDevice(arena, handle -> {
204             Pair<String, String> busIds = readBusIds(handle, arena);
205             if (busIds != null
206                     && (NvmlQuery.matches(busIds.getA(), needle) || NvmlQuery.matches(busIds.getB(), needle))) {
207                 found[0] = handle;
208                 return true;
209             }
210             return false;
211         });
212         return found[0];
213     }
214 
215     private static @Nullable MemorySegment acquireHandleByName(String gpuName, Arena arena) {
216         String needle = gpuName.toLowerCase(Locale.ROOT);
217         MemorySegment[] found = new MemorySegment[1];
218         forEachDevice(arena, handle -> {
219             String name = readName(handle, arena);
220             if (name != null && NvmlQuery.matches(name, needle)) {
221                 found[0] = handle;
222                 return true;
223             }
224             return false;
225         });
226         return found[0];
227     }

229     /**
230      * Counts the devices whose name matches, so an ambiguous match can be rejected rather than resolved arbitrarily.
231      *
232      * @param gpuName GPU name to match
233      * @param arena   allocates the out-parameters for the walk
234      * @return the number of matching devices, or {@code -1} if the devices could not be enumerated
235      */
236     private static int countMatchesByName(String gpuName, Arena arena) {
237         String needle = gpuName.toLowerCase(Locale.ROOT);
238         int[] matches = new int[1];
239         boolean enumerated = forEachDevice(arena, handle -> {
240             String name = readName(handle, arena);
241             if (name != null && NvmlQuery.matches(name, needle)) {
242                 matches[0]++;
243             }
244             return false;
245         });
246         return enumerated ? matches[0] : -1;
247     }

253     private static double readUtilization(Device device) {
254         MemorySegment utilSeg = device.arena.allocate(NvmlFunctions.UTILIZATION_LAYOUT);
255         if (NvmlFunctions.deviceGetUtilizationRates(device.handle, utilSeg) == NvmlFunctions.NVML_SUCCESS) {
256             return utilSeg.get(JAVA_INT, 0);
257         }
258         return -1d;
259     }

261     private static long readVramTotal(Device device) {
262         MemorySegment memSeg = device.arena.allocate(NvmlFunctions.MEMORY_LAYOUT);
263         if (NvmlFunctions.deviceGetMemoryInfo(device.handle, memSeg) == NvmlFunctions.NVML_SUCCESS) {
264             return memSeg.get(JAVA_LONG,
265                     NvmlFunctions.MEMORY_LAYOUT.byteOffset(MemoryLayout.PathElement.groupElement("total")));
266         }
267         return -1L;
268     }

325     /**
326      * Finds the stable PCI bus ID string for the NVML device whose bus ID contains the given fragment.
327      *
328      * @param pciBusId PCI bus ID fragment
329      * @return matched PCI bus ID string, or {@code null} if not found
330      */
331     public static @Nullable String findDevice(@Nullable String pciBusId) {
332         if (!NvmlFunctions.isAvailable() || pciBusId == null || pciBusId.isEmpty()) {
333             return null;
334         }
335         if (!nvmlInit()) {
336             return null;
337         }
338         try {
339             return NvmlQuery.matchBusId(DEVICE_CACHE.get(NvmlUtilFFM::enumerateDeviceBusIds), pciBusId);
340         } finally {
341             nvmlUninit();
342         }
343     }
344 
345     /**
346      * Finds the stable PCI bus ID string for the NVML device whose name matches the given GPU name.
347      *
348      * @param gpuName GPU name string
349      * @return PCI bus ID string, or {@code null} if not found
350      */
351     public static @Nullable String findDeviceByName(@Nullable String gpuName) {
352         if (!NvmlFunctions.isAvailable() || gpuName == null || gpuName.isEmpty()) {
353             return null;
354         }
355         if (!nvmlInit()) {
356             return null;
357         }
358         try (Arena arena = Arena.ofConfined()) {
359             int matchCount = countMatchesByName(gpuName, arena);
360             if (matchCount <= 0) {
361                 return null;
362             }
363             if (matchCount > 1) {
364                 LOG.warn("NVML name match for '{}' is ambiguous ({} devices match); use PCI bus ID for reliable"
365                         + " device identification", gpuName, matchCount);
366                 return null;
367             }
368             MemorySegment handle = acquireHandleByName(gpuName, arena);
369             if (handle == null) {
370                 return null;
371             }
372             Pair<String, String> busIds = readBusIds(handle, arena);
373             if (busIds == null) {
374                 return null;
375             }
376             // Prefer the modern form, matching the order enumerateDeviceBusIds records them in.
377             return busIds.getA().isEmpty() ? emptyToNull(busIds.getB()) : busIds.getA();
378         } finally {
379             nvmlUninit();
380         }
381     }

383     private static @Nullable String emptyToNull(String value) {
384         return value.isEmpty() ? null : value;
385     }
