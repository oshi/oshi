5   package oshi.driver.unix.bsd;
6   
7   import static oshi.ffm.ForeignFunctions.CAPTURED_STATE_LAYOUT;
8   import static oshi.ffm.ForeignFunctions.callInArenaOrDefault;
9   import static oshi.util.LogLevel.ERROR;
10  
11  import java.lang.foreign.MemorySegment;
12  import java.lang.foreign.ValueLayout;
13  
14  import org.slf4j.Logger;
15  import org.slf4j.LoggerFactory;
16  
17  import oshi.annotation.concurrent.ThreadSafe;
18  
19  /**
20   * Fetches the kernel's routing table through a {@code NET_RT_DUMP} sysctl. Shared by the BSDs, which take the same
21   * call; only the message layout inside the buffer differs between them.
22   */
23  @ThreadSafe
24  public final class BsdRouteDumpFFM {
25  
26      private static final Logger LOG = LoggerFactory.getLogger(BsdRouteDumpFFM.class);
27  
28      private static final int CTL_NET = 4;
29      private static final int PF_ROUTE = 17;
30      private static final int NET_RT_DUMP = 1;
31  
32      /**
33       * The platform's {@code sysctl}, which the BSDs each declare in their own bindings.
34       */
35      @FunctionalInterface
36      public interface Sysctl {
37          /**
38           * Calls {@code sysctl}.
39           *
40           * @param callState errno capture
41           * @param name      the MIB
42           * @param namelen   entries in the MIB
43           * @param oldp      buffer to fill, or {@link MemorySegment#NULL} to ask only for the size
44           * @param oldlenp   size of the buffer, and on return the bytes written
45           * @return 0 on success
46           * @throws Throwable on invocation error
47           */
48          int call(MemorySegment callState, MemorySegment name, int namelen, MemorySegment oldp, MemorySegment oldlenp)
49                  throws Throwable;
50      }
51  
52      private BsdRouteDumpFFM() {
53      }
54  
55      /**
56       * Fetches the routing table dump.
57       *
58       * @param sysctl This platform's {@code sysctl}
59       * @return The bytes the kernel returned, or an empty array if the query failed
60       */
61      public static byte[] queryRouteDump(Sysctl sysctl) {
62          return callInArenaOrDefault(arena -> {
63              int[] mib = { CTL_NET, PF_ROUTE, 0, 0, NET_RT_DUMP, 0 };
64              MemorySegment mibSeg = arena.allocate(ValueLayout.JAVA_INT, mib.length);
65              for (int i = 0; i < mib.length; i++) {
66                  mibSeg.setAtIndex(ValueLayout.JAVA_INT, i, mib[i]);
67              }
68              MemorySegment lenSeg = arena.allocate(ValueLayout.JAVA_LONG);
69              MemorySegment callState = arena.allocate(CAPTURED_STATE_LAYOUT);
70              if (0 != sysctl.call(callState, mibSeg, mib.length, MemorySegment.NULL, lenSeg)) {
71                  LOG.error("Didn't get buffer length for NET_RT_DUMP");
72                  return new byte[0];
73              }
74              long len = lenSeg.get(ValueLayout.JAVA_LONG, 0);
75              if (len <= 0) {
76                  return new byte[0];
77              }
78              MemorySegment buf = arena.allocate(len);
79              if (0 != sysctl.call(callState, mibSeg, mib.length, buf, lenSeg)) {
80                  LOG.error("Didn't get buffer for NET_RT_DUMP");
81                  return new byte[0];
82              }
83              // The second call may report fewer bytes than the first reserved
84              long actual = Math.min(len, lenSeg.get(ValueLayout.JAVA_LONG, 0));
85              return buf.asSlice(0, actual).toArray(ValueLayout.JAVA_BYTE);
86          }, LOG, ERROR, "Failed to read the routing table", new byte[0]);
87      }
88  }
