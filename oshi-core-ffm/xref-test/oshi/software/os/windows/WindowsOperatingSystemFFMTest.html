5   package oshi.software.os.windows;
6   
7   import static org.hamcrest.MatcherAssert.assertThat;
8   import static org.hamcrest.Matchers.greaterThan;
9   import static org.hamcrest.Matchers.is;
10  import static org.hamcrest.Matchers.notNullValue;
11  import static org.hamcrest.Matchers.oneOf;
12  
13  import java.util.List;
14  import java.util.Map;
15  
16  import org.jspecify.annotations.Nullable;
17  import org.junit.jupiter.api.Test;
18  import org.junit.jupiter.api.condition.EnabledForJreRange;
19  import org.junit.jupiter.api.condition.EnabledOnOs;
20  import org.junit.jupiter.api.condition.JRE;
21  import org.junit.jupiter.api.condition.OS;
22  
23  import oshi.driver.common.windows.registry.ProcessPerfCounterBlock;
24  import oshi.driver.common.windows.registry.ThreadPerfCounterBlock;
25  import oshi.software.os.OSProcess;
26  
27  @EnabledForJreRange(min = JRE.JAVA_25)
28  @EnabledOnOs(OS.WINDOWS)
29  class WindowsOperatingSystemFFMTest {
30  
31      private final TestWindowsOperatingSystemFFM os = new TestWindowsOperatingSystemFFM();
32  
33      @Test
34      void testQueryBitness() {
35          assertThat("64-bit JVM bitness should be returned directly", os.queryBitnessForTest(64), is(64));
36          assertThat("32-bit JVM bitness should be 32 or 64", os.queryBitnessForTest(32), is(oneOf(32, 64)));
37      }
38  
39      @Test
40      void testQueryChildProcesses() {
41          int pid = os.getProcessId();
42          assertThat("Current process id should be positive", pid, is(greaterThan(0)));
43          List<OSProcess> childProcesses = os.queryChildProcessesForTest(pid);
44          assertThat("Child process query should not be null", childProcesses, is(notNullValue()));
45          assertThat("Child process query should include the queried process",
46                  childProcesses.stream().anyMatch(p -> p.getProcessID() == pid), is(true));
47      }
48  
49      @Test
50      void testQueryDescendantProcesses() {
51          int pid = os.getProcessId();
52          assertThat("Current process id should be positive", pid, is(greaterThan(0)));
53          List<OSProcess> descendantProcesses = os.queryDescendantProcessesForTest(pid);
54          assertThat("Descendant process query should not be null", descendantProcesses, is(notNullValue()));
55          assertThat("Descendant process query should include the queried process",
56                  descendantProcesses.stream().anyMatch(p -> p.getProcessID() == pid), is(true));
57      }
58  
59      @Test
60      void testBuildMapsFromPerfCounters() {
61          assertThat("Process map from performance counters should not be null",
62                  os.buildProcessMapFromPerfCountersForTest(), is(notNullValue()));
63          assertThat("Thread map from performance counters should not be null",
64                  os.buildThreadMapFromPerfCountersForTest(), is(notNullValue()));
65      }
66  
67      @Test
68      void testQueryParentPidMap() {
69          Map<Integer, Integer> parentPidMap = os.queryParentPidMapForTest();
70          assertThat("Parent pid map should not be null", parentPidMap, is(notNullValue()));
71          assertThat("Parent pid map should include the current process", parentPidMap.containsKey(os.getProcessId()),
72                  is(true));
73      }
74  
75      private static final class TestWindowsOperatingSystemFFM extends WindowsOperatingSystemFFM {
76          private int queryBitnessForTest(int jvmBitness) {
77              return queryBitness(jvmBitness);
78          }
79  
80          private List<OSProcess> queryChildProcessesForTest(int parentPid) {
81              return queryChildProcesses(parentPid);
82          }
83  
84          private List<OSProcess> queryDescendantProcessesForTest(int parentPid) {
85              return queryDescendantProcesses(parentPid);
86          }
87  
88          private @Nullable Map<Integer, ProcessPerfCounterBlock> buildProcessMapFromPerfCountersForTest() {
89              return buildProcessMapFromPerfCounters(null);
90          }
91  
92          private @Nullable Map<Integer, ThreadPerfCounterBlock> buildThreadMapFromPerfCountersForTest() {
93              return buildThreadMapFromPerfCounters(null);
94          }
95  
96          private Map<Integer, Integer> queryParentPidMapForTest() {
97              return queryParentPidMap();
98          }
99      }
100 }
