5   package oshi.hardware;
6   
7   import java.util.Optional;
8   
9   import oshi.annotation.PublicApi;
10  import oshi.annotation.concurrent.Immutable;
11  import oshi.util.Constants;
12  
13  /**
14   * Display refers to the information regarding a video source and monitor identified by the EDID standard.
15   * <p>
16   * {@link #getDisplayInfo()} returns a {@link DisplayInfo} exposing the display's decoded attributes — manufacturer ID,
17   * product ID, serial number, physical dimensions, preferred resolution, model name, and the raw (or synthesized) EDID
18   * byte array.
19   * <p>
20   * Example: extracting monitor manufacturer and dimensions:
21   *
22   * <pre>{@code
23   * for (Display display : hal.getDisplays()) {
24   *     DisplayInfo info = display.getDisplayInfo();
25   *     System.out.println("Manufacturer: " + info.getManufacturerID());
26   *     System.out.println("Product ID:   " + info.getProductID());
27   *     System.out.printf("Size: %d cm x %d cm%n", info.getHcm(), info.getVcm());
28   * }
29   * }</pre>
30   *
31   * For displays that report their attributes without providing an EDID (such as a built-in macOS Retina panel),
32   * {@link DisplayInfo#isEdidSynthetic()} returns {@code true} and {@link DisplayInfo#getEdid()} returns an EDID
33   * synthesized from those attributes.
34   *
35   * @see DisplayInfo
36   * @see oshi.util.EdidUtil
37   */
38  @PublicApi
39  @Immutable
40  public interface Display {
41      /**
42       * The EDID byte array.
43       *
44       * @return The EDID byte array, either reported by the display or, when {@link DisplayInfo#isEdidSynthetic()} is
45       *         {@code true}, synthesized from the display's reported attributes.
46       * @deprecated As of 7.4.0, use {@link #getDisplayInfo()}.{@link DisplayInfo#getEdid() getEdid()} instead; the
47       *             decoded attributes are also available directly from {@link DisplayInfo}. Scheduled for removal in the
48       *             next major release.
49       */
50      @Deprecated
51      byte[] getEdid();
52  
53      /**
54       * The decoded display information.
55       *
56       * @return A {@link DisplayInfo} holding the display's decoded attributes.
57       */
58      DisplayInfo getDisplayInfo();
59  
60      /**
61       * The system-level device identification for this display. The form of the identifier is platform-specific:
62       * <ul>
63       * <li>Linux: the DRM connector name from sysfs (e.g. {@code HDMI-A-1}, {@code eDP-1}, {@code DP-2}), or the
64       * {@code xrandr} output name when DRM sysfs is not available.</li>
65       * <li>macOS: on Apple Silicon, the port named by an external monitor's framebuffer {@code TransportDescription}
66       * (e.g. {@code Port-HDMI@1}, {@code Port-USB-C@1}), or the built-in panel's device tree name (e.g. {@code disp0}).
67       * Intel Macs do not expose a port.</li>
68       * <li>Windows: the connector derived from the Connecting and Configuring Displays (CCD) API's output technology and
69       * connector instance (e.g. {@code HDMI}, {@code DisplayPort-1}).</li>
70       * <li>Other UNIX platforms: the {@code xrandr} output name, which is the same value {@link #getOutputName()}
71       * returns.</li>
72       * </ul>
73       *
74       * @return The device port identifier, or {@link Constants#UNKNOWN} if not available.
75       */
76      default String getDevicePort() {
77          return Constants.UNKNOWN;
78      }
79  
80      /**
81       * The X11 output name for this display as reported by {@code xrandr} (e.g. {@code HDMI-1}, {@code DP2}). This is
82       * the name to pass to {@code xrandr --output}. Implemented on Linux and the other UNIX platforms, and only
83       * available where an X server with the RandR extension is reachable. On Linux the display is matched to an X output
84       * by its DRM {@code CONNECTOR_ID}, falling back to a comparison of their EDIDs.
85       *
86       * @return An {@link Optional} containing the xrandr output name, or empty if not available.
87       */
88      default Optional<String> getOutputName() {
89          return Optional.empty();
90      }
91  }
