7   import java.util.List;
8   import java.util.Optional;

10  import oshi.annotation.PublicApi;
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.software.os.OSFileStore;

14  /**
15   * A storage mechanism where data are recorded by various electronic, magnetic, optical, or mechanical changes to a
16   * surface layer of one or more rotating disks or or flash storage such as a removable or solid state drive. In
17   * constrast to a File System, defining the way an Operating system uses the storage, the Disk Store represents the
18   * hardware which a FileSystem uses for its File Stores.
19   * <p>
20   * Each disk store contains a list of {@link HWPartition} objects. Partitions that are mounted have a
21   * {@link HWPartition#getMountPoint()} value that can be correlated with {@link OSFileStore#getMount()} to link hardware
22   * partitions to their corresponding file stores.
23   * <p>
24   * Example: finding the disk store and partition for a given file path:
25   *
26   * <pre>{@code
27   * String filePath = "/home/user/data.txt";
28   * for (HWDiskStore disk : hal.getDiskStores()) {
29   *     for (HWPartition part : disk.getPartitions()) {
30   *         if (!part.getMountPoint().isEmpty() && filePath.startsWith(part.getMountPoint())) {
31   *             System.out.printf("File is on disk %s, partition %s%n", disk.getName(), part.getName());
32   *         }
33   *     }
34   * }
35   * }</pre>
36   *
37   * Thread safe for the designed use of retrieving the most recent data. Users should be aware that the
38   * {@link #updateAttributes()} method may update attributes, including the time stamp, and should externally synchronize
39   * such usage to ensure consistent calculations.
40   *
41   * @see HWPartition
42   * @see OSFileStore
43   */
44  @PublicApi
45  @ThreadSafe
46  public interface HWDiskStore {
47  
48      /**
49       * The disk name
50       *
51       * @return the name
52       */
53      String getName();
54  
55      /**
56       * The type of the disk's media.
57       * <p>
58       * Determined on Linux from the device's {@code removable} and {@code queue/rotational} sysfs attributes, on macOS
59       * from the IOKit registry's {@code Removable} property and {@code Medium Type}, and on Windows from the storage
60       * driver's removable-media flag, bus type and seek penalty, falling back to {@code MSFT_PhysicalDisk}'s
61       * {@code MediaType} and {@code SpindleSpeed}, and on FreeBSD from the {@code rotationrate} the drive reports to
62       * {@code geom}. Other platforms always report {@code Unknown}.
63       * <p>
64       * {@code Virtual} is reported only for a Linux device-mapper device, a macOS APFS container or disk image, and a
65       * Windows file-backed virtual or Storage Spaces disk. A hypervisor's disk that the guest sees as an ordinary device
66       * is reported by its media technology, not as {@code Virtual}. Whether a disk duplicates the capacity and I/O of
67       * another entry in the list is reported by {@link #getBackingStores()}, not by this type.
68       *
69       * @return one of {@code SSD}, {@code HDD}, {@code Removable}, {@code Virtual}, or {@code Unknown} where the type
70       *         could not be determined, which includes a disk whose driver does not report it
71       */
72      default String getDiskType() {
73          return "Unknown";
74      }
75  
76      /**
77       * The disk model
78       *
79       * @return the model
80       */
81      String getModel();

83      /**
84       * The disk serial number, if available.
85       *
86       * @return the serial number
87       */
88      String getSerial();

90      /**
91       * The size of the disk
92       *
93       * @return the disk size, in bytes
94       */
95      long getSize();

105      * The number of bytes read from the disk

107      * @return the bytes read
108      */
109     long getReadBytes();
110 
111     /**
112      * The number of writes to the disk
113      *
114      * @return the writes

116     long getWrites();

119      * The number of bytes written to the disk
120      *
121      * @return the bytes written

125     /**
126      * The length of the disk queue (#I/O's in progress). Includes I/O requests that have been issued to the device

134      * The time spent reading or writing, in milliseconds.

136      * @return the transfer time
137      */
138     long getTransferTime();

140     /**
141      * The partitions on this disk.
142      *
143      * @return an {@code UnmodifiableList} of the partitions on this drive.
144      */
145     List<HWPartition> getPartitions();

147     /**
148      * The stores this disk is built on, when it is a layer over other storage rather than a device in its own right.
149      * <p>
150      * A layered disk is listed alongside the stores it is built on, so its capacity and I/O statistics duplicate
151      * theirs. This is the case for a device-mapper (LVM, LUKS, multipath) or md RAID device on Linux, an APFS container
152      * or a disk image on macOS, and a Storage Spaces or file-backed virtual disk on Windows.
153      * <p>
154      * An empty {@link Optional} means the backing stores are not reported, either because the platform does not report
155      * them or because they could not be read, and says nothing about whether the disk is layered. A present but empty
156      * list means the disk is not built on anything OSHI can see. Summing the I/O of only the disks whose result is
157      * present and empty therefore counts each transfer once, provided every disk's result is present and every store it
158      * names is itself listed. Windows does not list the physical disks of a Storage Spaces pool, so there the Storage
159      * Spaces disk is the only entry carrying their transfers.
160      * <p>
161      * Backing stores are reported on Linux, from each block device's sysfs {@code slaves} directory; on macOS, from the
162      * IOKit registry; and on Windows 8 and Server 2012 or later, from the Storage namespace of WMI. Other platforms
163      * return an empty {@link Optional}.
164      * <p>
165      * Each entry is a {@link StoreKind} prefix followed by a name that matches a getter on the corresponding class:
166      * <table>
167      * <caption>Backing store entries</caption>
168      * <tr>
169      * <th>Kind</th>
170      * <th>Name matches</th>
171      * <th>Examples</th>
172      * </tr>
173      * <tr>
174      * <td>{@link StoreKind#DISK}</td>
175      * <td>{@link HWDiskStore#getName()}</td>
176      * <td>{@code disk:/dev/sda}, {@code disk:\\.\PHYSICALDRIVE1}</td>
177      * </tr>
178      * <tr>
179      * <td>{@link StoreKind#PARTITION}</td>
180      * <td>{@link HWPartition#getIdentification()}</td>
181      * <td>{@code partition:/dev/sda2}, {@code partition:disk0s2}</td>
182      * </tr>
183      * <tr>
184      * <td>{@link StoreKind#FILE}</td>
185      * <td>{@link OSFileStore#getMount()} of the file store holding the backing file</td>
186      * <td>{@code file:/System/Volumes/Data}, {@code file:C:\}</td>
187      * </tr>
188      * </table>
189      * An entry may name a store that is not in the current list, such as a network file store when only local file
190      * stores were requested.
191      *
192      * @return An {@link Optional} containing the stores this disk is built on, sorted, which is an empty list for a
193      *         disk that is not layered; or empty if the backing stores are not reported.
194      */
195     default Optional<List<String>> getBackingStores() {
196         return Optional.empty();
197     }
198 
199     /**
200      * The time this disk's statistics were updated.
201      *
202      * @return the timeStamp, in milliseconds since the epoch.
203      */
204     long getTimeStamp();
205 
206     /**
207      * Make a best effort to update all the statistics about the drive without needing to recreate the drive list. This
208      * method provides for more frequent periodic updates of individual drive statistics but may be less efficient to
209      * use if updating all drives. It will not detect if a removable drive has been removed and replaced by a different
210      * drive in between method calls.
211      *
212      * @return True if the update was (probably) successful, false if the disk was not found
213      */
214     boolean updateAttributes();
215 
216     /**
217      * The kind of store named by an entry in {@link #getBackingStores()}, each identified by a prefix on the entry.
218      */
219     @PublicApi
220     enum StoreKind {
221         /**
222          * A disk store, named by {@link HWDiskStore#getName()}.
223          */
224         DISK("disk:"),
225         /**
226          * A partition, named by {@link HWPartition#getIdentification()}.
227          */
228         PARTITION("partition:"),
229         /**
230          * A file store, named by {@link OSFileStore#getMount()}.
231          */
232         FILE("file:");
233 
234         private final String prefix;
235 
236         StoreKind(String prefix) {
237             this.prefix = prefix;
238         }
239 
240         /**
241          * Gets the prefix that identifies this kind of entry, including its trailing colon.
242          *
243          * @return the prefix, such as {@code disk:}
244          */
245         public String getPrefix() {
246             return prefix;
247         }
248 
249         /**
250          * Creates an entry of this kind.
251          *
252          * @param name the name of the store, as returned by the getter this kind matches
253          * @return the entry, such as {@code disk:/dev/sda}
254          */
255         public String entry(String name) {
256             return prefix + name;
257         }
258 
259         /**
260          * Tests whether an entry is of this kind.
261          *
262          * @param entry an entry from {@link #getBackingStores()}
263          * @return true if the entry starts with this kind's prefix
264          */
265         public boolean matches(String entry) {
266             return entry.startsWith(prefix);
267         }
268 
269         /**
270          * Gets the name of the store an entry of this kind refers to.
271          *
272          * @param entry an entry from {@link #getBackingStores()}
273          * @return the name following the prefix, or an empty string if the entry is not of this kind
274          */
275         public String nameOf(String entry) {
276             return matches(entry) ? entry.substring(prefix.length()) : "";
277         }
278     }
