5   package oshi.hardware.platform.unix.dragonflybsd;
6   
7   import java.util.List;
8   
9   import com.sun.jna.Memory;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.hardware.CentralProcessor;
13  import oshi.hardware.HWDiskStore;
14  import oshi.hardware.common.platform.unix.dragonflybsd.DragonFlyBsdHWDiskStore;
15  import oshi.hardware.platform.unix.freebsd.FreeBsdHardwareAbstractionLayerJNA;
16  import oshi.util.platform.unix.freebsd.BsdSysctlUtil;
17  
18  /**
19   * DragonFly BSD hardware abstraction layer. Every component behaves as FreeBSD's does except two, so this inherits the
20   * FreeBSD wiring and replaces those: the processor, which reads its tick counters from {@code kern.cputime} rather than
21   * {@code kern.cp_time}, and the disks, whose statistics come from {@code kern.devstat.all} because DragonFly has no
22   * {@code geom} and no {@code iostat -x}.
23   */
24  @ThreadSafe
25  public final class DragonFlyBsdHardwareAbstractionLayerJNA extends FreeBsdHardwareAbstractionLayerJNA {
26  
27      @Override
28      public CentralProcessor createProcessor() {
29          return new DragonFlyBsdCentralProcessorJNA();
30      }
31  
32      @Override
33      public List<HWDiskStore> getDiskStores() {
34          return DragonFlyBsdHWDiskStore.getDisks(BsdSysctlUtil.sysctl("kern.disks", ""),
35                  DragonFlyBsdHardwareAbstractionLayerJNA::queryDevstat);
36      }
37  
38      private static byte[] queryDevstat() {
39          try (Memory m = BsdSysctlUtil.sysctl(DragonFlyBsdHWDiskStore.DEVSTAT_SYSCTL)) {
40              return m == null ? new byte[0] : m.getByteArray(0, (int) m.size());
41          }
42      }
43  }
