5   package oshi.hardware.platform.unix.netbsd;
6   
7   import static oshi.jna.platform.unix.NetBsdLibc.CTL_KERN;
8   import static oshi.jna.platform.unix.NetBsdLibc.KERN_CP_TIME;
9   
10  import java.util.Arrays;
11  
12  import com.sun.jna.Memory;
13  
14  import oshi.annotation.concurrent.ThreadSafe;
15  import oshi.hardware.common.platform.unix.netbsd.NetBsdCentralProcessor;
16  import oshi.jna.platform.unix.NetBsdLibc;
17  import oshi.util.platform.unix.netbsd.NetBsdSysctlUtil;
18  
19  /**
20   * A CentralProcessor for NetBSD that uses JNA native sysctl calls when the JNA native library is available, falling
21   * back to the command-line {@link NetBsdCentralProcessor} implementation otherwise.
22   */
23  @ThreadSafe
24  public class NetBsdCentralProcessorJNA extends NetBsdCentralProcessor {
25  
26      @Override
27      protected long[] queryCpTime(int cpu) {
28          if (!NetBsdSysctlUtil.JNA_AVAILABLE) {
29              return super.queryCpTime(cpu);
30          }
31          // On NetBSD, kern.cp_time with a third MIB element gives per-CPU data
32          int[] mib = cpu < 0 ? new int[] { CTL_KERN, KERN_CP_TIME } : new int[] { CTL_KERN, KERN_CP_TIME, cpu };
33          try (Memory m = NetBsdSysctlUtil.sysctl(mib)) {
34              if (m == null) {
35                  return new long[0];
36              }
37              return m.getLongArray(0, (int) (m.size() / Long.BYTES));
38          }
39      }
40  
41      @Override
42      protected double[] queryLoadAverage(int nelem) {
43          if (!NetBsdSysctlUtil.JNA_AVAILABLE) {
44              return super.queryLoadAverage(nelem);
45          }
46          double[] average = new double[nelem];
47          int retval = NetBsdLibc.INSTANCE.getloadavg(average, nelem);
48          if (retval < nelem) {
49              Arrays.fill(average, -1d);
50          }
51          return average;
52      }
53  }
