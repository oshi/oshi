10  import static oshi.ffm.platform.unix.netbsd.NetBsdLibcFunctions.CTL_KERN;
11  import static oshi.ffm.platform.unix.netbsd.NetBsdLibcFunctions.KERN_CP_TIME;
12  import static oshi.util.LogLevel.WARN;
13  
14  import java.lang.foreign.MemorySegment;
15  import java.util.Arrays;
16  
17  import org.slf4j.Logger;
18  import org.slf4j.LoggerFactory;
19  
20  import oshi.annotation.concurrent.ThreadSafe;
21  import oshi.ffm.platform.unix.netbsd.NetBsdLibcFunctions;
22  import oshi.ffm.util.platform.unix.netbsd.NetBsdSysctlUtilFFM;
23  import oshi.hardware.common.platform.unix.netbsd.NetBsdCentralProcessor;

25  /**
26   * A CentralProcessor for NetBSD that uses FFM native sysctl calls when FFM native access is available, falling back to
27   * the command-line {@link NetBsdCentralProcessor} implementation otherwise.
28   */
29  @ThreadSafe
30  public class NetBsdCentralProcessorFFM extends NetBsdCentralProcessor {
31  
32      private static final Logger LOG = LoggerFactory.getLogger(NetBsdCentralProcessorFFM.class);
33  
34      @Override
35      protected long[] queryCpTime(int cpu) {
36          if (!NetBsdSysctlUtilFFM.FFM_AVAILABLE) {
37              return super.queryCpTime(cpu);
38          }
39          // On NetBSD, kern.cp_time with a third MIB element gives per-CPU data
40          int[] mib = cpu < 0 ? new int[] { CTL_KERN, KERN_CP_TIME } : new int[] { CTL_KERN, KERN_CP_TIME, cpu };
41          MemorySegment m = NetBsdSysctlUtilFFM.sysctl(mib);
42          return m == null ? new long[0] : m.asSlice(0, m.byteSize() / Long.BYTES * Long.BYTES).toArray(JAVA_LONG);
43      }
44  
45      @Override
46      protected double[] queryLoadAverage(int nelem) {
47          if (!NetBsdSysctlUtilFFM.FFM_AVAILABLE) {
48              return super.queryLoadAverage(nelem);
49          }
50          double[] failed = new double[nelem];
51          Arrays.fill(failed, -1d);
52          return callInArenaOrDefault(arena -> {
53              MemorySegment loadavg = arena.allocate(JAVA_DOUBLE, nelem);
54              if (NetBsdLibcFunctions.getloadavg(loadavg, nelem) < nelem) {
55                  return failed;
56              }
57              return loadavg.toArray(JAVA_DOUBLE);
58          }, LOG, WARN, "Failed getloadavg", failed);
59      }
60  }
