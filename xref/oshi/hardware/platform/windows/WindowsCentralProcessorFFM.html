71      @Override
72      protected ProcessorIdentifier queryProcessorId() {
73          String cpuVendor = "";
74          String cpuName = "";
75          String cpuIdentifier = "";
76          String cpuFamily = "";
77          String cpuModel = "";
78          String cpuStepping = "";
79          long cpuVendorFreq = 0L;
80          String processorID;
81          boolean cpu64bit = false;
82  
83          final String cpuRegistryRoot = "HARDWARE\\DESCRIPTION\\System\\CentralProcessor\\";
84          try (Arena arena = Arena.ofConfined()) {
85              MemorySegment hKey = arena.allocate(ValueLayout.ADDRESS);
86              MemorySegment subKey = WindowsForeignFunctions.toWideString(arena, cpuRegistryRoot);
87              if (Advapi32FFM.RegOpenKeyEx(HKLM, subKey, 0, WinNTFFM.KEY_READ, hKey) == 0) {
88                  try (NativeHandle key = NativeHandle.of(hKey.get(ValueLayout.ADDRESS, 0), Advapi32FFM::RegCloseKey)) {
89                      // Get first subkey name
90                      MemorySegment nameBuffer = arena.allocate(256 * 2L); // WCHAR[256]
91                      MemorySegment nameLen = arena.allocate(ValueLayout.JAVA_INT);
92                      nameLen.set(ValueLayout.JAVA_INT, 0, 256);
93                      if (Advapi32FFM.RegEnumKeyEx(key.get(), 0, nameBuffer, nameLen, MemorySegment.NULL,
94                              MemorySegment.NULL, MemorySegment.NULL, MemorySegment.NULL) == 0) {
95                          String firstKey = WindowsForeignFunctions.readWideString(nameBuffer);
96                          String cpuRegistryPath = cpuRegistryRoot + firstKey;
97                          cpuVendor = registryGetString(arena, cpuRegistryPath, "VendorIdentifier");
98                          cpuName = registryGetString(arena, cpuRegistryPath, "ProcessorNameString");
99                          cpuIdentifier = registryGetString(arena, cpuRegistryPath, "Identifier");
100                         cpuVendorFreq = registryGetDword(arena, cpuRegistryPath, "~MHz") * 1_000_000L;
101                     }
102                 }
103             }
104         } catch (Throwable t) {
105             LOG.debug("Failed to read processor registry info", t);
106         }
107 
108         if (!cpuIdentifier.isEmpty()) {
109             cpuFamily = parseIdentifier(cpuIdentifier, "Family");
110             cpuModel = parseIdentifier(cpuIdentifier, "Model");
111             cpuStepping = parseIdentifier(cpuIdentifier, "Stepping");
112         }
113 
114         // GetNativeSystemInfo to determine 64-bit
115         try (Arena arena = Arena.ofConfined()) {
116             // SYSTEM_INFO is 48 bytes on x64
117             MemorySegment sysInfo = arena.allocate(48);
118             if (Kernel32FFM.GetNativeSystemInfo(sysInfo)) {
119                 // wProcessorArchitecture is at offset 0 (WORD in union)
120                 int arch = Short.toUnsignedInt(sysInfo.get(ValueLayout.JAVA_SHORT, 0));
121                 if (arch == 9 || arch == 12 || arch == 6) { // AMD64, ARM64, IA64
122                     cpu64bit = true;
123                 }
124             }
125         }
126 
127         var processorIdResult = Win32ProcessorFFM.queryProcessorId();
128         if (processorIdResult.getResultCount() > 0) {
129             processorID = WmiUtil.getString(processorIdResult, ProcessorIdProperty.PROCESSORID, 0);
130         } else {
131             processorID = createProcessorID(cpuStepping, cpuModel, cpuFamily,
132                     cpu64bit ? new String[] { "ia64" } : new String[0]);
133         }
134         return new ProcessorIdentifier(cpuVendor, cpuName, cpuFamily, cpuModel, cpuStepping, processorID, cpu64bit,
135                 cpuVendorFreq);
136     }

148     @Override
149     public long[] querySystemCpuLoadTicks() {
150         long[] ticks = new long[TickType.values().length];
151         if (useLegacySystemCounters()) {
152             try (Arena arena = Arena.ofConfined()) {
153                 // FILETIME is 8 bytes (dwLowDateTime + dwHighDateTime)
154                 MemorySegment lpIdleTime = arena.allocate(8);
155                 MemorySegment lpKernelTime = arena.allocate(8);
156                 MemorySegment lpUserTime = arena.allocate(8);
157                 if (!Kernel32FFM.GetSystemTimes(lpIdleTime, lpKernelTime, lpUserTime)) {
158                     LOG.error("Failed to update system idle/kernel/user times. Error code: {}",
159                             Kernel32FFM.GetLastError().orElse(-1));
160                     return ticks;
161                 }
162 
163                 Map<SystemTickCountProperty, Long> valueMap = ProcessorInformationFFM.querySystemCounters();
164                 ticks[TickType.IRQ.getIndex()] = valueMap.getOrDefault(SystemTickCountProperty.PERCENTINTERRUPTTIME, 0L)
165                         / 10_000L;
166                 ticks[TickType.SOFTIRQ.getIndex()] = valueMap.getOrDefault(SystemTickCountProperty.PERCENTDPCTIME, 0L)
167                         / 10_000L;
168 
169                 ticks[TickType.IDLE.getIndex()] = lpIdleTime.get(ValueLayout.JAVA_LONG, 0) / 10_000L;
170                 ticks[TickType.SYSTEM.getIndex()] = lpKernelTime.get(ValueLayout.JAVA_LONG, 0) / 10_000L
171                         - ticks[TickType.IDLE.getIndex()];
172                 ticks[TickType.USER.getIndex()] = lpUserTime.get(ValueLayout.JAVA_LONG, 0) / 10_000L;
173                 ticks[TickType.SYSTEM.getIndex()] -= ticks[TickType.IRQ.getIndex()]
174                         + ticks[TickType.SOFTIRQ.getIndex()];
175             }
176             return ticks;
177         }
178         long[][] procTicks = getProcessorCpuLoadTicks();
179         for (int i = 0; i < ticks.length; i++) {
180             for (long[] procTick : procTicks) {
181                 ticks[i] += procTick[i];
182             }
183         }
184         return ticks;
185     }

197     @Override
198     protected long[] queryNTPower(int fieldIndex) {
199         long[] freqs = new long[getLogicalProcessorCount()];
200         int totalSize = PPI_SIZE * freqs.length;
201         try (Arena arena = Arena.ofConfined()) {
202             MemorySegment buffer = arena.allocate(totalSize);
203             int status = Kernel32FFM.CallNtPowerInformation(PROCESSOR_INFORMATION_LEVEL, MemorySegment.NULL, 0, buffer,
204                     totalSize);
205             if (status != 0) {
206                 LOG.error("Unable to get Processor Information, status: {}", status);
207                 Arrays.fill(freqs, -1L);
208                 return freqs;
209             }
210             for (int i = 0; i < freqs.length; i++) {
211                 long offset = i * (long) PPI_SIZE;
212                 // ProcessorPowerInformation: number(4), maxMhz(4), currentMhz(4), mhzLimit(4), maxIdleState(4),
213                 // currentIdleState(4)
214                 freqs[i] = switch (fieldIndex) {
215                     case 1 -> Integer.toUnsignedLong(buffer.get(ValueLayout.JAVA_INT, offset + 4)) * 1_000_000L;
216                     case 2 -> Integer.toUnsignedLong(buffer.get(ValueLayout.JAVA_INT, offset + 8)) * 1_000_000L;
217                     default -> -1L;
218                 };
219                 if (freqs[i] == 0) {
220                     freqs[i] = getProcessorIdentifier().getVendorFreq();
221                 }
222             }
223         }
224         return freqs;
225     }

256     private static String registryGetString(Arena arena, String path, String valueName) {
257         return getOrDefault(() -> {
258             MemorySegment hKey = arena.allocate(ValueLayout.ADDRESS);
259             MemorySegment subKey = WindowsForeignFunctions.toWideString(arena, path);
260             if (Advapi32FFM.RegOpenKeyEx(HKLM, subKey, 0, WinNTFFM.KEY_READ, hKey) == 0) {
261                 try (NativeHandle key = NativeHandle.of(hKey.get(ValueLayout.ADDRESS, 0), Advapi32FFM::RegCloseKey)) {
262                     MemorySegment valueNameSeg = WindowsForeignFunctions.toWideString(arena, valueName);
263                     MemorySegment dataSize = arena.allocate(ValueLayout.JAVA_INT);
264                     dataSize.set(ValueLayout.JAVA_INT, 0, 512);
265                     MemorySegment data = arena.allocate(512);
266                     if (Advapi32FFM.RegQueryValueEx(key.get(), valueNameSeg, MemorySegment.NULL, data, dataSize) == 0) {
267                         return WindowsForeignFunctions.readWideString(data);
268                     }
269                 }
270             }
271             return "";
272         }, "", LOG, "Failed to read registry string {}/{}", path, valueName);
273     }

275     private static long registryGetDword(Arena arena, String path, String valueName) {
276         return getLongOrDefault(() -> {
277             MemorySegment hKey = arena.allocate(ValueLayout.ADDRESS);
278             MemorySegment subKey = WindowsForeignFunctions.toWideString(arena, path);
279             if (Advapi32FFM.RegOpenKeyEx(HKLM, subKey, 0, WinNTFFM.KEY_READ, hKey) == 0) {
280                 try (NativeHandle key = NativeHandle.of(hKey.get(ValueLayout.ADDRESS, 0), Advapi32FFM::RegCloseKey)) {
281                     MemorySegment valueNameSeg = WindowsForeignFunctions.toWideString(arena, valueName);
282                     MemorySegment dataSize = arena.allocate(ValueLayout.JAVA_INT);
283                     dataSize.set(ValueLayout.JAVA_INT, 0, 4);
284                     MemorySegment data = arena.allocate(4);
285                     if (Advapi32FFM.RegQueryValueEx(key.get(), valueNameSeg, MemorySegment.NULL, data, dataSize) == 0) {
286                         return Integer.toUnsignedLong(data.get(ValueLayout.JAVA_INT, 0));
287                     }
288                 }
289             }
290             return 0L;
291         }, 0L, LOG, "Failed to read registry DWORD {}/{}", path, valueName);
