25  final class WindowsLogicalVolumeGroupFFM extends WindowsLogicalVolumeGroup {
26  
27      private static final Logger LOG = LoggerFactory.getLogger(WindowsLogicalVolumeGroupFFM.class);
28  
29      private static final boolean IS_WINDOWS8_OR_GREATER = VersionHelpersFFM.IsWindows8OrGreater();
30  
31      WindowsLogicalVolumeGroupFFM(String name, Map<String, Set<String>> lvMap, Set<String> pvSet) {
32          super(name, lvMap, pvSet);
33      }
34  
35      static List<LogicalVolumeGroup> getLogicalVolumeGroups() {
36          // Storage Spaces requires Windows 8 or Server 2012
37          if (!IS_WINDOWS8_OR_GREATER) {
38              return Collections.emptyList();
39          }
40          WmiQueryHandlerFFM h = Objects.requireNonNull(WmiQueryHandlerFFM.createInstance());
41          boolean comInit = false;
42          try {
43              comInit = h.initCOM();
44              // Query Storage Pools first, so we can skip other queries if we have no pools
45              WmiResult<StoragePoolProperty> sp = MSFTStorage.queryStoragePools(h);
46              if (sp.getResultCount() == 0) {
47                  return Collections.emptyList();
48              }
49              return buildFromWmi(sp, MSFTStorage.queryVirtualDisks(h), MSFTStorage.queryPhysicalDisks(h),
50                      MSFTStorage.queryStoragePoolPhysicalDisks(h), WindowsLogicalVolumeGroupFFM::new);
51          } catch (FfmComException e) {
52              LOG.warn("COM exception", e);
53              return Collections.emptyList();
54          } finally {
55              if (comInit) {
56                  h.unInitCOM();
57              }
58          }
59      }
60  }
