136     @Override
137     public synchronized boolean updateAttributes() {
138         String index = null;
139         List<HWPartition> partitions = getPartitions();
140         if (!partitions.isEmpty()) {
141             index = Integer.toString(partitions.get(0).getMajor());
142         } else if (getName().startsWith(PHYSICALDRIVE_PREFIX)) {
143             index = getName().substring(PHYSICALDRIVE_PREFIX.length());
144         } else {
145             LOG.warn("Couldn't match index for {}", getName());
146             return false;
147         }
148         DiskStats stats = queryReadWriteStats(index);
149         if (stats.getReadMap().containsKey(index)) {
150             setDiskStats(stats, index);
151             return true;
152         } else {
153             return false;
154         }
155     }

166     /**
167      * Populates a DiskStats object from performance counter data.
168      *
169      * @param index             The disk index to filter on, or null for all
170      * @param instanceValuePair The raw PDH counter data
171      * @return A populated DiskStats object
172      */
173     protected static DiskStats populateDiskStats(@Nullable String index,
174             Pair<List<String>, Map<PhysicalDiskProperty, List<Long>>> instanceValuePair) {
175         DiskStats stats = new DiskStats();
176         List<String> instances = instanceValuePair.getA();
177         Map<PhysicalDiskProperty, List<Long>> valueMap = instanceValuePair.getB();
178         stats.setTimeStamp(System.currentTimeMillis());
179         List<Long> readList = valueMap.get(PhysicalDiskProperty.DISKREADSPERSEC);
180         List<Long> readByteList = valueMap.get(PhysicalDiskProperty.DISKREADBYTESPERSEC);
181         List<Long> writeList = valueMap.get(PhysicalDiskProperty.DISKWRITESPERSEC);
182         List<Long> writeByteList = valueMap.get(PhysicalDiskProperty.DISKWRITEBYTESPERSEC);
183         List<Long> queueLengthList = valueMap.get(PhysicalDiskProperty.CURRENTDISKQUEUELENGTH);
184         List<Long> diskTimeList = valueMap.get(PhysicalDiskProperty.PERCENTDISKTIME);
185 
186         if (instances.isEmpty() || readList == null || readByteList == null || writeList == null
187                 || writeByteList == null || queueLengthList == null || diskTimeList == null) {
188             LOG.debug("Empty disk stats for index {}: instances={}, counters null: r={} rb={} w={} wb={} ql={} dt={}",
189                     index, instances.isEmpty(), readList == null, readByteList == null, writeList == null,
190                     writeByteList == null, queueLengthList == null, diskTimeList == null);
191             return stats;
192         }
193         for (int i = 0; i < instances.size(); i++) {
194             String name = getIndexFromName(instances.get(i));
195             if (index != null && !index.equals(name)) {
196                 continue;
197             }
198             stats.getReadMap().put(name, readList.get(i));
199             stats.getReadByteMap().put(name, readByteList.get(i));
200             stats.getWriteMap().put(name, writeList.get(i));
201             stats.getWriteByteMap().put(name, writeByteList.get(i));
202             stats.getQueueLengthMap().put(name, queueLengthList.get(i));
203             stats.getDiskTimeMap().put(name, diskTimeList.get(i) / 10_000L);
204         }
205         return stats;
206     }

208     /**
209      * Populates the drive-to-partition map from WMI association data.
210      *
211      * @param maps       The PartitionMaps to populate
212      * @param antecedent The antecedent device ID string
213      * @param dependent  The dependent device ID string
214      */
215     protected static void mapDriveToPartition(PartitionMaps maps, String antecedent, String dependent) {
216         Matcher mAnt = DEVICE_ID.matcher(antecedent);
217         Matcher mDep = DEVICE_ID.matcher(dependent);
218         if (mAnt.matches() && mDep.matches()) {
219             maps.getDriveToPartitionMap().computeIfAbsent(mAnt.group(1).replace("\\\\", "\\"), x -> new ArrayList<>())
220                     .add(mDep.group(1));
221         }
222     }

224     /**
225      * Populates the partition-to-logical-drive map from WMI association data.
226      *
227      * @param maps       The PartitionMaps to populate
228      * @param antecedent The antecedent device ID string
229      * @param dependent  The dependent device ID string
230      * @param size       The partition size
231      */
232     protected static void mapPartitionToLogicalDrive(PartitionMaps maps, String antecedent, String dependent,
233             long size) {
234         Matcher mAnt = DEVICE_ID.matcher(antecedent);
235         Matcher mDep = DEVICE_ID.matcher(dependent);
236         if (mAnt.matches() && mDep.matches()) {
237             maps.getPartitionToLogicalDriveMap().computeIfAbsent(mAnt.group(1), x -> new ArrayList<>())
238                     .add(new Pair<>(mDep.group(1) + "\\", size));
239         }
240     }

242     /**
243      * Builds an unmodifiable sorted partition list from the maps.
244      *
245      * @param maps     The partition maps
246      * @param diskName The disk name to look up
247      * @return A sorted unmodifiable list of partitions
248      */
249     protected static List<HWPartition> buildPartitionList(PartitionMaps maps, String diskName) {
250         List<HWPartition> partitions = new ArrayList<>();
251         List<String> partList = maps.getDriveToPartitionMap().get(diskName);
252         if (partList != null && !partList.isEmpty()) {
253             for (String part : partList) {
254                 if (maps.getPartitionMap().containsKey(part)) {
255                     partitions.addAll(maps.getPartitionMap().get(part));
256                 }
257             }
258         }
259         return Collections.unmodifiableList(
260                 partitions.stream().sorted(Comparator.comparing(HWPartition::getName)).collect(Collectors.toList()));
261     }

402     /**
403      * Parses the {@code Win32_DiskDrive} MediaType string into a disk type.
404      * <p>
405      * This property is mapped from {@code DISK_GEOMETRY.MediaType} and describes only removability and form factor; its
406      * entire value domain is "Fixed hard disk media", "Removable media other than floppy", "External hard disk media"
407      * and "Format is unknown". It therefore cannot distinguish an SSD from an HDD, and a fixed or external disk is
408      * reported as {@code Unknown} rather than guessed at. Use {@link #classifyDiskType(StorageProperties, String)} to
409      * reach the media technology.
410      *
411      * @param mediaType the WMI MediaType value from {@code Win32_DiskDrive}
412      * @return {@code Removable} for removable media, otherwise {@code Unknown}
413      */
414     protected static String parseWindowsMediaType(@Nullable String mediaType) {
415         if (mediaType != null && mediaType.toLowerCase(Locale.ROOT).contains("removable")) {
416             return DISK_TYPE_REMOVABLE;
417         }
418         return DISK_TYPE_UNKNOWN;
419     }
420 
421     /**
422      * Classifies a disk from the storage driver's response, falling back to the WMI media type string when the driver
423      * did not answer.
424      *
425      * @param props     the storage properties read from the device, or {@code null} if the device could not be opened
426      *                  or did not answer
427      * @param mediaType the WMI MediaType value from {@code Win32_DiskDrive}
428      * @return {@code SSD}, {@code HDD}, {@code Removable}, {@code Virtual}, or {@code Unknown}
429      */
430     protected static String classifyDiskType(@Nullable StorageProperties props, @Nullable String mediaType) {
431         if (props == null) {
432             // The device did not answer at all, so the WMI string is the only source left. It identifies
433             // removable media and nothing else.
434             return parseWindowsMediaType(mediaType);
435         }
436         // Checked before the media technology to match the Linux implementation, where a removable
437         // device's rotational attribute is likewise not consulted.
438         if (props.isRemovableMedia()) {
439             return DISK_TYPE_REMOVABLE;
440         }
441         if (props.getBusType() == BUS_TYPE_FILE_BACKED_VIRTUAL || props.getBusType() == BUS_TYPE_SPACES) {
442             return DISK_TYPE_VIRTUAL;
443         }
444         Boolean seekPenalty = props.getIncursSeekPenalty();
445         if (seekPenalty != null) {
446             // A device with no seek penalty is solid state. A RAM disk also answers no here, which
447             // matches what Linux reports for one from queue/rotational.
448             return seekPenalty.booleanValue() ? DISK_TYPE_HDD : DISK_TYPE_SSD;
449         }
450         // The descriptor was read but the driver does not implement the seek penalty property. The WMI
451         // string is not consulted here: the descriptor already answered the only question it can.
452         return DISK_TYPE_UNKNOWN;
453     }

455     /**
456      * Converts an {@code MSFT_PhysicalDisk} MediaType, and the SpindleSpeed that accompanies it, into a disk type.
457      *
458      * @param mediaType    the MediaType value: 3 is HDD, 4 is SSD, 5 is storage class memory, 0 is unspecified
459      * @param spindleSpeed the SpindleSpeed in RPM, as an unsigned value. {@code UINT32_MAX} means rotating at an
460      *                     unknown speed.
461      * @return {@code SSD}, {@code HDD}, or {@code Unknown}
462      */
463     protected static String diskTypeFromMsftMediaType(int mediaType, long spindleSpeed) {
464         switch (mediaType) {
465             case MSFT_MEDIA_TYPE_HDD:
466                 return DISK_TYPE_HDD;
467             case MSFT_MEDIA_TYPE_SSD:
468             case MSFT_MEDIA_TYPE_SCM:
469                 return DISK_TYPE_SSD;
470             default:
471                 // For an unspecified MediaType a nonzero SpindleSpeed still identifies rotating media. A zero
472                 // is not read as solid state, because that is also what an absent property reads as.
473                 return spindleSpeed > 0L ? DISK_TYPE_HDD : DISK_TYPE_UNKNOWN;
474         }
475     }

477     /**
478      * Maps each {@code MSFT_PhysicalDisk} DeviceId to a disk type. The DeviceId of a directly attached disk is its
479      * physical drive number, matching {@code Win32_DiskDrive}'s Index.
480      *
481      * @param pd the physical disk query result
482      * @return a map of DeviceId to disk type, omitting disks whose type could not be determined
483      */
484     protected static Map<String, String> buildMsftDiskTypeMap(WmiResult<PhysicalDiskMediaTypeProperty> pd) {
485         Map<String, String> diskTypeMap = new HashMap<>();
486         for (int i = 0; i < pd.getResultCount(); i++) {
487             String deviceId = WmiUtil.getString(pd, PhysicalDiskMediaTypeProperty.DEVICEID, i);
488             String diskType = diskTypeFromMsftMediaType(
489                     WmiUtil.getUint16(pd, PhysicalDiskMediaTypeProperty.MEDIATYPE, i),
490                     WmiUtil.getUint32asLong(pd, PhysicalDiskMediaTypeProperty.SPINDLESPEED, i));
491             if (!deviceId.isEmpty() && !DISK_TYPE_UNKNOWN.equals(diskType)) {
492                 diskTypeMap.put(deviceId, diskType);
493             }
494         }
495         return diskTypeMap;
496     }

498     /**
499      * Classifies every disk in a {@code Win32_DiskDrive} query result from the storage driver.
500      *
501      * @param vals     the disk drive query result
502      * @param resolver reads a physical drive's storage properties, which requires the binding's native calls
503      * @return a disk type for each result, in query order
504      */
505     protected static String[] classifyDiskTypes(WmiResult<DiskDriveProperty> vals, StoragePropertiesResolver resolver) {
506         int count = vals.getResultCount();
507         String[] diskTypes = new String[count];
508         for (int i = 0; i < count; i++) {
509             diskTypes[i] = classifyDiskType(resolver.resolve(WmiUtil.getString(vals, DiskDriveProperty.NAME, i)),
510                     WmiUtil.getString(vals, DiskDriveProperty.MEDIATYPE, i));
511         }
512         return diskTypes;
513     }

515     /**
516      * Tests whether any disk type is still undetermined, and so whether the Storage namespace fallback is worth its
517      * cost. On hardware whose driver answers the seek penalty query this is {@code false} and the fallback is skipped.
518      *
519      * @param diskTypes the disk types from {@link #classifyDiskTypes(WmiResult, StoragePropertiesResolver)}
520      * @return {@code true} if at least one disk type is {@code Unknown}
521      */
522     protected static boolean anyDiskTypeUnknown(String[] diskTypes) {
523         for (String diskType : diskTypes) {
524             if (DISK_TYPE_UNKNOWN.equals(diskType)) {
525                 return true;
526             }
527         }
528         return false;
529     }

531     /**
532      * Fills in the disk types the storage driver could not determine from {@code MSFT_PhysicalDisk}. Types the driver
533      * did determine are left alone, so a removable or virtual disk is not relabeled by its media technology.
534      *
535      * @param diskTypes the disk types to update in place
536      * @param vals      the disk drive query result the types were derived from
537      * @param pd        the physical disk media type query result
538      */
539     protected static void fillUnknownDiskTypes(String[] diskTypes, WmiResult<DiskDriveProperty> vals,
540             WmiResult<PhysicalDiskMediaTypeProperty> pd) {
541         Map<String, String> diskTypeMap = buildMsftDiskTypeMap(pd);
542         if (diskTypeMap.isEmpty()) {
543             return;
544         }
545         for (int i = 0; i < diskTypes.length; i++) {
546             if (DISK_TYPE_UNKNOWN.equals(diskTypes[i])) {
547                 // MSFT_PhysicalDisk's DeviceId is the physical drive number, as Win32_DiskDrive's Index is
548                 String deviceId = Integer.toString(WmiUtil.getUint32(vals, DiskDriveProperty.INDEX, i));
549                 diskTypes[i] = diskTypeMap.getOrDefault(deviceId, DISK_TYPE_UNKNOWN);
550             }
551         }
552     }
