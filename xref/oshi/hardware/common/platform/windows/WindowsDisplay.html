5   package oshi.hardware.common.platform.windows;
6   
7   import java.util.Optional;
8   
9   import org.jspecify.annotations.Nullable;
10  
11  import oshi.annotation.concurrent.Immutable;
12  import oshi.driver.common.windows.DisplayConnector.Connector;
13  import oshi.hardware.Display;
14  import oshi.hardware.DisplayMode;
15  import oshi.hardware.common.AbstractDisplay;
16  import oshi.util.Constants;
17  
18  /**
19   * Common logic for a Windows {@link Display}. Both bindings read the EDID from the registry and describe its connector
20   * from the Connecting and Configuring Displays (CCD) API; only the native calls differ.
21   */
22  @Immutable
23  public abstract class WindowsDisplay extends AbstractDisplay {
24  
25      private final @Nullable Connector connector;
26  
27      /**
28       * Constructor for WindowsDisplay.
29       *
30       * @param edid      a byte array representing a display EDID
31       * @param connector the connector this display is attached to, or {@code null} if Windows cannot resolve it
32       */
33      protected WindowsDisplay(byte[] edid, @Nullable Connector connector) {
34          super(edid);
35          this.connector = connector;
36      }
37  
38      @Override
39      public String getDevicePort() {
40          return connector == null ? Constants.UNKNOWN : connector.getName();
41      }
42  
43      @Override
44      public Optional<DisplayMode> getCurrentMode() {
45          return connector == null ? Optional.empty() : connector.getMode();
46      }
47  
48      @Override
49      public Optional<Boolean> isBuiltIn() {
50          return connector == null ? Optional.empty() : Optional.of(connector.isBuiltIn());
51      }
52  
53      @Override
54      public Optional<Boolean> isPrimary() {
55          // Windows defines the primary display as the one whose desktop origin is (0, 0). Displays cloning it share its
56          // source mode and position, so each of them reports primary too.
57          return getCurrentMode().map(mode -> mode.getX() == 0 && mode.getY() == 0);
58      }
59  }
