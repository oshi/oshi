12  import java.util.regex.Pattern;

18  import oshi.annotation.concurrent.ThreadSafe;

22  /**
23   * Common logic for a macOS {@link GpuStats} session.
24   *
25   * <p>
26   * On Apple Silicon, GPU ticks, utilization, and power are sourced from an {@link IOReportSampler} subscription.
27   * Utilization falls back to IOAccelerator PerformanceStatistics when the subscription fails or returns a negative
28   * value. Temperature is read from the SMC first, then falls back to IOAccelerator {@code Temperature(C)}.
29   *
30   * <p>
31   * On Intel Mac, utilization and VRAM used are sourced from IOAccelerator PerformanceStatistics.
32   *
33   * <p>
34   * Clock speeds, fan speed, and shared memory are not available on any macOS path and always return -1.
35   *
36   * <p>
37   * Only the two native reads differ between the bindings, so subclasses implement {@link #queryPerfStats} and
38   * {@link #queryGpuTemperatureFromSmc} and inherit everything else.
39   */
40  @ThreadSafe
41  public abstract class MacGpuStats implements GpuStats {
42  
43      private static final Logger LOG = LoggerFactory.getLogger(MacGpuStats.class);
44  
45      // IOAccelerator PerformanceStatistics keys. Private because subclasses never name a key themselves: this class
46      // decides which to ask for and passes them to queryPerfStats.
47      private static final String GPU_CORE_UTIL_KEY = "GPU Core Utilization";
48      private static final String DEVICE_UTIL_KEY = "Device Utilization %";
49      private static final String VRAM_USED_KEY = "vramUsedBytes";
50      private static final String VRAM_USED_KEY_AS = "In use system memory";
51      private static final String TEMPERATURE_KEY = "Temperature(C)";
52  
53      private static final double GPU_UTIL_DIVISOR = 0xFFFFFFFFL;
54  
55      /**
56       * Cards whose IOAccelerator statistics have no Temperature(C) key, so the lookup is not repeated. Static because a
57       * GpuStats session is short-lived, and keyed by card because one card lacking the sensor says nothing about another
58       * on the same machine.
59       */
60      private static final Set<String> PERF_STATS_TEMP_ABSENT = ConcurrentHashMap.newKeySet();
61  
62      private static final Pattern TRADEMARK_PATTERN = Pattern.compile("[®™]|\\([Rr]\\)|\\([Tt][Mm]\\)");
63  
64      private final boolean isAppleSilicon;
65      private final String normCardName;
66      private final Pattern cardNamePattern;
67  
68      /** Non-null only on Apple Silicon, where the IOReport GPU channels exist. */
69      private final @Nullable IOReportSampler sampler;
70  
71      private boolean closed;
72  
73      /**
74       * Creates a session for one card.
75       *
76       * @param isAppleSilicon whether this machine is Apple Silicon
77       * @param cardName       the card's reported name, matched against IOAccelerator model names
78       * @param samplerFactory creates the IOReport subscription, returning {@code null} if it could not be established.
79       *                       It is consulted only on Apple Silicon: the IOReport GPU channels do not exist on Intel, so
80       *                       subscribing there would attempt a subscription that can never yield a sample.
81       */
82      protected MacGpuStats(boolean isAppleSilicon, String cardName, Supplier<@Nullable IOReportSampler> samplerFactory) {
83          this.isAppleSilicon = isAppleSilicon;
84          this.normCardName = normalize(cardName);
85          this.cardNamePattern = Pattern.compile("\\b" + Pattern.quote(normCardName) + "\\b");
86          this.sampler = isAppleSilicon ? samplerFactory.get() : null;
87          if (isAppleSilicon && sampler == null) {
88              LOG.warn("IOReport subscription failed for '{}'; GPU ticks and power will be unavailable."
89                      + " Utilization will fall back to IOAccelerator PerformanceStatistics.", cardName);
90          }
91      }
92  
93      /**
94       * Reads IOAccelerator PerformanceStatistics values for this card, in a single registry walk.
95       * <p>
96       * Returning {@code null} rather than an empty map when the card's statistics cannot be read at all is load-bearing:
97       * {@link #getTemperature()} latches a card as having no temperature sensor only when the dictionary was present and
98       * the key was absent, never when the dictionary itself was missing, which can be transient.
99       *
100      * @param keys the keys to read
101      * @return the requested keys that were present, or {@code null} if this card's statistics could not be read
102      */
103     protected abstract @Nullable Map<String, Long> queryPerfStats(String... keys);
104 
105     /**
106      * Reads the GPU temperature from the SMC. Called only on Apple Silicon.
107      *
108      * @return the temperature in degrees Celsius, or a value at or below zero if unavailable
109      */
110     protected abstract double queryGpuTemperatureFromSmc();
111 
112     /**
113      * Tests whether an IOAccelerator model name refers to this card, ignoring case and trademark symbols.
114      *
115      * @param model the model name read from the registry
116      * @return true if it names this card
117      */
118     protected final boolean matchesName(String model) {
119         if (model == null || model.isEmpty()) {
120             return false;
121         }
122         String normModel = normalize(model);
123         return normModel.equals(normCardName) || cardNamePattern.matcher(normModel).find();
124     }
125 
126     private static String normalize(String name) {
127         return TRADEMARK_PATTERN.matcher(name.toLowerCase(Locale.ROOT)).replaceAll("").trim();
128     }
129 
130     @Override
131     public synchronized void close() {
132         closed = true;
133         if (sampler != null) {
134             sampler.close();
135         }
136     }
137 
138     @Override
139     public synchronized boolean isClosed() {
140         return closed;
141     }
142 
143     @Override
144     public synchronized GpuTicks getGpuTicks() {
145         checkOpen();
146         if (sampler != null) {
147             return sampler.sampleGpuTicks();
148         }
149         return new GpuTicks(0L, 0L);
150     }
151 
152     @Override
153     public synchronized double getGpuUtilization() {
154         checkOpen();
155         if (sampler != null) {
156             double util = sampler.sampleGpuUtilization();
157             if (util >= 0) {
158                 return util;
159             }
160         }
161         Map<String, Long> stats = queryPerfStats(GPU_CORE_UTIL_KEY, DEVICE_UTIL_KEY);
162         if (stats == null) {
163             return -1d;
164         }
165         Long coreUtil = stats.get(GPU_CORE_UTIL_KEY);
166         if (coreUtil != null) {
167             return coreUtil / GPU_UTIL_DIVISOR * 100.0;
168         }
169         Long deviceUtil = stats.get(DEVICE_UTIL_KEY);
170         return deviceUtil == null ? -1d : deviceUtil;
171     }
172 
173     @Override
174     public synchronized long getVramUsed() {
175         checkOpen();
176         // Apple Silicon reports GPU memory as a share of unified system memory, so its key differs from the discrete
177         // VRAM key an Intel Mac reports. Each is tried as the other's fallback.
178         String primaryKey = isAppleSilicon ? VRAM_USED_KEY_AS : VRAM_USED_KEY;
179         String fallbackKey = isAppleSilicon ? VRAM_USED_KEY : VRAM_USED_KEY_AS;
180         Map<String, Long> stats = queryPerfStats(primaryKey, fallbackKey);
181         if (stats == null) {
182             return -1L;
183         }
184         Long used = stats.get(primaryKey);
185         if (used == null) {
186             used = stats.get(fallbackKey);
187         }
188         return used == null ? -1L : used;
189     }
190 
191     @Override
192     public synchronized long getSharedMemoryUsed() {
193         checkOpen();
194         return -1L;
195     }
196 
197     @Override
198     public synchronized double getTemperature() {
199         checkOpen();
200         if (isAppleSilicon) {
201             double temp = queryGpuTemperatureFromSmc();
202             if (temp > 0) {
203                 return temp;
204             }
205         }
206         // IOAccelerator statistics, not the SMC: a different sensor that is not power-gated with the GPU cluster, so
207         // the SMC plausibility floor deliberately does not apply here. Unavailable is -1, not 0. This key does not
208         // exist on Apple Silicon, so it is tried once per card and then latched off rather than repeating an IOKit
209         // registry walk on every call. Latch only on structural absence, not on a low reading.
210         if (PERF_STATS_TEMP_ABSENT.contains(normCardName)) {
211             return -1d;
212         }
213         Map<String, Long> stats = queryPerfStats(TEMPERATURE_KEY);
214         if (stats == null) {
215             // Not latched: a missing accelerator entry can be transient (driver reset, eGPU unplugged), unlike a
216             // dictionary that is present but has no temperature key.
217             return -1d;
218         }
219         Long temp = stats.get(TEMPERATURE_KEY);
220         if (temp == null) {
221             LOG.debug("No Temperature(C) in IOAccelerator statistics for {}; not retrying.", normCardName);
222             PERF_STATS_TEMP_ABSENT.add(normCardName);
223             return -1d;
224         }
225         return temp > 0 ? temp : -1d;
226     }
227 
228     @Override
229     public synchronized double getPowerDraw() {
230         checkOpen();
231         if (sampler != null) {
232             return sampler.samplePowerWatts();
233         }
234         return -1d;
235     }
236 
237     @Override
238     public synchronized long getCoreClockMhz() {
239         checkOpen();
240         return -1L;
241     }
242 
243     @Override
244     public synchronized long getMemoryClockMhz() {
245         checkOpen();
246         return -1L;
247     }
248 
249     @Override
250     public synchronized double getFanSpeedPercent() {
251         checkOpen();
252         return -1d;
253     }
254 
255     private void checkOpen() {
256         if (closed) {
257             throw new IllegalStateException(
258                     "GpuStats session has been closed. Obtain a new session via GraphicsCard.createStatsSession().");
259         }
260     }
