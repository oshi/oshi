19  /**
20   * Base class for macOS HWDiskStore implementations. Subclasses provide platform-specific disk enumeration and
21   * statistics updates.
22   */
23  @ThreadSafe
24  public abstract class MacHWDiskStore extends AbstractHWDiskStore {
25  
26      /** IORegistry property naming a media object's BSD device. */
27      protected static final String BSD_NAME = "BSD Name";
28      /** IORegistry property marking a media object as a whole disk rather than a partition. */
29      protected static final String WHOLE = "Whole";
30      /** IOKit class of a block storage medium. */
31      protected static final String IO_MEDIA = "IOMedia";
32      /** IORegistry property on a disk image device naming the image it was attached from. */
33      protected static final String DISK_IMAGE_URL = "DiskImageURL";
34      /** IORegistry class of the device a disk image is attached as. */
35      protected static final String DISK_IMAGE_DEVICE = "AppleDiskImageDevice";
36      /**
37       * IORegistry class of the device a disk image was attached as before macOS 11, which does not publish
38       * {@link #DISK_IMAGE_URL}.
39       */
40      protected static final String LEGACY_DISK_IMAGE_DEVICE = "IOHDIXHDDrive";
41      /**
42       * IORegistry class at the top of the IOService plane, below its root. Every device descends from it, so reaching it
43       * means a walk up the plane found no backing ancestor.
44       */
45      protected static final String PLATFORM_EXPERT = "IOPlatformExpertDevice";
46  
47      private static final String VIRTUAL = "Virtual";
48  
49      private final Optional<List<String>> backingStores;

51      /**
52       * Creates a MacHWDiskStore with unknown disk type.
53       *
54       * @param name   the disk name
55       * @param model  the disk model
56       * @param serial the serial number
57       * @param size   the disk size in bytes
58       */
59      protected MacHWDiskStore(String name, String model, String serial, long size) {
60          super(name, model, serial, size);
61          this.backingStores = Optional.empty();
62      }
63  
64      /**
65       * Creates a MacHWDiskStore.
66       *
67       * @param name     the disk name
68       * @param model    the disk model
69       * @param serial   the serial number
70       * @param size     the disk size in bytes
71       * @param diskType the disk type
72       */
73      protected MacHWDiskStore(String name, String model, String serial, long size, String diskType) {
74          super(name, model, serial, size, diskType);
75          this.backingStores = Optional.empty();
76      }
77  
78      /**
79       * Creates a MacHWDiskStore built on other storage. A disk with at least one backing store, such as an APFS
80       * container or a disk image, is typed {@code Virtual} whatever its media reports.
81       *
82       * @param name          the disk name
83       * @param model         the disk model
84       * @param serial        the serial number
85       * @param size          the disk size in bytes
86       * @param diskType      the disk type its media reports
87       * @param backingStores the stores this disk is built on, or empty if they could not be read
88       */
89      protected MacHWDiskStore(String name, String model, String serial, long size, String diskType,
90              Optional<List<String>> backingStores) {
91          super(name, model, serial, size, backingStores.map(List::isEmpty).orElse(true) ? diskType : VIRTUAL);
92          this.backingStores = backingStores;
93      }
94  
95      @Override
96      public Optional<List<String>> getBackingStores() {
97          return backingStores;
98      }
99  
100     /**
101      * Builds the backing store of a disk whose nearest IOMedia ancestor in the IOService plane is another media object:
102      * the partition an APFS container is synthesized on, or the disk it is carved from.
103      *
104      * @param bsdName the ancestor's {@code BSD Name}, or {@code null} if it has none
105      * @param whole   the ancestor's {@code Whole} property, or {@code null} if absent
106      * @return the backing store, or empty if the ancestor has no BSD name
107      */
108     protected static Optional<List<String>> mediaBackingStore(@Nullable String bsdName, @Nullable Boolean whole) {
109         if (bsdName == null) {
110             return Optional.empty();
111         }
112         StoreKind kind = Boolean.TRUE.equals(whole) ? StoreKind.DISK : StoreKind.PARTITION;
113         return Optional.of(Collections.singletonList(kind.entry(bsdName)));
114     }
115 
116     /**
117      * Builds the backing store of a disk image: the file store holding the image file.
118      *
119      * @param imageUrl the image device's {@code DiskImageURL}
120      * @param mountOf  returns the mount point of the file store holding a path, or an empty string if it cannot be read
121      * @return the backing store, or empty if the image is not a local file or its file store could not be read
122      */
123     protected static Optional<List<String>> imageBackingStore(String imageUrl, UnaryOperator<String> mountOf) {
124         String path = imagePath(imageUrl);
125         if (path.isEmpty()) {
126             return Optional.empty();
127         }
128         String mount = mountOf.apply(path);
129         if (mount.isEmpty()) {
130             return Optional.empty();
131         }
132         return Optional.of(Collections.singletonList(StoreKind.FILE.entry(mount)));
133     }
134 
135     /**
136      * Converts a disk image URL to the path of the image file. Package-private for testing.
137      *
138      * @param imageUrl the image device's {@code DiskImageURL}, such as {@code file:///Users/x/a%20b.dmg}
139      * @return the decoded file path, or an empty string if the URL is not a {@code file} URL
140      */
141     static String imagePath(String imageUrl) {
142         try {
143             URI uri = new URI(imageUrl);
144             return "file".equals(uri.getScheme()) && uri.getPath() != null ? uri.getPath() : "";
145         } catch (URISyntaxException e) {
146             return "";
147         }
148     }
149 
150     /**
151      * Classifies a disk from the IOKit {@code Medium Type} device characteristic.
152      *
153      * @param mediumType the {@code Medium Type} value from the device's {@code Device Characteristics} dictionary, or
154      *                   {@code null} if the property is absent
155      * @return {@code "SSD"}, {@code "HDD"}, or {@code "Unknown"} if the medium type is absent or unrecognized
156      */
157     protected static String parseMediumType(@Nullable String mediumType) {
158         if (mediumType != null) {
159             if (mediumType.contains("Solid State") || mediumType.contains("SSD")) {
160                 return "SSD";
161             } else if (mediumType.contains("Rotational")) {
162                 return "HDD";
163             }
164         }
165         return "Unknown";
166     }
