5   package oshi.hardware.common.platform.unix.netbsd;
6   
7   import java.util.Map;
8   
9   import org.jspecify.annotations.Nullable;
10  
11  import oshi.annotation.concurrent.Immutable;
12  import oshi.hardware.common.platform.unix.BsdFirmware;
13  import oshi.util.ParseUtil;
14  import oshi.util.tuples.Triplet;
15  
16  /**
17   * NetBSD Firmware implementation, from the SMBIOS strings the kernel publishes under {@code machdep.dmi}. These are
18   * absent on machines without SMBIOS BIOS information, which includes many ARM systems.
19   */
20  @Immutable
21  public class NetBsdFirmware extends BsdFirmware {
22  
23      @Override
24      protected Triplet<@Nullable String, @Nullable String, @Nullable String> readFirmware() {
25          Map<String, String> dmi = NetBsdComputerSystem.queryDmi();
26          return new Triplet<>(dmi.get("bios-vendor"), dmi.get("bios-version"),
27                  ParseUtil.parseYyyyMmDdToIso(dmi.getOrDefault("bios-date", "")));
28      }
29  }
