23  /**
24   * NetBSD Central Processor implementation. Shares the {@code dmesg}-based topology and cache parsing with the other
25   * BSDs via {@link BsdCentralProcessor}; the processor identification, frequency, CPU ticks, and load average are read
26   * from command output and name-based sysctls specific to NetBSD.
27   */
28  @ThreadSafe
29  public class NetBsdCentralProcessor extends BsdCentralProcessor {
30      /** The first 64-bit member of a hex dump. */
31      private static final Pattern CPU_ID_FIRST_LONG = Pattern.compile("\\p{XDigit}{16}");
32      /** The x86 model line, which ends with the CPUID signature: {@code "cpu0: AMD EPYC 7763 ..., id 0xa00f11"}. */
33      private static final Pattern CPU_SIGNATURE = Pattern.compile("cpu0: .*, id 0x(\\p{XDigit}++)");
34  
35      private static final int CPUSTATES = 5;
36      private static final int CP_USER = 0;
37      private static final int CP_NICE = 1;
38      private static final int CP_SYS = 2;
39      private static final int CP_INTR = 3;
40      private static final int CP_IDLE = 4;
41  
42      @Override
43      protected int sysctl(String name, int def) {
44          return BsdSysctlUtil.sysctl(name, def);
45      }

47      @Override
48      protected ProcessorIdentifier queryProcessorId() {
49          // The architecture of the processor, where hw.machine is the port: evbarm rather than aarch64
50          String machineArch = BsdSysctlUtil.sysctl("hw.machine_arch", "");
51          boolean cpu64bit = machineArch.contains("64");
52          List<String> dmesg = queryDmesg();
53          String cpuName = queryCpuName();
54          long cpuFreq = queryFreq(cpuName);
55          // Off x86 the brand is absent and hw.model names the board, so take the processor's own line
56          String dmesgName = parseDmesgCpuName(dmesg);
57          if (BsdSysctlUtil.sysctl("machdep.cpu_brand", "").isEmpty() && !dmesgName.isEmpty()) {
58              cpuName = dmesgName;
59          }
60          if (machineArch.startsWith("aarch64")) {
61              long midr = parseMidr(ExecutingCommand.getFirstAnswer("sysctl -nx machdep.cpu0.cpu_id"),
62                      machineArch.endsWith("eb"));
63              if (midr >= 0) {
64                  return createArmIdentifier(midr, cpuName, cpuFreq);
65              }
66          }
67          String cpuVendor = ExecutingCommand.getFirstAnswer("sysctl -n machdep.dmi.processor-vendor").trim();
68          String[] fms = parseFamilyModelStepping(dmesg);
69          return new ProcessorIdentifier(cpuVendor, cpuName, fms[0], fms[1], fms[2], "", cpu64bit, cpuFreq);
70      }

72      /**
73       * Reads the Main ID Register from the hex dump of {@code machdep.cpu0.cpu_id}, a {@code struct
74       * aarch64_sysctl_cpu_id} whose first member is the register as a 64-bit integer.
75       *
76       * @param cpuId     the output of {@code sysctl -nx machdep.cpu0.cpu_id}
77       * @param bigEndian whether the kernel is big-endian ({@code aarch64eb})
78       * @return the register, or -1 if the output does not carry one
79       */
80      static long parseMidr(String cpuId, boolean bigEndian) {
81          Matcher m = CPU_ID_FIRST_LONG.matcher(cpuId.trim());
82          if (!m.lookingAt()) {
83              return -1L;
84          }
85          // The dump is in memory order, so a little-endian kernel's register reads byte-reversed
86          long first = ParseUtil.hexStringToLong(m.group(), 0L);
87          return (bigEndian ? first : Long.reverseBytes(first)) & 0xffff_ffffL;
88      }

90      /**
91       * Creates the identifier of an ARM processor from its Main ID Register, in the form Linux reports it: the
92       * implementer as the vendor, the part number as the model, and the variant and revision as the stepping.
93       *
94       * @param midr    the Main ID Register
95       * @param cpuName the processor name
96       * @param cpuFreq the processor frequency in Hz, or -1 if unknown
97       * @return the processor identifier
98       */
99      static ProcessorIdentifier createArmIdentifier(long midr, String cpuName, long cpuFreq) {
100         String vendor = String.format(Locale.ROOT, "0x%02x", (midr >> 24) & 0xff);
101         String model = String.format(Locale.ROOT, "0x%03x", (midr >> 4) & 0xfff);
102         String stepping = "r" + ((midr >> 20) & 0xf) + "p" + (midr & 0xf);
103         String processorID = String.format(Locale.ROOT, "%08X00000000", midr);
104         // Every AArch64 processor reports architecture 0xF, "defined by the ID registers"; Linux shows 8
105         return new ProcessorIdentifier(vendor, cpuName, "8", model, stepping, processorID, true, cpuFreq);
106     }

113     private static String queryCpuName() {
114         String cpuName = BsdSysctlUtil.sysctl("machdep.cpu_brand", "");
115         if (cpuName.isEmpty()) {
116             cpuName = BsdSysctlUtil.sysctl("hw.model", "");
117         }
118         return cpuName;
119     }
120 
121     private static long queryFreq(String cpuName) {
122         // machdep.tsc_freq gives frequency in Hz on x86
123         long freq = BsdSysctlUtil.sysctl("machdep.tsc_freq", 0L);
124         if (freq > 0L) {
125             return freq;
126         }
127         return parseFreq(cpuName, BsdSysctlUtil.sysctl("machdep.dmi.processor-frequency", ""));
128     }

130     /**
131      * Parses the processor frequency from the processor name, e.g. {@code "Intel ... @ 2.10GHz"}, or failing that from
132      * the SMBIOS processor frequency, e.g. {@code "2000 MHz"}, which is the only source on non-x86 machines.
133      *
134      * @param cpuName      the processor name
135      * @param dmiFrequency the value of {@code machdep.dmi.processor-frequency}
136      * @return the frequency in Hz, or -1 if neither string carries one
137      */
138     static long parseFreq(String cpuName, String dmiFrequency) {
139         long freq = ParseUtil.parseHertz(cpuName);
140         return freq > 0L ? freq : ParseUtil.parseHertz(dmiFrequency);
141     }

148     /**
149      * Parses the output of {@code vmstat -s} to extract context switch and interrupt counts.
150      *
151      * @param vmstat the lines emitted by {@code vmstat -s}
152      * @return a {@link Pair} of (context switches, interrupts)
153      */
154     static Pair<Long, Long> parseVmStats(List<String> vmstat) {
155         long contextSwitches = 0L;
156         long interrupts = 0L;
157         for (String line : vmstat) {
158             if (line.endsWith("CPU context switches")) {
159                 contextSwitches = ParseUtil.parseLongOrDefault(ParseUtil.whitespaces.split(line.trim(), -1)[0], 0L);
160             } else if (line.endsWith("device interrupts")) {
161                 interrupts = ParseUtil.parseLongOrDefault(ParseUtil.whitespaces.split(line.trim(), -1)[0], 0L);
162             }
163         }
164         return new Pair<>(contextSwitches, interrupts);
165     }

167     /**
168      * Parses family, model, and stepping from the CPUID signature at the end of the x86 model line in {@code dmesg},
169      * {@code "cpu0: AMD EPYC 7763 64-Core Processor, id 0xa00f11"}.
170      *
171      * @param dmesg the lines emitted by {@code dmesg}, without timestamps
172      * @return a 3-element array of [family, model, stepping] in decimal, each empty string if not found
173      */
174     static String[] parseFamilyModelStepping(List<String> dmesg) {
175         for (String line : dmesg) {
176             Matcher m = CPU_SIGNATURE.matcher(line);
177             if (m.matches()) {
178                 Triplet<Integer, Integer, Integer> fms = cpuidToFamilyModelStepping(
179                         ParseUtil.hexStringToInt(m.group(1), 0));
180                 return new String[] { fms.getA().toString(), fms.getB().toString(), fms.getC().toString() };
181             }
182         }
183         return new String[] { "", "", "" };
184     }

191     @Override
192     protected long[][] queryProcessorCpuLoadTicks() {
193         long[][] ticks = new long[getLogicalProcessorCount()][];
194         for (int cpu = 0; cpu < ticks.length; cpu++) {
195             ticks[cpu] = cpTimeToTicks(queryCpTime(cpu));
196         }
197         return ticks;
198     }

200     /**
201      * Queries the {@code kern.cp_time} counters, for the whole system or for one processor.
202      *
203      * @param cpu The processor number, or a negative value for the system total
204      * @return The counters indexed by the {@code CP_*} states, {@code [user, nice, sys, intr, idle]}; shorter than
205      *         {@code CPUSTATES} if they could not be read
206      */
207     protected long[] queryCpTime(int cpu) {
208         // "kern.cp_time: user = N, nice = N, sys = N, intr = N, idle = N", or "kern.cp_time.N: ..." per processor
209         return parseCpTime(
210                 ExecutingCommand.getFirstAnswer(cpu < 0 ? "sysctl kern.cp_time" : "sysctl kern.cp_time." + cpu));
211     }
212 
213     /**
214      * Maps the {@code kern.cp_time} counters onto the {@link TickType} indices.
215      *
216      * @param cpTime The counters indexed by the {@code CP_*} states
217      * @return The ticks, all zero if {@code cpTime} is shorter than {@code CPUSTATES}
218      */
219     static long[] cpTimeToTicks(long[] cpTime) {
220         long[] ticks = new long[TickType.values().length];
221         if (cpTime.length >= CPUSTATES) {
222             ticks[TickType.USER.getIndex()] = cpTime[CP_USER];
223             ticks[TickType.NICE.getIndex()] = cpTime[CP_NICE];
224             ticks[TickType.SYSTEM.getIndex()] = cpTime[CP_SYS];
225             ticks[TickType.IRQ.getIndex()] = cpTime[CP_INTR];
226             ticks[TickType.IDLE.getIndex()] = cpTime[CP_IDLE];
227         }
228         return ticks;
229     }

231     /**
232      * Parse sysctl kern.cp_time output to an array of tick values.
233      * <p>
234      * Input format: "kern.cp_time: user = 2930, nice = 42, sys = 1334, intr = 1877, idle = 46354"
235      *
236      * @param cpTimeStr the sysctl output string
237      * @return array of [user, nice, sys, intr, idle] tick values
238      */
239     static long[] parseCpTime(@Nullable String cpTimeStr) {
240         long[] ticks = new long[CPUSTATES];
241         if (cpTimeStr == null || cpTimeStr.isEmpty()) {
242             return ticks;
243         }
244         // Extract the values after the colon
245         int colonIdx = cpTimeStr.indexOf(':');
246         if (colonIdx < 0) {
247             return ticks;
248         }
249         String[] pairs = cpTimeStr.substring(colonIdx + 1).split(",", -1);
250         for (String pair : pairs) {
251             // Both halves are trimmed below, so a plain "=" split is equivalent to matching the spaces here
252             String[] kv = pair.split("=", -1);
253             if (kv.length == 2) {
254                 long val = ParseUtil.parseLongOrDefault(kv[1].trim(), 0L);
255                 switch (kv[0].trim()) {
256                     case "user":
257                         ticks[CP_USER] = val;
258                         break;
259                     case "nice":
260                         ticks[CP_NICE] = val;
261                         break;
262                     case "sys":
263                         ticks[CP_SYS] = val;
264                         break;
265                     case "intr":
266                         ticks[CP_INTR] = val;
267                         break;
268                     case "idle":
269                         ticks[CP_IDLE] = val;
270                         break;
271                     default:
272                         break;
273                 }
274             }
275         }
276         return ticks;
277     }

279     /**
280      * Returns the system load average for the number of elements specified, up to 3, representing 1, 5, and 15 minutes.
281      * The system load average is the sum of the number of runnable entities queued to the available processors and the
282      * number of runnable entities running on the available processors averaged over a period of time. The way in which
283      * the load average is calculated is operating system specific but is typically a damped time-dependent average. If
284      * the load average is not available, a negative value is returned. This method is designed to provide a hint about
285      * the system load and may be queried frequently.
286      * <p>
287      * The load average may be unavailable on some platforms (e.g., Windows) where it is expensive to implement this
288      * method.
289      *
290      * @param nelem Number of elements to return.
291      * @return an array of the system load averages for 1, 5, and 15 minutes with the size of the array specified by
292      *         nelem; or negative values if not available.
293      */
294     @Override
295     public double[] getSystemLoadAverage(int nelem) {
296         if (nelem < 1 || nelem > 3) {
297             throw new IllegalArgumentException("Must include from one to three elements.");
298         }
299         return queryLoadAverage(nelem);
300     }
301 
302     /**
303      * Queries the system load average.
304      *
305      * @param nelem Number of elements to return, already validated to be from one to three
306      * @return the load averages for 1, 5, and 15 minutes, truncated to {@code nelem}; -1 for any not available
307      */
308     protected double[] queryLoadAverage(int nelem) {
309         double[] average = new double[nelem];
310         Arrays.fill(average, -1d);
311         // Parse "vm.loadavg: 1.59 0.47 0.18"
312         String loadavg = ExecutingCommand.getFirstAnswer("sysctl -n vm.loadavg");
313         if (!loadavg.isEmpty()) {
314             String[] loads = ParseUtil.whitespaces.split(loadavg.trim(), -1);
315             for (int i = 0; i < nelem && i < loads.length; i++) {
316                 average[i] = ParseUtil.parseDoubleOrDefault(loads[i], -1d);
317             }
318         }
319         return average;
320     }
