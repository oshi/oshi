27      // The rate of an Ethernet media type (1000baseT, 10Gbase-T, 2500Base-T) or a wireless one (OFDM/54Mbps)
28      private static final Pattern MEDIA_SPEED = Pattern.compile("(\\d++(?:\\.\\d++)?+)([GM]?+)(?:base|bps)",
29              Pattern.CASE_INSENSITIVE);

32       * Constructs a new {@code BsdNetworkIF}.

36       */
37      public BsdNetworkIF(NetworkInterface netint) throws InstantiationException {
38          super(netint);
39          updateAttributes();
40      }

52      @Override
53      public synchronized boolean updateAttributes() {
54          Map<String, Long> stats = parseNetstat(ExecutingCommand.runNative("netstat -bnI " + getName()));
55          this.timeStamp = System.currentTimeMillis();
56          if (!stats.containsKey("Ibytes")) {
57              // No update
58              return false;
59          }
60          if (!stats.containsKey("Ipkts")) {
61              // OpenBSD's and NetBSD's -b prints only the byte counts
62              stats.putAll(parseNetstat(ExecutingCommand.runNative("netstat -nI " + getName())));
63          }
64          this.bytesRecv = stats.getOrDefault("Ibytes", 0L);
65          this.bytesSent = stats.getOrDefault("Obytes", 0L);
66          this.packetsRecv = stats.getOrDefault("Ipkts", 0L);
67          this.packetsSent = stats.getOrDefault("Opkts", 0L);
68          // OpenBSD names the error columns differently, and OpenBSD and NetBSD the collision column
69          this.inErrors = stats.getOrDefault("Ierrs", stats.getOrDefault("Ifail", 0L));
70          this.outErrors = stats.getOrDefault("Oerrs", stats.getOrDefault("Ofail", 0L));
71          this.collisions = stats.getOrDefault("Coll", stats.getOrDefault("Colls", 0L));
72          this.inDrops = stats.getOrDefault("Idrop", 0L);
73          this.speed = parseMediaSpeed(ExecutingCommand.runNative("/sbin/ifconfig " + getName()));
74          return true;
75      }

77      /**
78       * Parses the link speed from the {@code media:} line of {@code ifconfig <interface>}. Package-private for testing.
79       * <p>
80       * The active media type is in parentheses when the interface autoselects, as in
81       * {@code media: Ethernet autoselect (10Gbase-T <full-duplex>)} on FreeBSD or
82       * {@code media: Ethernet autoselect (1000baseT full-duplex,rxpause,txpause)} on OpenBSD; otherwise the line names
83       * the fixed media type itself. An Ethernet type gives its rate before {@code base}, in megabits unless marked
84       * {@code G}, and a wireless one in {@code Mbps}. An interface with no link shows {@code (none)} or no parentheses,
85       * and one without media, such as a virtio interface on NetBSD, no line at all. A fixed media type names its rate
86       * whether or not anything is connected, so a {@code status:} of {@code no carrier}, or {@code no network} for a
87       * wireless interface, reports 0 as well.
88       *
89       * @param ifconfig the lines emitted by {@code ifconfig <interface>}
90       * @return the link speed in bits per second, or 0 if none is shown
91       */
92      static long parseMediaSpeed(List<String> ifconfig) {
93          long speed = 0L;
94          for (String line : ifconfig) {
95              String trimmed = line.trim();
96              if (trimmed.startsWith("status: no ")) {
97                  // "no carrier" or "no network": fixed media still names a rate with nothing plugged in
98                  return 0L;
99              }
100             if (speed == 0L && trimmed.startsWith("media:")) {
101                 speed = parseMediaRate(trimmed);
102             }
103         }
104         return speed;
105     }
106 
107     private static long parseMediaRate(String mediaLine) {
108         String media = mediaLine;
109         int open = media.indexOf('(');
110         int close = media.lastIndexOf(')');
111         if (open >= 0 && close > open) {
112             media = media.substring(open + 1, close);
113         }
114         Matcher m = MEDIA_SPEED.matcher(media);
115         if (m.find()) {
116             double rate = ParseUtil.parseDoubleOrDefault(m.group(1), 0d);
117             return (long) (rate * ("G".equalsIgnoreCase(m.group(2)) ? 1_000_000_000L : 1_000_000L));
118         }
119         return 0L;
120     }

122     /**
123      * Parses the link-level row of {@code netstat -I <interface>} into its counters, keyed by column heading.
124      * <p>
125      * The counters follow the Name, Mtu, Network and Address columns, but which ones appear varies: FreeBSD prints an
126      * Idrop column that DragonFly BSD does not, and OpenBSD and NetBSD print only the byte counts with {@code -b}. The
127      * values are matched to the headings from the end of the row, because the Address column is empty on some loopback
128      * rows.
129      *
130      * @param netstat the lines emitted by {@code netstat -I <interface>}, with or without {@code -b}
131      * @return a map of column headings to counter values, empty if the output has no link-level row
132      */
133     static Map<String, Long> parseNetstat(List<String> netstat) {
134         Map<String, Long> stats = new HashMap<>();
135         if (netstat.size() < 2) {
136             return stats;
137         }
138         String[] header = ParseUtil.whitespaces.split(netstat.get(0).trim(), -1);
139         String[] row = ParseUtil.whitespaces.split(netstat.get(1).trim(), -1);
140         int firstCounter = Arrays.asList(header).indexOf("Address") + 1;
141         int counters = header.length - firstCounter;
142         // The row needs at least the name and every counter
143         if (firstCounter == 0 || row.length <= counters) {
144             return stats;
145         }
146         for (int i = 0; i < counters; i++) {
147             stats.put(header[firstCounter + i],
148                     ParseUtil.parseUnsignedLongOrDefault(row[row.length - counters + i], 0L));
149         }
150         return stats;
151     }
