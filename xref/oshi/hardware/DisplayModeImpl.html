5   package oshi.hardware;
6   
7   import java.util.Locale;
8   import java.util.Objects;
9   
10  import org.jspecify.annotations.Nullable;
11  
12  import oshi.annotation.concurrent.Immutable;
13  
14  /**
15   * The default {@link DisplayMode} implementation. This is not part of the OSHI public API and may change between minor
16   * releases.
17   */
18  @Immutable
19  public final class DisplayModeImpl implements DisplayMode {
20  
21      private final int width;
22      private final int height;
23      private final int pixelWidth;
24      private final int pixelHeight;
25      private final double refreshRate;
26      private final int rotation;
27      private final int x;
28      private final int y;
29  
30      /**
31       * Constructs a {@code DisplayModeImpl}. All sizes are in the display's current orientation.
32       *
33       * @param x           the x coordinate of the display's top-left corner on the desktop, in logical units
34       * @param y           the y coordinate of the display's top-left corner on the desktop, in logical units
35       * @param width       the logical width
36       * @param height      the logical height
37       * @param pixelWidth  the width in native pixels
38       * @param pixelHeight the height in native pixels
39       * @param refreshRate the refresh rate in hertz, or 0 if not known
40       * @param rotation    the rotation in degrees clockwise; normalized to 0, 90, 180 or 270
41       */
42      public DisplayModeImpl(int x, int y, int width, int height, int pixelWidth, int pixelHeight, // NOSONAR java:S107
43              double refreshRate, int rotation) {
44          this.x = x;
45          this.y = y;
46          this.width = width;
47          this.height = height;
48          this.pixelWidth = pixelWidth;
49          this.pixelHeight = pixelHeight;
50          this.refreshRate = refreshRate > 0 ? refreshRate : 0d;
51          this.rotation = normalizeRotation(rotation);
52      }
53  
54      /**
55       * Normalizes a rotation to the nearest quarter turn in the range 0 to 270.
56       *
57       * @param degrees a rotation in degrees clockwise, possibly negative or a multiple of a full turn
58       * @return 0, 90, 180 or 270
59       */
60      static int normalizeRotation(int degrees) {
61          int quarterTurns = Math.round(degrees / 90f) % 4;
62          return (quarterTurns < 0 ? quarterTurns + 4 : quarterTurns) * 90;
63      }
64  
65      @Override
66      public int getWidth() {
67          return width;
68      }
69  
70      @Override
71      public int getHeight() {
72          return height;
73      }
74  
75      @Override
76      public int getPixelWidth() {
77          return pixelWidth;
78      }
79  
80      @Override
81      public int getPixelHeight() {
82          return pixelHeight;
83      }
84  
85      @Override
86      public double getRefreshRate() {
87          return refreshRate;
88      }
89  
90      @Override
91      public int getRotation() {
92          return rotation;
93      }
94  
95      @Override
96      public int getX() {
97          return x;
98      }
99  
100     @Override
101     public int getY() {
102         return y;
103     }
104 
105     @Override
106     public boolean equals(@Nullable Object obj) {
107         if (this == obj) {
108             return true;
109         }
110         if (!(obj instanceof DisplayModeImpl)) {
111             return false;
112         }
113         DisplayModeImpl other = (DisplayModeImpl) obj;
114         return width == other.width && height == other.height && pixelWidth == other.pixelWidth
115                 && pixelHeight == other.pixelHeight && Double.compare(refreshRate, other.refreshRate) == 0
116                 && rotation == other.rotation && x == other.x && y == other.y;
117     }
118 
119     @Override
120     public int hashCode() {
121         return Objects.hash(width, height, pixelWidth, pixelHeight, refreshRate, rotation, x, y);
122     }
123 
124     @Override
125     public String toString() {
126         StringBuilder sb = new StringBuilder();
127         sb.append(width).append('x').append(height);
128         if (pixelWidth != width || pixelHeight != height) {
129             sb.append(" (").append(pixelWidth).append('x').append(pixelHeight).append(" pixels)");
130         }
131         if (refreshRate > 0) {
132             sb.append(String.format(Locale.ROOT, " @ %.2f Hz", refreshRate));
133         }
134         if (rotation != 0) {
135             sb.append(", rotated ").append(rotation).append('°');
136         }
137         sb.append(" at (").append(x).append(',').append(y).append(')');
138         return sb.toString();
139     }
140 }
