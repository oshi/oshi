20  /**
21   * Utility to query xrandr
22   */
23  @ThreadSafe
24  public final class Xrandr {
25  
26      private static final String[] XRANDR_VERBOSE = { "xrandr", "--verbose" };
27  
28      /**
29       * Property names an X server may publish the EDID under, each as it appears in {@code xrandr --verbose} output,
30       * including the trailing colon. {@code EDID} is the name in randrproto 1.3 and later; {@code RANDR_EDID} is the
31       * name it replaced; {@code EDID_DATA} is the driver-side atom X.Org Server used through 1.6.
32       */
33      private static final String[] EDID_PROPERTIES = { "EDID:", "RANDR_EDID:", "EDID_DATA:" };
34  
35      private Xrandr() {
36      }
37  
38      /**
39       * Tests whether a property line names the EDID, under any of the property names an X server may use for it.
40       *
41       * @param trimmed a whitespace-trimmed line of {@code xrandr --verbose} output
42       * @return true if the line is the header of an EDID property block
43       */
44      private static boolean isEdidProperty(String trimmed) {
45          for (String property : EDID_PROPERTIES) {
46              if (property.equals(trimmed)) {
47                  return true;
48              }
49          }
50          return false;
51      }
52  
53      /**
54       * Gets EDID byte arrays from the running X server via xrandr.
55       *
56       * @return a list of EDID byte arrays
57       */
58      public static List<byte[]> getEdidArrays() {
59          // Special handling for X commands, don't use LC_ALL
60          return getEdidArrays(runXrandr());
61      }
62  
63      /**
64       * Parse EDID arrays from xrandr verbose output.
65       *
66       * @param xrandr output of {@code xrandr --verbose}
67       * @return a list of EDID byte arrays (at least 128 bytes each)
68       */
69      static List<byte[]> getEdidArrays(List<String> xrandr) {
70          Map<String, Pair<Integer, byte[]>> data = getDisplayData(xrandr);
71          List<byte[]> edids = new ArrayList<>(data.size());
72          for (Pair<Integer, byte[]> pair : data.values()) {
73              edids.add(pair.getB());
74          }
75          return Collections.unmodifiableList(edids);
76      }
77  
78      /**
79       * Gets display data from the running X server via xrandr, mapping each connected output's xrandr port name to its
80       * connector ID and EDID.
81       *
82       * @return an ordered map of xrandr port name to {@link Pair} of connector ID ({@code -1} if not available) and EDID
83       *         byte array
84       */
85      public static Map<String, Pair<Integer, byte[]>> getDisplayData() {
86          return getDisplayData(runXrandr());
87      }
88  
89      /**
90       * Parse display data from xrandr verbose output. For each connected output, extracts the xrandr port name (the
91       * first whitespace-delimited token on the output header line), the {@code CONNECTOR_ID} property (if present,
92       * requires Linux 6.5+), and the EDID byte array. The parser is order-independent: {@code CONNECTOR_ID} may appear
93       * before or after {@code EDID:}.
94       *
95       * @param xrandr output of {@code xrandr --verbose}
96       * @return an ordered map of xrandr port name to {@link Pair} of connector ID ({@code -1} if not available) and EDID
97       *         byte array (at least 128 bytes). Only connected outputs with a valid EDID are included.
98       */
99      static Map<String, Pair<Integer, byte[]>> getDisplayData(List<String> xrandr) {
100         if (xrandr.isEmpty()) {
101             return Collections.emptyMap();
102         }
103         Map<String, Pair<Integer, byte[]>> results = new LinkedHashMap<>();
104         String currentPort = "";
105         boolean currentConnected = false;
106         int currentConnectorId = -1;
107         byte[] currentEdid = null;
108         StringBuilder sb = null;
109         for (String s : xrandr) {
110             // Output headers start at column 0; properties and modes are always indented
111             if (!s.isEmpty() && !Character.isWhitespace(s.charAt(0))) {
112                 // Flush the previous output before starting a new one
113                 if (currentConnected && currentEdid != null) {
114                     results.put(currentPort, new Pair<>(currentConnectorId, currentEdid));
115                 }
116                 String[] words = ParseUtil.whitespaces.split(s.trim(), -1);
117                 currentPort = words[0];
118                 currentConnected = words.length > 1 && "connected".equals(words[1]);
119                 currentConnectorId = -1;
120                 currentEdid = null;
121                 sb = null;
122                 continue;
123             }
124             String trimmed = s.trim();
125             if (trimmed.startsWith("CONNECTOR_ID:")) {
126                 currentConnectorId = ParseUtil.parseLastInt(trimmed, -1);
127             } else if (isEdidProperty(trimmed)) {
128                 sb = new StringBuilder();
129             } else if (sb != null) {
130                 sb.append(trimmed);
131                 if (sb.length() < 256) {
132                     continue;
133                 }
134                 currentEdid = ParseUtil.hexStringToByteArray(sb.toString());
135                 if (currentEdid.length < 128) {
136                     currentEdid = null;
137                 }
138                 sb = null;
139             }
140         }
141         // Flush the last output
142         if (currentConnected && currentEdid != null) {
143             results.put(currentPort, new Pair<>(currentConnectorId, currentEdid));
144         }
145         return Collections.unmodifiableMap(results);
146     }

148     /**
149      * Finds the xrandr output name for a display identified by its DRM connector ID and/or EDID, matching by
150      * {@code CONNECTOR_ID} first (Linux 6.5+) and falling back to EDID comparison.
151      *
152      * @param xrandrData  xrandr display data as returned by {@link #getDisplayData()}, which the caller is expected to
153      *                    share among the displays it is naming rather than querying per display
154      * @param connectorId the DRM connector ID ({@code -1} if not available)
155      * @param edid        the EDID byte array from DRM sysfs
156      * @return an {@link Optional} containing the xrandr output name, or empty if no X server is available or no match
157      *         is found
158      */
159     public static Optional<String> findOutputName(Map<String, Pair<Integer, byte[]>> xrandrData, int connectorId,
160             byte[] edid) {
161         if (xrandrData.isEmpty()) {
162             return Optional.empty();
163         }
164         // First try matching by CONNECTOR_ID (Linux 6.5+)
165         if (connectorId >= 0) {
166             for (Map.Entry<String, Pair<Integer, byte[]>> entry : xrandrData.entrySet()) {
167                 if (entry.getValue().getA() == connectorId) {
168                     return Optional.of(entry.getKey());
169                 }
170             }
171         }
172         // Fallback: match by first 128 bytes of EDID
173         if (edid.length >= 128) {
174             byte[] edid128 = Arrays.copyOf(edid, 128);
175             for (Map.Entry<String, Pair<Integer, byte[]>> entry : xrandrData.entrySet()) {
176                 byte[] xrandrEdid = entry.getValue().getB();
177                 if (xrandrEdid.length >= 128 && Arrays.equals(edid128, Arrays.copyOf(xrandrEdid, 128))) {
178                     return Optional.of(entry.getKey());
179                 }
180             }
181         }
182         return Optional.empty();
183     }
184 
185     private static List<String> runXrandr() {
186         if (System.getenv("DISPLAY") == null) {
187             return Collections.emptyList();
188         }
189         return ExecutingCommand.runNative(XRANDR_VERBOSE, null);
190     }
