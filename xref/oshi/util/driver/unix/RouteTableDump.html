20  /**
21   * Reads the routing table out of the buffer a {@code sysctl} of {@code NET_RT_DUMP} returns on macOS and the BSDs.
22   * <p>
23   * The buffer is a run of variable-length messages, each a fixed header followed by the addresses named in its
24   * {@code rtm_addrs} bitmask, one per set bit in ascending {@code RTAX} order. Every address is a {@code sockaddr}
25   * stating its own length in its first byte and padded up to a platform-dependent unit.
26   * <p>
27   * Only four numbers differ between the platforms, which {@link Layout} carries; the walk itself is the same everywhere.
28   * The padding unit is the one to respect: it is four bytes on macOS and eight on the BSDs, and reading a message with
29   * the wrong unit yields addresses from the middle of the next field rather than an error.
30   */
31  @ThreadSafe
32  public final class RouteTableDump {
33  
34      private static final int RTM_GET = 4;
35      private static final int RTF_GATEWAY = 0x2;
36      private static final int RTF_HOST = 0x4;
37  
38      private static final int RTAX_DST = 0;
39      private static final int RTAX_GATEWAY = 1;
40      private static final int RTAX_NETMASK = 2;
41  
42      private static final int AF_INET = 2;
43  
44      // Offsets shared by every platform measured
45      private static final int OFF_MSGLEN = 0;
46      private static final int OFF_TYPE = 3;
47      private static final int OFF_ADDRS = 12;
48  
49      /**
50       * The parts of a routing message whose position or size depends on the platform, and the platforms' values.
51       * <p>
52       * They are declared together so they can be read as a table, because reading down a column is the only way to
53       * notice that macOS pads addresses to four bytes where the BSDs pad to eight. That one is the trap: the wrong unit
54       * does not fail, it reads addresses from the middle of the next field.
55       * <p>
56       * OpenBSD is otherwise the odd one, stating its header length per message and carrying a route priority usable as a
57       * metric where the others have none.
58       * <p>
59       * DragonFly BSD differs from FreeBSD only in the last two columns, and only because FreeBSD stopped cloning routes:
60       * it defines neither {@code RTF_WASCLONED} nor {@code RTF_CLONED}, while DragonFly kept both. The flag values are
61       * not interchangeable across these platforms either -- OpenBSD's {@code RTF_CLONED} is {@code 0x10000}, where
62       * {@code 0x20000} is {@code RTF_CACHED}.
63       */
64      public enum Layout {
65          // header hdrLen flags index priority padding rtaxMax AF_INET6 cloned linkInfo
66          MACOS(92, -1, 8, 4, -1, 4, 8, 30, 0x20000, 0x400), //
67          FREEBSD(152, -1, 8, 4, -1, 8, 8, 28, 0, 0), //
68          DRAGONFLY(152, -1, 8, 4, -1, 8, 11, 28, 0x20000, 0x400), //
69          NETBSD(120, -1, 8, 4, -1, 8, 9, 24, 0, 0), //
70          OPENBSD(96, 4, 16, 6, 10, 8, 15, 24, 0, 0);
71  
72          private final int headerSize;
73          private final int hdrLenOffset;
74          private final int flagsOffset;
75          private final int indexOffset;
76          private final int priorityOffset;
77          private final int paddingUnit;
78          private final int rtaxMax;
79          private final int afInet6;
80          private final int clonedFlag;
81          private final int linkInfoFlag;
82  
83          Layout(int headerSize, int hdrLenOffset, int flagsOffset, int indexOffset, int priorityOffset, int paddingUnit,
84                  int rtaxMax, int afInet6, int clonedFlag, int linkInfoFlag) {
85              this.headerSize = headerSize;
86              this.hdrLenOffset = hdrLenOffset;
87              this.flagsOffset = flagsOffset;
88              this.indexOffset = indexOffset;
89              this.priorityOffset = priorityOffset;
90              this.paddingUnit = paddingUnit;
91              this.rtaxMax = rtaxMax;
92              this.afInet6 = afInet6;
93              this.clonedFlag = clonedFlag;
94              this.linkInfoFlag = linkInfoFlag;
95          }
96      }
97  
98      private RouteTableDump() {
99      }

101     /**
102      * Parses a routing table dump.
103      *
104      * @param buffer      The bytes {@code sysctl} returned
105      * @param layout      The platform's message layout
106      * @param ifNameByIdx Interface names by index, used to name each route's interface
107      * @return The routes read, skipping any message whose destination will not parse
108      */
109     public static List<IPRoute> parse(byte[] buffer, Layout layout, Map<Integer, String> ifNameByIdx) {
110         List<IPRoute> routes = new ArrayList<>();
111         ByteBuffer bb = ByteBuffer.wrap(buffer).order(ByteOrder.nativeOrder());
112         int offset = 0;
113         while (offset < buffer.length) {
114             // Anything the kernel did not lay out as whole messages is a truncated buffer, not a table with a little
115             // rubbish on the end. Give up on all of it rather than return part: the caller reads the table by running
116             // a command when this yields nothing, which is a better answer than a partial one.
117             int msgLen = buffer.length - offset < layout.headerSize ? 0 : bb.getShort(offset + OFF_MSGLEN) & 0xFFFF;
118             if (msgLen < layout.headerSize || offset + msgLen > buffer.length
119                     || !addressesFit(bb, buffer, offset, msgLen, layout)) {
120                 routes.clear();
121                 return routes;
122             }
123             if ((buffer[offset + OFF_TYPE] & 0xFF) == RTM_GET) {
124                 IPRoute route = parseMessage(bb, buffer, offset, msgLen, layout, ifNameByIdx);
125                 if (route != null) {
126                     routes.add(route);
127                 }
128             }
129             offset += msgLen;
130         }
131         return routes;
132     }

134     /**
135      * Checks that a message's addresses lie inside it, so the reads that follow cannot run into the next message or off
136      * the end of the buffer. A sockaddr states its own length, and nothing else constrains it.
137      */
138     private static boolean addressesFit(ByteBuffer bb, byte[] buffer, int offset, int msgLen, Layout layout) {
139         int headerSize = layout.hdrLenOffset < 0 ? layout.headerSize
140                 : bb.getShort(offset + layout.hdrLenOffset) & 0xFFFF;
141         if (headerSize < layout.headerSize || headerSize > msgLen) {
142             return false;
143         }
144         int addrs = bb.getInt(offset + OFF_ADDRS);
145         int end = offset + msgLen;
146         int sa = offset + headerSize;
147         for (int rtax = 0; rtax < layout.rtaxMax; rtax++) {
148             if ((addrs & (1 << rtax)) == 0) {
149                 continue;
150             }
151             if (sa >= end) {
152                 // The mask names an address the message does not carry. Every message on every platform measured
153                 // ends exactly on its last address, so this is a truncated message rather than a shorthand
154                 return false;
155             }
156             if (sa + 2 > end) {
157                 return false;
158             }
159             int saLen = buffer[sa] & 0xFF;
160             if (saLen > end - sa) {
161                 return false;
162             }
163             sa += saLen == 0 ? layout.paddingUnit : roundUp(saLen, layout.paddingUnit);
164             if (sa > end) {
165                 return false;
166             }
167         }
168         return true;
169     }

171     private static @Nullable IPRoute parseMessage(ByteBuffer bb, byte[] buffer, int offset, int msgLen, Layout layout,
172             Map<Integer, String> ifNameByIdx) {
173         int addrs = bb.getInt(offset + OFF_ADDRS);
174         int flags = bb.getInt(offset + layout.flagsOffset);
175         // A route cloned from another is a cache entry for one host rather than a table entry, and is not listed --
176         // except a link-layer one, which is how the neighbour cache appears and which the table does show
177         if (layout.clonedFlag != 0 && (flags & layout.clonedFlag) != 0 && (flags & layout.linkInfoFlag) == 0) {
178             return null;
179         }
180         int index = bb.getShort(offset + layout.indexOffset) & 0xFFFF;
181         int headerSize = layout.hdrLenOffset < 0 ? layout.headerSize
182                 : bb.getShort(offset + layout.hdrLenOffset) & 0xFFFF;
183 
184         byte[] destination = new byte[0];
185         byte[] gateway = new byte[0];
186         int maskStart = -1;
187         int maskLen = 0;
188 
189         int end = offset + msgLen;
190         int sa = offset + headerSize;
191         for (int rtax = 0; rtax < layout.rtaxMax && sa < end; rtax++) {
192             if ((addrs & (1 << rtax)) == 0) {
193                 continue;
194             }
195             int saLen = buffer[sa] & 0xFF;
196             int saFamily = buffer[sa + 1] & 0xFF;
197             if (rtax == RTAX_DST) {
198                 destination = readAddress(buffer, sa, saLen, saFamily, layout);
199             } else if (rtax == RTAX_GATEWAY) {
200                 gateway = readAddress(buffer, sa, saLen, saFamily, layout);
201             } else if (rtax == RTAX_NETMASK) {
202                 // The netmask is a truncated sockaddr carrying only the bytes it needs, and its family is not the
203                 // address family, so where the mask starts follows from the destination rather than from this header
204                 maskStart = sa;
205                 maskLen = saLen;
206             }
207             sa += saLen == 0 ? layout.paddingUnit : roundUp(saLen, layout.paddingUnit);
208         }
209 
210         if (destination.length == 0) {
211             return null;
212         }
213         boolean isGateway = (flags & RTF_GATEWAY) != 0;
214         int prefixLength = prefixFromMask(buffer, maskStart, maskLen, destination.length);
215         boolean isHost = (flags & RTF_HOST) != 0 || prefixLength == destination.length * 8;
216         long metric = layout.priorityOffset < 0 ? -1L : buffer[offset + layout.priorityOffset] & 0xFFL;
217         String name = ifNameByIdx.get(index);
218         return new IPRoute(destination, prefixLength, isGateway ? gateway : new byte[0], name == null ? "" : name,
219                 index, metric, isGateway, isHost);
220     }

222     /**
223      * Reads an address out of a sockaddr, yielding nothing for a link-layer or truncated one.
224      */
225     private static byte[] readAddress(byte[] buffer, int sa, int saLen, int saFamily, Layout layout) {
226         if (saFamily == AF_INET && saLen >= 8) {
227             return Arrays.copyOfRange(buffer, sa + 4, sa + 8);
228         }
229         if (saFamily == layout.afInet6 && saLen >= 24) {
230             byte[] address = Arrays.copyOfRange(buffer, sa + 8, sa + 24);
231             clearEmbeddedScope(address);
232             return address;
233         }
234         return new byte[0];
235     }

237     /**
238      * Clears the interface index a routing socket embeds in the third and fourth bytes of an address whose scope is
239      * narrower than the whole site. The address is carried that way inside the kernel; the index belongs in the scope,
240      * which this API does not report, and leaving it in place renders {@code fe80::1} as {@code fe80:17::1} and turns
241      * one multicast route into one per interface.
242      * <p>
243      * It applies to link-local unicast, and to multicast scoped to an interface or a link.
244      */
245     private static void clearEmbeddedScope(byte[] address) {
246         if (address.length != 16) {
247             return;
248         }
249         boolean linkLocalUnicast = (address[0] & 0xFF) == 0xFE && (address[1] & 0xC0) == 0x80;
250         boolean scopedMulticast = (address[0] & 0xFF) == 0xFF && (address[1] & 0x0F) <= 2;
251         if (linkLocalUnicast || scopedMulticast) {
252             address[2] = 0;
253             address[3] = 0;
254         }
255     }

257     /**
258      * Derives a prefix length from a truncated netmask sockaddr. The bytes present start where the address would in a
259      * sockaddr of the destination's family, and any byte the kernel left off is zero.
260      */
261     private static int prefixFromMask(byte[] buffer, int maskStart, int maskLen, int addressBytes) {
262         if (maskStart < 0) {
263             // No netmask supplied at all: a host route unless its flags say otherwise
264             return addressBytes * 8;
265         }
266         int addrOffset = addressBytes == 4 ? 4 : 8;
267         // maskLen comes from the message; keep the read inside the buffer whatever it claims
268         int available = Math.min(maskLen, buffer.length - maskStart);
269         byte[] mask = new byte[addressBytes];
270         for (int i = 0; i < addressBytes && addrOffset + i < available; i++) {
271             mask[i] = buffer[maskStart + addrOffset + i];
272         }
273         return ParseUtil.netmaskToPrefixLength(mask);
274     }
275 
276     private static int roundUp(int len, int unit) {
277         return 1 + ((len - 1) | (unit - 1));
278     }
