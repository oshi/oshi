20  /**
21   * Utility to query the routing table from {@code netstat}.
22   * <p>
23   * Two layouts are handled. The BSD-derived one, used by macOS, the BSDs and AIX, always places the destination, gateway
24   * and flags in the first three whitespace-delimited columns and varies only in where the interface name sits. The SVR4
25   * one, used by Solaris and illumos, differs between address families and can shift its trailing columns, so it gets its
26   * own parser.
27   */
28  @ThreadSafe
29  public final class NetstatRoute {
30  
31      /**
32       * An interface name is a letter followed by name characters. Used to reject a numeric or decorative token that a
33       * miscounted column index would otherwise report as an interface.
34       */
35      private static final Pattern INTERFACE_NAME = Pattern.compile("[A-Za-z][A-Za-z0-9._:@-]*");
36  
37      /** Column headers under which each platform prints the interface name. */
38      private static final String[] INTERFACE_HEADERS = { "Netif", "Iface", "Interface", "If", "Device" };
39  
40      private NetstatRoute() {
41      }
42  
43      /**
44       * Queries both address families from a BSD-derived {@code netstat} and returns their routes as one list.
45       * <p>
46       * Each family is queried with its own explicitly-scoped command rather than reading a combined table, which removes
47       * any need to detect the section banner separating the two halves and makes the literal {@code default}
48       * unambiguous.
49       *
50       * @param ipv4Command        the command listing the IPv4 table
51       * @param ipv6Command        the command listing the IPv6 table
52       * @param defaultIfNameIndex the token index of the interface name, used when no column header is recognized
53       * @param ifIndexByName      a map of interface name to index; an empty map reports every index as -1
54       * @return the parsed routes for both families
55       */
56      public static List<IPRoute> queryRoutes(String ipv4Command, String ipv6Command, int defaultIfNameIndex,
57              Map<String, Integer> ifIndexByName) {
58          List<IPRoute> routes = new ArrayList<>(queryRoutes(ipv4Command, false, defaultIfNameIndex, ifIndexByName));
59          routes.addAll(queryRoutes(ipv6Command, true, defaultIfNameIndex, ifIndexByName));
60          return routes;
61      }

63      /**
64       * Queries and parses a BSD-derived {@code netstat} routing table.
65       *
66       * @param command            the command to run, which must name a single address family
67       * @param ipv6               whether the command selects the IPv6 table
68       * @param defaultIfNameIndex the token index of the interface name, used when no column header is recognized
69       * @param ifIndexByName      a map of interface name to index; an empty map reports every index as -1
70       * @return the parsed routes
71       */
72      public static List<IPRoute> queryRoutes(String command, boolean ipv6, int defaultIfNameIndex,
73              Map<String, Integer> ifIndexByName) {
74          return parseRoutes(ExecutingCommand.runNative(command), ipv6, defaultIfNameIndex, ifIndexByName);
75      }

77      /**
78       * Parses the BSD-derived {@code netstat} routing table layout used by macOS, FreeBSD, DragonFly BSD, NetBSD,
79       * OpenBSD and AIX, in which the destination, gateway and flags are always the first three whitespace-delimited
80       * tokens.
81       * <p>
82       * Rows are recognized structurally rather than by matching header text, because the three verified platforms print
83       * their banner, column header and separator in three different orders, and header text may be localized.
84       *
85       * @param netstat            the lines of {@code netstat} output for one address family
86       * @param ipv6               whether this is the IPv6 table
87       * @param defaultIfNameIndex the token index of the interface name, used when no column header is recognized
88       * @param ifIndexByName      a map of interface name to index; an empty map reports every index as -1
89       * @return the parsed routes
90       */
91      static List<IPRoute> parseRoutes(List<String> netstat, boolean ipv6, int defaultIfNameIndex,
92              Map<String, Integer> ifIndexByName) {
93          int ifNameIndex = defaultIfNameIndex;
94          int metricIndex = -1;
95          List<IPRoute> routes = new ArrayList<>();
96          for (String line : netstat) {
97              String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
98              if (fields.length < 3) {
99                  continue;
100             }
101             // A column header both identifies itself and tells us where this platform puts the interface name, which
102             // is the only thing that varies across the six platforms sharing this layout.
103             if (isColumnHeader(fields[0])) {
104                 int headerIfIndex = indexOfAny(fields, INTERFACE_HEADERS);
105                 if (headerIfIndex >= 0) {
106                     ifNameIndex = headerIfIndex;
107                 }
108                 metricIndex = indexOfAny(fields, "Prio");
109                 continue;
110             }
111             IPRoute route = parseBsdRow(fields, ipv6, ifNameIndex, metricIndex, ifIndexByName);
112             if (route != null) {
113                 routes.add(route);
114             }
115         }
116         return routes;
117     }
118 
119     private static @Nullable IPRoute parseBsdRow(String[] fields, boolean ipv6, int ifNameIndex, int metricIndex,
120             Map<String, Integer> ifIndexByName) {
121         // The flags column is a run of letters and the destination is an address. Together these identify a data row
122         // and reject every banner, header and separator line the three verified platforms emit.
123         if (!ParseUtil.isRouteFlags(fields[2])) {
124             return null;
125         }
126         Pair<byte[], Integer> dest = ParseUtil.parseRouteDestination(fields[0], ipv6);
127         byte[] destination = dest.getA();
128         if (destination.length == 0) {
129             return null;
130         }
131         String flags = fields[2];
132         boolean isGateway = flags.indexOf('G') >= 0;
133         boolean hostFlag = flags.indexOf('H') >= 0;
134         int prefixLength = resolvePrefixLength(dest.getB(), hostFlag, destination.length);
135         // Only a gateway route has a next hop. On a directly attached route this column holds the interface's own
136         // address, a link-layer address, or a symbolic name such as "link#14", none of which is a gateway.
137         byte[] gateway = isGateway ? parseAddress(fields[1], ipv6) : new byte[0];
138         String interfaceName = readInterfaceName(fields, ifNameIndex);
139         long metric = metricIndex >= 0 && metricIndex < fields.length
140                 ? ParseUtil.parseLongOrDefault(fields[metricIndex], -1L)
141                 : -1L;
142         return new IPRoute(destination, prefixLength, gateway, interfaceName,
143                 indexOfInterface(interfaceName, ifIndexByName), metric, isGateway,
144                 hostFlag || prefixLength == destination.length * 8);
145     }

164      *
165      * @param command       the command to run, which must name a single address family

174     /**
175      * Parses the Solaris {@code netstat -rnv} IRE table.
176      * <p>
177      * The verbose form is required because {@code netstat -rn -f inet} publishes no netmask at all, leaving the prefix
178      * length permanently unknown. The verbose form adds a Mask column for IPv4, but reorders the remaining columns and
179      * places the flags differently for the two address families:
180      *
181      * <pre>
182      * IPv4: Destination Mask Gateway Device MTU Ref Flg Out In/Fwd
183      * IPv6: Destination/Mask Gateway If MTU Ref Flags Out In/Fwd
184      * </pre>
185      *
186      * The optional Device column may be empty, which whitespace splitting collapses, shifting every following token
187      * left by one. The flags column is therefore located by scanning from the right, where the trailing columns are
188      * always numeric, rather than by a fixed index.
189      *
190      * @param netstat       the lines of {@code netstat -rnv} output for one address family
191      * @param ipv6          whether this is the IPv6 table
192      * @param ifIndexByName a map of interface name to index; an empty map reports every index as -1
193      * @return the parsed routes
194      */
195     static List<IPRoute> parseSolarisRoutes(List<String> netstat, boolean ipv6, Map<String, Integer> ifIndexByName) {
196         // IPv4 carries a Mask column between the destination and the gateway; IPv6 folds the prefix into the
197         // destination and has none.
198         int gatewayIndex = ipv6 ? 1 : 2;
199         int deviceIndex = gatewayIndex + 1;
200         List<IPRoute> routes = new ArrayList<>();
201         for (String line : netstat) {
202             String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
203             if (fields.length < deviceIndex + 1) {
204                 continue;
205             }
206             int flagsIndex = findFlagsFromRight(fields, deviceIndex);
207             if (flagsIndex < 0) {
208                 continue;
209             }
210             Pair<byte[], Integer> dest = ParseUtil.parseRouteDestination(fields[0], ipv6);
211             byte[] destination = dest.getA();
212             if (destination.length == 0) {
213                 continue;
214             }
215             String flags = fields[flagsIndex];
216             boolean isGateway = flags.indexOf('G') >= 0;
217             boolean hostFlag = flags.indexOf('H') >= 0;
218             int prefixLength = dest.getB();
219             if (prefixLength < 0 && !ipv6) {
220                 prefixLength = ParseUtil.netmaskToPrefixLength(fields[1]);
221             }
222             prefixLength = resolvePrefixLength(prefixLength, hostFlag, destination.length);
223             byte[] gateway = isGateway ? parseAddress(fields[gatewayIndex], ipv6) : new byte[0];
224             // With every column present the flags sit three past the device. Anything less means the optional device
225             // column was empty and collapsed under whitespace splitting, so there is no interface to report. Do not
226             // derive the device index from the flags index arithmetically: on a collapsed row that lands on the
227             // gateway.
228             String interfaceName = flagsIndex == deviceIndex + 3 ? readInterfaceName(fields, deviceIndex) : "";
229             routes.add(new IPRoute(destination, prefixLength, gateway, interfaceName,
230                     indexOfInterface(interfaceName, ifIndexByName), -1L, isGateway,
231                     hostFlag || prefixLength == destination.length * 8));
232         }
233         return routes;
234     }

236     /**
237      * Locates the flags column by scanning right to left. Every column to its right is numeric, so the first token of
238      * letters found is the flags field.
239      */
240     private static int findFlagsFromRight(String[] fields, int floor) {
241         for (int i = fields.length - 1; i >= floor; i--) {
242             if (ParseUtil.isRouteFlags(fields[i])) {
243                 return i;
244             }
245         }
246         return -1;
247     }
248 
249     /**
250      * Applies the H flag to a destination token that stated no prefix. A bare address in a routing table is a host
251      * route, which covers the whole address.
252      */
253     private static int resolvePrefixLength(int statedPrefix, boolean hostFlag, int addressBytes) {
254         if (statedPrefix >= 0) {
255             return statedPrefix;
256         }
257         return hostFlag ? addressBytes * 8 : -1;
258     }
259 
260     private static byte[] parseAddress(String token, boolean ipv6) {
261         return ipv6 ? ParseUtil.parseIpv6AddressToBytes(token) : ParseUtil.parseIpv4AddressToBytes(token);
262     }
263 
264     /**
265      * Reads the interface name, rejecting anything not shaped like one. A wrong column index therefore degrades to an
266      * empty name rather than reporting a reference count or an expiry time as an interface.
267      */
268     private static String readInterfaceName(String[] fields, int index) {
269         if (index < 0 || index >= fields.length) {
270             return "";
271         }
272         String token = fields[index];
273         return INTERFACE_NAME.matcher(token).matches() ? token : "";
274     }
275 
276     private static int indexOfInterface(String name, Map<String, Integer> ifIndexByName) {
277         if (name.isEmpty()) {
278             return -1;
279         }
280         Integer index = ifIndexByName.get(name);
281         return index == null ? -1 : index;
282     }
283 
284     private static boolean isColumnHeader(String firstToken) {
285         return "Destination".equals(firstToken) || "Destination/Mask".equals(firstToken);
286     }
287 
288     private static int indexOfAny(String[] fields, String... candidates) {
289         for (int i = 0; i < fields.length; i++) {
290             for (String candidate : candidates) {
291                 if (candidate.equals(fields[i])) {
292                     return i;
293                 }
294             }
295         }
296         return -1;
297     }
