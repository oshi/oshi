7   import java.util.ArrayList;
8   import java.util.List;
9   import java.util.Map;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.software.os.NetworkParams.IPRoute;
13  import oshi.util.FileUtil;
14  import oshi.util.ParseUtil;
15  import oshi.util.linux.ProcPath;

17  /**
18   * Utility to read the routing table from {@code /proc/net/route} and {@code /proc/net/ipv6_route}.
19   * <p>
20   * The two files share no format. The IPv4 one has a header line, eleven columns, the interface first, and addresses as
21   * eight little-endian hex digits with a separate mask column. The IPv6 one has no header, ten columns, the interface
22   * last, addresses as thirty-two hex digits in network order, and the prefix length beside the destination.
23   */
24  @ThreadSafe
25  public final class RouteTable {
26  
27      /** Route is a gateway route, {@code RTF_GATEWAY}. */
28      private static final int RTF_GATEWAY = 0x0002;
29      /** Route is to a single host, {@code RTF_HOST}. */
30      private static final int RTF_HOST = 0x0004;
31  
32      private static final int IPV4_FIELD_COUNT = 11;
33      private static final int IPV6_FIELD_COUNT = 10;
34  
35      private RouteTable() {
36      }
37  
38      /**
39       * Queries and parses {@code /proc/net/route}.
40       *
41       * @param ifIndexByName a map of interface name to index; an empty map reports every index as -1
42       * @return the parsed IPv4 routes
43       */
44      public static List<IPRoute> queryIpv4Routes(Map<String, Integer> ifIndexByName) {
45          return parseIpv4Routes(FileUtil.readFile(ProcPath.ROUTE), ifIndexByName);
46      }
47  
48      /**
49       * Parses {@code /proc/net/route} content.
50       * <p>
51       * Columns are {@code Iface Destination Gateway Flags RefCnt Use Metric Mask MTU Window IRTT}. Addresses are the
52       * kernel's {@code %08X} of a {@code __be32}, which on a little-endian host is exactly what
53       * {@link ParseUtil#parseIntToIP(int)} inverts.
54       *
55       * @param lines         the lines of {@code /proc/net/route}
56       * @param ifIndexByName a map of interface name to index; an empty map reports every index as -1
57       * @return the parsed IPv4 routes
58       */
59      static List<IPRoute> parseIpv4Routes(List<String> lines, Map<String, Integer> ifIndexByName) {
60          List<IPRoute> routes = new ArrayList<>();
61          for (String line : lines) {
62              String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
63              if (fields.length < IPV4_FIELD_COUNT) {
64                  continue;
65              }
66              // The header line's Destination column is the word "Destination", which is not eight hex digits. Testing
67              // the shape rather than a parse result keeps a genuine 0xFFFFFFFF destination from being mistaken for a
68              // parse failure.
69              if (!isHexWord(fields[1], 8)) {
70                  continue;
71              }
72              byte[] destination = ParseUtil.parseIntToIP((int) ParseUtil.hexStringToLong(fields[1], 0L));
73              int flags = ParseUtil.hexStringToInt(fields[3], 0);
74              boolean isGateway = (flags & RTF_GATEWAY) != 0;
75              int prefixLength = ParseUtil
76                      .netmaskToPrefixLength(ParseUtil.parseIntToIP((int) ParseUtil.hexStringToLong(fields[7], 0L)));
77              byte[] gateway = isGateway ? ParseUtil.parseIntToIP((int) ParseUtil.hexStringToLong(fields[2], 0L))
78                      : new byte[0];
79              String interfaceName = fields[0];
80              routes.add(new IPRoute(destination, prefixLength, gateway, interfaceName,
81                      indexOfInterface(interfaceName, ifIndexByName), ParseUtil.parseLongOrDefault(fields[6], -1L),
82                      isGateway, (flags & RTF_HOST) != 0 || prefixLength == 32));
83          }
84          return routes;
85      }
86  
87      /**
88       * Queries and parses {@code /proc/net/ipv6_route}.
89       *
90       * @param ifIndexByName a map of interface name to index; an empty map reports every index as -1
91       * @return the parsed IPv6 routes
92       */
93      public static List<IPRoute> queryIpv6Routes(Map<String, Integer> ifIndexByName) {
94          return parseIpv6Routes(FileUtil.readFile(ProcPath.IPV6_ROUTE), ifIndexByName);
95      }
96  
97      /**
98       * Parses {@code /proc/net/ipv6_route} content.
99       * <p>
100      * There is no header line. Columns are {@code destination prefixLength source sourcePrefixLength nextHop metric
101      * refCount use flags interface}, with addresses as thirty-two hex digits.
102      *
103      * @param lines         the lines of {@code /proc/net/ipv6_route}
104      * @param ifIndexByName a map of interface name to index; an empty map reports every index as -1
105      * @return the parsed IPv6 routes
106      */
107     static List<IPRoute> parseIpv6Routes(List<String> lines, Map<String, Integer> ifIndexByName) {
108         List<IPRoute> routes = new ArrayList<>();
109         for (String line : lines) {
110             String[] fields = ParseUtil.whitespaces.split(line.trim(), -1);
111             if (fields.length < IPV6_FIELD_COUNT) {
112                 continue;
113             }
114             if (!isHexWord(fields[0], 32)) {
115                 continue;
116             }
117             // The kernel prints these addresses one byte at a time, so they are already in network order and need
118             // none of the per-word reversal that /proc/net/tcp6 requires.
119             byte[] destination = ParseUtil.hexStringToByteArray(fields[0]);
120             long flags = ParseUtil.hexStringToLong(fields[8], 0L);
121             boolean isGateway = (flags & RTF_GATEWAY) != 0;
122             int prefixLength = ParseUtil.hexStringToInt(fields[1], -1);
123             byte[] nextHop = ParseUtil.hexStringToByteArray(fields[4]);
124             byte[] gateway = isGateway && nextHop.length == 16 ? nextHop : new byte[0];
125             String interfaceName = fields[9];
126             // A /128 local route does not reliably carry RTF_HOST, so the prefix width is checked as well
127             routes.add(new IPRoute(destination, prefixLength, gateway, interfaceName,
128                     indexOfInterface(interfaceName, ifIndexByName), ParseUtil.hexStringToLong(fields[5], -1L),
129                     isGateway, (flags & RTF_HOST) != 0 || prefixLength == 128));
130         }
131         return routes;
132     }
133 
134     private static boolean isHexWord(String token, int length) {
135         if (token.length() != length) {
136             return false;
137         }
138         for (int i = 0; i < token.length(); i++) {
139             char c = token.charAt(i);
140             if ((c < '0' || c > '9') && (c < 'a' || c > 'f') && (c < 'A' || c > 'F')) {
141                 return false;
142             }
143         }
144         return true;
145     }
146 
147     private static int indexOfInterface(String name, Map<String, Integer> ifIndexByName) {
148         if (name.isEmpty()) {
149             return -1;
150         }
151         Integer index = ifIndexByName.get(name);
152         return index == null ? -1 : index;
153     }
