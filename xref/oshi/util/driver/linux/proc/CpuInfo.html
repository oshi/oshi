20  /**
21   * Utility to read CPU info from {@code /proc/cpuinfo}
22   */
23  @ThreadSafe
24  public final class CpuInfo {
25  
26      private static final String ARCHITECTURE_PROPERTIES = "oshi.architecture.properties";
27  
28      private CpuInfo() {
29      }
30  
31      /**
32       * Gets the CPU manufacturer from {@code /proc/cpuinfo}
33       *
34       * @return The manufacturer if known, null otherwise
35       */
36      public static @Nullable String queryCpuManufacturer() {
37          return queryCpuManufacturer(FileUtil.readFile(CPUINFO));
38      }
39  
40      /**
41       * Parse the CPU manufacturer from /proc/cpuinfo lines.
42       *
43       * @param cpuInfo lines from /proc/cpuinfo
44       * @return The manufacturer if known, null otherwise
45       */
46      static @Nullable String queryCpuManufacturer(List<String> cpuInfo) {
47          for (String line : cpuInfo) {
48              if (line.startsWith("CPU implementer")) {
49                  // The implementer codes are those ProcessorIdentifier translates, from the same table
50                  String key = String.format(Locale.ROOT, "hw_impl.0x%02x", ParseUtil.parseLastInt(line, 0));
51                  return FileUtil.readPropertiesFromFilename(ARCHITECTURE_PROPERTIES).getProperty(key);
52              }
53          }
54          return null;
55      }
56  
57      /**
58       * Gets the board manufacturer, model, version, and serial number from {@code /proc/cpuinfo}
59       *
60       * @return A quartet of strings for manufacturer, model, version, and serial number. Each one may be null if
61       *         unknown.
62       */
63      public static Quartet<@Nullable String, @Nullable String, @Nullable String, @Nullable String> queryBoardInfo() {
64          return queryBoardInfo(FileUtil.readFile(CPUINFO));
65      }
66  
67      /**
68       * Parse board info from /proc/cpuinfo lines.
69       *
70       * @param cpuInfo lines from /proc/cpuinfo
71       * @return A quartet of strings for manufacturer, model, version, and serial number
72       */
73      static Quartet<@Nullable String, @Nullable String, @Nullable String, @Nullable String> queryBoardInfo(
74              List<String> cpuInfo) {
75          String pcManufacturer = null;
76          String pcModel = null;
77          String pcVersion = null;
78          String pcSerialNumber = null;
79  
80          for (String line : cpuInfo) {
81              String[] splitLine = ParseUtil.whitespacesColonWhitespace.split(line, -1);
82              if (splitLine.length < 2) {
83                  continue;
84              }
85              switch (splitLine[0]) {
86                  case "Hardware":
87                      pcModel = splitLine[1];
88                      break;
89                  case "Revision":
90                      pcVersion = splitLine[1];
91                      pcManufacturer = queryBoardManufacturer(pcVersion);
92                      break;
93                  case "Serial":
94                      pcSerialNumber = splitLine[1];
95                      break;
96                  default:
97                      // Do nothing
98              }
99          }
100         return new Quartet<>(pcManufacturer, pcModel, pcVersion, pcSerialNumber);
101     }
102 
103     /**
104      * Decodes the manufacturer of a Raspberry Pi from its revision code. Only the new-style codes, which set bit 23,
105      * carry one, in bits 16 to 19. Other ARM kernels print a revision too, usually {@code 0000}, that means nothing of
106      * the kind.
107      *
108      * @param revision the {@code Revision} value, in hex
109      * @return the manufacturer, or null if the code does not name one
110      */
111     private static @Nullable String queryBoardManufacturer(String revision) {
112         long code = ParseUtil.hexStringToLong(revision, 0L);
113         if ((code & 0x80_0000L) == 0) {
114             return null;
115         }
116         switch ((int) (code >> 16) & 0xF) {
117             case 0:
118                 return "Sony UK";
119             case 1:
120                 return "Egoman";
121             case 2:
122             case 4:
123                 return "Embest";
124             case 3:
125                 return "Sony Japan";
126             case 5:
127                 return "Stadium";
128             default:
129                 return null;
130         }
131     }

133     /**
134      * Queries CPU feature flags from /proc/cpuinfo.
135      *
136      * @return a list of feature flag strings
137      */
138     public static List<String> queryFeatureFlags() {
139         return queryFeatureFlags(FileUtil.readFile(CPUINFO));
140     }
141 
142     /**
143      * Parse feature flag lines from /proc/cpuinfo.
144      *
145      * @param cpuInfo lines from /proc/cpuinfo
146      * @return deduplicated lines starting with "flags" or "features"
147      */
148     static List<String> queryFeatureFlags(List<String> cpuInfo) {
149         return cpuInfo.stream().filter(f -> {
150             String s = f.toLowerCase(Locale.ROOT);
151             return s.startsWith("flags") || s.startsWith("features");
152         }).distinct().collect(Collectors.toList());
153     }
