5   package oshi.util.linux;
6   
7   import java.util.Collections;
8   import java.util.HashMap;
9   import java.util.List;
10  import java.util.Map;
11  
12  import oshi.annotation.concurrent.ThreadSafe;
13  import oshi.util.FileUtil;
14  
15  /**
16   * Reads device properties from the udev database that {@code systemd-udevd} keeps under {@code /run/udev/data}. The
17   * files are world-readable and hold the same {@code E:} properties libudev returns from
18   * {@code udev_device_get_property_value}, so code that cannot load libudev can still read them.
19   * <p>
20   * Each device has one file, named by subsystem: {@code b<major>:<minor>} for a block device, {@code n<ifindex>} for a
21   * network interface, and {@code +<subsystem>:<name>} for a device with no device node, such as a PCI function. A host
22   * without {@code udevd}, such as most containers, has no database; every query then returns an empty map.
23   */
24  @ThreadSafe
25  public final class UdevData {
26  
27      /** The udev database directory. */
28      public static final String RUN_UDEV_DATA = "/run/udev/data/";
29  
30      private UdevData() {
31      }
32  
33      /**
34       * Reads the udev properties of a block device.
35       *
36       * @param major the device's major number
37       * @param minor the device's minor number
38       * @return the properties, or an empty map if udev has no record of the device
39       */
40      public static Map<String, String> queryBlockDevice(int major, int minor) {
41          return queryBlockDevice(RUN_UDEV_DATA, major, minor);
42      }
43  
44      /**
45       * Reads the udev properties of a block device from a given database directory. Used for testing and for a database
46       * copied from another host.
47       *
48       * @param dataDir the database directory, ending in a slash
49       * @param major   the device's major number
50       * @param minor   the device's minor number
51       * @return the properties, or an empty map if there is no record
52       */
53      public static Map<String, String> queryBlockDevice(String dataDir, int major, int minor) {
54          if (major <= 0) {
55              return Collections.emptyMap();
56          }
57          return parseProperties(FileUtil.readFile(dataDir + "b" + major + ":" + minor, false));
58      }
59  
60      /**
61       * Reads the udev properties of a network interface.
62       *
63       * @param ifindex the interface index, as in {@code /sys/class/net/<name>/ifindex}
64       * @return the properties, or an empty map if udev has no record of the interface
65       */
66      public static Map<String, String> queryNetworkInterface(int ifindex) {
67          return queryNetworkInterface(RUN_UDEV_DATA, ifindex);
68      }
69  
70      /**
71       * Reads the udev properties of a network interface from a given database directory. Used for testing and for a
72       * database copied from another host.
73       *
74       * @param dataDir the database directory, ending in a slash
75       * @param ifindex the interface index
76       * @return the properties, or an empty map if there is no record
77       */
78      public static Map<String, String> queryNetworkInterface(String dataDir, int ifindex) {
79          if (ifindex <= 0) {
80              return Collections.emptyMap();
81          }
82          return parseProperties(FileUtil.readFile(dataDir + "n" + ifindex, false));
83      }
84  
85      /**
86       * Reads the udev properties of a device that has no device node, such as a PCI function.
87       *
88       * @param subsystem the device's subsystem, such as {@code pci}
89       * @param sysname   the device's name in that subsystem, such as {@code 0000:01:00.0}
90       * @return the properties, or an empty map if udev has no record of the device
91       */
92      public static Map<String, String> queryDevice(String subsystem, String sysname) {
93          return queryDevice(RUN_UDEV_DATA, subsystem, sysname);
94      }
95  
96      /**
97       * Reads the udev properties of a device that has no device node from a given database directory. Used for testing
98       * and for a database copied from another host.
99       *
100      * @param dataDir   the database directory, ending in a slash
101      * @param subsystem the device's subsystem
102      * @param sysname   the device's name in that subsystem
103      * @return the properties, or an empty map if there is no record
104      */
105     public static Map<String, String> queryDevice(String dataDir, String subsystem, String sysname) {
106         if (subsystem.isEmpty() || sysname.isEmpty()) {
107             return Collections.emptyMap();
108         }
109         return parseProperties(FileUtil.readFile(dataDir + "+" + subsystem + ":" + sysname, false));
110     }
111 
112     /**
113      * Parses the property lines of a udev database file. A property line is {@code E:KEY=VALUE}; the other records
114      * ({@code S:} symlinks, {@code G:} tags, {@code I:} initialization time and so on) are ignored.
115      *
116      * @param lines the lines of the file
117      * @return the properties keyed by name
118      */
119     static Map<String, String> parseProperties(List<String> lines) {
120         Map<String, String> properties = new HashMap<>();
121         for (String line : lines) {
122             if (line.startsWith("E:")) {
123                 int eq = line.indexOf('=');
124                 if (eq > 2) {
125                     properties.put(line.substring(2, eq), line.substring(eq + 1));
126                 }
127             }
128         }
129         return properties;
130     }
131 }
