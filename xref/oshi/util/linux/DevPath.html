5   package oshi.util.linux;
6   
7   import java.io.File;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.util.GlobalConfig;
11  
12  /**
13   * Provides constants for paths in the {@code /dev} filesystem on Linux.
14   * <p>
15   * Two roots are involved. {@link #DEV} is where OSHI reads the device filesystem, and can be configured elsewhere, for
16   * example to a host's {@code /dev} mounted into a container. {@link #NODE_ROOT} is the root of the device names the
17   * kernel, udev, {@code /proc/mounts} and tools such as {@code pvs} report, which is always {@code /dev/}. Device names
18   * OSHI returns use {@link #NODE_ROOT}, so they match those sources wherever OSHI reads the files from; the prefix
19   * constants other than {@link #DISK_BY_UUID} are names, and {@link #toFilePath(String)} gives the path to read one.
20   * <p>
21   * If the user desires to configure a custom {@code /dev} path, it must be declared in the OSHI configuration file or
22   * updated in the {@link GlobalConfig} class prior to initializing this class.
23   */
24  @ThreadSafe
25  public final class DevPath {
26  
27      /**
28       * The /dev filesystem location, where device files are read.
29       */
30      public static final String DEV = queryDevConfig() + "/";
31  
32      /**
33       * The root of device node names as the kernel and udev report them, regardless of {@link #DEV}.
34       */
35      public static final String NODE_ROOT = "/dev/";
36  
37      /** Path to disk-by-uuid directory, below {@link #DEV}. */
38      public static final String DISK_BY_UUID = DEV + "disk/by-uuid";
39      /** Name prefix for device-mapper devices. */
40      public static final String DM = NODE_ROOT + "dm";
41      /** Name prefix for loop devices. */
42      public static final String LOOP = NODE_ROOT + "loop";
43      /** Name prefix for device-mapper mapped names. */
44      public static final String MAPPER = NODE_ROOT + "mapper/";
45      /** Name prefix for RAM disk devices. */
46      public static final String RAM = NODE_ROOT + "ram";
47  
48      private DevPath() {
49      }
50  
51      private static String queryDevConfig() {
52          String devPath = GlobalConfig.get(GlobalConfig.OSHI_UTIL_DEV_PATH, "/dev");
53          // Ensure prefix begins with path separator, but doesn't end with one
54          devPath = '/' + devPath.replaceAll("/$|^/", "");
55          if (!new File(devPath).exists()) {
56              throw new GlobalConfig.PropertyException(GlobalConfig.OSHI_UTIL_DEV_PATH, "The path does not exist");
57          }
58          return devPath;
59      }
60  
61      /**
62       * Converts a device name below {@link #NODE_ROOT} to the path below {@link #DEV} where its file can be read.
63       *
64       * @param nodeName a device name such as {@code /dev/mapper/vg-root}
65       * @return the path to read, or {@code nodeName} unchanged if it is not below {@link #NODE_ROOT}
66       */
67      public static String toFilePath(String nodeName) {
68          return reroot(nodeName, NODE_ROOT, DEV);
69      }
70  
71      /**
72       * Converts a path below {@link #DEV}, such as a resolved symbolic link, to the device name below
73       * {@link #NODE_ROOT}.
74       *
75       * @param filePath a path such as {@code /host/dev/sda1}
76       * @return the device name, or {@code filePath} unchanged if it is not below {@link #DEV}
77       */
78      public static String toNodeName(String filePath) {
79          return reroot(filePath, DEV, NODE_ROOT);
80      }
81  
82      /**
83       * Replaces one root prefix with another. Package-private for testing, since {@link #DEV} is fixed at startup.
84       *
85       * @param path the path to convert
86       * @param from the root to replace, ending in a slash
87       * @param to   the root to replace it with, ending in a slash
88       * @return the converted path, or {@code path} unchanged if it is not below {@code from}
89       */
90      static String reroot(String path, String from, String to) {
91          return path.startsWith(from) ? to + path.substring(from.length()) : path;
92      }
93  }
