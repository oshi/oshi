34  /**
35   * Optional runtime binding to the NVIDIA Management Library (NVML). All methods return sentinel values ({@code -1} or
36   * {@code -1L}) when NVML is unavailable or a specific query fails.
37   *
38   * <p>
39   * The native library is loaded once at class initialization. Each metric call pairs {@code nvmlInit_v2} with
40   * {@code nvmlShutdown} to correctly manage NVML's internal reference count, ensuring OSHI does not interfere with other
41   * code in the same process that may also be managing the NVML lifecycle.
42   *
43   * <p>
44   * Device handles are enumerated once on first successful init and cached by PCI bus ID string for correlation with OSHI
45   * GraphicsCard instances. The query skeleton shared with the FFM binding lives in {@link NvmlQuery}.
46   */
47  @ThreadSafe
48  public final class NvmlUtilJNA {
49  
50      private static final Logger LOG = LoggerFactory.getLogger(NvmlUtilJNA.class);
51  
52      // -------------------------------------------------------------------------
53      // Library loading (holder pattern — loads the .dll/.so once)
54      // -------------------------------------------------------------------------
55  
56      private static final class Holder {
57          static final NvmlLibrary LIB;
58          static final boolean LIBRARY_LOADED;
59  
60          static {
61              NvmlLibrary lib = null;
62              boolean loaded = false;
63              try {
64                  lib = Native.load(Platform.isWindows() ? "nvml" : "nvidia-ml", NvmlLibrary.class);
65                  loaded = true;
66                  LOG.debug("NVML library loaded");
67              } catch (UnsatisfiedLinkError | NoClassDefFoundError e) {
68                  LOG.debug("NVML library not available: {}", e.getMessage());
69              }
70              LIB = lib;
71              LIBRARY_LOADED = loaded;
72          }
73      }
74  
75      // Stores PCI bus ID strings (stable identifiers) rather than Pointer handles, which are only valid within a
76      // single nvmlInit/nvmlShutdown scope.
77      private static final NvmlDeviceCache DEVICE_CACHE = new NvmlDeviceCache("JNA");
78  
79      private static final NvmlScope<Pointer> SCOPE = new NvmlScope<Pointer>() {
80          @Override
81          public boolean init() {
82              return nvmlInit();
83          }
84  
85          @Override
86          public void uninit() {
87              nvmlUninit();
88          }
89  
90          @Override
91          public <R> R withDevice(String deviceId, Function<Pointer, R> body, R sentinel) {
92              Pointer device = acquireHandleByBusId(deviceId);
93              return device == null ? sentinel : body.apply(device);
94          }
95      };

104     /**
105      * Calls {@code nvmlInit_v2}, incrementing NVML's internal reference count. Every successful call must be paired
106      * with exactly one call to {@link #nvmlUninit()}, which decrements the same counter. This ensures OSHI does not
107      * permanently hold a reference that would interfere with other code in the process managing the NVML lifecycle.
108      *
109      * @return true if this call successfully initialized NVML and must be paired with {@link #nvmlUninit()}
110      */
111     private static boolean nvmlInit() {
112         if (!Holder.LIBRARY_LOADED) {
113             return false;
114         }
115         int ret = Holder.LIB.nvmlInit_v2();
116         if (ret == Nvml.NVML_SUCCESS) {
117             return true;
118         }
119         LOG.debug("nvmlInit_v2 failed with code {}", ret);
120         return false;
121     }
122 
123     /**
124      * Calls {@code nvmlShutdown}, decrementing the same internal reference count that {@link #nvmlInit()} incremented.
125      * Must be called exactly once for each successful call to {@link #nvmlInit()}.
126      */
127     private static void nvmlUninit() {
128         Holder.LIB.nvmlShutdown();
129     }

135     /**
136      * Applies {@code visitor} to each NVML device handle in index order, stopping early if it returns true. Devices
137      * whose handle cannot be acquired are skipped. Must be called while NVML is initialized.
138      *
139      * @param visitor applied to each handle, returning true to stop the walk
140      * @return false if the device count could not be read, letting callers distinguish an NVML failure from a machine
141      *         with no devices
142      */
143     private static boolean forEachDevice(Predicate<Pointer> visitor) {
144         IntByReference countRef = new IntByReference();
145         if (Holder.LIB.nvmlDeviceGetCount_v2(countRef) != Nvml.NVML_SUCCESS) {
146             return false;
147         }
148         int count = countRef.getValue();
149         for (int i = 0; i < count; i++) {
150             PointerByReference handleRef = new PointerByReference();
151             if (Holder.LIB.nvmlDeviceGetHandleByIndex_v2(i, handleRef) != Nvml.NVML_SUCCESS) {
152                 continue;
153             }
154             if (visitor.test(handleRef.getValue())) {
155                 return true;
156             }
157         }
158         return true;
159     }

161     /**
162      * Reads a device's two PCI bus ID forms, lowercased.
163      *
164      * @param handle the device handle
165      * @return the modern and legacy bus IDs, or {@code null} if the PCI info could not be read
166      */
167     private static Pair<String, String> readBusIds(Pointer handle) {
168         NvmlPciInfo pci = new NvmlPciInfo();
169         if (Holder.LIB.nvmlDeviceGetPciInfo_v3(handle, pci) != Nvml.NVML_SUCCESS) {
170             return null;
171         }
172         pci.read();
173         return new Pair<>(Native.toString(pci.busId).toLowerCase(Locale.ROOT),
174                 Native.toString(pci.busIdLegacy).toLowerCase(Locale.ROOT));
175     }
176 
177     /**
178      * Reads a device's name, lowercased.
179      *
180      * @param handle the device handle
181      * @return the name, or {@code null} if it could not be read
182      */
183     private static String readName(Pointer handle) {
184         byte[] nameBuf = new byte[Nvml.NVML_DEVICE_NAME_BUFFER_SIZE];
185         if (Holder.LIB.nvmlDeviceGetName(handle, nameBuf, nameBuf.length) != Nvml.NVML_SUCCESS) {
186             return null;
187         }
188         return Native.toString(nameBuf).toLowerCase(Locale.ROOT);
189     }
190 
191     /**
192      * Returns a set of PCI bus ID strings for all NVML devices, or {@code null} on NVML error (so the caller can
193      * distinguish a real failure from a legitimate empty result).
194      *
195      * @return set of PCI bus ID strings, or {@code null} on NVML error
196      */
197     private static Set<String> enumerateDeviceBusIds() {
198         Set<String> ids = new HashSet<>();
199         boolean enumerated = forEachDevice(handle -> {
200             Pair<String, String> busIds = readBusIds(handle);
201             if (busIds != null) {
202                 addIfNotEmpty(ids, busIds.getA());
203                 addIfNotEmpty(ids, busIds.getB());
204             }
205             return false;
206         });
207         return enumerated ? Collections.unmodifiableSet(ids) : null;
208     }
209 
210     private static void addIfNotEmpty(Set<String> ids, String id) {
211         if (!id.isEmpty()) {
212             ids.add(id);
213         }
214     }

216     /**
217      * Acquires a fresh device handle within the current init scope by matching the given PCI bus ID fragment. Must be
218      * called while NVML is initialized.
219      *
220      * @param pciBusId PCI bus ID fragment to match
221      * @return device handle Pointer, or {@code null} if not found
222      */
223     private static Pointer acquireHandleByBusId(String pciBusId) {
224         String needle = pciBusId.toLowerCase(Locale.ROOT);
225         Pointer[] found = new Pointer[1];
226         forEachDevice(handle -> {
227             Pair<String, String> busIds = readBusIds(handle);
228             if (busIds != null
229                     && (NvmlQuery.matches(busIds.getA(), needle) || NvmlQuery.matches(busIds.getB(), needle))) {
230                 found[0] = handle;
231                 return true;
232             }
233             return false;
234         });
235         return found[0];
236     }
237 
238     /**
239      * Acquires a fresh device handle within the current init scope by matching the GPU name. Must be called while NVML
240      * is initialized.
241      *
242      * @param gpuName GPU name to match (case-insensitive substring)
243      * @return device handle Pointer, or {@code null} if not found
244      */
245     private static Pointer acquireHandleByName(String gpuName) {
246         String needle = gpuName.toLowerCase(Locale.ROOT);
247         Pointer[] found = new Pointer[1];
248         forEachDevice(handle -> {
249             String name = readName(handle);
250             if (name != null && NvmlQuery.matches(name, needle)) {
251                 found[0] = handle;
252                 return true;
253             }
254             return false;
255         });
256         return found[0];
257     }
258 
259     /**
260      * Counts the devices whose name matches, so an ambiguous match can be rejected rather than resolved arbitrarily.
261      *
262      * @param gpuName GPU name to match
263      * @return the number of matching devices, or {@code -1} if the devices could not be enumerated
264      */
265     private static int countMatchesByName(String gpuName) {
266         String needle = gpuName.toLowerCase(Locale.ROOT);
267         int[] matches = new int[1];
268         boolean enumerated = forEachDevice(handle -> {
269             String name = readName(handle);
270             if (name != null && NvmlQuery.matches(name, needle)) {
271                 matches[0]++;
272             }
273             return false;
274         });
275         return enumerated ? matches[0] : -1;
276     }

282     private static double readUtilization(Pointer device) {
283         NvmlUtilization util = new NvmlUtilization();
284         if (Holder.LIB.nvmlDeviceGetUtilizationRates(device, util) == Nvml.NVML_SUCCESS) {
285             util.read();
286             return util.gpu;
287         }
288         return -1d;
289     }

291     private static long readVramUsed(Pointer device) {
292         NvmlMemory mem = new NvmlMemory();
293         if (Holder.LIB.nvmlDeviceGetMemoryInfo(device, mem) == Nvml.NVML_SUCCESS) {
294             mem.read();
295             return mem.used;
296         }
297         return -1L;
298     }

346     /**
347      * Finds the stable PCI bus ID string for the NVML device whose bus ID contains the given fragment. The match is
348      * case-insensitive and uses substring matching to accommodate domain-qualified vs. non-qualified forms.
349      *
350      * <p>
351      * Returns a stable string identifier rather than a device handle. Handles are only valid within a single
352      * {@code nvmlInit}/{@code nvmlShutdown} scope; returning one across that boundary would leave the caller with a
353      * stale pointer. Callers should pass the returned string to the metric methods, which re-acquire a fresh handle
354      * internally.
355      *
356      * @param pciBusId PCI bus ID fragment (e.g. {@code "0000:01:00.0"} or {@code "01:00.0"})
357      * @return matched PCI bus ID string, or {@code null} if not found or NVML unavailable
358      */
359     public static String findDevice(String pciBusId) {
360         if (!Holder.LIBRARY_LOADED || pciBusId == null || pciBusId.isEmpty()) {
361             return null;
362         }
363         if (!nvmlInit()) {
364             return null;
365         }
366         try {
367             // Return the canonical bus ID from the enumerated set that matches, not a handle.
368             return NvmlQuery.matchBusId(DEVICE_CACHE.get(NvmlUtilJNA::enumerateDeviceBusIds), pciBusId);
369         } finally {
370             nvmlUninit();
371         }
372     }
373 
374     /**
375      * Finds the stable PCI bus ID string for the NVML device whose name matches the given GPU name. Used as a fallback
376      * when PCI bus ID is unavailable.
377      *
378      * <p>
379      * Returns a stable string identifier rather than a device handle for the same reason as {@link #findDevice}.
380      *
381      * @param gpuName GPU name string (case-insensitive substring match)
382      * @return PCI bus ID string of the matched device, or {@code null} if not found or NVML unavailable
383      */
384     public static String findDeviceByName(String gpuName) {
385         if (!Holder.LIBRARY_LOADED || gpuName == null || gpuName.isEmpty()) {
386             return null;
387         }
388         if (!nvmlInit()) {
389             return null;
390         }
391         try {
392             // Check for ambiguous name matches before committing to the first hit.
393             int matchCount = countMatchesByName(gpuName);
394             if (matchCount <= 0) {
395                 // Zero matches, or nvmlDeviceGetCount_v2 failed so the devices could not be enumerated.
396                 return null;
397             }
398             if (matchCount > 1) {
399                 LOG.warn("NVML name match for '{}' is ambiguous ({} devices match); use PCI bus ID for reliable"
400                         + " device identification", gpuName, matchCount);
401                 return null;
402             }
403             // Acquire a handle by name to confirm the device exists, then extract its bus ID.
404             Pointer handle = acquireHandleByName(gpuName);
405             if (handle == null) {
406                 return null;
407             }
408             Pair<String, String> busIds = readBusIds(handle);
409             if (busIds == null) {
410                 return null;
411             }
412             // Prefer the modern form, matching the order enumerateDeviceBusIds records them in.
413             return busIds.getA().isEmpty() ? emptyToNull(busIds.getB()) : busIds.getA();
414         } finally {
415             nvmlUninit();
416         }
417     }

419     private static String emptyToNull(String value) {
420         return value.isEmpty() ? null : value;
421     }
