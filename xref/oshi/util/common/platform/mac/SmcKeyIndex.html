25  /**
26   * Connection-independent logic for locating macOS SMC sensor keys by index.
27   * <p>
28   * The SMC exposes its keys as a list addressed by index, and returns them in ascending order of the key's four
29   * characters. That ordering lets a caller binary-search for a prefix rather than reading every key: on an M2 Max the
30   * full index is 2409 keys, of which 297 begin with {@code T} and 16 with {@code Tg}.
31   * <p>
32   * These methods take the index size and a lookup function rather than an SMC connection, so the JNA and FFM backends
33   * can share this logic despite their different connection handle types, and so it can be tested without Mac hardware.
34   */
35  @ThreadSafe
36  public final class SmcKeyIndex {
37  
38      private static final Logger LOG = LoggerFactory.getLogger(SmcKeyIndex.class);
39  
40      /**
41       * Apple Silicon GPU cluster temperature keys, e.g. {@code Tg0f} or {@code Tg1A}.
42       * <p>
43       * The third character is always a digit, but the fourth varies: across the sensor keys published for M1 through M5
44       * and A18 it is uppercase in 42% of cases, lowercase in 43%, and a digit in 15%. A mask requiring an uppercase
45       * fourth character would miss {@code Tg0f}, which is the only GPU key present on some M2 machines.
46       */
47      private static final Pattern GPU_TEMPERATURE_KEY = Pattern.compile("^Tg\\d[\\dA-Za-z]$");
48  
49      /**
50       * Fan current-speed keys, e.g. {@code F0Ac} or {@code F1Ac}.
51       * <p>
52       * Unlike {@link #GPU_TEMPERATURE_KEY} the fourth character is fixed, not variable: the published fan keys are
53       * {@code F%dAc} current, {@code F%dMn} minimum, {@code F%dMx} maximum, {@code F%dSf} safe and {@code F%dTg} target,
54       * so only the {@code Ac} suffix names the current speed. The mask matters because the other fan keys sort inside
55       * the same {@code F} prefix block: an M3 Pro's block is
56       * {@code F0Ac F0CR F0Dc F0Fb F0Fc F0Md F0Mn F0Mx F0Sf F0St F0Tg}, the same again for {@code F1}, then
57       * {@code FBAC FBAD FNum FOFC FOff FRmp Fpds Frqd Ftst}.
58       */
59      private static final Pattern FAN_SPEED_KEY = Pattern.compile("^F\\dAc$");
60  
61      /** SMC keys are four characters. */
62      private static final int KEY_LENGTH = 4;
63  
64      /**
65       * Upper bound on a fan count read from {@code FNum}.
66       * <p>
67       * The binding constraint is the key format, not the hardware: {@code String.format("F%dAc", 10)} produces the
68       * five-character {@code "F10Ac"}, and reading a key truncates it to four characters, so an eleventh fan would
69       * silently read {@code "F10A"} instead. Ten is therefore the highest index this naming scheme can express. It also
70       * sits above the hardware bound, since no Mac has more than eight fans.
71       */
72      public static final int MAX_FANS = 10;
73  
74      /**
75       * Upper bound on a plausible {@code #KEY} count, guarding against a garbage read. Observed counts are in the low
76       * thousands.
77       */
78      private static final int MAX_KEY_COUNT = 65536;
79  
80      /** Upper bound on the forward scan, so an unsorted index cannot cause a runaway read. */
81      private static final int MAX_SCAN = 256;
82  
83      private SmcKeyIndex() {
84      }
85  
86      /**
87       * Locates the keys sharing a prefix, by binary searching the sorted key index and then scanning forward.
88       * <p>
89       * Returns {@code null}, rather than an empty list, if the index could not be read reliably. Callers must not cache
90       * a {@code null} result: an empty list means "this machine has no such keys", while {@code null} means "ask again
91       * later".
92       *
93       * @param keyCount   the number of keys in the index, from the SMC's {@code #KEY} key
94       * @param keyAtIndex looks up the key name at an index, returning {@code null} if that read fails
95       * @param prefix     the key prefix to locate, e.g. {@code "Tg"}
96       * @param mask       an additional test each candidate key must pass
97       * @return the matching keys in index order, or {@code null} if the index could not be read
98       */
99      public static @Nullable List<String> findKeys(int keyCount, IntFunction<@Nullable String> keyAtIndex, String prefix,
100             Predicate<String> mask) {
101         if (keyCount <= 0 || keyCount > MAX_KEY_COUNT) {
102             LOG.debug("Implausible SMC key count {}; skipping key discovery.", keyCount);
103             return null; // NOSONAR java:S1168 - null and empty are different answers; see the javadoc
104         }
105         // Any failed read, in the binary search or in the forward scan, could have hidden a matching key: the search
106         // substitutes a neighbour for an unreadable probe, which can move the landing point past the block entirely,
107         // and the scan skips an unreadable index outright. Track failures through one wrapper so no path is missed.
108         boolean[] readFailed = new boolean[1];
109         IntFunction<@Nullable String> tracked = i -> {
110             String key = keyAtIndex.apply(i);
111             if (key == null) {
112                 readFailed[0] = true;
113             }
114             return key;
115         };
116         int start = lowerBound(keyCount, tracked, prefix);
117         if (start < 0) {
118             return null; // NOSONAR java:S1168 - null and empty are different answers; see the javadoc
119         }
120         Set<String> found = new LinkedHashSet<>();
121         int limit = Math.min(keyCount, start + MAX_SCAN);
122         for (int i = start; i < limit; i++) {
123             String key = tracked.apply(i);
124             if (key == null) {
125                 key = tracked.apply(i); // one retry, in case the failure was transient
126             }
127             if (key == null) {
128                 // Skip rather than stop: an unreadable key in the middle of the block would otherwise discard every
129                 // key after it. The scan is bounded, and the prefix test below still ends it at the block boundary.
130                 LOG.debug("Could not read SMC key at index {}; continuing the scan.", i);
131             } else {
132                 if (!key.startsWith(prefix)) {
133                     break;
134                 }
135                 if (mask.test(key)) {
136                     found.add(key);
137                 }
138             }
139         }
140         if (found.isEmpty() && readFailed[0]) {
141             // Nothing matched, but a read failed, so "no such keys" cannot be distinguished from "the one key this
142             // machine has was unreadable". Returning null keeps the caller from caching an empty set and disabling the
143             // sensor for the JVM lifetime. An empty result from a run with no failed read is different: that is a
144             // genuine "this machine has none", and is a cacheable answer.
145             LOG.debug("No SMC keys matched '{}' and at least one read failed; skipping key discovery.", prefix);
146             return null; // NOSONAR java:S1168 - null and empty are different answers; see the javadoc
147         }
148         return Collections.unmodifiableList(new ArrayList<>(found));
149     }

151     /**
152      * Binary searches for the first index whose key sorts at or after the prefix.
153      *
154      * @param keyCount   the number of keys in the index
155      * @param keyAtIndex looks up the key name at an index
156      * @param prefix     the prefix to locate
157      * @return that index, or {@code -1} if the index could not be read
158      */
159     private static int lowerBound(int keyCount, IntFunction<@Nullable String> keyAtIndex, String prefix) {
160         int lo = 0;
161         int hi = keyCount;
162         while (lo < hi) {
163             int mid = (lo + hi) >>> 1;
164             String key = probe(keyAtIndex, mid, keyCount);
165             if (key == null) {
166                 return -1;
167             }
168             // A four-character key is never equal to the shorter prefix, so this also skips the prefix itself.
169             if (key.compareTo(prefix) > 0) {
170                 hi = mid;
171             } else {
172                 lo = mid + 1;
173             }
174         }
175         return lo;
176     }

178     /**
179      * Reads the key at an index, retrying at neighbouring indices if that read fails, so a single unreadable key does
180      * not abort discovery. Walking outward keeps the binary search's ordering assumption intact, since neighbours sort
181      * adjacently.
182      *
183      * @param keyAtIndex looks up the key name at an index
184      * @param index      the index to read
185      * @param keyCount   the number of keys in the index, bounding the outward walk
186      * @return a key name, or {@code null} if nothing nearby could be read
187      */
188     private static @Nullable String probe(IntFunction<@Nullable String> keyAtIndex, int index, int keyCount) {
189         String key = keyAtIndex.apply(index);
190         if (key != null) {
191             return key;
192         }
193         for (int delta = 1; delta <= 4; delta++) {
194             if (index - delta >= 0) {
195                 key = keyAtIndex.apply(index - delta);
196                 if (key != null) {
197                     return key;
198                 }
199             }
200             if (index + delta < keyCount) {
201                 key = keyAtIndex.apply(index + delta);
202                 if (key != null) {
203                     return key;
204                 }
205             }
206         }
207         return null;
208     }

210     /**
211      * Tests whether a key names an Apple Silicon GPU cluster temperature sensor.
212      *
213      * @param key the four-character SMC key
214      * @return true if the key matches the GPU temperature naming convention
215      */
216     public static boolean isGpuTemperatureKey(@Nullable String key) {
217         return key != null && GPU_TEMPERATURE_KEY.matcher(key).matches();
218     }

220     /**
221      * Tests whether a key names a fan's current speed.
222      *
223      * @param key the four-character SMC key
224      * @return true if the key matches the fan current-speed naming convention
225      */
226     public static boolean isFanSpeedKey(@Nullable String key) {
227         return key != null && FAN_SPEED_KEY.matcher(key).matches();
228     }

230     /**
231      * Builds the fan current-speed keys for a fan count, as the naming convention implies them.
232      *
233      * @param fanCount the number of fans, from the SMC's {@code FNum} key
234      * @return the keys {@code F0Ac} through {@code F(n-1)Ac}, clamped to {@link #MAX_FANS}, never null
235      */
236     public static List<String> fanSpeedKeys(long fanCount) {
237         // Clamp before narrowing: FNum is a single byte, but a mis-sized read returns whatever the buffer held.
238         int fans = (int) Math.max(0, Math.min(MAX_FANS, fanCount));
239         if (fans < fanCount) {
240             LOG.warn("Ignoring an implausible SMC fan count of {}; using {}.", fanCount, fans);
241         }
242         List<String> keys = new ArrayList<>(fans);
243         for (int i = 0; i < fans; i++) {
244             keys.add(String.format(Locale.ROOT, "F%dAc", i));
245         }
246         return Collections.unmodifiableList(keys);
247     }

249     /**
250      * Reconciles the fan keys found in the key index against the count reported by {@code FNum}.
251      * <p>
252      * Discovery is preferred where it produced anything, because the index lists the keys that actually exist. Where it
253      * did not, a positive {@code FNum} still implies the conventionally named keys, which is what earlier versions of
254      * OSHI read directly, so this never reports fewer fans than they did.
255      * <p>
256      * Returns {@code null} when the two sources together cannot distinguish "this machine has no fans" from "the keys
257      * could not be read", so that the caller does not cache the answer. That happens when discovery failed and
258      * {@code FNum} read as zero.
259      *
260      * @param discovered the keys found by {@link #findKeys}, or {@code null} if the index could not be read
261      * @param fanCount   the count from {@code FNum}, or 0 if that read failed
262      * @return the keys to read, or {@code null} if the answer is not yet known
263      */
264     public static @Nullable List<String> reconcileFanKeys(@Nullable List<String> discovered, long fanCount) {
265         if (discovered != null && !discovered.isEmpty()) {
266             if (fanCount > 0 && discovered.size() != fanCount) {
267                 LOG.debug("Found {} fan speed keys {} but FNum reports {} fans; using the discovered keys.",
268                         discovered.size(), discovered, fanCount);
269             }
270             return discovered;
271         }
272         if (fanCount > 0) {
273             List<String> keys = fanSpeedKeys(fanCount);
274             LOG.debug("Fan speed key discovery found none; using the {} keys FNum implies: {}", keys.size(), keys);
275             return keys;
276         }
277         if (discovered == null) {
278             LOG.debug("Neither the SMC key index nor FNum could be read; deferring the fan count.");
279             return null; // NOSONAR java:S1168 - null and empty are different answers; see the javadoc
280         }
281         return Collections.emptyList();
282     }

284     /**
285      * Parses a user-supplied comma-separated list of SMC keys, ignoring blanks and warning about malformed entries.
286      *
287      * @param csv the configured value, which may be null or empty
288      * @return the keys, never null
289      */
290     public static List<String> parseConfiguredKeys(@Nullable String csv) {
291         if (csv == null || csv.trim().isEmpty()) {
292             return Collections.emptyList();
293         }
294         List<String> keys = new ArrayList<>();
295         for (String token : csv.split(",", -1)) {
296             String key = token.trim();
297             if (key.length() == KEY_LENGTH) {
298                 keys.add(key);
299             } else if (!key.isEmpty()) {
300                 LOG.warn("Ignoring configured SMC key '{}': keys are exactly {} characters.", key, KEY_LENGTH);
301             }
302         }
303         return Collections.unmodifiableList(keys);
304     }

306     /**
307      * Returns the first plausible reading among the given keys, in the order given.
308      * <p>
309      * Plausibility is applied here, at read time, for the same reason as in {@link #maxPlausible}: whether a key exists
310      * is a property of the hardware and does not change, but whether it currently reports a usable value is not.
311      *
312      * @param keys        the keys to read, in preference order
313      * @param reader      reads a key and returns the value in its final units, 0 if unavailable
314      * @param isPlausible tests whether a reading is usable
315      * @param description what is being read, for log messages, e.g. {@code "temperature"}
316      * @return the first plausible reading, or 0 if none were plausible
317      */
318     public static double firstPlausible(List<String> keys, ToDoubleFunction<String> reader, DoublePredicate isPlausible,
319             String description) {
320         for (String key : keys) {
321             double value = reader.applyAsDouble(key);
322             if (isPlausible.test(value)) {
323                 return value;
324             }
325             // A zero means the key was absent or undecodable, which is expected while scanning a candidate list; a
326             // non-zero implausible value means a sensor answered with something unusable, which is worth a note.
327             if (value != 0d) {
328                 LOG.debug("Ignoring implausible {} {} from SMC key {}.", description, value, key);
329             }
330         }
331         return 0d;
332     }

334     /**
335      * Returns the highest plausible temperature among the given keys.
336      * <p>
337      * Plausibility is applied here, at read time, rather than when the keys were discovered. Whether a key exists is a
338      * property of the hardware and does not change; whether it currently reports a usable value is not, because an idle
339      * sensor may report a sentinel below ambient. Filtering at discovery time would let a single unlucky first read
340      * cache an empty set and disable the sensor for the lifetime of the JVM.
341      *
342      * @param keys        the keys to read
343      * @param reader      reads a key, returning the temperature in degrees Celsius
344      * @param isPlausible tests whether a reading is usable
345      * @return the highest plausible reading, or 0 if none were plausible
346      */
347     public static double maxPlausible(List<String> keys, ToDoubleFunction<String> reader, DoublePredicate isPlausible) {
348         double max = 0d;
349         for (String key : keys) {
350             double value = reader.applyAsDouble(key);
351             if (isPlausible.test(value) && value > max) {
352                 max = value;
353             }
354         }
355         if (max == 0d && !keys.isEmpty()) {
356             LOG.debug("No plausible temperature among SMC keys {}; sensors are likely idle-gated.", keys);
357         }
358         return max;
359     }
