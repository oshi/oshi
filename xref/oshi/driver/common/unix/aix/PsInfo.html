5   package oshi.driver.common.unix.aix;
6   
7   import java.nio.ByteBuffer;
8   import java.util.Locale;
9   
10  import org.jspecify.annotations.Nullable;
11  import org.slf4j.Logger;
12  import org.slf4j.LoggerFactory;
13  
14  import oshi.annotation.concurrent.ThreadSafe;
15  import oshi.util.FileUtil;
16  import oshi.util.tuples.Triplet;
17  
18  /**
19   * Shared utility to query the pure-Java parts of AIX {@code /proc/<pid>/psinfo} and
20   * {@code /proc/<pid>/lwp/<tid>/lwpsinfo}.
21   * <p>
22   * The address-space reads {@code queryArgsEnv} performs need libc {@code open}/{@code pread} and live in the JNA
23   * ({@code PsInfoJNA}) and FFM ({@code PsInfoFFM}) drivers respectively.
24   */
25  @ThreadSafe
26  public final class PsInfo {
27  
28      private static final Logger LOG = LoggerFactory.getLogger(PsInfo.class);
29  
30      private PsInfo() {
31      }
32  
33      /**
34       * Reads {@code /proc/<pid>/psinfo}.
35       *
36       * @param pid the process ID
37       * @return a structure containing information for the requested process, or {@code null} if the file could not be
38       *         read or could not be parsed
39       */
40      public static @Nullable AixPsInfo queryPsInfo(int pid) {
41          String path = String.format(Locale.ROOT, "/proc/%d/psinfo", pid);
42          ByteBuffer buff = FileUtil.readAllBytesAsBuffer(path);
43          if (buff.remaining() == 0) {
44              // Short-lived processes commonly disappear from /proc between enumeration and read, so this is expected
45              // often enough that it belongs at debug rather than warn.
46              LOG.debug("psinfo file empty or unreadable for pid {} ({})", pid, path);
47              return null;
48          }
49          int sz = buff.remaining();
50          try {
51              return new AixPsInfo(buff);
52          } catch (RuntimeException e) {
53              LOG.warn("Failed to parse psinfo for pid {} (file size {} bytes)", pid, sz, e);
54              return null;
55          }
56      }
57  
58      /**
59       * Reads {@code /proc/<pid>/lwp/<tid>/lwpsinfo}.
60       *
61       * @param pid the process ID
62       * @param tid the thread ID (lwpid)
63       * @return a structure containing information for the requested thread, or {@code null} if the file could not be
64       *         read or could not be parsed
65       */
66      public static @Nullable AixLwpsInfo queryLwpsInfo(int pid, int tid) {
67          String path = String.format(Locale.ROOT, "/proc/%d/lwp/%d/lwpsinfo", pid, tid);
68          ByteBuffer buff = FileUtil.readAllBytesAsBuffer(path);
69          if (buff.remaining() == 0) {
70              LOG.debug("lwpsinfo file empty or unreadable for pid {} tid {} ({})", pid, tid, path);
71              return null;
72          }
73          try {
74              return new AixLwpsInfo(buff);
75          } catch (RuntimeException e) {
76              LOG.debug("Failed to parse lwpsinfo for pid {} tid {}", pid, tid, e);
77              return null;
78          }
79      }
80  
81      /**
82       * Reads the {@code pr_argc}, {@code pr_argv}, and {@code pr_envp} fields from a {@code psinfo_t}.
83       *
84       * @param pid    the process ID (used for diagnostic logging only)
85       * @param psinfo a populated {@link AixPsInfo} containing the offset pointers, or {@code null} if absent
86       * @return a triplet of {@code (argc, argv, envp)}, or {@code null} if the psinfo is unusable
87       */
88      public static @Nullable Triplet<Integer, Long, Long> queryArgsEnvAddrs(int pid, @Nullable AixPsInfo psinfo) {
89          if (psinfo != null) {
90              int argc = psinfo.pr_argc;
91              // Must have at least one argc (the command itself) so failure here means exit
92              if (argc > 0) {
93                  return new Triplet<>(argc, psinfo.pr_argv, psinfo.pr_envp);
94              }
95              LOG.trace("Failed argc sanity check: argc={}", argc);
96              return null;
97          }
98          LOG.trace("Failed to read psinfo file for pid: {} ", pid);
99          return null;
100     }
101 }
