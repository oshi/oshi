22  /**
23   * Utility to query geom part list
24   */
25  @ThreadSafe
26  public final class GeomPartList {
27  
28      private static final String GEOM_PART_LIST = "geom part list";
29      private static final String STAT_FILESIZE = "stat -f %i /dev/";
30  
31      private GeomPartList() {
32      }
33  
34      /**
35       * Queries partition data using geom, mount, and stat commands
36       *
37       * @return A map with disk name as the key and a List of partitions as the value
38       */
39      public static Map<String, List<HWPartition>> queryPartitions() {
40          // FreeBSD has no major numbers. The inode of a devfs entry is the device's minor number.
41          return parsePartitions(ExecutingCommand.runNative(GEOM_PART_LIST), Mount.queryPartitionToMountMap(),
42                  part -> ParseUtil.parseIntOrDefault(ExecutingCommand.getFirstAnswer(STAT_FILESIZE + part), 0));
43      }
44  
45      /**
46       * Parses the output of {@code geom part list}. Each geom lists its partitions under {@code Providers:} and the
47       * device holding them under {@code Consumers:}; only the providers are partitions. A partition can itself hold a
48       * scheme, as an MBR slice holds a BSD label, so its geom is named after the slice; its partitions are listed under
49       * the physical disk.
50       *
51       * @param geom        the lines of output from {@code geom part list}
52       * @param mountMap    a map of {@code /dev} device name to mount point
53       * @param minorLookup returns the minor device number of a partition
54       * @return A map with disk name as the key and a List of partitions, sorted by name, as the value
55       */
56      static Map<String, List<HWPartition>> parsePartitions(List<String> geom, Map<String, String> mountMap,
57              ToIntFunction<String> minorLookup) {
58          // Partitions by the geom listing them, and the geom providing each partition
59          Map<String, List<HWPartition>> byGeom = new HashMap<>();
60          Map<String, String> geomOf = new HashMap<>();
61          PartitionBuilder part = null;
62          String geomName = null;
63          boolean inProviders = false;
64          for (String raw : geom) {
65              String line = raw.trim();
66              if (line.startsWith("Geom name:") || line.equals("Providers:") || line.equals("Consumers:")) {
67                  addPartition(byGeom, geomOf, geomName, part, mountMap, minorLookup);
68                  part = null;
69                  inProviders = line.equals("Providers:");
70                  if (line.startsWith("Geom name:")) {
71                      geomName = line.substring(line.lastIndexOf(' ') + 1);
72                  }
73              } else if (inProviders && geomName != null) {
74                  String[] split = ParseUtil.whitespaces.split(line, -1);
75                  if (line.contains("Name:")) {
76                      addPartition(byGeom, geomOf, geomName, part, mountMap, minorLookup);
77                      part = new PartitionBuilder(split[split.length - 1]);
78                  } else if (part != null && split.length >= 2) {
79                      if (line.startsWith("Mediasize:")) {
80                          part.size = ParseUtil.parseLongOrDefault(split[1], 0L);
81                      } else if (line.startsWith("rawuuid:")) {
82                          part.uuid = split[1];
83                      } else if (line.startsWith("type:")) {
84                          part.type = split[1];
85                      } else if (line.startsWith("label:") && !"(null)".equals(split[1])) {
86                          part.label = split[1];
87                      }
88                  }
89              }
90          }
91          addPartition(byGeom, geomOf, geomName, part, mountMap, minorLookup);
92  
93          Map<String, List<HWPartition>> partitionMap = new HashMap<>();
94          for (Map.Entry<String, List<HWPartition>> entry : byGeom.entrySet()) {
95              partitionMap.computeIfAbsent(physicalDisk(entry.getKey(), geomOf), d -> new ArrayList<>())
96                      .addAll(entry.getValue());
97          }
98          for (List<HWPartition> partitions : partitionMap.values()) {
99              partitions.sort(Comparator.comparing(HWPartition::getName));
100         }
101         return partitionMap;
102     }
103 
104     /** Follows a geom named after a partition, such as an MBR slice holding a BSD label, up to the disk. */
105     private static String physicalDisk(String geomName, Map<String, String> geomOf) {
106         String disk = geomName;
107         // Bounded, in case the output is malformed into a cycle
108         for (int depth = 0; depth < 8; depth++) {
109             String parent = geomOf.get(disk);
110             if (parent == null) {
111                 break;
112             }
113             disk = parent;
114         }
115         return disk;
116     }
117 
118     private static void addPartition(Map<String, List<HWPartition>> byGeom, Map<String, String> geomOf,
119             @Nullable String geomName, @Nullable PartitionBuilder part, Map<String, String> mountMap,
120             ToIntFunction<String> minorLookup) {
121         if (geomName != null && part != null) {
122             geomOf.put(part.name, geomName);
123             byGeom.computeIfAbsent(geomName, g -> new ArrayList<>())
124                     .add(new HWPartition(part.name, part.name, part.type, part.uuid, part.label, part.size, 0,
125                             minorLookup.applyAsInt(part.name), mountPoint(part, mountMap)));
126         }
127     }
128 
129     /** The mount point of a partition, mounted by its own name or by its GPT label or GPT ID. */
130     private static String mountPoint(PartitionBuilder part, Map<String, String> mountMap) {
131         String mountPoint = mountMap.get(part.name);
132         if (mountPoint == null && !part.label.isEmpty()) {
133             mountPoint = mountMap.get("gpt/" + part.label);
134         }
135         if (mountPoint == null && !Constants.UNKNOWN.equals(part.uuid)) {
136             mountPoint = mountMap.get("gptid/" + part.uuid);
137         }
138         return mountPoint == null ? "" : mountPoint;
139     }
140 
141     /** The fields of a partition collected from its lines of {@code geom} output. */
142     private static final class PartitionBuilder {
143         private final String name;
144         private String type = Constants.UNKNOWN;
145         private String uuid = Constants.UNKNOWN;
146         private String label = "";
147         private long size;
148 
149         private PartitionBuilder(String name) {
150             this.name = name;
151         }
152     }
153 }
