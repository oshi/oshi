35  /**
36   * Reads sensor values from the web server of a running LibreHardwareMonitor application.
37   * <p>
38   * LibreHardwareMonitor 0.9.5 and later no longer publish the {@code ROOT\LibreHardwareMonitor} WMI namespace; their
39   * built-in web server replaces it. The server is off by default. A user turns it on in the application (Options, Remote
40   * Web Server, Run) and gives OSHI its address in {@link GlobalConfig#OSHI_OS_WINDOWS_LHM_URL}.
41   * <p>
42   * The {@code /metrics} endpoint is read rather than {@code /data.json}. It is one line per sensor, it gives every value
43   * in base units with an invariant decimal point, and its format is the same in every release that has it. The raw value
44   * in {@code data.json} changed from a localized string with units in 0.9.6 to a number afterward.
45   */
46  @ThreadSafe
47  public final class LhmMetrics {
48  
49      private static final Logger LOG = LoggerFactory.getLogger(LhmMetrics.class);
50  
51      /** The processor's temperature sensors, in degrees Celsius. */
52      public static final String CPU_TEMPERATURE = "lhm_cpu_temperature_celsius";
53      /** The processor's voltage sensors, in volts. */
54      public static final String CPU_VOLTAGE = "lhm_cpu_voltage_volts";
55      /**
56       * Fan sensors on the motherboard, in RPM. A Super I/O chip is a child of the motherboard, and the server names its
57       * sensors after the parent's hardware type.
58       */
59      public static final String MOTHERBOARD_FAN = "lhm_motherboard_fan_rpm";
60      /** Voltage sensors on the motherboard, including the Super I/O chip's {@code Vcore}, in volts. */
61      public static final String MOTHERBOARD_VOLTAGE = "lhm_motherboard_voltage_volts";
62  
63      private static final String ENDPOINT = "/metrics";
64  
65      // Short, because a refused connection to a closed port on Windows retries for two seconds before failing
66      private static final int CONNECT_TIMEOUT_MS = 500;
67      // The body must be complete within BODY_DEADLINE_MS. The deadline is checked between reads, and a blocked read
68      // gives up after READ_TIMEOUT_MS of silence, so a slow server is abandoned within the sum of the two.
69      private static final int READ_TIMEOUT_MS = 500;
70      private static final int BODY_DEADLINE_MS = 2000;
71      // A real response is tens of kilobytes; anything far larger is not a LibreHardwareMonitor server
72      private static final int MAX_BODY_CHARS = 1 << 20;
73  
74      // lhm_cpu_temperature_celsius {"sensorName"="CPU Package", ..., "hardwareId"="/intelcpu/0", ...} 45.5
75      // A timestamp in milliseconds follows the value in 0.9.5, which lists up to five samples per sensor, and in
76      // later versions when the request asks for one.
77      private static final Pattern SAMPLE = Pattern.compile("^(lhm_\\w++)\\s*+\\{(.*)\\}\\s++(\\S++)(?:\\s++(\\d++))?$");
78      private static final Pattern LABEL = Pattern.compile("\"(\\w++)\"=\"([^\"]*+)\"");
79  
80      private LhmMetrics() {
81      }
82  
83      /**
84       * Fetches and parses the {@code /metrics} endpoint of a LibreHardwareMonitor web server.
85       *
86       * @param baseUrl the server's address, such as {@code http://127.0.0.1:8085}; a URL already ending in
87       *                {@code /metrics} is used as given
88       * @return the readings, empty if the server could not be reached, is not an {@code http} or {@code https} address,
89       *         or returned none
90       */
91      public static List<Reading> query(String baseUrl) {
92          String url = metricsUrl(baseUrl);
93          HttpURLConnection conn = null;
94          try {
95              URI uri = URI.create(url);
96              String scheme = uri.getScheme();
97              if (!"http".equalsIgnoreCase(scheme) && !"https".equalsIgnoreCase(scheme)) {
98                  LOG.debug("Not an http or https address for LibreHardwareMonitor: {}", url);
99                  return Collections.emptyList();
100             }
101             // The server is local, or on a host the user named; a proxy configured for the JVM is not meant for it
102             conn = (HttpURLConnection) uri.toURL().openConnection(Proxy.NO_PROXY);
103             conn.setConnectTimeout(CONNECT_TIMEOUT_MS);
104             conn.setReadTimeout(READ_TIMEOUT_MS);
105             conn.setUseCaches(false);
106             int status = conn.getResponseCode();
107             if (status != HttpURLConnection.HTTP_OK) {
108                 LOG.debug("LibreHardwareMonitor at {} returned HTTP {}", url, status);
109                 return Collections.emptyList();
110             }
111             try (Reader reader = new InputStreamReader(conn.getInputStream(), UTF_8)) {
112                 String body = readBounded(reader, MAX_BODY_CHARS,
113                         System.nanoTime() + TimeUnit.MILLISECONDS.toNanos(BODY_DEADLINE_MS));
114                 if (body == null) {
115                     LOG.debug("LibreHardwareMonitor at {} sent more than {} characters or took too long", url,
116                             MAX_BODY_CHARS);
117                     return Collections.emptyList();
118                 }
119                 return parse(Arrays.asList(body.split("\n", -1)));
120             }
121         } catch (IOException | IllegalArgumentException e) {
122             LOG.debug("Unable to read LibreHardwareMonitor at {}", url, e);
123             return Collections.emptyList();
124         } finally {
125             if (conn != null) {
126                 conn.disconnect();
127             }
128         }
129     }

131     /**
132      * Reads a response body, giving up once it exceeds a size or a deadline.
133      *
134      * @param reader        the body
135      * @param maxChars      the largest body to accept
136      * @param deadlineNanos the {@link System#nanoTime()} by which the body must be complete
137      * @return the body, or {@code null} if it was too large or too slow
138      * @throws IOException if reading fails
139      */
140     static @Nullable String readBounded(Reader reader, int maxChars, long deadlineNanos) throws IOException {
141         StringBuilder sb = new StringBuilder();
142         char[] buf = new char[8192];
143         int n;
144         while ((n = reader.read(buf)) != -1) {
145             sb.append(buf, 0, n);
146             if (sb.length() > maxChars || System.nanoTime() - deadlineNanos > 0) {
147                 return null;
148             }
149         }
150         return System.nanoTime() - deadlineNanos > 0 ? null : sb.toString();
151     }

153     /**
154      * Builds the endpoint address from the configured one.
155      *
156      * @param baseUrl the configured address
157      * @return the address of the {@code /metrics} endpoint
158      */
159     static String metricsUrl(String baseUrl) {
160         String url = baseUrl.trim();
161         if (url.endsWith(ENDPOINT)) {
162             return url;
163         }
164         while (url.endsWith("/")) {
165             url = url.substring(0, url.length() - 1);
166         }
167         return url + ENDPOINT;
168     }
169 
170     /**
171      * Parses the {@code /metrics} response. Comment lines and samples whose value is not a number are skipped. Where a
172      * sensor appears more than once, as in 0.9.5, which lists its recent history, only the sample with the latest
173      * timestamp is kept.
174      *
175      * @param lines the response body, one sample per line
176      * @return one reading per sensor, in the order each sensor first appears
177      */
178     public static List<Reading> parse(List<String> lines) {
179         Map<String, Reading> readings = new LinkedHashMap<>();
180         Map<String, Long> timestamps = new HashMap<>();
181         for (String line : lines) {
182             Matcher sample = SAMPLE.matcher(line.trim());
183             if (!sample.matches()) {
184                 continue;
185             }
186             double value = ParseUtil.parseDoubleOrDefault(sample.group(3), Double.NaN);
187             if (Double.isNaN(value)) {
188                 continue;
189             }
190             String hardwareId = "";
191             String sensorId = "";
192             String sensorName = "";
193             Matcher label = LABEL.matcher(sample.group(2));
194             while (label.find()) {
195                 switch (label.group(1)) {
196                     case "hardwareId":
197                         hardwareId = label.group(2);
198                         break;
199                     case "sensorId":
200                         sensorId = label.group(2);
201                         break;
202                     case "sensorName":
203                         sensorName = label.group(2);
204                         break;
205                     default:
206                         break;
207                 }
208             }
209             String key = sample.group(1) + ' ' + hardwareId + ' ' + sensorId + ' ' + sensorName;
210             long timestamp = ParseUtil.parseLongOrDefault(sample.group(4), -1L);
211             Long previous = timestamps.get(key);
212             if (previous == null || timestamp > previous) {
213                 readings.put(key, new Reading(sample.group(1), hardwareId, sensorName, value));
214                 timestamps.put(key, timestamp);
215             }
216         }
217         return Collections.unmodifiableList(new ArrayList<>(readings.values()));
218     }

220     /**
221      * One sensor reading from the {@code /metrics} endpoint.
222      */
223     @Immutable
224     public static final class Reading {
225         private final String metric;
226         private final String hardwareId;
227         private final String sensorName;
228         private final double value;
229 
230         /**
231          * Creates a reading.
232          *
233          * @param metric     the metric name, which encodes the hardware type, the sensor type and the unit
234          * @param hardwareId the hardware identifier, such as {@code /intelcpu/0}
235          * @param sensorName the sensor name, such as {@code CPU Package}
236          * @param value      the value, in the unit the metric name states
237          */
238         public Reading(String metric, String hardwareId, String sensorName, double value) {
239             this.metric = metric;
240             this.hardwareId = hardwareId;
241             this.sensorName = sensorName;
242             this.value = value;
243         }
244 
245         /**
246          * Gets the metric name.
247          *
248          * @return the metric name, such as {@link LhmMetrics#CPU_TEMPERATURE}
249          */
250         public String getMetric() {
251             return metric;
252         }
253 
254         /**
255          * Gets the hardware identifier.
256          *
257          * @return the identifier of the hardware carrying the sensor, such as {@code /intelcpu/0}
258          */
259         public String getHardwareId() {
260             return hardwareId;
261         }
262 
263         /**
264          * Gets the sensor name.
265          *
266          * @return the sensor name, such as {@code CPU Package}
267          */
268         public String getSensorName() {
269             return sensorName;
270         }
271 
272         /**
273          * Gets the value.
274          *
275          * @return the value, in the unit the metric name states
276          */
277         public double getValue() {
278             return value;
279         }
280 
281         @Override
282         public String toString() {
283             return metric + '{' + hardwareId + ' ' + sensorName + "}=" + value;
284         }
285     }
