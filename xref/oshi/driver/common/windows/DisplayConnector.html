20  /**
21   * Shared logic for mapping a Windows display to the physical connector it is attached to, using the Connecting and
22   * Configuring Displays (CCD) APIs in {@code user32.dll}.
23   * <p>
24   * The struct sizes and field offsets the two native backends read are declared here as constants so the JNA and FFM
25   * implementations cannot drift apart, and the pure parsing (connector naming, built-in classification, the current
26   * mode, device-path normalization) is shared and unit-tested. The native calls themselves -
27   * {@code GetDisplayConfigBufferSizes}, {@code QueryDisplayConfig}, and {@code DisplayConfigGetDeviceInfo} - are made by
28   * each backend.
29   * <p>
30   * A {@code LUID} is two {@code DWORD}s, so it is only 4-byte aligned and the {@code adapterId} fields below can land on
31   * an offset that is not a multiple of 8. An 8-byte read there must therefore be unaligned, which the FFM backend
32   * enforces at runtime.
33   */
34  @ThreadSafe
35  public final class DisplayConnector {
36  
37      private DisplayConnector() {
38      }

52      public static final int PATH_TARGET_ID_OFFSET = 28;
53      /**
54       * Offset of {@code targetInfo.rotation} (a {@code DISPLAYCONFIG_ROTATION}) within a

63      public static final int PATH_FLAGS_OFFSET = 68;
64      /** {@code DISPLAYCONFIG_PATH_ACTIVE} bit within the path {@code flags}. */
65      public static final int PATH_ACTIVE = 0x00000001;

118     /**
119      * Names the connector a display is attached through, from a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}'s
120      * {@code outputTechnology} and {@code connectorInstance}. The {@code connectorInstance} is Windows' own
121      * disambiguator, zero when the adapter has a single connector of that type and one-based when it has several, so it
122      * is appended verbatim only when non-zero (e.g. {@code HDMI}, {@code DisplayPort-1}, {@code DisplayPort-2}).
123      *
124      * @param outputTechnology  the {@code DISPLAYCONFIG_VIDEO_OUTPUT_TECHNOLOGY} value
125      * @param connectorInstance the connector instance number
126      * @return a connector name such as {@code HDMI} or {@code DisplayPort-1}
127      */
128     public static String connectorName(int outputTechnology, int connectorInstance) {
129         String base = technologyName(outputTechnology);
130         return connectorInstance > 0 ? base + "-" + connectorInstance : base;
131     }
132 
133     private static String technologyName(int outputTechnology) {
134         switch (outputTechnology) {
135             case VOT_HD15:
136                 return "VGA";
137             case VOT_SVIDEO:
138                 return "S-Video";
139             case VOT_COMPOSITE_VIDEO:
140                 return "Composite";
141             case VOT_COMPONENT_VIDEO:
142                 return "Component";
143             case VOT_DVI:
144                 return "DVI";
145             case VOT_HDMI:
146                 return "HDMI";
147             case VOT_LVDS:
148                 return "LVDS";
149             case VOT_SDI:
150                 return "SDI";
151             case VOT_DISPLAYPORT_EXTERNAL:
152                 return "DisplayPort";
153             case VOT_DISPLAYPORT_EMBEDDED:
154                 return "eDP";
155             case VOT_UDI_EXTERNAL:
156             case VOT_UDI_EMBEDDED:
157                 return "UDI";
158             case VOT_SDTVDONGLE:
159                 return "SDTV";
160             case VOT_MIRACAST:
161                 return "Miracast";
162             case VOT_INTERNAL:
163                 return "Internal";
164             default:
165                 return "Other";
166         }
167     }

169     /**
170      * Tests whether an output technology connects a display built into the device.
171      *
172      * @param outputTechnology the {@code DISPLAYCONFIG_VIDEO_OUTPUT_TECHNOLOGY} value
173      * @return true for LVDS, embedded DisplayPort, embedded UDI, and internal connections
174      */
175     public static boolean isBuiltIn(int outputTechnology) {
176         switch (outputTechnology) {
177             case VOT_LVDS:
178             case VOT_DISPLAYPORT_EMBEDDED:
179             case VOT_UDI_EMBEDDED:
180             case VOT_INTERNAL:
181                 return true;
182             default:
183                 return false;
184         }
185     }

187     /**
188      * Reads the current mode of one active path from the buffers {@code QueryDisplayConfig} filled. The desktop
189      * position and size come from the source mode the path indexes into the mode array, and the rotation and refresh
190      * rate from the path's target. Windows reports the desktop in physical pixels, so the logical and pixel sizes are
191      * equal.
192      *
193      * @param path      reads a {@code UINT32} at an offset within the path's {@code DISPLAYCONFIG_PATH_INFO}
194      * @param modes     reads a {@code UINT32} at an offset within the {@code DISPLAYCONFIG_MODE_INFO} array
195      * @param modeCount the number of entries in the mode array
196      * @return the path's current mode, or {@code null} if it does not index a source mode
197      */
198     public static @Nullable DisplayMode readMode(LongToIntFunction path, LongToIntFunction modes, int modeCount) {
199         // A flat UINT32 because QueryDisplayConfig is called without QDC_VIRTUAL_MODE_AWARE; with that flag the field
200         // becomes a union of cloneGroupId and sourceModeInfoIdx bitfields
201         int index = path.applyAsInt(PATH_SOURCE_MODE_IDX_OFFSET);
202         // DISPLAYCONFIG_PATH_MODE_IDX_INVALID is 0xffffffff, negative as an int
203         if (index < 0 || index >= modeCount) {
204             return null;
205         }
206         long base = (long) index * MODE_INFO_SIZE;
207         if (modes.applyAsInt(base + MODE_INFO_TYPE_OFFSET) != MODE_INFO_TYPE_SOURCE) {
208             return null;
209         }
210         int width = modes.applyAsInt(base + SOURCE_MODE_WIDTH_OFFSET);
211         int height = modes.applyAsInt(base + SOURCE_MODE_HEIGHT_OFFSET);
212         if (width <= 0 || height <= 0) {
213             return null;
214         }
215         int x = modes.applyAsInt(base + SOURCE_MODE_POSITION_X_OFFSET);
216         int y = modes.applyAsInt(base + SOURCE_MODE_POSITION_Y_OFFSET);
217         int rotation = rotationDegrees(path.applyAsInt(PATH_TARGET_ROTATION_OFFSET));
218         double refreshRate = refreshRate(path.applyAsInt(PATH_TARGET_REFRESH_NUMERATOR_OFFSET),
219                 path.applyAsInt(PATH_TARGET_REFRESH_DENOMINATOR_OFFSET));
220         return new DisplayModeImpl(x, y, width, height, width, height, refreshRate, rotation);
221     }

223     /**
224      * Converts a {@code DISPLAYCONFIG_ROTATION} to degrees clockwise.
225      *
226      * @param rotation the {@code DISPLAYCONFIG_ROTATION} value
227      * @return 0, 90, 180 or 270; 0 for an unrecognized value
228      */
229     static int rotationDegrees(int rotation) {
230         switch (rotation) {
231             case ROTATION_ROTATE90:
232                 return 90;
233             case ROTATION_ROTATE180:
234                 return 180;
235             case ROTATION_ROTATE270:
236                 return 270;
237             case ROTATION_IDENTITY:
238             default:
239                 return 0;
240         }
241     }
242 
243     /**
244      * Converts a {@code DISPLAYCONFIG_RATIONAL} refresh rate to hertz. The fields are unsigned.
245      *
246      * @param numerator   the numerator
247      * @param denominator the denominator
248      * @return the refresh rate in hertz, or 0 if the denominator is 0
249      */
250     static double refreshRate(int numerator, int denominator) {
251         if (denominator == 0) {
252             return 0d;
253         }
254         return (double) Integer.toUnsignedLong(numerator) / Integer.toUnsignedLong(denominator);
255     }

257     /**
258      * A display's connector as the CCD API describes it, and the mode it is currently driven in.
259      */
260     @Immutable
261     public static final class Connector {
262 
263         private final String name;
264         private final boolean builtIn;
265         private final @Nullable DisplayMode mode;
266 
267         /**
268          * Describes a connector from a {@code DISPLAYCONFIG_TARGET_DEVICE_NAME}'s fields and the path's mode.
269          *
270          * @param outputTechnology  the {@code DISPLAYCONFIG_VIDEO_OUTPUT_TECHNOLOGY} value
271          * @param connectorInstance the connector instance number
272          * @param mode              the path's current mode, as read by {@link #readMode}, or {@code null} if unknown
273          */
274         public Connector(int outputTechnology, int connectorInstance, @Nullable DisplayMode mode) {
275             this.name = connectorName(outputTechnology, connectorInstance);
276             this.builtIn = DisplayConnector.isBuiltIn(outputTechnology);
277             this.mode = mode;
278         }
279 
280         /**
281          * The connector name, as named by {@link DisplayConnector#connectorName(int, int)}.
282          *
283          * @return the connector name, e.g. {@code HDMI} or {@code DisplayPort-1}
284          */
285         public String getName() {
286             return name;
287         }
288 
289         /**
290          * Whether the connector attaches a display built into the device.
291          *
292          * @return true if built in
293          */
294         public boolean isBuiltIn() {
295             return builtIn;
296         }
297 
298         /**
299          * The mode the display on this connector is currently driven in.
300          *
301          * @return the current mode, or empty if unknown
302          */
303         public Optional<DisplayMode> getMode() {
304             return Optional.ofNullable(mode);
305         }
306     }
307 
308     /**
309      * Normalizes a monitor device interface path for case-insensitive matching. The path returned by
310      * {@code SetupDiGetDeviceInterfaceDetail} and the {@code monitorDevicePath} returned by
311      * {@code DisplayConfigGetDeviceInfo} are the same string but may differ in case.
312      *
313      * @param devicePath the device interface path, or {@code null}
314      * @return the lower-cased path, or {@link Constants#UNKNOWN} if the input is null or empty
315      */
316     public static String normalizePath(String devicePath) {
317         if (Util.isBlank(devicePath)) {
318             return Constants.UNKNOWN;
319         }
320         return devicePath.toLowerCase(Locale.ROOT);
321     }
