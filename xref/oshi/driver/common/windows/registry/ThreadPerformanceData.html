29  /**
30   * Common logic for building thread performance data maps from registry or performance counter results. Callers (JNA/FFM
31   * variants) supply the platform-specific pre-fetched registry or performance-counter data.
32   */
33  @ThreadSafe
34  public final class ThreadPerformanceData {
35  
36      /**
37       * The performance object name for thread counters.
38       */
39      public static final String THREAD = "Thread";
40  
41      private ThreadPerformanceData() {
42      }
43  
44      /**
45       * Builds a thread map from registry performance data that has already been read.
46       *
47       * @param pids       An optional collection of process IDs to filter the list to. May be null for no filtering.
48       * @param threadData The raw registry data triplet (instance maps, perfTime100nSec, now in ms)
49       * @return A map with Thread ID as the key and a {@link ThreadPerfCounterBlock} object populated with performance
50       *         counter information, or null if threadData is null.
51       */
52      private static @Nullable Map<Integer, ThreadPerfCounterBlock> mapFromRegistryData(
53              @Nullable Collection<Integer> pids,
54              @Nullable Triplet<List<Map<ThreadPerformanceProperty, Object>>, Long, Long> threadData) {
55          if (threadData == null) {
56              return null;
57          }
58          List<Map<ThreadPerformanceProperty, Object>> threadInstanceMaps = threadData.getA();
59          long perfTime100nSec = threadData.getB(); // 1601
60          long now = threadData.getC(); // 1970 epoch
61  
62          Map<Integer, ThreadPerfCounterBlock> threadMap = new HashMap<>();
63          for (Map<ThreadPerformanceProperty, Object> threadInstanceMap : threadInstanceMaps) {
64              int pid = intValue(threadInstanceMap, ThreadPerformanceProperty.IDPROCESS);
65              int tid = intValue(threadInstanceMap, ThreadPerformanceProperty.IDTHREAD);
66              // TID 0 is never a real thread -- thread IDs come from the same pool as PIDs -- so never key the map
67              // on it; 0 is the "ID Thread" sentinel the perf-counter path can report for the _Total aggregate or
68              // an exiting thread.
69              if (tid != 0 && (pids == null || pids.contains(pid)) && pid > 0) {
70                  String name = stringValue(threadInstanceMap, ThreadPerformanceProperty.NAME);
71                  long upTime = (perfTime100nSec - longValue(threadInstanceMap, ThreadPerformanceProperty.ELAPSEDTIME))
72                          / 10_000L;
73                  if (upTime < 1) {
74                      upTime = 1;
75                  }
76                  long user = longValue(threadInstanceMap, ThreadPerformanceProperty.PERCENTUSERTIME) / 10_000L;
77                  long kernel = longValue(threadInstanceMap, ThreadPerformanceProperty.PERCENTPRIVILEGEDTIME) / 10_000L;
78                  int priority = intValue(threadInstanceMap, ThreadPerformanceProperty.PRIORITYCURRENT);
79                  int threadState = intValue(threadInstanceMap, ThreadPerformanceProperty.THREADSTATE);
80                  int threadWaitReason = intValue(threadInstanceMap, ThreadPerformanceProperty.THREADWAITREASON);
81                  // Start address is pointer sized when fetched from registry, so this could be
82                  // either Integer (uint32) or Long depending on OS bitness
83                  long startAddr = pointerValue(threadInstanceMap, ThreadPerformanceProperty.STARTADDRESS);
84                  long contextSwitches = Integer
85                          .toUnsignedLong(intValue(threadInstanceMap, ThreadPerformanceProperty.CONTEXTSWITCHESPERSEC));
86                  threadMap.put(tid, new ThreadPerfCounterBlock(name, tid, pid, now - upTime, user, kernel, priority,
87                          threadState, threadWaitReason, startAddr, contextSwitches));
88              }
89          }
90          return threadMap;
91      }

93      /**
94       * Builds a thread map from performance counter query results.
95       *
96       * @param pids           An optional collection of process IDs to filter the list to. May be null for no filtering.
97       * @param instanceValues The query results as a pair of (instances, valueMap)
98       * @return A map with Thread ID as the key and a {@link ThreadPerfCounterBlock} object populated with performance
99       *         counter information, or null if instanceValues is null.
100      */
101     private static @Nullable Map<Integer, ThreadPerfCounterBlock> mapFromCounterValues(
102             @Nullable Collection<Integer> pids,
103             @Nullable Pair<List<String>, Map<ThreadPerformanceProperty, List<Long>>> instanceValues) {
104         if (instanceValues == null) {
105             return null;
106         }
107         Map<Integer, ThreadPerfCounterBlock> threadMap = new HashMap<>();
108         long now = System.currentTimeMillis(); // 1970 epoch
109         List<String> instances = instanceValues.getA();
110         Map<ThreadPerformanceProperty, List<Long>> valueMap = instanceValues.getB();
111         List<Long> tidList = counterList(valueMap, ThreadPerformanceProperty.IDTHREAD);
112         List<Long> pidList = counterList(valueMap, ThreadPerformanceProperty.IDPROCESS);
113         List<Long> userList = counterList(valueMap, ThreadPerformanceProperty.PERCENTUSERTIME); // 100-nsec
114         List<Long> kernelList = counterList(valueMap, ThreadPerformanceProperty.PERCENTPRIVILEGEDTIME); // 100-nsec
115         List<Long> startTimeList = counterList(valueMap, ThreadPerformanceProperty.ELAPSEDTIME); // filetime
116         List<Long> priorityList = counterList(valueMap, ThreadPerformanceProperty.PRIORITYCURRENT);
117         List<Long> stateList = counterList(valueMap, ThreadPerformanceProperty.THREADSTATE);
118         List<Long> waitReasonList = counterList(valueMap, ThreadPerformanceProperty.THREADWAITREASON);
119         List<Long> startAddrList = counterList(valueMap, ThreadPerformanceProperty.STARTADDRESS);
120         List<Long> contextSwitchesList = counterList(valueMap, ThreadPerformanceProperty.CONTEXTSWITCHESPERSEC);
121 
122         int nameIndex = 0;
123         for (int inst = 0; inst < instances.size(); inst++) {
124             int pid = pidList.get(inst).intValue();
125             int tid = tidList.get(inst).intValue();
126             // TID 0 is never a real thread (thread IDs share the PID pool). PDH reports it as the "ID Thread"
127             // sentinel for the _Total aggregate and exiting threads; keying the map on 0 would clobber a real
128             // thread with another.
129             if (tid != 0 && (pids == null || pids.contains(pid))) {
130                 String name = Integer.toString(nameIndex++);
131                 long startTime = startTimeList.get(inst);
132                 startTime = ParseUtil.filetimeToUtcMs(startTime, false);
133                 if (startTime > now) {
134                     startTime = now - 1;
135                 }
136                 long user = userList.get(inst) / 10_000L;
137                 long kernel = kernelList.get(inst) / 10_000L;
138                 int priority = priorityList.get(inst).intValue();
139                 int threadState = stateList.get(inst).intValue();
140                 int threadWaitReason = waitReasonList.get(inst).intValue();
141                 long startAddr = startAddrList.get(inst).longValue();
142                 long contextSwitches = contextSwitchesList.get(inst).longValue();
143 
144                 threadMap.put(tid, new ThreadPerfCounterBlock(name, tid, pid, startTime, user, kernel, priority,
145                         threadState, threadWaitReason, startAddr, contextSwitches));
146             }
147         }
148         return threadMap;
149     }

151     /**
152      * Reads thread performance data from the registry.
153      *
154      * @param executor The backend performing the native queries
155      * @param pids     Process IDs to filter to, or null for all
156      * @return A map of thread ID to counter block, or null if the read failed
157      */
158     public static @Nullable Map<Integer, ThreadPerfCounterBlock> buildThreadMapFromRegistry(
159             PerfCounterQueryExecutor executor, @Nullable Collection<Integer> pids) {
160         if (!HkeyPerformanceDataUtil.PERFDATA) {
161             return null;
162         }
163         return mapFromRegistryData(pids, executor.readPerfDataFromRegistry(THREAD, ThreadPerformanceProperty.class));
164     }
165 
166     /**
167      * Reads thread performance data for every thread from performance counters.
168      *
169      * @param executor The backend performing the native queries
170      * @param pids     Process IDs to filter to, or null for all
171      * @return A map of thread ID to counter block, or null if the read failed
172      */
173     public static @Nullable Map<Integer, ThreadPerfCounterBlock> buildThreadMapFromPerfCounters(
174             PerfCounterQueryExecutor executor, @Nullable Collection<Integer> pids) {
175         return buildThreadMapFromPerfCounters(executor, pids, null, -1);
176     }
177 
178     /**
179      * Reads thread performance data from performance counters, optionally narrowed to one process.
180      *
181      * @param executor  The backend performing the native queries
182      * @param pids      Process IDs to filter to, or null for all
183      * @param procName  Limit the matches to threads of this process, or null for all
184      * @param threadNum Limit the matches to this thread, or -1 for all
185      * @return A map of thread ID to counter block, or null if the read failed
186      */
187     public static @Nullable Map<Integer, ThreadPerfCounterBlock> buildThreadMapFromPerfCounters(
188             PerfCounterQueryExecutor executor, @Nullable Collection<Integer> pids, @Nullable String procName,
189             int threadNum) {
190         if (executor.isPerfProcDisabled()) {
191             return Collections.emptyMap();
192         }
193         Pair<List<String>, Map<ThreadPerformanceProperty, List<Long>>> instanceValues = procName == null
194                 || procName.isEmpty() ? ThreadInformation.queryThreadCounters(executor)
195                         : ThreadInformation.queryThreadCounters(executor, procName, threadNum);
196         return mapFromCounterValues(pids, instanceValues);
197     }
