5   package oshi.driver.common.windows.registry;
6   
7   import java.util.List;
8   import java.util.Map;
9   
10  import oshi.annotation.concurrent.ThreadSafe;
11  
12  /**
13   * Typed lookups into the property maps produced for a performance counter object.
14   * <p>
15   * Every property of the counter enum is present or none of them are: {@link HkeyPerformanceDataUtil#getCounterIndices}
16   * returns {@code null} unless it resolved an index for each one, and the perf-counter path fills its value map from the
17   * same all-or-nothing query. A missing key therefore means the data was assembled inconsistently rather than that a
18   * counter is optional, so these throw rather than substitute a value, naming the property instead of failing later with
19   * a bare {@code NullPointerException}.
20   */
21  @ThreadSafe
22  final class PerfCounterValues {
23  
24      private PerfCounterValues() {
25      }
26  
27      private static <T extends Enum<T>> Object require(Map<T, ?> map, T key) {
28          Object value = map.get(key);
29          if (value == null) {
30              throw new IllegalStateException("Missing performance counter value for " + key.name());
31          }
32          return value;
33      }
34  
35      /**
36       * Fetches a property whose value is stored as an {@code Integer}.
37       *
38       * @param <T> the counter property enum type
39       * @param map the property map for one instance
40       * @param key the property to fetch
41       * @return the value as an {@code int}
42       */
43      static <T extends Enum<T>> int intValue(Map<T, Object> map, T key) {
44          return ((Integer) require(map, key)).intValue();
45      }
46  
47      /**
48       * Fetches a property whose value is stored as a {@code Long}.
49       *
50       * @param <T> the counter property enum type
51       * @param map the property map for one instance
52       * @param key the property to fetch
53       * @return the value as a {@code long}
54       */
55      static <T extends Enum<T>> long longValue(Map<T, Object> map, T key) {
56          return ((Long) require(map, key)).longValue();
57      }
58  
59      /**
60       * Fetches a property whose value is stored as a {@code String}.
61       *
62       * @param <T> the counter property enum type
63       * @param map the property map for one instance
64       * @param key the property to fetch
65       * @return the value as a {@code String}
66       */
67      static <T extends Enum<T>> String stringValue(Map<T, Object> map, T key) {
68          return (String) require(map, key);
69      }
70  
71      /**
72       * Fetches a pointer-sized property, which the registry reports as an {@code Integer} on a 32-bit system and a
73       * {@code Long} on a 64-bit one.
74       *
75       * @param <T> the counter property enum type
76       * @param map the property map for one instance
77       * @param key the property to fetch
78       * @return the value widened to a {@code long}, without sign extension
79       */
80      static <T extends Enum<T>> long pointerValue(Map<T, Object> map, T key) {
81          Object value = require(map, key);
82          return value instanceof Long ? (Long) value : Integer.toUnsignedLong((Integer) value);
83      }
84  
85      /**
86       * Fetches the per-instance values of one counter from a performance counter query result.
87       *
88       * @param <T>      the counter property enum type
89       * @param valueMap the query's value map
90       * @param key      the property to fetch
91       * @return the list of values, one per instance
92       */
93      static <T extends Enum<T>> List<Long> counterList(Map<T, List<Long>> valueMap, T key) {
94          @SuppressWarnings("unchecked")
95          List<Long> values = (List<Long>) require(valueMap, key);
96          return values;
97      }
98  }
