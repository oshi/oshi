25  /**
26   * Base class for HKEY_PERFORMANCE_DATA utilities. Contains the common logic for building the counter name/index map and
27   * resolving counter indices from enum definitions. Subclasses provide the platform-specific native call to read the
28   * registry string array.
29   */
30  @ThreadSafe
31  public abstract class HkeyPerformanceDataUtil {
32  
33      private static final Logger LOG = LoggerFactory.getLogger(HkeyPerformanceDataUtil.class);
34  
35      /**
36       * Subclass-only constructor.
37       */
38      protected HkeyPerformanceDataUtil() {
39      }
40  
41      // Field offsets within the HKEY_PERFORMANCE_DATA structures, from winperf.h on 64-bit Windows. The walk below
42      // needs a handful of fields from five structs; naming them here keeps the arithmetic in one place rather than
43      // once per native backend. HkeyPerformanceDataUtilTest lays out a synthetic block at these offsets written
44      // independently, so a wrong constant reads the wrong place and fails there.
45      private static final long DATA_HEADER_LENGTH = 24;
46      private static final long DATA_NUM_OBJECT_TYPES = 28;
47      private static final long DATA_PERF_TIME_100N_SEC = 72;
48  
49      private static final long OBJECT_TOTAL_BYTE_LENGTH = 0;
50      private static final long OBJECT_DEFINITION_LENGTH = 4;
51      private static final long OBJECT_HEADER_LENGTH = 8;
52      private static final long OBJECT_NAME_TITLE_INDEX = 12;
53      private static final long OBJECT_NUM_COUNTERS = 32;
54      private static final long OBJECT_NUM_INSTANCES = 40;
55  
56      private static final long COUNTER_DEF_BYTE_LENGTH = 0;
57      private static final long COUNTER_DEF_NAME_TITLE_INDEX = 4;
58      private static final long COUNTER_DEF_COUNTER_SIZE = 32;
59      private static final long COUNTER_DEF_COUNTER_OFFSET = 36;
60  
61      private static final long INSTANCE_DEF_BYTE_LENGTH = 0;
62      private static final long INSTANCE_DEF_NAME_OFFSET = 16;
63  
64      private static final long COUNTER_BLOCK_BYTE_LENGTH = 0;
65  
66      /**
67       * Whether HKEY_PERFORMANCE_DATA may be read at all, per {@link GlobalConfig#OSHI_OS_WINDOWS_HKEYPERFDATA}. Lives
68       * here rather than on one caller because the setting governs the registry itself, and is documented as covering
69       * both processes and threads.
70       */
71      static final boolean PERFDATA = GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_HKEYPERFDATA, true);
72  
73      /**
74       * Registry key containing English counter name/index pairs.
75       */
76      public static final String HKEY_PERFORMANCE_TEXT = "SOFTWARE\\Microsoft\\Windows NT\\CurrentVersion\\Perflib\\009";
77  
78      /**
79       * Registry value name for the counter text.
80       */
81      public static final String COUNTER = "Counter";

83      /**
84       * Builds the counter index map from a string array of alternating index/name pairs.
85       * <p>
86       * The array format is pairs of {@code "<index>", "<name>"}. The first pair is a length/count entry (e.g.,
87       * {@code "1", "1847"}) followed by actual counter definitions such as {@code "2", "System", "4", "Memory", "6",
88       * "% Processor Time", ...}. Even positions (0, 2, 4, ...) hold numeric index strings and odd positions (1, 3, 5,
89       * ...) hold the corresponding counter names.
90       *
91       * @param counterText The string array from the registry, or null if the read failed
92       * @return An unmodifiable map of counter name to index
93       */
94      protected static Map<String, Integer> buildCounterIndexMap(String @Nullable [] counterText) {
95          HashMap<String, Integer> indexMap = new HashMap<>();
96          if (counterText != null && counterText.length > 1) {
97              for (int i = 1; i < counterText.length; i += 2) {
98                  int idx = ParseUtil.parseIntOrDefault(counterText[i - 1], 0);
99                  if (idx > 0) {
100                     indexMap.putIfAbsent(counterText[i], idx);
101                 }
102             }
103         }
104         return Collections.unmodifiableMap(indexMap);
105     }

107     /**
108      * Looks up the counter index values for the given counter object and the enum of counter names.
109      *
110      * @param <T>             An enum containing the counters, whose class is passed as {@code counterEnum}
111      * @param objectName      The counter object to look up the index for
112      * @param counterEnum     The {@link Enum} containing counters to look up the indices for. The first Enum value will
113      *                        be ignored.
114      * @param counterIndexMap The map of counter names to their indices
115      * @return A {@link Pair} containing the index of the counter object as the first element, and an {@link EnumMap}
116      *         mapping counter enum values to their index as the second element, if the lookup is successful; null
117      *         otherwise.
118      */
119     protected static <T extends Enum<T> & PdhCounterWildcardProperty> @Nullable Pair<Integer, EnumMap<T, Integer>> getCounterIndices(
120             String objectName, Class<T> counterEnum, Map<String, Integer> counterIndexMap) {
121         Integer counterIndex = counterIndexMap.get(objectName);
122         if (counterIndex == null) {
123             LOG.debug("Couldn't find counter index of {}.", objectName);
124             return null;
125         }
126         T[] enumConstants = counterEnum.getEnumConstants();
127         EnumMap<T, Integer> indexMap = new EnumMap<>(counterEnum);
128         // Start iterating at 1 because first Enum value defines the name/instance and
129         // is not a counter name
130         for (int i = 1; i < enumConstants.length; i++) {
131             T key = enumConstants[i];
132             Integer idx = counterIndexMap.get(key.getCounter());
133             if (idx == null) {
134                 LOG.debug("Couldn't find counter index of {}.", key.getCounter());
135                 return null;
136             }
137             indexMap.put(key, idx);
138         }
139         // We have all the pieces! Return them.
140         return new Pair<>(counterIndex, indexMap);
141     }

143     /**
144      * Walks a populated HKEY_PERFORMANCE_DATA block and extracts the counters named by an enum.
145      * <p>
146      * The block is a flat buffer of variable-length records: a header, then object types, each holding counter
147      * definitions followed by instances, each instance followed by its counter block. Every step is offset arithmetic,
148      * so this is shared rather than written once per native backend.
149      *
150      * @param <T>          The counter enum type. Its first constant names the instance; the rest name counters.
151      * @param buffer       The populated performance data block
152      * @param objectIndex  Index of the object type to extract
153      * @param enumIndexMap Counter enum values mapped to their registry index
154      * @param counterEnum  The counter enum class
155      * @return A triplet of the per-instance counter maps, the timestamp in 100nSec units of the Windows 1601 epoch, and
156      *         the same timestamp in milliseconds of the 1970 epoch; or null if the object was not present or a counter
157      *         could not be read
158      */
159     protected static <T extends Enum<T> & PdhCounterWildcardProperty> @Nullable Triplet<List<Map<T, Object>>, Long, Long> parsePerfData(
160             PerfDataBuffer buffer, int objectIndex, EnumMap<T, Integer> enumIndexMap, Class<T> counterEnum) {
161         long perfTime100nSec = buffer.getLong(DATA_PERF_TIME_100N_SEC); // 1601
162         long now = ParseUtil.filetimeToUtcMs(perfTime100nSec, false); // 1970
163 
164         // Iterate object types
165         long perfObjectOffset = buffer.getInt(DATA_HEADER_LENGTH);
166         int numObjectTypes = buffer.getInt(DATA_NUM_OBJECT_TYPES);
167         for (int obj = 0; obj < numObjectTypes; obj++) {
168             if (buffer.getInt(perfObjectOffset + OBJECT_NAME_TITLE_INDEX) == objectIndex) {
169                 // Counter definitions start after the object header
170                 long perfCounterOffset = perfObjectOffset + buffer.getInt(perfObjectOffset + OBJECT_HEADER_LENGTH);
171                 int numCounters = buffer.getInt(perfObjectOffset + OBJECT_NUM_COUNTERS);
172                 Map<Integer, Integer> counterOffsetMap = new HashMap<>();
173                 Map<Integer, Integer> counterSizeMap = new HashMap<>();
174                 for (int counter = 0; counter < numCounters; counter++) {
175                     int nameTitleIndex = buffer.getInt(perfCounterOffset + COUNTER_DEF_NAME_TITLE_INDEX);
176                     counterOffsetMap.put(nameTitleIndex, buffer.getInt(perfCounterOffset + COUNTER_DEF_COUNTER_OFFSET));
177                     counterSizeMap.put(nameTitleIndex, buffer.getInt(perfCounterOffset + COUNTER_DEF_COUNTER_SIZE));
178                     perfCounterOffset += buffer.getInt(perfCounterOffset + COUNTER_DEF_BYTE_LENGTH);
179                 }
180 
181                 // Instances start after all the object definitions
182                 long perfInstanceOffset = perfObjectOffset + buffer.getInt(perfObjectOffset + OBJECT_DEFINITION_LENGTH);
183                 int numInstances = buffer.getInt(perfObjectOffset + OBJECT_NUM_INSTANCES);
184                 T[] counterKeys = counterEnum.getEnumConstants();
185                 List<Map<T, Object>> counterMaps = new ArrayList<>(numInstances);
186                 for (int inst = 0; inst < numInstances; inst++) {
187                     long perfCounterBlockOffset = perfInstanceOffset
188                             + buffer.getInt(perfInstanceOffset + INSTANCE_DEF_BYTE_LENGTH);
189                     Map<T, Object> counterMap = new EnumMap<>(counterEnum);
190                     // The first enum constant names the instance rather than a counter
191                     long nameOffset = buffer.getInt(perfInstanceOffset + INSTANCE_DEF_NAME_OFFSET);
192                     counterMap.put(counterKeys[0], buffer.getWideString(perfInstanceOffset + nameOffset));
193                     for (int i = 1; i < counterKeys.length; i++) {
194                         T key = counterKeys[i];
195                         int keyIndex = enumIndexMap.getOrDefault(key, -1);
196                         Integer offset = counterOffsetMap.get(keyIndex);
197                         if (offset == null) {
198                             return null;
199                         }
200                         int size = counterSizeMap.getOrDefault(keyIndex, 0);
201                         if (size == 4) {
202                             counterMap.put(key, buffer.getInt(perfCounterBlockOffset + offset));
203                         } else if (size == 8) {
204                             counterMap.put(key, buffer.getLong(perfCounterBlockOffset + offset));
205                         } else {
206                             return null;
207                         }
208                     }
209                     counterMaps.add(counterMap);
210                     perfInstanceOffset = perfCounterBlockOffset
211                             + buffer.getInt(perfCounterBlockOffset + COUNTER_BLOCK_BYTE_LENGTH);
212                 }
213                 return new Triplet<>(counterMaps, perfTime100nSec, now);
214             }
215             perfObjectOffset += buffer.getInt(perfObjectOffset + OBJECT_TOTAL_BYTE_LENGTH);
216         }
217         return null;
218     }
