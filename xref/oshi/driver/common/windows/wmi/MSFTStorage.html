5   package oshi.driver.common.windows.wmi;
6   
7   import java.util.regex.Pattern;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  
11  /**
12   * Constants and property enums for WMI classes in the Storage namespace associated with Storage Pools.
13   */
14  @ThreadSafe
15  public class MSFTStorage {
16  
17      /**
18       * The WMI namespace for storage.
19       */
20      public static final String STORAGE_NAMESPACE = "ROOT\\Microsoft\\Windows\\Storage";
21  
22      /** Matches a storage pool reference or object ID, capturing the pool's GUID. */
23      public static final Pattern SP_OBJECT_ID = Pattern.compile("SP:(\\{[^}]*\\})");
24      /** Matches a physical disk reference or object ID, capturing the disk's GUID. */
25      public static final Pattern PD_OBJECT_ID = Pattern.compile("PD:(\\{[^}]*\\})");
26      /** Matches a virtual disk reference or object ID, capturing its pool's GUID and then its own. */
27      public static final Pattern VD_OBJECT_ID = Pattern.compile("VD:(\\{[^}]*\\})(\\{[^}]*\\})");
28  
29      /**
30       * The WMI class name for non-primordial storage pools.
31       */
32      public static final String MSFT_STORAGE_POOL_WHERE_IS_PRIMORDIAL_FALSE = "MSFT_StoragePool WHERE IsPrimordial=FALSE";
33  
34      /**
35       * The WMI class name for storage pool to physical disk associations.
36       */
37      public static final String MSFT_STORAGE_POOL_TO_PHYSICAL_DISK = "MSFT_StoragePoolToPhysicalDisk";
38  
39      /**
40       * The WMI class name for physical disks.
41       */
42      public static final String MSFT_PHYSICAL_DISK = "MSFT_PhysicalDisk";
43  
44      /**
45       * The WMI class name for virtual disks.
46       */
47      public static final String MSFT_VIRTUAL_DISK = "MSFT_VirtualDisk";
48  
49      /**
50       * The WMI class name for virtual disk to physical disk associations.
51       */
52      public static final String MSFT_VIRTUAL_DISK_TO_PHYSICAL_DISK = "MSFT_VirtualDiskToPhysicalDisk";
53  
54      /**
55       * The WMI class name for disks, the storage namespace's view of each disk Windows exposes.
56       */
57      public static final String MSFT_DISK = "MSFT_Disk";
58  
59      /**
60       * Properties to identify the storage pool. The Object ID uniquely defines the pool.
61       */
62      public enum StoragePoolProperty {
63          /** Friendly name of the storage pool. */
64          FRIENDLYNAME,
65          /** Object ID of the storage pool. */
66          OBJECTID;
67      }
68  
69      /**
70       * Properties to link a storage pool with a physical disk. OSHI parses these references to strings that can match
71       * the object IDs.
72       */
73      public enum StoragePoolToPhysicalDiskProperty {
74          /** Storage pool reference. */
75          STORAGEPOOL,
76          /** Physical disk reference. */
77          PHYSICALDISK;
78      }
79  
80      /**
81       * Properties for a physical disk. The Object ID uniquely defines the disk.
82       */
83      public enum PhysicalDiskProperty {
84          /** Friendly name of the physical disk. */
85          FRIENDLYNAME,
86          /** Physical location of the disk. */
87          PHYSICALLOCATION,
88          /** Object ID of the physical disk. */
89          OBJECTID;
90      }
91  
92      /**
93       * Properties for a physical disk's media type. Unlike {@code Win32_DiskDrive}'s MediaType, which reports only
94       * removability, this MediaType reports the media technology and so distinguishes an SSD from an HDD.
95       */
96      public enum PhysicalDiskMediaTypeProperty {
97          /** Device ID of the physical disk. For a directly attached disk this is its physical drive number. */
98          DEVICEID,
99          /** Media type: 3 is HDD, 4 is SSD, 5 is storage class memory, 0 is unspecified. */
100         MEDIATYPE,
101         /** Spindle speed in RPM. Zero for non-rotational media, {@code UINT32_MAX} if rotating at unknown speed. */
102         SPINDLESPEED;
103     }
104 
105     /**
106      * Properties to identify a physical disk by the device number Windows exposes it as.
107      */
108     public enum PhysicalDiskIdProperty {
109         /** Device ID of the physical disk: its physical drive number. */
110         DEVICEID,
111         /** Object ID of the physical disk. */
112         OBJECTID;
113     }
114 
115     /**
116      * Properties to link a virtual disk with the physical disks holding its data. OSHI parses these references to
117      * strings that can match the object IDs.
118      */
119     public enum VirtualDiskToPhysicalDiskProperty {
120         /** Virtual disk reference. */
121         VIRTUALDISK,
122         /** Physical disk reference. */
123         PHYSICALDISK;
124     }
125 
126     /**
127      * Properties for a disk, to identify what it is built on.
128      */
129     public enum DiskProperty {
130         /** The disk number, matching {@code Win32_DiskDrive}'s Index. */
131         NUMBER,
132         /** The bus type: 15 is a file-backed virtual disk, 16 a Storage Spaces disk. */
133         BUSTYPE,
134         /** For a file-backed virtual disk, the path of its backing file. */
135         LOCATION,
136         /** The unique ID, which for a Storage Spaces disk matches its virtual disk's. */
137         UNIQUEID;
138     }
139 
140     /**
141      * Properties for a virtual disk's identity, to match it with the disk Windows exposes it as.
142      */
143     public enum VirtualDiskIdProperty {
144         /** The unique ID, matching the {@code MSFT_Disk} exposing this virtual disk. */
145         UNIQUEID,
146         /** Object ID of the virtual disk. */
147         OBJECTID;
148     }
149 
150     /**
151      * Properties for a virtual disk. The Object ID uniquely defines the disk.
152      */
153     public enum VirtualDiskProperty {
154         /** Friendly name of the virtual disk. */
155         FRIENDLYNAME,
156         /** Object ID of the virtual disk. */
157         OBJECTID;
158     }
159 
160     /**
161      * Constructor.
162      */
163     protected MSFTStorage() {
164     }
165 
166     /**
167      * Queries storage pools.
168      *
169      * @param h An instantiated {@link WmiQueryExecutor}. User should have already initialized COM.
170      * @return Storage pool information.
171      */
172     public static WmiResult<StoragePoolProperty> queryStoragePools(WmiQueryExecutor h) {
173         WmiQuery<StoragePoolProperty> query = new WmiQuery<>(STORAGE_NAMESPACE,
174                 MSFT_STORAGE_POOL_WHERE_IS_PRIMORDIAL_FALSE, StoragePoolProperty.class);
175         return h.queryWMI(query, false);
176     }
177 
178     /**
179      * Queries storage pool to physical disk mapping.
180      *
181      * @param h An instantiated {@link WmiQueryExecutor}. User should have already initialized COM.
182      * @return Storage pool to physical disk mapping.
183      */
184     public static WmiResult<StoragePoolToPhysicalDiskProperty> queryStoragePoolPhysicalDisks(WmiQueryExecutor h) {
185         WmiQuery<StoragePoolToPhysicalDiskProperty> query = new WmiQuery<>(STORAGE_NAMESPACE,
186                 MSFT_STORAGE_POOL_TO_PHYSICAL_DISK, StoragePoolToPhysicalDiskProperty.class);
187         return h.queryWMI(query, false);
188     }
