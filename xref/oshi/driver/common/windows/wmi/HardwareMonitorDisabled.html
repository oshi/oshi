5   package oshi.driver.common.windows.wmi;
6   
7   import oshi.annotation.concurrent.ThreadSafe;
8   import oshi.util.GlobalConfig;
9   
10  /**
11   * Tests whether the Open Hardware Monitor and Libre Hardware Monitor sources of sensor data are disabled by
12   * configuration.
13   * <p>
14   * Either application publishes its data only while it is running, and a user may start or stop it at any time, so OSHI
15   * queries for it on every read rather than caching its availability. These switches let a user who does not run one of
16   * them say so up front, so the query is never attempted. They are read on each call for the same reason the queries
17   * are: configuration may be changed before the first query is made.
18   *
19   * @see GlobalConfig#OSHI_OS_WINDOWS_OHM_DISABLED
20   * @see GlobalConfig#OSHI_OS_WINDOWS_LHM_DISABLED
21   */
22  @ThreadSafe
23  public final class HardwareMonitorDisabled {
24  
25      private HardwareMonitorDisabled() {
26          throw new AssertionError();
27      }
28  
29      /**
30       * Tests whether queries to a hardware monitor's WMI namespace are disabled.
31       *
32       * @param namespace the WMI namespace, either {@link OhmHardware#OHM_NAMESPACE} or {@link LhmSensor#LHM_NAMESPACE}
33       * @return {@code true} if the namespace should not be queried, {@code false} otherwise, including for any other
34       *         namespace
35       */
36      public static boolean isWmiDisabled(String namespace) {
37          if (OhmHardware.OHM_NAMESPACE.equals(namespace)) {
38              return GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_OHM_DISABLED, false);
39          }
40          if (LhmSensor.LHM_NAMESPACE.equals(namespace)) {
41              return GlobalConfig.get(GlobalConfig.OSHI_OS_WINDOWS_LHM_DISABLED, false);
42          }
43          return false;
44      }
45  }
