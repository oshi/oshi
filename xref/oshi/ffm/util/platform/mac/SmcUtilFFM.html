152     /**
153      * Open a connection to SMC.
154      *
155      * @return The io_connect_t port (nonzero) if successful, 0 if failure. Caller must close with {@link #smcClose}.
156      */
157     public static int smcOpen() {
158         IOService smcService = IOKitUtilFFM.getMatchingService("AppleSMC");
159         if (smcService == null) {
160             OPEN_FAILURE.serviceNotFound();
161             return 0;
162         }
163         try {
164             return callInArenaIntOrDefault(arena -> {
165                 MemorySegment connPtr = arena.allocate(JAVA_INT);
166                 int task = mach_task_self();
167                 int result = IOServiceOpen(smcService.handle(), task, 0, connPtr);
168                 if (result == 0) {
169                     int conn = connPtr.get(JAVA_INT, 0);
170                     if (conn == 0) {
171                         OPEN_FAILURE.nullConnection();
172                         return 0;
173                     }
174                     return conn;
175                 }
176                 OPEN_FAILURE.openFailed(result);
177                 return 0;
178             }, LOG, ERROR, "Exception opening SMC connection", 0);
179         } finally {
180             smcService.release();
181         }
182     }

194     /**
195      * Get a value from SMC which is in a floating point datatype (SP78, FPE2, FLT)
196      *
197      * @param conn The connection
198      * @param key  The key to retrieve
199      * @return Double representing the value
200      */
201     public static double smcGetFloat(int conn, String key) {
202         return callInArenaDoubleOrDefault(arena -> {
203             MemorySegment val = arena.allocate(SMC_VAL);
204             int result = smcReadKey(conn, key, val, arena);
205             if (result == 0) {
206                 int dataSize = val.get(JAVA_INT, SMC_VAL.byteOffset(SMC_VAL_DATA_SIZE));
207                 if (dataSize > 0) {
208                     byte[] dataType = readByteArray(val, SMC_VAL.byteOffset(SMC_VAL_DATA_TYPE), 5);
209                     byte[] bytes = readByteArray(val, SMC_VAL.byteOffset(SMC_VAL_BYTES), dataSize);
210                     if (Arrays.equals(dataType, DATATYPE_SP78) && dataSize == 2) {
211                         return bytes[0] + (bytes[1] & 0xFF) / 256d;
212                     } else if (Arrays.equals(dataType, DATATYPE_FPE2) && dataSize == 2) {
213                         return ParseUtil.byteArrayToFloat(bytes, dataSize, 2);
214                     } else if (Arrays.equals(dataType, DATATYPE_FLT) && dataSize == 4) {
215                         return ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN).getFloat();
216                     }
217                 }
218             }
219             return 0d;
220         }, 0d, LOG, WARN, "Failed to read SMC float key {}", key);
221     }

223     /**
224      * The lowest reading accepted as a genuine temperature, in degrees Celsius.
225      * <p>
226      * Apple Silicon power-gates a CPU core cluster when it is idle, and the SMC then reports a fixed sentinel for each
227      * die sensor in that cluster instead of a reading. Observed sentinels are 6.7 and 4.633 on an M2 Max (Tp01/Tp09 and
228      * Tp05/Tp0D), up to 8.425 across all of that machine's sensors, and -4.0 through 5.2 in the
229      * <a href="https://github.com/dkorunic/iSMC">iSMC</a> sample reports, which independently describe them as
230      * "firmware sentinels from inactive sensor slots". They are below room ambient and so cannot be real die
231      * temperatures, but most are positive, so a simple {@code > 0} test accepts them.
232      * <p>
233      * Across those reports, covering M1 through M5, A18 and Intel T2 machines, no genuine sensor read between 8.425 and
234      * 21, so this floor sits in an empty band with margin on both sides. It is a plausibility guard, not a hardware
235      * specification.
236      */
237     public static final double MIN_PLAUSIBLE_TEMPERATURE = 15d;
238 
239     /**
240      * Tests whether a reading is plausible as a temperature, rejecting the sentinel an idle-gated sensor reports.
241      *
242      * @param celsius the reading to test, in degrees Celsius
243      * @return true if the reading is at least {@link #MIN_PLAUSIBLE_TEMPERATURE}
244      */
245     public static boolean isPlausibleTemperature(double celsius) {
246         return celsius >= MIN_PLAUSIBLE_TEMPERATURE;
247     }

262     /**
263      * Get the highest plausible temperature among a list of SMC keys.
264      * <p>
265      * Plausibility is applied per read rather than when the keys were discovered, because whether a key exists is a
266      * property of the hardware while whether it currently reports a usable value is not.
267      * <p>
268      * Clusters expose their sensors in pairs at different locations on the die, offset by a few degrees and tracking
269      * each other as load changes; verified on an M2 Max where a pair moved 44.8/51.4 to 56.1/62.6 under load and fell
270      * together afterwards. They are not an instantaneous/peak-hold pair, so taking the maximum consistently reports the
271      * hotter location rather than a latched peak.
272      *
273      * @param conn The connection
274      * @param keys The keys to read
275      * @return The highest reading at or above {@link #MIN_PLAUSIBLE_TEMPERATURE}, or 0 if none were plausible
276      */
277     public static double smcGetMaxTemperature(int conn, List<String> keys) {
278         return SmcKeyIndex.maxPlausible(keys, key -> smcGetFloat(conn, key), SmcUtilFFM::isPlausibleTemperature);
279     }

281     /**
282      * Get the name of the key at an index in the SMC key index.
283      *
284      * @param conn  The connection
285      * @param index The index, from 0 to the value of {@link #SMC_KEY_COUNT}
286      * @return The four-character key name, or null if it could not be read
287      */
288     public static @Nullable String smcReadKeyAtIndex(int conn, int index) {
289         return callInArenaOrDefault(arena -> {
290             MemorySegment input = arena.allocate(SMC_KEY_DATA);
291             MemorySegment output = arena.allocate(SMC_KEY_DATA);
292             input.set(JAVA_BYTE, SMC_KEY_DATA.byteOffset(SMC_DATA8), SMC_CMD_READ_INDEX);
293             input.set(JAVA_INT, SMC_KEY_DATA.byteOffset(SMC_DATA32), index);
294             if (smcCall(conn, KERNEL_INDEX_SMC, input, output) != 0) {
295                 return null;
296             }
297             int key = output.get(JAVA_INT, SMC_KEY_DATA.byteOffset(SMC_KEY));
298             byte[] keyBytes = ParseUtil.longToByteArray(key, 4, 4);
299             StringBuilder sb = new StringBuilder(4);
300             for (byte b : keyBytes) {
301                 sb.append((char) (b & 0xFF));
302             }
303             return sb.toString();
304         }, null, LOG, DEBUG, "Failed to read SMC key at index {}", index);
305     }

307     /**
308      * Get the SMC data type of a key, e.g. {@code flt} or {@code sp78}.
309      *
310      * @param conn The connection
311      * @param key  The key to query
312      * @return The data type, or an empty string if it could not be read
313      */
314     public static String smcGetDataType(int conn, String key) {
315         return callInArenaOrDefault(arena -> {
316             MemorySegment val = arena.allocate(SMC_VAL);
317             if (smcReadKey(conn, key, val, arena) != 0) {
318                 return "";
319             }
320             byte[] type = readByteArray(val, SMC_VAL.byteOffset(SMC_VAL_DATA_TYPE), 5);
321             StringBuilder sb = new StringBuilder(4);
322             for (byte b : type) {
323                 if (b == 0) {
324                     break;
325                 }
326                 sb.append((char) b);
327             }
328             return sb.toString().trim();
329         }, "", LOG, DEBUG, "Failed to read SMC data type for key {}", key);
330     }

332     /**
333      * Get the Apple Silicon GPU temperature keys for this machine, discovering them from the SMC on first use and
334      * caching the result.
335      * <p>
336      * Discovery binary searches the sorted key index for the {@code Tg} block and keeps the keys matching the GPU
337      * naming convention. It does not filter by plausibility: an idle sensor reports a sentinel below ambient, so
338      * filtering here would let one unlucky first call cache an empty set and disable GPU temperature permanently.
339      * Plausibility is applied per read by {@link #smcGetMaxTemperature}.
340      * <p>
341      * Can be overridden with the {@link GlobalConfig#OSHI_OS_MAC_SENSORS_GPUTEMPERATURE_KEYS} configuration property,
342      * which bypasses discovery entirely. Falls back to {@link #SMC_KEYS_GPU_TEMP_AS} if discovery cannot complete,
343      * without caching, so a later call retries.
344      *
345      * @return The keys to read for GPU temperature, never null
346      */
347     public static List<String> getGpuTemperatureKeys() {
348         return GPU_KEYS.get(SmcUtilFFM::discoverGpuTemperatureKeys);
349     }

351     /**
352      * Get the fan speed keys for this machine, discovering them from the SMC on first use and caching the result.
353      * <p>
354      * Discovery binary searches the sorted key index for the {@code F} block and keeps the keys matching the fan
355      * current-speed naming convention, then reconciles the result against {@code FNum}: the index is authoritative, but
356      * a positive {@code FNum} still implies the conventionally named keys where discovery finds none, so this never
357      * reports fewer fans than reading {@code FNum} directly did. See
358      * {@link SmcKeyIndex#reconcileFanKeys(java.util.List, long)}.
359      * <p>
360      * Can be overridden with the {@link GlobalConfig#OSHI_OS_MAC_SENSORS_FANSPEED_KEYS} configuration property, which
361      * bypasses discovery entirely.
362      *
363      * @return The keys to read for fan speeds, never null
364      */
365     public static List<String> getFanSpeedKeys() {
366         return FAN_KEYS.get(SmcUtilFFM::discoverFanSpeedKeys);
367     }

369     /**
370      * @return the keys to read, or null if neither the key index nor {@code FNum} could be read
371      */
372     private static @Nullable List<String> discoverFanSpeedKeys() {
373         int conn = smcOpen();
374         if (conn == 0) {
375             return null; // NOSONAR java:S1168 - null means "could not read", which the caller must not cache
376         }
377         try {
378             int keyCount = (int) smcGetLong(conn, SMC_KEY_COUNT);
379             // Scanning the index rather than probing F0Ac through F9Ac directly, because smcGetDataType cannot tell an
380             // absent key from a failed read: both return an empty string. findKeys tracks read failures and so can
381             // distinguish "this machine has no fans" from "ask again later", which a direct probe cannot.
382             List<String> discovered = SmcKeyIndex.findKeys(keyCount, i -> smcReadKeyAtIndex(conn, i), FAN_KEY_PREFIX,
383                     SmcKeyIndex::isFanSpeedKey);
384             // Read FNum on the same connection, so a machine whose index is unreadable still reports its fans.
385             return SmcKeyIndex.reconcileFanKeys(discovered, smcGetLong(conn, SMC_KEY_FAN_NUM));
386         } finally {
387             smcClose(conn);
388         }
389     }

392      * Get the keys to read for CPU voltage, in the order they should be tried.
393      * <p>
394      * Unlike the GPU temperature keys these are not discovered from the key index, because the {@code V} prefix is not

398      * name the key on hardware where neither default works.

405         return configured.isEmpty() ? SMC_KEYS_CPU_VOLTAGE : configured;
406     }

438     /**
439      * @return the discovered keys, or null if the SMC key index could not be read
440      */
441     private static @Nullable List<String> discoverGpuTemperatureKeys() {
442         int conn = smcOpen();
443         if (conn == 0) {
444             return null; // NOSONAR java:S1168 - null means "could not read", which the caller must not cache
445         }
446         try {
447             int keyCount = (int) smcGetLong(conn, SMC_KEY_COUNT);
448             // No data-type filter: smcGetFloat already decodes every temperature encoding the SMC uses (flt, sp78,
449             // fpe2) and returns 0 for anything else, which the plausibility floor then rejects.
450             return SmcKeyIndex.findKeys(keyCount, i -> smcReadKeyAtIndex(conn, i), GPU_KEY_PREFIX,
451                     SmcKeyIndex::isGpuTemperatureKey);
452         } finally {
453             smcClose(conn);
454         }
455     }

464     public static long smcGetLong(int conn, String key) {
465         return callInArenaLongOrDefault(arena -> {
466             MemorySegment val = arena.allocate(SMC_VAL);
467             int result = smcReadKey(conn, key, val, arena);
468             if (result == 0) {

477     private static int smcReadKey(int conn, String key, MemorySegment val, Arena arena) throws Throwable {
478         MemorySegment input = arena.allocate(SMC_KEY_DATA);
479         MemorySegment output = arena.allocate(SMC_KEY_DATA);
480 
481         input.set(JAVA_INT, SMC_KEY_DATA.byteOffset(SMC_KEY), (int) ParseUtil.strToLong(key, 4));
482 
483         int result = smcGetKeyInfo(conn, input, output);
484         if (result == 0) {
485             int dataSize = output.get(JAVA_INT, KEY_INFO_DATA_SIZE_OFFSET);
486             int dataType = output.get(JAVA_INT, KEY_INFO_DATA_TYPE_OFFSET);
487 
488             val.set(JAVA_INT, SMC_VAL.byteOffset(SMC_VAL_DATA_SIZE), dataSize);
489             byte[] typeBytes = ParseUtil.longToByteArray(dataType, 4, 5);
490             writeByteArray(val, SMC_VAL.byteOffset(SMC_VAL_DATA_TYPE), typeBytes);
491 
492             // Reset input and re-populate for the actual read
493             input.fill((byte) 0);
494             input.set(JAVA_INT, SMC_KEY_DATA.byteOffset(SMC_KEY), (int) ParseUtil.strToLong(key, 4));
495             input.set(JAVA_INT, KEY_INFO_DATA_SIZE_OFFSET, dataSize);
496             input.set(JAVA_BYTE, SMC_KEY_DATA.byteOffset(SMC_DATA8), SMC_CMD_READ_BYTES);
497 
498             result = smcCall(conn, KERNEL_INDEX_SMC, input, output);
499             if (result == 0) {
500                 byte[] outBytes = readByteArray(output, SMC_KEY_DATA.byteOffset(SMC_BYTES), 32);
501                 writeByteArray(val, SMC_VAL.byteOffset(SMC_VAL_BYTES), outBytes);
502             }
503         }
504         return result;
505     }

507     private static int smcGetKeyInfo(int conn, MemorySegment input, MemorySegment output) throws Throwable {
508         int key = input.get(JAVA_INT, SMC_KEY_DATA.byteOffset(SMC_KEY));
509         int[] cached = KEY_INFO_CACHE.get(key);
510         if (cached != null) {
511             output.set(JAVA_INT, KEY_INFO_DATA_SIZE_OFFSET, cached[0]);
512             output.set(JAVA_INT, KEY_INFO_DATA_TYPE_OFFSET, cached[1]);
513             output.set(JAVA_BYTE, KEY_INFO_DATA_ATTR_OFFSET, (byte) cached[2]);
514             return 0;
515         }
516         input.set(JAVA_BYTE, SMC_KEY_DATA.byteOffset(SMC_DATA8), SMC_CMD_READ_KEYINFO);
517         int result = smcCall(conn, KERNEL_INDEX_SMC, input, output);
518         if (result == 0) {
519             int dataSize = output.get(JAVA_INT, KEY_INFO_DATA_SIZE_OFFSET);
520             int dataType = output.get(JAVA_INT, KEY_INFO_DATA_TYPE_OFFSET);
521             byte dataAttr = output.get(JAVA_BYTE, KEY_INFO_DATA_ATTR_OFFSET);
522             KEY_INFO_CACHE.put(key, new int[] { dataSize, dataType, dataAttr & 0xFF });
523         }
524         return result;
525     }

527     private static int smcCall(int conn, int index, MemorySegment input, MemorySegment output) throws Throwable {
528         try (Arena arena = Arena.ofConfined()) {
529             MemorySegment outputSize = arena.allocate(JAVA_LONG);
530             outputSize.set(JAVA_LONG, 0, SMC_KEY_DATA.byteSize());
531             return IOConnectCallStructMethod(conn, index, input, SMC_KEY_DATA.byteSize(), output, outputSize);
532         }
533     }

535     private static byte[] readByteArray(MemorySegment seg, long offset, int length) {
536         byte[] result = new byte[length];
537         for (int i = 0; i < length; i++) {
538             result[i] = seg.get(JAVA_BYTE, offset + i);
539         }
540         return result;
541     }

543     private static void writeByteArray(MemorySegment seg, long offset, byte[] data) {

545             seg.set(JAVA_BYTE, offset + i, data[i]);
546         }
547     }
