25  /**
26   * Utility class for working with the Foreign Function and Memory API. Provides helpers for library lookup, downcalls,
27   * and UTF-16 string conversion.
28   */
29  public abstract class WindowsForeignFunctions extends ForeignFunctions {
30  
31      protected WindowsForeignFunctions() {
32      }
33  
34      /**
35       * Validates a Windows API return code, allowing {@code ERROR_SUCCESS} or specified codes.
36       *
37       * @param rc            return code from a Windows API call
38       * @param allowedErrors optional additional codes
39       * @return the original code if valid
40       * @throws Win32Exception if not successful
41       */
42      public static int checkSuccess(int rc, int... allowedErrors) {
43          if (rc == ERROR_SUCCESS) {
44              return rc;
45          }
46          for (int allowed : allowedErrors) {
47              if (rc == allowed) {
48                  return rc;
49              }
50          }
51          throw new Win32Exception(rc);
52      }
53  
54      /**
55       * Converts a Windows BOOL (0 or non-zero) to a Java boolean. In Windows API, 0 means FALSE and non-zero means TRUE.
56       *
57       * @param winBool the BOOL value returned by a Windows API
58       * @return {@code true} if non-zero, {@code false} if zero
59       */
60      public static boolean isSuccess(int winBool) {
61          return winBool != 0;
62      }
63  
64      /**
65       * Passes through the result of a Windows API call, logging the name of the function and the value of
66       * {@code GetLastError()} if it failed. Intended to be used as a guard condition so that the failure branch of a
67       * native call is a single expression:
68       *
69       * <pre>
70       * if (!succeededOrLog(Advapi32FFM.OpenProcessToken(hProcess, access, hToken), LOG, "OpenProcessToken")) {
71       *     return false;
72       * }
73       * </pre>
74       *
75       * @param success  the result of the native call
76       * @param logger   the logger of the calling class, so the message is attributed to the caller
77       * @param function the name of the native function that was called
78       * @return {@code success}, unchanged
79       */
80      public static boolean succeededOrLog(boolean success, Logger logger, String function) {
81          if (!success) {
82              logger.error("{} failed, error: {}", function, Kernel32FFM.GetLastError().orElse(-1));
83          }
84          return success;
85      }
86  
87      /**
88       * Reads a null-terminated UTF-16 wide string from the given memory segment.
89       *
90       * @param seg the memory segment containing the UTF-16 string
91       * @return the decoded Java string
92       */
93      public static String readWideString(MemorySegment seg) {
94          StringBuilder sb = new StringBuilder();
95          for (int offset = 0;; offset += 2) { // 2 bytes per UTF-16 char
96              char c = seg.get(JAVA_CHAR, offset);
97              if (c == '\0') {
98                  break; // null terminator
99              }
100             sb.append(c);
101         }
102         return sb.toString();
103     }
104 
105     // Bound for reading a UTF-16 string from a raw pointer: 512 characters, comfortably covering identifier-length
106     // strings such as a Windows username (UNLEN = 256). A fixed bound fails safe (catchable) on non-terminated data.
107     private static final long WIDE_POINTER_STRING_BYTES = 1024;
108 
109     /**
110      * Reads a null-terminated UTF-16 wide string from a raw native pointer, sizing the read to accommodate
111      * identifier-length strings (e.g. a Windows username up to {@code UNLEN} characters) plus a null terminator. This
112      * is the wide-string counterpart to {@link oshi.ffm.ForeignFunctions#getStringFromNativePointer}.
113      *
114      * @param pointer the native pointer to a wide string
115      * @return the decoded Java string, or {@code ""} if the pointer is null or {@link MemorySegment#NULL}
116      */
117     public static String readWideStringFromPointer(MemorySegment pointer) {
118         if (pointer == null || pointer.equals(MemorySegment.NULL)) {
119             return "";
120         }
121         return readWideString(pointer.reinterpret(WIDE_POINTER_STRING_BYTES));
122     }
123 
124     /**
125      * Reads a null-terminated ANSI (single-byte) string from the given memory segment.
126      *
127      * @param seg    the memory segment containing the ANSI string
128      * @param maxLen the maximum number of bytes to read
129      * @return the decoded Java string
130      */
131     public static String readAnsiString(MemorySegment seg, int maxLen) {
132         StringBuilder sb = new StringBuilder();
133         for (int i = 0; i < maxLen; i++) {
134             byte b = seg.get(JAVA_BYTE, i);
135             if (b == 0) {
136                 break;
137             }
138             sb.append((char) b);
139         }
140         return sb.toString();
141     }
142 
143     /**
144      * Creates and initializes a {@code TOKEN_PRIVILEGES} structure with one privilege enabled.
145      *
146      * @param arena the memory arena used for allocation
147      * @param luid  the LUID of the privilege to enable
148      * @return a memory segment containing the initialized {@code TOKEN_PRIVILEGES} structure
149      */
150     public static MemorySegment setupTokenPrivileges(Arena arena, MemorySegment luid) {
151         MemorySegment tkp = arena.allocate(TOKEN_PRIVILEGES);
152 
153         tkp.set(JAVA_INT, TOKEN_PRIVILEGES_PRIVILEGE_COUNT_OFFSET, 1);
154 
155         MemorySegment luidSegment = tkp.asSlice(TOKEN_PRIVILEGES_LUID_OFFSET, luid.byteSize());
156         luidSegment.copyFrom(luid);
157 
158         tkp.set(JAVA_INT, TOKEN_PRIVILEGES_ATTRIBUTES_OFFSET, SE_PRIVILEGE_ENABLED);
159 
160         return tkp;
161     }
162 
163     /**
164      * Converts a Java string into a null-terminated UTF-16LE wide string memory segment.
165      *
166      * @param arena the memory arena used for allocation
167      * @param s     the Java string to convert
168      * @return a memory segment containing the UTF-16LE encoded string with a null terminator
169      */
170     public static MemorySegment toWideString(Arena arena, String s) {
171         return arena.allocateFrom(s + "\0", StandardCharsets.UTF_16LE);
172     }
