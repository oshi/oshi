5   package oshi.ffm.platform.unix.netbsd;
6   
7   import static java.lang.foreign.ValueLayout.ADDRESS;
8   import static java.lang.foreign.ValueLayout.JAVA_INT;
9   
10  import java.lang.foreign.FunctionDescriptor;
11  import java.lang.foreign.MemorySegment;
12  import java.lang.foreign.ValueLayout;
13  import java.lang.invoke.MethodHandle;
14  
15  import oshi.ffm.ForeignFunctions;
16  import oshi.ffm.platform.unix.PosixLibcFunctions;
17  
18  /**
19   * FFM bindings for NetBSD libc functions used by OSHI.
20   * <p>
21   * Binds {@code sysctl}, {@code _lwp_self} and {@code getloadavg}; the POSIX bindings ({@code getpid},
22   * {@code getrlimit}) are inherited from {@link PosixLibcFunctions}.
23   */
24  public final class NetBsdLibcFunctions extends PosixLibcFunctions {
25  
26      private NetBsdLibcFunctions() {
27      }
28  
29      /** Layout of the C {@code size_t} type on NetBSD (8 bytes on all supported 64-bit archs). */
30      public static final ValueLayout.OfLong SIZE_T = ValueLayout.JAVA_LONG;
31  
32      /** {@code getrlimit} resource: maximum number of open file descriptors. NetBSD value (8), not Linux's 7. */
33      public static final int RLIMIT_NOFILE = 8;
34  
35      // int sysctl(const int *name, u_int namelen, void *oldp, size_t *oldlenp, const void *newp, size_t newlen);
36      private static final MethodHandle sysctl = LINKER.downcallHandle(LIBC.findOrThrow("sysctl"),
37              FunctionDescriptor.of(JAVA_INT, ADDRESS, JAVA_INT, ADDRESS, ADDRESS, ADDRESS, SIZE_T), CAPTURE_CALL_STATE);
38  
39      /**
40       * Calls {@code sysctl(name, namelen, oldp, oldlenp, newp, newlen)} with errno capture.
41       *
42       * @param callState segment allocated with {@link ForeignFunctions#CAPTURED_STATE_LAYOUT} for errno capture
43       * @param name      MIB array segment (sequence of ints)
44       * @param namelen   number of ints in {@code name}
45       * @param oldp      output buffer for the current value, or {@link MemorySegment#NULL} to query size only
46       * @param oldlenp   pointer to a {@code size_t}: on input, the size of {@code oldp}; on output, the bytes written
47       * @param newp      new value buffer, or {@link MemorySegment#NULL} when reading
48       * @param newlen    size of {@code newp} in bytes, or 0 when reading
49       * @return 0 on success, -1 on error
50       * @throws Throwable on FFM invocation error
51       */
52      public static int sysctl(MemorySegment callState, MemorySegment name, int namelen, MemorySegment oldp,
53              MemorySegment oldlenp, MemorySegment newp, long newlen) throws Throwable {
54          return (int) sysctl.invokeExact(callState, name, namelen, oldp, oldlenp, newp, newlen);
55      }
56  
57      // lwpid_t _lwp_self(void);
58      private static final MethodHandle lwpSelf = LINKER.downcallHandle(LIBC.findOrThrow("_lwp_self"),
59              FunctionDescriptor.of(JAVA_INT));
60  
61      /**
62       * Calls {@code _lwp_self()} — returns the LWP (lightweight process/thread) ID of the calling thread.
63       *
64       * @return the LWP ID of the calling thread
65       * @throws Throwable on FFM invocation error
66       */
67      public static int lwpSelf() throws Throwable {
68          return (int) lwpSelf.invokeExact();
69      }
70  
71      // int getloadavg(double loadavg[], int nelem);
72      private static final MethodHandle getloadavg = LINKER.downcallHandle(LIBC.findOrThrow("getloadavg"),
73              FunctionDescriptor.of(JAVA_INT, ADDRESS, JAVA_INT));
74  
75      /**
76       * Calls {@code getloadavg(loadavg, nelem)}.
77       *
78       * @param loadavg pre-allocated segment of {@code nelem} doubles
79       * @param nelem   number of load average values to retrieve (1-3)
80       * @return number of samples set, or -1 on error
81       * @throws Throwable on FFM invocation error
82       */
83      public static int getloadavg(MemorySegment loadavg, int nelem) throws Throwable {
84          return (int) getloadavg.invokeExact(loadavg, nelem);
85      }
86  
87      // --- Sysctl MIB constants, from <sys/sysctl.h> ---
88  
89      public static final int CTL_KERN = 1;
90  
91      public static final int KERN_ARGMAX = 8;
92      public static final int KERN_CP_TIME = 51;
93      public static final int KERN_PROC_ARGS = 48;
94      public static final int KERN_PROC_ENV = 3;
95  }
