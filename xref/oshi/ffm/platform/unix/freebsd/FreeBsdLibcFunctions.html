25  /**
26   * FFM bindings for FreeBSD libc functions used by OSHI.
27   * <p>
28   * Covers {@code sysctl} with an int MIB, {@code sysctlbyname}, {@code getloadavg}, {@code thr_self}, the {@code utmpx}
29   * family, and the {@code getaddrinfo}/{@code freeaddrinfo}/{@code gai_strerror} surface used by network params. The
30   * POSIX bindings ({@code getpid}, {@code getrlimit}, {@code gethostname}) are inherited from
31   * {@link PosixLibcFunctions}.

50      /**
51       * Layout of FreeBSD's {@code struct addrinfo} ({@code <netdb.h>}).
52       *
53       * <pre>
54       *   int    ai_flags      (4) @ 0
55       *   int    ai_family     (4) @ 4
56       *   int    ai_socktype   (4) @ 8
57       *   int    ai_protocol   (4) @ 12
58       *   int    ai_addrlen    (4) @ 16   -- socklen_t (uint32_t on FreeBSD)
59       *   pad                  (4) @ 20
60       *   ptr    ai_canonname  (8) @ 24   -- note: canonname BEFORE addr, unlike Linux glibc
61       *   ptr    ai_addr       (8) @ 32
62       *   ptr    ai_next       (8) @ 40
63       *   total = 48 bytes
64       * </pre>
65       */
66      public static final StructLayout ADDRINFO_LAYOUT = MemoryLayout.structLayout(JAVA_INT.withName("ai_flags"),
67              JAVA_INT.withName("ai_family"), JAVA_INT.withName("ai_socktype"), JAVA_INT.withName("ai_protocol"),
68              JAVA_INT.withName("ai_addrlen"), MemoryLayout.paddingLayout(4), ADDRESS.withName("ai_canonname"),
69              ADDRESS.withName("ai_addr"), ADDRESS.withName("ai_next"));
70  
71      private static final VarHandle ADDRINFO_CANONNAME = ADDRINFO_LAYOUT
72              .varHandle(PathElement.groupElement("ai_canonname"));

87      /**
88       * Layout of FreeBSD's {@code struct utmpx}.
89       *
90       * <pre>
91       *   short            ut_type      (2)
92       *   pad                           (6)
93       *   long             ut_tv.tv_sec (8)
94       *   long             ut_tv.tv_usec(8)
95       *   char[8]          ut_id        (8)
96       *   int              ut_pid       (4)
97       *   char[32]         ut_user      (32)
98       *   char[16]         ut_line      (16)
99       *   char[128]        ut_host      (128)
100      *   char[64]         __ut_spare   (64)
101      *   pad                           (4)
102      *   total = 280 bytes
103      * </pre>
104      *
105      * The trailing padding is real: {@code ut_tv} aligns the struct to 8, and {@code structLayout} never infers tail
106      * padding. {@code ut_tv} is nested rather than flattened so that the audit in {@code .github/ffm-audit} can check
107      * its offset against the header, which it cannot do for a member the header does not declare.
108      */
109     public static final StructLayout UTMPX_LAYOUT = MemoryLayout.structLayout(JAVA_SHORT.withName("ut_type"),
110             MemoryLayout.paddingLayout(6),
111             MemoryLayout.structLayout(JAVA_LONG.withName("tv_sec"), JAVA_LONG.withName("tv_usec")).withName("ut_tv"),
112             MemoryLayout.sequenceLayout(UTX_IDSIZE, JAVA_BYTE).withName("ut_id"), JAVA_INT.withName("ut_pid"),
113             MemoryLayout.sequenceLayout(UTX_USERSIZE, JAVA_BYTE).withName("ut_user"),
114             MemoryLayout.sequenceLayout(UTX_LINESIZE, JAVA_BYTE).withName("ut_line"),
115             MemoryLayout.sequenceLayout(UTX_HOSTSIZE, JAVA_BYTE).withName("ut_host"),
116             MemoryLayout.sequenceLayout(UTX_SPARESIZE, JAVA_BYTE).withName("__ut_spare"),
117             MemoryLayout.paddingLayout(4));

121             PathElement.groupElement("tv_sec"));
122     private static final VarHandle UTMPX_TV_USEC = UTMPX_LAYOUT.varHandle(PathElement.groupElement("ut_tv"),
123             PathElement.groupElement("tv_usec"));

127     private static final long UTMPX_HOST_OFFSET = UTMPX_LAYOUT.byteOffset(PathElement.groupElement("ut_host"));

239     /**
240      * Reads the login time from a utmpx segment as epoch milliseconds.
241      *
242      * @param ut segment populated by {@link #getutxent()} and reinterpreted to {@link #UTMPX_LAYOUT}
243      * @return login time in milliseconds since epoch
244      */
245     public static long utmpxLoginTime(MemorySegment ut) {
246         long tvSec = (long) UTMPX_TV_SEC.get(ut, 0L);
247         long tvUsec = (long) UTMPX_TV_USEC.get(ut, 0L);
248         return tvSec * 1000L + tvUsec / 1000L;
249     }

301     /**
302      * Reads {@code ai_canonname} from the first addrinfo result pointer.
303      *
304      * @param resPtr the raw pointer value stored in the output segment after {@link #getaddrinfo}
305      * @param arena  arena to scope the reinterpret
306      * @return the canonical name string, or {@code null} if the pointer is NULL
307      */
308     public static String addrinfoCanonname(MemorySegment resPtr, java.lang.foreign.Arena arena) {
309         if (resPtr == null || MemorySegment.NULL.equals(resPtr)) {
310             return null;
311         }
312         MemorySegment addrinfo = MemorySegment.ofAddress(resPtr.address()).reinterpret(ADDRINFO_LAYOUT.byteSize(),
313                 arena, null);
314         MemorySegment canonPtr = (MemorySegment) ADDRINFO_CANONNAME.get(addrinfo, 0L);
315         return getStringFromNativePointer(canonPtr, arena);
316     }
