5   package oshi.jna.platform.windows;
6   
7   import com.sun.jna.Native;
8   import com.sun.jna.Pointer;
9   import com.sun.jna.ptr.IntByReference;
10  import com.sun.jna.win32.StdCallLibrary;
11  
12  import oshi.driver.common.windows.DisplayConnector;
13  
14  /**
15   * Binding to the {@code user32.dll} Connecting and Configuring Displays (CCD) functions, which are not mapped by JNA's
16   * platform {@code User32}. These functions have no {@code A}/{@code W} variants, so the library is loaded without a
17   * name-mangling function mapper. Buffers are passed as raw pointers and their fields read by offset (see
18   * {@link DisplayConnector}), rather than mapping the {@code DISPLAYCONFIG_*} structs.
19   * <p>
20   * This class should be considered non-API as it may be removed if/when its code is incorporated into the JNA project.
21   */
22  public interface User32 extends StdCallLibrary {
23  
24      User32 INSTANCE = Native.load("user32", User32.class);
25  
26      /**
27       * Retrieves the size of the buffers required to call {@code QueryDisplayConfig}.
28       *
29       * @param flags                    a combination of {@code QDC_*} flags
30       * @param numPathArrayElements     receives the number of path elements
31       * @param numModeInfoArrayElements receives the number of mode info elements
32       * @return {@code ERROR_SUCCESS} (0) on success, otherwise a Win32 error code
33       */
34      int GetDisplayConfigBufferSizes(int flags, IntByReference numPathArrayElements,
35              IntByReference numModeInfoArrayElements);
36  
37      /**
38       * Retrieves the active display paths and their modes.
39       *
40       * @param flags                    a combination of {@code QDC_*} flags
41       * @param numPathArrayElements     in/out count of path array elements
42       * @param pathArray                buffer receiving {@code DISPLAYCONFIG_PATH_INFO} elements
43       * @param numModeInfoArrayElements in/out count of mode info array elements
44       * @param modeInfoArray            buffer receiving {@code DISPLAYCONFIG_MODE_INFO} elements
45       * @param currentTopologyId        optional topology id output, may be {@code null}
46       * @return {@code ERROR_SUCCESS} (0) on success, otherwise a Win32 error code
47       */
48      int QueryDisplayConfig(int flags, IntByReference numPathArrayElements, Pointer pathArray,
49              IntByReference numModeInfoArrayElements, Pointer modeInfoArray, Pointer currentTopologyId);
50  
51      /**
52       * Retrieves display configuration information about a device, given a request packet whose header identifies the
53       * target.
54       *
55       * @param requestPacket a {@code DISPLAYCONFIG_DEVICE_INFO_HEADER} followed by the type-specific payload
56       * @return {@code ERROR_SUCCESS} (0) on success, otherwise a Win32 error code
57       */
58      int DisplayConfigGetDeviceInfo(Pointer requestPacket);
59  }
