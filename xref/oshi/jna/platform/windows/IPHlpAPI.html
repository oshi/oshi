15  /**
16   * Extends JNA's IP Helper API mapping with the routing table calls it does not provide.
17   * <p>
18   * This class should be considered non-API as it may be removed if/when its code is incorporated into the JNA project.
19   */
20  public interface IPHlpAPI extends com.sun.jna.platform.win32.IPHlpAPI {
21  
22      IPHlpAPI INSTANCE = Native.load("IPHlpAPI", IPHlpAPI.class, W32APIOptions.DEFAULT_OPTIONS);
23  
24      /**
25       * A union of {@code SOCKADDR_IN}, {@code SOCKADDR_IN6} and a bare {@code ADDRESS_FAMILY}, mapped as its largest arm
26       * so the layout is 28 bytes with 4-byte alignment.
27       * <p>
28       * The IPv4 arm's {@code sin_addr} occupies the four bytes at offset 4, which is where the IPv6 arm's
29       * {@code sin6_flowinfo} sits, so one field serves both and {@link #si_family} selects between them.
30       */
31      @FieldOrder({ "si_family", "port", "ipv4AddrOrFlowInfo", "ipv6Addr", "scopeId" })
32      class SOCKADDR_INET extends Structure {
33          public short si_family;
34          public short port;
35          public int ipv4AddrOrFlowInfo;
36          public byte[] ipv6Addr = new byte[16];
37          public int scopeId;
38      }
39  
40      /** An address paired with the number of leading bits that are significant. */
41      @FieldOrder({ "Prefix", "PrefixLength" })
42      class IP_ADDRESS_PREFIX extends Structure {
43          public SOCKADDR_INET Prefix = new SOCKADDR_INET();
44          public byte PrefixLength;
45      }
46  
47      /**
48       * A single routing table row.
49       * <p>
50       * {@code NET_LUID} is a union of a {@code ULONG64} and a bitfield over the same 64 bits. Since OSHI never reads it,
51       * mapping it as a plain {@code long} is layout-identical and avoids a nested structure.
52       */
53      @FieldOrder({ "InterfaceLuid", "InterfaceIndex", "DestinationPrefix", "NextHop", "SitePrefixLength",
54              "ValidLifetime", "PreferredLifetime", "Metric", "Protocol", "Loopback", "AutoconfigureAddress", "Publish",
55              "Immortal", "Age", "Origin" })
56      class MIB_IPFORWARD_ROW2 extends Structure {
57          public long InterfaceLuid;
58          public int InterfaceIndex;
59          public IP_ADDRESS_PREFIX DestinationPrefix = new IP_ADDRESS_PREFIX();
60          public SOCKADDR_INET NextHop = new SOCKADDR_INET();
61          public byte SitePrefixLength;
62          public int ValidLifetime;
63          public int PreferredLifetime;
64          public int Metric;
65          public int Protocol;
66          public byte Loopback;
67          public byte AutoconfigureAddress;
68          public byte Publish;
69          public byte Immortal;
70          public int Age;
71          public int Origin;
72  
73          public MIB_IPFORWARD_ROW2() {
74              super();
75          }
76  
77          public MIB_IPFORWARD_ROW2(Pointer p) {
78              super(p);
79          }
80      }
81  
82      /**
83       * Retrieves the IP route entries on the local computer. The table is allocated by the system and must be released
84       * with {@link #FreeMibTable(Pointer)}.
85       *
86       * @param Family the inherited {@link #AF_INET}, {@link #AF_INET6}, or {@link #AF_UNSPEC} for both, narrowed to the
87       *               {@code ADDRESS_FAMILY} this call declares
88       * @param Table  receives a pointer to a {@code MIB_IPFORWARD_TABLE2}, whose {@code ULONG NumEntries} is followed by
89       *               the row array. The rows are 8-byte aligned, so they begin at offset 8 rather than 4.
90       * @return {@code NO_ERROR} on success
91       * @see <a href=
92       *      "https://learn.microsoft.com/en-us/windows/win32/api/netioapi/nf-netioapi-getipforwardtable2">GetIpForwardTable2</a>
93       */
94      int GetIpForwardTable2(short Family, PointerByReference Table);
95  
96      /**
97       * Frees a table allocated by the IP Helper API.
98       *
99       * @param Memory the table to free
100      * @see <a href=
101      *      "https://learn.microsoft.com/en-us/windows/win32/api/netioapi/nf-netioapi-freemibtable">FreeMibTable</a>
102      */
103     void FreeMibTable(Pointer Memory);
104 
105     /** {@code GetAdaptersAddresses} flag to leave out unicast addresses. */
106     int GAA_FLAG_SKIP_UNICAST = 0x1;
107     /** {@code GetAdaptersAddresses} flag to leave out anycast addresses. */
108     int GAA_FLAG_SKIP_ANYCAST = 0x2;
109     /** {@code GetAdaptersAddresses} flag to leave out multicast addresses. */
110     int GAA_FLAG_SKIP_MULTICAST = 0x4;
111     /** {@code GetAdaptersAddresses} flag to leave out the adapter's friendly name. */
112     int GAA_FLAG_SKIP_FRIENDLY_NAME = 0x20;
113 
114     /**
115      * The leading fields of {@code IP_ADAPTER_ADDRESSES_LH}, through {@code OperStatus}. The structure is read in place
116      * from the buffer {@link #GetAdaptersAddresses} fills, so the fields after these need not be mapped.
117      */
118     @FieldOrder({ "Length", "IfIndex", "Next", "AdapterName", "FirstUnicastAddress", "FirstAnycastAddress",
119             "FirstMulticastAddress", "FirstDnsServerAddress", "DnsSuffix", "Description", "FriendlyName",
120             "PhysicalAddress", "PhysicalAddressLength", "Flags", "Mtu", "IfType", "OperStatus" })
121     class IP_ADAPTER_ADDRESSES extends Structure {
122         public int Length;
123         public int IfIndex;
124         public Pointer Next;
125         public Pointer AdapterName;
126         public Pointer FirstUnicastAddress;
127         public Pointer FirstAnycastAddress;
128         public Pointer FirstMulticastAddress;
129         public Pointer FirstDnsServerAddress;
130         public Pointer DnsSuffix;
131         public Pointer Description;
132         public Pointer FriendlyName;
133         public byte[] PhysicalAddress = new byte[8];
134         public int PhysicalAddressLength;
135         public int Flags;
136         public int Mtu;
137         public int IfType;
138         public int OperStatus;
139 
140         public IP_ADAPTER_ADDRESSES(Pointer p) {
141             super(p);
142             read();
143         }
144     }
145 
146     /** One DNS server address of an adapter, in a linked list. */
147     @FieldOrder({ "Length", "Reserved", "Next", "lpSockaddr", "iSockaddrLength" })
148     class IP_ADAPTER_DNS_SERVER_ADDRESS extends Structure {
149         public int Length;
150         public int Reserved;
151         public Pointer Next;
152         public Pointer lpSockaddr;
153         public int iSockaddrLength;
154 
155         public IP_ADAPTER_DNS_SERVER_ADDRESS(Pointer p) {
156             super(p);
157             read();
158         }
159     }
160 
161     /**
162      * Retrieves the addresses associated with the adapters on the local computer.
163      *
164      * @param Family           {@link #AF_INET}, {@link #AF_INET6}, or {@link #AF_UNSPEC} for both
165      * @param Flags            the {@code GAA_FLAG_*} flags
166      * @param Reserved         must be null
167      * @param AdapterAddresses the buffer to fill with a linked list of {@link IP_ADAPTER_ADDRESSES}
168      * @param SizePointer      the buffer size, updated to the size required on {@code ERROR_BUFFER_OVERFLOW}
169      * @return {@code ERROR_SUCCESS} on success
170      * @see <a href=
171      *      "https://learn.microsoft.com/en-us/windows/win32/api/iphlpapi/nf-iphlpapi-getadaptersaddresses">GetAdaptersAddresses</a>
172      */
173     int GetAdaptersAddresses(int Family, int Flags, Pointer Reserved, Pointer AdapterAddresses,
174             IntByReference SizePointer);
175 }
