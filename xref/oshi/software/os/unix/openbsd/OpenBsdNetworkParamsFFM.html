23  /**
24   * FFM-backed OpenBSD network params. The gateway lookup and the {@code getHostName} fallback live on
25   * {@link OpenBsdNetworkParams}; only the {@code gethostname} binding is FFM-specific.
26   */
27  @ThreadSafe
28  public class OpenBsdNetworkParamsFFM extends OpenBsdNetworkParams {
29  
30      private static final Logger LOG = LoggerFactory.getLogger(OpenBsdNetworkParamsFFM.class);
31  
32      @Override
33      protected @Nullable String queryHostName() {
34          return callInArenaOrDefault(arena -> {
35              MemorySegment buf = arena.allocate(HOST_NAME_MAX + 1L);
36              if (0 != PosixLibcFunctions.gethostname(buf, HOST_NAME_MAX + 1L)) {
37                  return null;
38              }
39              return buf.getString(0);
40          }, LOG, LogLevel.WARN, "Failed to get hostname", null);
41      }
42  
43      @Override
44      protected byte[] queryRouteDump() {
45          return BsdRouteDumpFFM.queryRouteDump((state, name, namelen, oldp, oldlenp) -> OpenBsdLibcFunctions
46                  .sysctl(state, name, namelen, oldp, oldlenp, MemorySegment.NULL, 0));
47      }
48  }
