5   package oshi.software.os.unix.freebsd;
6   
7   import com.sun.jna.Memory;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.software.common.os.unix.freebsd.FreeBsdInternetProtocolStats;
11  import oshi.util.platform.unix.freebsd.BsdSysctlUtil;
12  
13  /**
14   * Internet Protocol Stats implementation
15   */
16  @ThreadSafe
17  public class FreeBsdInternetProtocolStatsJNA extends FreeBsdInternetProtocolStats {
18  
19      @Override
20      protected byte[] querySysctl(String name) {
21          try (Memory m = BsdSysctlUtil.sysctl(name)) {
22              return m == null ? new byte[0] : m.getByteArray(0, (int) m.size());
23          }
24      }
25  }
