18  /**
19   * Provides key statistics for TCP and UDP network protocols, including aggregate counters and per-connection details.
20   * <p>
21   * Use {@link #getTCPv4Stats()} and {@link #getTCPv6Stats()} for aggregate TCP counters (connections established,
22   * active, passive, failures, resets, and segment counts). Use {@link #getUDPv4Stats()} and {@link #getUDPv6Stats()} for
23   * aggregate UDP counters (datagrams sent, received, and errors).
24   * <p>
25   * Use {@link #getConnections()} to list individual active TCP and UDP connections, each represented as an
26   * {@link IPConnection} with local/remote addresses, ports, state, and owning process ID.
27   * <p>
28   * Example: listing active TCP connections:
29   *
30   * <pre>{@code
31   * InternetProtocolStats ipStats = os.getInternetProtocolStats();
32   * for (IPConnection conn : ipStats.getConnections()) {
33   *     if (conn.getType().startsWith("tcp")) {
34   *         String local = addrToString(conn.getLocalAddress());
35   *         String foreign = addrToString(conn.getForeignAddress());
36   *         System.out.printf("%s:%d -> %s:%d (%s)%n", local, conn.getLocalPort(), foreign, conn.getForeignPort(),
37   *                 conn.getState());
38   *     }
39   * }
40   *
41   * // Helper to safely convert address bytes to a string
42   * static String addrToString(byte[] addr) {
43   *     try {
44   *         return addr.length > 0 ? InetAddress.getByAddress(addr).getHostAddress() : "*";
45   *     } catch (UnknownHostException e) {
46   *         return "?";
47   *     }
48   * }
49   * }</pre>
50   *
51   * <b>Platform notes:</b> On macOS, connection information requires elevated permissions. Without elevated permissions,
52   * TCP segment data is estimated.
53   *
54   * @see TcpStats
55   * @see UdpStats
56   * @see IPConnection
57   */
58  @PublicApi
59  @ThreadSafe
60  public interface InternetProtocolStats {
61  
62      /**
63       * Get the TCP stats for IPv4 connections.
64       * <p>
65       * On macOS connection information requires elevated permissions. Without elevated permissions, segment data is
66       * estimated.
67       *
68       * @return a {@link TcpStats} object encapsulating the stats.
69       */
70      TcpStats getTCPv4Stats();
71  
72      /**
73       * Get the TCP stats for IPv6 connections, if available. If not available separately, these may be 0 and included in
74       * IPv4 connections.
75       *
76       * @return a {@link TcpStats} object encapsulating the stats.
77       */
78      TcpStats getTCPv6Stats();
79  
80      /**
81       * Get the UDP stats for IPv4 datagrams.
82       *
83       * @return a {@link UdpStats} object encapsulating the stats.
84       */
85      UdpStats getUDPv4Stats();
86  
87      /**
88       * Get the UDP stats for IPv6 datagrams, if available. If not available separately, these may be 0 and included in
89       * IPv4 datagrams.
90       *
91       * @return a {@link UdpStats} object encapsulating the stats.
92       */
93      UdpStats getUDPv6Stats();
94  
95      /**
96       * Gets a list of TCP and UDP connections.
97       *
98       * @return A list of {@link IPConnection} objects for TCP and UDP connections.
99       */
100     List<IPConnection> getConnections();

102     /**
103      * Encapsulates statistics associated with a TCP connection.
104      */
105     @PublicApi
106     @Immutable
107     final class TcpStats {
108         private final long connectionsEstablished;
109         private final long connectionsActive;
110         private final long connectionsPassive;
111         private final long connectionFailures;
112         private final long connectionsReset;
113         private final long segmentsSent;
114         private final long segmentsReceived;
115         private final long segmentsRetransmitted;
116         private final long inErrors;
117         private final long outResets;
118 
119         /**
120          * Constructs a new TcpStats instance with the given counters.
121          *
122          * @param connectionsEstablished TCP connections currently established
123          * @param connectionsActive      active connection openings
124          * @param connectionsPassive     passive connection openings
125          * @param connectionFailures     failed connection attempts
126          * @param connectionsReset       connections reset
127          * @param segmentsSent           segments sent
128          * @param segmentsReceived       segments received
129          * @param segmentsRetransmitted  segments retransmitted
130          * @param inErrors               segments received in error
131          * @param outResets              resets sent
132          */
133         public TcpStats(long connectionsEstablished, long connectionsActive, long connectionsPassive,
134                 long connectionFailures, long connectionsReset, long segmentsSent, long segmentsReceived,
135                 long segmentsRetransmitted, long inErrors, long outResets) {
136             this.connectionsEstablished = connectionsEstablished;
137             this.connectionsActive = connectionsActive;
138             this.connectionsPassive = connectionsPassive;
139             this.connectionFailures = connectionFailures;
140             this.connectionsReset = connectionsReset;
141             this.segmentsSent = segmentsSent;
142             this.segmentsReceived = segmentsReceived;
143             this.segmentsRetransmitted = segmentsRetransmitted;
144             this.inErrors = inErrors;
145             this.outResets = outResets;
146         }
147 
148         /**
149          * Connections Established is the number of TCP connections for which the current state is either ESTABLISHED or
150          * CLOSE-WAIT
151          *
152          * @return the connectionsEstablished
153          */
154         public long getConnectionsEstablished() {
155             return connectionsEstablished;
156         }
157 
158         /**
159          * Connections Active is the number of times TCP connections have made a direct transition to the SYN-SENT state
160          * from the CLOSED state. In other words, it shows a number of connections which are initiated by the local
161          * computer. The value is a cumulative total.
162          *
163          * @return the connectionsActive
164          */
165         public long getConnectionsActive() {
166             return connectionsActive;
167         }
168 
169         /**
170          * Connections Passive is the number of times TCP connections have made a direct transition to the SYN-RCVD
171          * state from the LISTEN state. In other words, it shows a number of connections to the local computer, which
172          * are initiated by remote computers. The value is a cumulative total.
173          *
174          * @return the connectionsPassive
175          */
176         public long getConnectionsPassive() {
177             return connectionsPassive;
178         }
179 
180         /**
181          * Connection Failures is the number of times TCP connections have made a direct transition to the CLOSED state
182          * from the SYN-SENT state or the SYN-RCVD state, plus the number of times TCP connections have made a direct
183          * transition to the LISTEN state from the SYN-RCVD state.
184          *
185          * @return the connectionFailures
186          */
187         public long getConnectionFailures() {
188             return connectionFailures;
189         }
190 
191         /**
192          * Connections Reset is the number of times TCP connections have made a direct transition to the CLOSED state
193          * from either the ESTABLISHED state or the CLOSE-WAIT state.
194          *
195          * @return the connectionsReset
196          */
197         public long getConnectionsReset() {
198             return connectionsReset;
199         }
200 
201         /**
202          * Segments Sent is the number of segments sent, including those on current connections, but excluding those
203          * containing only retransmitted bytes.
204          *
205          * @return the segmentsSent
206          */
207         public long getSegmentsSent() {
208             return segmentsSent;
209         }
210 
211         /**
212          * Segments Received is the number of segments received, including those received in error. This count includes
213          * segments received on currently established connections.
214          *
215          * @return the segmentsReceived
216          */
217         public long getSegmentsReceived() {
218             return segmentsReceived;
219         }
220 
221         /**
222          * Segments Retransmitted is the number of segments retransmitted, that is, segments transmitted containing one
223          * or more previously transmitted bytes.
224          *
225          * @return the segmentsRetransmitted
226          */
227         public long getSegmentsRetransmitted() {
228             return segmentsRetransmitted;
229         }
230 
231         /**
232          * The number of errors received.
233          *
234          * @return the inErrors
235          */
236         public long getInErrors() {
237             return inErrors;
238         }
239 
240         /**
241          * The number of segments transmitted with the reset flag set.
242          *
243          * @return the outResets
244          */
245         public long getOutResets() {
246             return outResets;
247         }
248 
249         @Override
250         public String toString() {
251             return "TcpStats [connectionsEstablished=" + connectionsEstablished + ", connectionsActive="
252                     + connectionsActive + ", connectionsPassive=" + connectionsPassive + ", connectionFailures="
253                     + connectionFailures + ", connectionsReset=" + connectionsReset + ", segmentsSent=" + segmentsSent
254                     + ", segmentsReceived=" + segmentsReceived + ", segmentsRetransmitted=" + segmentsRetransmitted
255                     + ", inErrors=" + inErrors + ", outResets=" + outResets + "]";
256         }
257     }

259     /**
260      * Encapsulates statistics associated with a UDP connection.
261      */
262     @PublicApi
263     @Immutable
264     final class UdpStats {
265         private final long datagramsSent;
266         private final long datagramsReceived;
267         private final long datagramsNoPort;
268         private final long datagramsReceivedErrors;
269 
270         /**
271          * Constructs a new UdpStats instance with the given counters.
272          *
273          * @param datagramsSent           datagrams sent
274          * @param datagramsReceived       datagrams received
275          * @param datagramsNoPort         datagrams received with no application at destination port
276          * @param datagramsReceivedErrors datagrams that could not be delivered
277          */
278         public UdpStats(long datagramsSent, long datagramsReceived, long datagramsNoPort,
279                 long datagramsReceivedErrors) {
280             this.datagramsSent = datagramsSent;
281             this.datagramsReceived = datagramsReceived;
282             this.datagramsNoPort = datagramsNoPort;
283             this.datagramsReceivedErrors = datagramsReceivedErrors;
284         }
285 
286         /**
287          * Datagrams Sent is the number of UDP datagrams sent from the entity.
288          *
289          * @return the datagramsSent
290          */
291         public long getDatagramsSent() {
292             return datagramsSent;
293         }
294 
295         /**
296          * Datagrams Received is the number of UDP datagrams delivered to UDP users.
297          *
298          * @return the datagramsReceived
299          */
300         public long getDatagramsReceived() {
301             return datagramsReceived;
302         }
303 
304         /**
305          * Datagrams No Port is the number of received UDP datagrams for which there was no application at the
306          * destination port.
307          *
308          * @return the datagramsNoPort
309          */
310         public long getDatagramsNoPort() {
311             return datagramsNoPort;
312         }
313 
314         /**
315          * Datagrams Received Errors is the number of received UDP datagrams that could not be delivered for reasons
316          * other than the lack of an application at the destination port.
317          *
318          * @return the datagramsReceivedErrors
319          */
320         public long getDatagramsReceivedErrors() {
321             return datagramsReceivedErrors;
322         }
323 
324         @Override
325         public String toString() {
326             return "UdpStats [datagramsSent=" + datagramsSent + ", datagramsReceived=" + datagramsReceived
327                     + ", datagramsNoPort=" + datagramsNoPort + ", datagramsReceivedErrors=" + datagramsReceivedErrors
328                     + "]";
329         }

332     /**
333      * The TCP connection state as described in RFC 793.
334      */
335     @PublicApi
336     enum TcpState {
337         /** Unknown state. */
338         UNKNOWN,
339         /** Connection closed. */
340         CLOSED,
341         /** Waiting for a connection request. */
342         LISTEN,
343         /** Waiting for acknowledgment of a connection request. */
344         SYN_SENT,
345         /** Waiting for acknowledgment of a connection request after having received one. */
346         SYN_RECV,
347         /** Connection established. */
348         ESTABLISHED,
349         /** Waiting for a connection termination request from the remote TCP. */
350         FIN_WAIT_1,
351         /** Waiting for a connection termination request from the remote TCP after sending one. */
352         FIN_WAIT_2,
353         /** Waiting for a connection termination request from the local user. */
354         CLOSE_WAIT,
355         /** Waiting for acknowledgment of a connection termination request. */
356         CLOSING,
357         /** Waiting for acknowledgment of the connection termination request sent to the remote TCP. */
358         LAST_ACK,
359         /** Waiting for enough time to pass to be sure the remote TCP received acknowledgment. */
360         TIME_WAIT,
361         /** No state. */
362         NONE;
363 
364         /**
365          * Maps a BSD TCP state code to a {@link TcpState}. This is the ordering defined by the BSD {@code tcp_fsm.h}
366          * {@code TCPS_*} constants, as reported by the macOS {@code tcpsi_state} field.
367          *
368          * @param state the BSD TCP state code
369          * @return the corresponding {@link TcpState}, or {@link #UNKNOWN} if the code is not recognized
370          */
371         public static TcpState fromBsdState(int state) {
372             switch (state) {
373                 case 0:
374                     return CLOSED;
375                 case 1:
376                     return LISTEN;
377                 case 2:
378                     return SYN_SENT;
379                 case 3:
380                     return SYN_RECV;
381                 case 4:
382                     return ESTABLISHED;
383                 case 5:
384                     return CLOSE_WAIT;
385                 case 6:
386                     return FIN_WAIT_1;
387                 case 7:
388                     return CLOSING;
389                 case 8:
390                     return LAST_ACK;
391                 case 9:
392                     return FIN_WAIT_2;
393                 case 10:
394                     return TIME_WAIT;
395                 default:
396                     return UNKNOWN;
397             }
398         }
399 
400         /**
401          * Maps a Windows {@code MIB_TCP_STATE} code to a {@link TcpState}. This 1-based enumeration is defined by the
402          * Windows IP Helper API; the DELETE_TCB pseudo-state (12) is treated as {@link #CLOSED}.
403          *
404          * @param state the Windows {@code MIB_TCP_STATE} code
405          * @return the corresponding {@link TcpState}, or {@link #UNKNOWN} if the code is not recognized
406          */
407         public static TcpState fromWindowsMibState(int state) {
408             switch (state) {
409                 case 1:
410                 case 12:
411                     return CLOSED;
412                 case 2:
413                     return LISTEN;
414                 case 3:
415                     return SYN_SENT;
416                 case 4:
417                     return SYN_RECV;
418                 case 5:
419                     return ESTABLISHED;
420                 case 6:
421                     return FIN_WAIT_1;
422                 case 7:
423                     return FIN_WAIT_2;
424                 case 8:
425                     return CLOSE_WAIT;
426                 case 9:
427                     return CLOSING;
428                 case 10:
429                     return LAST_ACK;
430                 case 11:
431                     return TIME_WAIT;
432                 default:
433                     return UNKNOWN;
434             }
435         }
436 
437         /**
438          * Maps a Linux TCP state code to a {@link TcpState}. This is the ordering defined by the Linux kernel
439          * {@code TCP_*} constants, as reported in the state field of {@code /proc/net/tcp} and {@code /proc/net/tcp6}.
440          *
441          * @param state the Linux TCP state code
442          * @return the corresponding {@link TcpState}, or {@link #UNKNOWN} if the code is not recognized
443          */
444         public static TcpState fromLinuxState(int state) {
445             switch (state) {
446                 case 0x01:
447                     return ESTABLISHED;
448                 case 0x02:
449                     return SYN_SENT;
450                 case 0x03:
451                     return SYN_RECV;
452                 case 0x04:
453                     return FIN_WAIT_1;
454                 case 0x05:
455                     return FIN_WAIT_2;
456                 case 0x06:
457                     return TIME_WAIT;
458                 case 0x07:
459                     return CLOSED;
460                 case 0x08:
461                     return CLOSE_WAIT;
462                 case 0x09:
463                     return LAST_ACK;
464                 case 0x0A:
465                     return LISTEN;
466                 case 0x0B:
467                     return CLOSING;
468                 default:
469                     return UNKNOWN;
470             }
471         }
472     }

474     /**
475      * Encapsulates information associated with an IP connection.
476      */
477     @PublicApi
478     @Immutable
479     final class IPConnection {
480         private final String type;
481         private final byte[] localAddress;
482         private final int localPort;
483         private final byte[] foreignAddress;
484         private final int foreignPort;
485         private final @Nullable TcpState state;
486         private final int transmitQueue;
487         private final int receiveQueue;
488         private int owningProcessId;
489 
490         /**
491          * Constructs a new IPConnection instance.
492          *
493          * @param type            the protocol type (e.g., tcp4, tcp6, udp4, udp6)
494          * @param localAddress    the local address bytes
495          * @param localPort       the local port
496          * @param foreignAddress  the remote address bytes
497          * @param foreignPort     the remote port
498          * @param state           the TCP state, or {@code null} for a UDP connection or when not known
499          * @param transmitQueue   the transmit queue size
500          * @param receiveQueue    the receive queue size
501          * @param owningProcessId the PID of the owning process
502          */
503         public IPConnection(String type, byte[] localAddress, int localPort, byte[] foreignAddress, int foreignPort,
504                 @Nullable TcpState state, int transmitQueue, int receiveQueue, int owningProcessId) {
505             this.type = type;
506             this.localAddress = Arrays.copyOf(localAddress, localAddress.length);
507             this.localPort = localPort;
508             this.foreignAddress = Arrays.copyOf(foreignAddress, foreignAddress.length);
509             this.foreignPort = foreignPort;
510             this.state = state;
511             this.transmitQueue = transmitQueue;
512             this.receiveQueue = receiveQueue;
513             this.owningProcessId = owningProcessId;
514         }
515 
516         /**
517          * Returns the connection protocol type, e.g., tcp4, tcp6, tcp46, udp4, udp6, udp46
518          *
519          * @return The protocol type
520          */
521         public String getType() {
522             return type;
523         }
524 
525         /**
526          * Gets the local address. For IPv4 addresses this is a 4-byte array. For IPv6 addresses this is a 16-byte
527          * array.
528          * <p>
529          * On Unix operating systems, the 16-bit value may be truncated, giving only the high order bytes. IPv6
530          * addresses ending in zeroes should be considered suspect.
531          *
532          * @return The local address, or an empty array if the listener can accept a connection on any interface.
533          */
534         public byte[] getLocalAddress() {
535             return Arrays.copyOf(localAddress, localAddress.length);
536         }
537 
538         /**
539          * Gets the local port.
540          *
541          * @return The local port, or 0 if unknown, or any port.
542          */
543         public int getLocalPort() {
544             return localPort;
545         }
546 
547         /**
548          * Gets the foreign/remote address. For IPv4 addresses this is a 4-byte array. For IPv6 addresses this is a
549          * 16-byte array.
550          * <p>
551          * On Unix operating systems, this value may be truncated. IPv6 addresses ending in zeroes should be considered
552          * suspect.
553          *
554          * @return The foreign/remote address, or an empty array if unknown. An empty array will also result if
555          */
556         public byte[] getForeignAddress() {
557             return Arrays.copyOf(foreignAddress, foreignAddress.length);
558         }
559 
560         /**
561          * Gets the foreign/remote port.
562          *
563          * @return The foreign/remote port, or 0 if unknown.
564          */
565         public int getForeignPort() {
566             return foreignPort;
567         }
568 
569         /**
570          * Gets the connection state (TCP connections only).
571          *
572          * @return The connection state if known or relevant, null otherwise.
573          */
574         public @Nullable TcpState getState() {
575             return state;
576         }
577 
578         /**
579          * Gets the size of the transmit queue. Not available on Windows.
580          *
581          * @return The size of the transmit queue, or 0 if unknown.
582          */
583         public int getTransmitQueue() {
584             return transmitQueue;
585         }
586 
587         /**
588          * Gets the size of the receive queue. Not available on Windows.
589          *
590          * @return The size of the receive queue, or 0 if unknown.
591          */
592         public int getReceiveQueue() {
593             return receiveQueue;
594         }
595 
596         /**
597          * Gets the id of the process which holds this connection.
598          *
599          * @return The process id of the process which holds this connection if known, -1 otherwise.
600          */
601         public int getowningProcessId() {
602             return owningProcessId;
603         }
604 
605         @Override
606         public String toString() {
607             String localIp = "*";
608             try {
609                 localIp = InetAddress.getByAddress(localAddress).toString();
610             } catch (UnknownHostException e) { // NOSONAR java:S108
611             }
612             String foreignIp = "*";
613             try {
614                 foreignIp = InetAddress.getByAddress(foreignAddress).toString();
615             } catch (UnknownHostException e) { // NOSONAR java:S108
616             }
617             return "IPConnection [type=" + type + ", localAddress=" + localIp + ", localPort=" + localPort
618                     + ", foreignAddress=" + foreignIp + ", foreignPort=" + foreignPort + ", state=" + state
619                     + ", transmitQueue=" + transmitQueue + ", receiveQueue=" + receiveQueue + ", owningProcessId="
620                     + owningProcessId + "]";
621         }
622     }
