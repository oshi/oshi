62      /**
63       * Ordered {@code ps} columns queried for each process. Shared by NetBsdOSProcess and the NetBSD OperatingSystem so
64       * the column list and parsing stay in lockstep. {@code ARGS} must remain last.
65       */
66      public static final List<BsdPsKeyword> PS_KEYWORDS = Collections.unmodifiableList(Arrays.asList(STATE, PID, PPID,

69      public static final String PS_COMMAND_ARGS = PS_KEYWORDS.stream().map(Enum::name)
70              .map(name -> name.toLowerCase(Locale.ROOT)).collect(Collectors.joining(","));

113     @Override
114     protected String queryPath(Map<BsdPsKeyword, String> psMap) {
115         String path = queryExecutablePath();
116         if (!path.isEmpty()) {
117             return path;
118         }
119         // Without the kernel's answer, the path is argv[0]. NetBSD's ps prints argv[0] in the comm column, the same
120         // value the args column begins with, but clips it to the width of its "COMMAND" header - seven characters -
121         // because only the last column is left unbounded. So argv[0] is taken from the args column, which is queried
122         // last and is therefore printed in full.
123         String argv = parseArgv0(psMap.get(BsdPsKeyword.ARGS));
124         // The placeholder says the arguments were unavailable, so no path is known
125         return isPlaceholder(argv) ? "" : argv;
126     }

149     /**
150      * Extracts the command line from a {@code ps} args column.
151      *
152      * @param args the args column, the process's arguments separated by spaces
153      * @return the arguments, or an empty string where {@code ps} substituted the parenthesized command because the
154      *         kernel would not release them, which is a placeholder rather than a command line
155      */
156     static String parseCommandLine(@Nullable String args) {
157         String argv = ParseUtil.getStringValueOrEmpty(args);
158         return isPlaceholder(parseArgv0(argv)) ? "" : argv;
159     }

161     @Override
162     protected String queryName(Map<BsdPsKeyword, String> psMap, String path) {
163         // From argv[0] rather than the path, so that it matches the name each of the process's threads is given
164         return parseName(psMap.get(BsdPsKeyword.ARGS));
165     }
166 
167     /**
168      * Extracts the process name from a {@code ps} args column: the last path component of {@code argv[0]}.
169      * <p>
170      * The placeholder {@code ps} substitutes when the kernel refuses the arguments wraps the kernel's own name for the
171      * process, which is the name whether or not the arguments could be read, so it is unwrapped rather than reporting
172      * {@code ps}'s punctuation as the name. Either way two readings of one process give the same name.
173      *
174      * @param args the args column, the process's arguments separated by spaces
175      * @return the process name
176      */
177     static String parseName(@Nullable String args) {
178         String argv = parseArgv0(args);
179         return isPlaceholder(argv) ? argv.substring(1, argv.length() - 1) : argv.substring(argv.lastIndexOf('/') + 1);
180     }
181 
182     /**
183      * Extracts {@code argv[0]} from a {@code ps} args column.
184      *
185      * @param args the args column, the process's arguments separated by spaces
186      * @return {@code argv[0]}, which is a path when the process was started by one, or the command in parentheses where
187      *         the kernel would not release the arguments
188      */
189     static String parseArgv0(@Nullable String args) {
190         String argv = ParseUtil.getStringValueOrEmpty(args).trim();
191         int space = argv.indexOf(' ');
192         return space < 0 ? argv : argv.substring(0, space);
193     }
194 
195     /**
196      * Whether a {@code ps} args column holds the parenthesized command that {@code ps} substitutes when the kernel
197      * refuses a process's arguments, which it does while that process is inside {@code posix_spawn}.
198      *
199      * @param argv the value from {@link #parseArgv0(String)}
200      * @return {@code true} if it is that placeholder rather than a real {@code argv[0]}
201      */
202     static boolean isPlaceholder(String argv) {
203         return argv.length() > 2 && argv.charAt(0) == '(' && argv.charAt(argv.length() - 1) == ')';
204     }

206     @Override
207     protected List<String> queryArguments() {
208         // NetBSD provides command line via /proc filesystem
209         String cmdline = "/proc/" + getProcessID() + "/cmdline";
210         // The kernel refuses to read a process's arguments while that process is inside posix_spawn, which a JVM is
211         // whenever it runs a command, so an empty read of a file that exists is retried. This process is the case OSHI
212         // provokes itself, by running ps while another thread spawns, and so is given the longer budget; for any other
213         // process an empty read is usually genuine. The result is memoized for the life of this object, so a single
214         // refusal would otherwise stick.
215         int attempts = getProcessID() == this.os.getProcessId() ? ARGS_READ_ATTEMPTS_SELF : ARGS_READ_ATTEMPTS_OTHER;
216         for (int attempt = 1;; attempt++) {
217             byte[] cmdBytes = FileUtil.readAllBytes(cmdline, false);
218             if (cmdBytes.length > 0) {
219                 return Collections.unmodifiableList(ParseUtil.parseByteArrayToStrings(cmdBytes));
220             }
221             if (attempt >= attempts || !new File(cmdline).exists()) {
222                 return Collections.emptyList();
223             }
224             Util.sleep(ARGS_READ_RETRY_MILLIS);
225         }
226     }

228     @Override
229     protected Map<String, String> queryEnvironmentVariables() {
230         // For the current process, use Java's System.getenv()
231         if (getProcessID() == this.os.getProcessId()) {
232             return System.getenv();
233         }
234         return parseEnvironment(queryEnvironmentBytes());
235     }

237     /**
238      * Queries the environment of this process from the kernel, the {@code kern.proc_args} sysctl with
239      * {@code KERN_PROC_ENV}.
240      *
241      * @return The NUL-delimited {@code KEY=VALUE} entries, or an empty array if this build cannot make the call; NetBSD
242      *         offers no command that reads another process's environment
243      */
244     protected byte[] queryEnvironmentBytes() {
245         return new byte[0];
246     }
247 
248     /**
249      * Parses the environment returned by the {@code kern.proc_args} sysctl.
250      *
251      * @param envBytes NUL-delimited {@code KEY=VALUE} entries
252      * @return the environment, empty if there were no entries
253      */
254     static Map<String, String> parseEnvironment(byte[] envBytes) {
255         // Parse by raw bytes rather than by String offsets, which would misalign on a multibyte charset
256         Map<String, String> env = new LinkedHashMap<>();
257         for (String envStr : ParseUtil.parseByteArrayToStrings(envBytes)) {
258             int idx = envStr.indexOf('=');
259             if (idx > 0) {
260                 env.put(envStr.substring(0, idx), envStr.substring(idx + 1));
261             }
262         }
263         return env.isEmpty() ? Collections.emptyMap() : Collections.unmodifiableMap(env);
264     }

287     private long queryOwnOpenFileLimit(boolean soft) {
288         if (getProcessID() == this.os.getProcessId()) {
289             // getrlimit where a backend can call it; otherwise the sysctl a child process inherits, then the system max
290             long limit = queryRlimitNofile(soft);
291             if (limit >= 0) {
292                 return limit;
293             }
294             limit = BsdSysctlUtil.sysctl("proc.curproc.rlimit.descriptors." + (soft ? "soft" : "hard"), 0L);
295             if (limit <= 0) {
296                 limit = BsdSysctlUtil.sysctl("kern.maxfiles", 0L);
297             }
298             return limit > 0 ? limit : -1L;
299         }
300         return -1L;
301     }

303     @Override
304     protected int queryBitness() {
305         // NetBSD does not maintain a compatibility layer.
306         // Process bitness is OS bitness
307         return System.getProperty("os.arch", "").contains("64") ? 64 : 32;
308     }

310     @Override
311     public long getAffinityMask() {
312         long bitMask = 0L;
313         // schedctl -p <pid> shows affinity; -A cpus would set it
314         List<String> schedctl = ExecutingCommand.runNative("schedctl -p " + getProcessID());
315         for (String line : schedctl) {
316             // Output includes "Affinity: <list>" when bound
317             if (line.contains("Affinity:")) {
318                 String[] parts = COMMA_OR_WHITESPACE.split(AFFINITY.split(line, -1)[1].trim(), -1);
319                 for (String part : parts) {
320                     int bitToSet = ParseUtil.parseIntOrDefault(part.trim(), -1);
321                     if (bitToSet >= 0) {
322                         bitMask |= 1L << bitToSet;
323                     }
324                 }
325                 return bitMask;
326             }
327         }
328         // Not bound — return all CPUs
329         int ncpu = BsdSysctlUtil.sysctl("hw.ncpuonline", 1);
330         return ncpu < 64 ? (1L << ncpu) - 1 : -1L;
331     }

333     @Override
334     public List<OSThread> getThreadDetails() {
335         String psCommand = NetBsdOSThread.PS_THREAD_COMMAND;
336         if (getProcessID() >= 0) {
337             psCommand += " -p " + getProcessID();
338         }
339         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsArgs = threadMap -> threadMap
340                 .containsKey(BsdPsThreadKeyword.ARGS);
341         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
342                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class, NetBsdOSThread.PS_THREAD_KEYWORDS,
343                         thread.trim(), ' '))
344                 .filter(hasColumnsArgs).map(threadMap -> new NetBsdOSThread(getProcessID(), threadMap))
345                 .filter(VALID_THREAD).collect(Collectors.toList());
346     }

348     @Override
349     protected void updateThreadCount() {
350         // Use nlwp keyword to get LWP count for this process
351         List<String> nlwpList = ExecutingCommand.runNative("ps -o nlwp -p " + getProcessID());
352         if (nlwpList.size() > 1) {
353             this.threadCount = ParseUtil.parseIntOrDefault(nlwpList.get(1).trim(), 1);
354         } else {
355             this.threadCount = 1;
356         }
357     }
