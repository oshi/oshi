7   import java.io.File;
8   import java.util.ArrayList;
9   import java.util.LinkedHashMap;
10  import java.util.List;
11  import java.util.Map;
12  import java.util.function.ToLongFunction;
13  import java.util.regex.Matcher;
14  import java.util.regex.Pattern;
15  
16  import oshi.annotation.concurrent.ThreadSafe;
17  import oshi.software.os.ApplicationInfo;
18  import oshi.util.ExecutingCommand;
19  import oshi.util.ParseUtil;
20  
21  /**
22   * Queries the packages installed on OpenBSD and NetBSD from {@code pkg_info}, which both systems' package tools
23   * provide.
24   */
25  @ThreadSafe
26  public final class PkgInfoInstalledApps {
27  
28      /** OpenBSD's package database: one directory per installed package, named for it. */
29      public static final String OPENBSD_PKG_DB = "/var/db/pkg";
30      /** NetBSD's package database since NetBSD 9, the {@code PKG_DBDIR} pkgsrc uses by default. */
31      public static final String NETBSD_PKG_DB = "/usr/pkg/pkgdb";
32  
33      /** A {@code pkg_info} line: the full package name, then its comment after one or more spaces. */
34      private static final Pattern PKG_INFO_LINE = Pattern.compile("(\\S++)\\s++(.*+)");
35      /**
36       * OpenBSD's rule from packages-specs(7): the version starts at the first dash followed by a digit, and may carry a
37       * flavor after it, as in {@code vim-9.1.1-no_x11}.
38       */
39      private static final Pattern OPENBSD_NAME = Pattern.compile("([^-]++(?:-(?!\\d)[^-]*+)*+)-(\\d.*+)");
40  
41      private PkgInfoInstalledApps() {
42      }
43  
44      /**
45       * Queries {@code pkg_info} for the installed packages. Neither the command nor the package database needs
46       * privilege.
47       *
48       * @param pkgDb   the package database directory, {@link #OPENBSD_PKG_DB} or {@link #NETBSD_PKG_DB}
49       * @param openBsd whether to split names by OpenBSD's rule, rather than NetBSD's
50       * @return the installed applications
51       */
52      public static List<ApplicationInfo> queryInstalledApps(String pkgDb, boolean openBsd) {
53          return parsePkgInfo(ExecutingCommand.runNative("pkg_info"), openBsd,
54                  pkg -> new File(pkgDb, pkg).lastModified());
55      }
56  
57      /**
58       * Parses the output of {@code pkg_info}. Package-private for testing.
59       * <p>
60       * Each line is a full package name, {@code name-version}, and its one-line comment. NetBSD's pkgsrc never puts a
61       * dash in a version, so the name ends at the last dash; OpenBSD's versions may carry a flavor after a further dash,
62       * so the name ends at the first dash followed by a digit. Neither tool records an install time, but each installs a
63       * package by creating its directory in the package database, so the directory's modification time is the timestamp.
64       *
65       * @param lines       the lines emitted by {@code pkg_info}
66       * @param openBsd     whether to split names by OpenBSD's rule, rather than NetBSD's
67       * @param installTime the modification time in milliseconds of a package's database directory, or 0 if it has none
68       * @return the installed applications
69       */
70      static List<ApplicationInfo> parsePkgInfo(List<String> lines, boolean openBsd, ToLongFunction<String> installTime) {
71          List<ApplicationInfo> apps = new ArrayList<>();
72          for (String line : lines) {
73              Matcher m = PKG_INFO_LINE.matcher(line.trim());
74              if (!m.matches()) {
75                  continue;
76              }
77              String pkg = m.group(1);
78              String name = pkg;
79              String version = "";
80              if (openBsd) {
81                  Matcher n = OPENBSD_NAME.matcher(pkg);
82                  if (n.matches()) {
83                      name = n.group(1);
84                      version = n.group(2);
85                  }
86              } else {
87                  int dash = pkg.lastIndexOf('-');
88                  if (dash > 0) {
89                      name = pkg.substring(0, dash);
90                      version = pkg.substring(dash + 1);
91                  }
92              }
93              Map<String, String> additionalInfo = new LinkedHashMap<>();
94              additionalInfo.put("description", ParseUtil.getStringValueOrUnknown(m.group(2).trim()));
95              apps.add(new ApplicationInfo(name, version, "", installTime.applyAsLong(pkg), additionalInfo));
96          }
97          return apps;
98      }
99  }
