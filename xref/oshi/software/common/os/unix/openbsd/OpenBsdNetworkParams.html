5   package oshi.software.common.os.unix.openbsd;
6   
7   import java.util.List;
8   
9   import org.jspecify.annotations.Nullable;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.software.common.AbstractNetworkParams;
13  import oshi.software.os.NetworkParams;
14  import oshi.util.ExecutingCommand;
15  import oshi.util.driver.unix.NetstatRoute;
16  
17  /**
18   * Abstract base for the OpenBSD NetworkParams. Holds the command-line gateway lookup and the host name fallback; the
19   * JNA and FFM subclasses supply the {@code gethostname} binding.
20   */
21  @ThreadSafe
22  public abstract class OpenBsdNetworkParams extends AbstractNetworkParams {
23  
24      /**
25       * OpenBSD prints the interface in the eighth column: Destination Gateway Flags Refs Use Mtu Prio Iface. The header
26       * scan corrects this if the layout differs on another release.
27       */
28      private static final int IF_NAME_INDEX = 7;
29  
30      @Override
31      public String getHostName() {
32          String hn = queryHostName();
33          return hn != null ? hn : super.getHostName();
34      }
35  
36      @Override
37      public String getIpv4DefaultGateway() {
38          return searchGateway(ExecutingCommand.runNative("route -n get default"));
39      }
40  
41      @Override
42      public String getIpv6DefaultGateway() {
43          return searchGateway(ExecutingCommand.runNative("route -n get -inet6 default"));
44      }
45  
46      @Override
47      public List<NetworkParams.IPRoute> getRoutes() {
48          return NetstatRoute.queryRoutes("netstat -rn -f inet", "netstat -rn -f inet6", IF_NAME_INDEX,
49                  queryInterfaceIndexByName());
50      }
51  
52      /**
53       * Reads the host name from libc, avoiding the name resolution that the {@link AbstractNetworkParams} fallback
54       * performs.
55       *
56       * @return the native host name, or {@code null} to fall back to the InetAddress lookup
57       */
58      protected abstract @Nullable String queryHostName();
59  }
