5   package oshi.software.common.os.unix.openbsd;
6   
7   import oshi.annotation.concurrent.ThreadSafe;
8   import oshi.software.common.AbstractNetworkParams;
9   import oshi.util.ExecutingCommand;
10  
11  /**
12   * Abstract base for the OpenBSD NetworkParams. Holds the command-line gateway lookup and the host name fallback; the
13   * JNA and FFM subclasses supply the {@code gethostname} binding.
14   */
15  @ThreadSafe
16  public abstract class OpenBsdNetworkParams extends AbstractNetworkParams {
17  
18      @Override
19      public String getHostName() {
20          String hn = queryHostName();
21          return hn != null ? hn : super.getHostName();
22      }
23  
24      @Override
25      public String getIpv4DefaultGateway() {
26          return searchGateway(ExecutingCommand.runNative("route -n get default"));
27      }
28  
29      @Override
30      public String getIpv6DefaultGateway() {
31          return searchGateway(ExecutingCommand.runNative("route -n get default"));
32      }
33  
34      /**
35       * Reads the host name from libc, avoiding the name resolution that the {@link AbstractNetworkParams} fallback
36       * performs.
37       *
38       * @return the native host name, or {@code null} to fall back to the InetAddress lookup
39       */
40      protected abstract String queryHostName();
41  }
