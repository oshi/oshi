44  /**
45   * Abstract base for a DragonFly BSD OSProcess. Inherits the shared BSD {@code ps} parsing and adds the fields whose
46   * column set differs on DragonFly.
47   */
48  public abstract class DragonFlyBsdOSProcess extends BsdOSProcess {
49  
50      /**
51       * Ordered {@code ps} columns queried for each process. Shared by the DragonFlyBsdOSProcess subclasses (JNA/FFM) and
52       * the DragonFly OperatingSystem classes so the column list and parsing stay in lockstep. {@code COMMAND} must
53       * remain last.
54       */
55      public static final List<BsdPsKeyword> PS_KEYWORDS = Collections.unmodifiableList(Arrays.asList(STATE, PID, PPID,
56              USER, UID, RGID, NLWP, PRI, VSZ, RSS, TIME, MAJFLT, MINFLT, NVCSW, NIVCSW, UCOMM, COMMAND));
57  
58      /**
59       * The {@code ps -o} column list for {@link #PS_KEYWORDS}. DragonFly's {@code ps} prints a size wider than its
60       * column scaled, as {@code 5.65G}, and a column is as wide as its header, so {@code vsz} and {@code rss} get
61       * headers wide enough for any count of KiB.
62       */
63      public static final String PS_COMMAND_ARGS = PS_KEYWORDS.stream().map(DragonFlyBsdOSProcess::psColumn)
64              .collect(Collectors.joining(","));
65  
66      private static String psColumn(BsdPsKeyword keyword) {
67          String column = keyword.name().toLowerCase(Locale.ROOT);
68          return keyword == VSZ || keyword == RSS ? column + "=" + keyword.name() + "_IN_KIB_UNSCALED" : column;
69      }

71      /**
72       * The {@code KERN_PROC} sysctl subtype that returns a process's executable path (12 on FreeBSD).
73       */
74      protected static final int KERN_PROC_PATHNAME = 9;

76      /**
77       * The size of the buffer for an executable path, the platform's {@code PATH_MAX}.

79      protected static final int PATH_MAX = 1024;

81      /**
82       * The longest command name the kernel keeps, {@code MAXCOMLEN}; {@code ucomm} is cut to this length.
83       */
84      private static final int MAXCOMLEN = 16;
85  
86      /**
87       * The longest {@code ps} command column that {@code MAXCOMLEN} bytes of arguments can print as: every byte escaped
88       * as {@code \ooo}, followed by a {@code " (ucomm)"} suffix.
89       */
90      private static final int MAX_PRINTED_COMLEN = 4 * MAXCOMLEN + MAXCOMLEN + 3;

132     /**
133      * {@inheritDoc}
134      * <p>
135      * On DragonFly BSD only the current process is asked; every other process falls back to the {@code ps} column. The
136      * kernel's {@code KERN_PROC_PATHNAME} handler dereferences the target's file-descriptor table without checking it,
137      * and an exiting process frees that table before it drops its executable's name, so querying another process at the
138      * wrong moment panics the kernel.
139      */
140     @Override
141     protected final String queryExecutablePath() {
142         return getProcessID() == queryOwnProcessId() ? queryOwnExecutablePath() : "";
143     }
144 
145     /**
146      * Returns the full path of the current process's executable through the {@code KERN_PROC_PATHNAME} sysctl. Only
147      * called when this process is the current one; see {@link #queryExecutablePath()}.
148      *
149      * @return the path, or an empty string if it is unavailable
150      */
151     protected abstract String queryOwnExecutablePath();
152 
153     /**
154      * {@inheritDoc}
155      * <p>
156      * Every process but the current one falls back to {@code argv[0]} from {@code /proc/<pid>/cmdline}, which holds the
157      * path when the process was started by one and is not cut to {@code MAXCOMLEN} as {@code ucomm} is. A process can
158      * rewrite {@code argv[0]}, so it is used only when it names the same program as {@code ucomm}; otherwise, or
159      * without procfs, the result is {@code ucomm}.
160      */
161     @Override
162     protected String queryPath(Map<BsdPsKeyword, String> psMap) {
163         String path = queryExecutablePath();
164         if (!path.isEmpty()) {
165             return path;
166         }
167         String command = ParseUtil.getStringValueOrEmpty(psMap.get(UCOMM));
168         // The NUL-separated arguments keep a path with spaces whole, which ps's space-separated column does not
169         List<String> args = queryArguments();
170         String argv0 = args.isEmpty() ? "" : args.get(0);
171         String argv0Name = argv0.substring(argv0.lastIndexOf('/') + 1);
172         return !command.isEmpty() && namesProgram(argv0Name, command) ? argv0 : command;
173     }
174 
175     /**
176      * Whether the last element of {@code argv[0]} names the program whose kernel command name is {@code ucomm}.
177      *
178      * @param argv0Name the last element of {@code argv[0]}
179      * @param ucomm     the kernel's command name, cut to {@code MAXCOMLEN}
180      * @return {@code true} if it is the same name, or extends a truncated {@code ucomm} with more of a file name
181      */
182     static boolean namesProgram(String argv0Name, String ucomm) {
183         if (argv0Name.equals(ucomm)) {
184             return true;
185         }
186         // A setproctitle() title, "name: status", also extends ucomm, but a file name rarely holds a colon or a space
187         return ucomm.length() == MAXCOMLEN && argv0Name.startsWith(ucomm)
188                 && argv0Name.substring(MAXCOMLEN).chars().noneMatch(c -> c == ':' || Character.isWhitespace(c));
189     }
190 
191     /**
192      * {@inheritDoc}
193      * <p>
194      * DragonFly's {@code /proc/<pid>/cmdline} holds the kernel's argument cache, which is dropped for a command line
195      * longer than {@code kern.ps_arg_cache_limit}. The file then holds another process's {@code MAXCOMLEN} command name
196      * buffer, and the current process loses its arguments past the limit. {@code ps} reads the arguments from the
197      * process's memory instead, so where the two disagree its command column is used, split on whitespace. That split
198      * cannot keep an argument containing a space whole, and keeps the escapes {@code ps} prints for unprintable
199      * characters.
200      */
201     @Override
202     protected List<String> queryArguments() {
203         byte[] cmdline = FileUtil.readAllBytes("/proc/" + getProcessID() + "/cmdline", false);
204         return Collections.unmodifiableList(reconcileArguments(cmdline, this.commandLineBackup, this.ucomm));
205     }

209         this.ucomm = ParseUtil.getStringValueOrEmpty(psMap.get(UCOMM));
210         return super.queryCommandLineBackup(psMap);
211     }

213     /**
214      * Chooses between the arguments in {@code /proc/<pid>/cmdline} and the {@code ps} command column.
215      *
216      * @param cmdline   the bytes of {@code /proc/<pid>/cmdline}
217      * @param psCommand the {@code ps} command column, or an empty string if unknown
218      * @param ucomm     the {@code ps} command name, or an empty string if unknown
219      * @return the arguments: the file's when it has none or they agree with {@code ps}, since only the file keeps an
220      *         argument that contains a space whole; otherwise those {@code ps} printed, or none if it could read none
221      *         either
222      */
223     static List<String> reconcileArguments(byte[] cmdline, String psCommand, String ucomm) {
224         List<String> args = ParseUtil.parseByteArrayToStrings(cmdline);
225         String joined = String.join(" ", args);
226         // Without the file, a process title cannot be told from arguments; the command line falls back to ps's
227         if (args.isEmpty() || psCommand.isEmpty() || psCommand.equals(joined)) {
228             return args;
229         }
230         // ps prints "(ucomm)" for a process whose arguments it cannot read, so the file holds the name buffer too
231         if (psCommand.startsWith("(") && psCommand.endsWith(")")) {
232             return cmdline.length == MAXCOMLEN ? Collections.emptyList() : args;
233         }
234         // The kernel drops its argument cache only past kern.ps_arg_cache_limit (256 by default), so the file holds the
235         // name buffer only when ps prints more than any MAXCOMLEN bytes of arguments could. That buffer holds ucomm,
236         // not argv[0], so argv[0] then comes from ps
237         boolean nameBuffer = cmdline.length == MAXCOMLEN && psCommand.length() > MAX_PRINTED_COMLEN;
238         String argv0 = nameBuffer ? ParseUtil.whitespaces.split(psCommand.trim(), 2)[0] : args.get(0);
239         // ps follows the arguments with " (ucomm)" when the last element of argv[0] does not name the command in its
240         // first MAXCOMLEN characters, as with a setproctitle() title; otherwise a trailing "(ucomm)" is an argument
241         String suffix = " (" + ucomm + ")";
242         boolean decorated = !ucomm.isEmpty() && psCommand.endsWith(suffix)
243                 && !firstComlen(argv0.substring(argv0.lastIndexOf('/') + 1)).equals(firstComlen(ucomm));
244         String printed = decorated ? psCommand.substring(0, psCommand.length() - suffix.length()) : psCommand;
245         if (printed.equals(joined)) {
246             return args;
247         }
248         // The current process's file stops at an argument boundary; ps escaping a character only looks like a prefix
249         boolean truncated = printed.startsWith(joined + " ");
250         if (nameBuffer || truncated) {
251             return Arrays.asList(ParseUtil.whitespaces.split(printed.trim(), -1));
252         }
253         return args;
254     }
255 
256     private static String firstComlen(String name) {
257         return name.length() > MAXCOMLEN ? name.substring(0, MAXCOMLEN) : name;
258     }

260     @Override
261     protected int queryBitness() {
262         // DragonFly BSD has no KERN_PROC_SV_NAME (its KERN_PROC subtype 9 is the executable's path) and no ABI
263         // compatibility layer: an x86_64 kernel runs only 64-bit programs (kern.elf64 exists, kern.elf32 does not), and
264         // the i386 kernel of 3.8 and earlier only 32-bit ones. So every process has the operating system's bitness.
265         return this.os.getBitness();
266     }

278     /**
279      * Parses the kernel and user time from DragonFly's {@code /proc/<pid>/status}, which ps does not separate.
280      * Package-private for testing.
281      * <p>
282      * The ninth and tenth fields are the user and system time as {@code seconds,microseconds}. The command name, the
283      * first field, has its spaces escaped, so whitespace separates every field.
284      *
285      * @param status the lines of {@code /proc/<pid>/status}, empty if the process has exited
286      * @return the kernel and user time in milliseconds, in that order, or an empty array if they could not be read
287      */
288     static long[] parseKernelUserTime(List<String> status) {
289         if (status.isEmpty()) {
290             return new long[0];
291         }
292         String[] split = ParseUtil.whitespaces.split(status.get(0).trim(), -1);
293         if (split.length < 10) {
294             return new long[0];
295         }
296         long user = secUsecToMillis(split[8]);
297         long system = secUsecToMillis(split[9]);
298         if (user < 0L || system < 0L) {
299             return new long[0];
300         }
301         return new long[] { system, user };
302     }
303 
304     private static long secUsecToMillis(String secUsec) {
305         String[] parts = secUsec.split(",", -1);
306         if (parts.length != 2) {
307             return -1L;
308         }
309         long seconds = ParseUtil.parseLongOrDefault(parts[0], -1L);
310         long micros = ParseUtil.parseLongOrDefault(parts[1], -1L);
311         return seconds < 0L || micros < 0L ? -1L : seconds * 1000L + micros / 1000L;
312     }

314     private static long queryStartTimeFromProc(int pid) {
315         List<String> status = FileUtil.readFile("/proc/" + pid + "/status", false);
316         if (!status.isEmpty()) {
317             // Format: name pid ... startSec,startUsec ...
318             String[] split = ParseUtil.whitespaces.split(status.get(0).trim(), -1);
319             if (split.length >= 8) {
320                 String[] timeParts = split[7].split(",", -1);
321                 long seconds = ParseUtil.parseLongOrDefault(timeParts[0], 0L);
322                 if (seconds > 0) {
323                     return seconds * 1000L;
324                 }
325             }
326         }
327         // Unknown start time: return the sentinel so updateAttributes() falls back to elapsed time
328         return -1L;
329     }

331     @Override
332     public List<OSThread> getThreadDetails() {
333         String psCommand = "ps -wwxo " + DragonFlyBsdOSThread.PS_THREAD_COLUMNS + " -H";
334         if (getProcessID() >= 0) {
335             psCommand += " -p " + getProcessID();
336         }
337         Predicate<Map<BsdPsThreadKeyword, String>> hasColumnsPri = threadMap -> threadMap
338                 .containsKey(BsdPsThreadKeyword.PRI);
339         return ExecutingCommand.runNative(psCommand).stream().skip(1).parallel()
340                 .map(thread -> ParseUtil.stringToEnumMap(BsdPsThreadKeyword.class,
341                         DragonFlyBsdOSThread.PS_THREAD_KEYWORDS, thread.trim(), ' '))
342                 .filter(hasColumnsPri).map(threadMap -> new DragonFlyBsdOSThread(getProcessID(), threadMap))
343                 .filter(VALID_THREAD).collect(Collectors.toList());
344     }
