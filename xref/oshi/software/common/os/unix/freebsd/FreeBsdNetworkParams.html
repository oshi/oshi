5   package oshi.software.common.os.unix.freebsd;
6   
7   import java.util.List;
8   
9   import org.jspecify.annotations.Nullable;
10  
11  import oshi.software.common.AbstractNetworkParams;
12  import oshi.software.os.NetworkParams;
13  import oshi.util.ExecutingCommand;
14  import oshi.util.ParseUtil;
15  import oshi.util.driver.unix.NetstatRoute;
16  import oshi.util.driver.unix.RouteTableDump;
17  
18  /**
19   * Abstract base for the FreeBSD NetworkParams. Resolves the host and domain names and the default gateways from command
20   * output; the JNA and FFM subclasses supply the resolver calls.
21   * <p>
22   * DragonFly BSD uses this class too. Its {@code netstat} prints Refs and Use columns FreeBSD's does not, which the
23   * routing table parser absorbs by reading the interface column from the header rather than a fixed index.
24   */
25  public abstract class FreeBsdNetworkParams extends AbstractNetworkParams {
26  
27      /** FreeBSD prints the interface in the fourth column: Destination Gateway Flags Netif Expire. */
28      private static final int IF_NAME_INDEX = 3;
29  
30      @Override
31      public String getDomainName() {
32          return ParseUtil.getStringValueOrEmpty(queryDomainName());
33      }
34  
35      @Override
36      public String getHostName() {
37          String hn = queryHostName();
38          return hn != null ? hn : super.getHostName();
39      }
40  
41      @Override
42      public String getIpv4DefaultGateway() {
43          return searchGateway(ExecutingCommand.runNative("route -4 get default"));
44      }
45  
46      @Override
47      public String getIpv6DefaultGateway() {
48          return searchGateway(ExecutingCommand.runNative("route -6 get default"));
49      }
50  
51      @Override
52      public List<NetworkParams.IPRoute> getRoutes() {
53          byte[] dump = queryRouteDump();
54          if (dump.length > 0) {
55              List<IPRoute> routes = RouteTableDump.parse(dump, routeLayout(), queryInterfaceNameByIndex());
56              if (!routes.isEmpty()) {
57                  return routes;
58              }
59          }
60          return NetstatRoute.queryRoutes("netstat -rn -f inet", "netstat -rn -f inet6", IF_NAME_INDEX,
61                  queryInterfaceIndexByName());
62      }
63  
64      /**
65       * Resolves the domain name via the subclass's getaddrinfo binding.
66       *
67       * @return the resolved canonical domain name, or {@code null} if the native call failed. {@link #getDomainName()}
68       *         converts that to the empty string its own contract promises.
69       */
70      protected abstract @Nullable String queryDomainName();
71  
72      /**
73       * Returns the hostname via the subclass's gethostname binding, or {@code null} to fall back to the InetAddress
74       * lookup inherited from {@link AbstractNetworkParams#getHostName()}.
75       *
76       * @return the native hostname, or {@code null} to fall back to the InetAddress lookup
77       */
78      protected abstract @Nullable String queryHostName();
79  
80      /**
81       * Fetches the kernel's routing table dump.
82       *
83       * @return The bytes of a {@code NET_RT_DUMP} sysctl, or an empty array if this build cannot make the call, in which
84       *         case the routing table is read by running a command instead
85       */
86      protected byte[] queryRouteDump() {
87          return new byte[0];
88      }
89  
90      /**
91       * The routing message layout to read the dump with. DragonFly BSD shares this class but not every value.
92       *
93       * @return This platform's layout
94       */
95      protected RouteTableDump.Layout routeLayout() {
96          return RouteTableDump.Layout.FREEBSD;
97      }
98  }
