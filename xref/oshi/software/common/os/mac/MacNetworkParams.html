5   package oshi.software.common.os.mac;
6   
7   import java.util.List;
8   
9   import oshi.annotation.concurrent.ThreadSafe;
10  import oshi.software.common.AbstractNetworkParams;
11  import oshi.software.os.NetworkParams;
12  import oshi.util.ExecutingCommand;
13  import oshi.util.ParseUtil;
14  import oshi.util.driver.unix.NetstatRoute;
15  import oshi.util.driver.unix.RouteTableDump;
16  
17  /**
18   * Common Mac NetworkParams logic shared between the JNA and FFM implementations. The default-gateway lookups are
19   * command-line based and live here; hostname/domain resolution is native and provided by the subclasses.
20   */
21  @ThreadSafe
22  public abstract class MacNetworkParams extends AbstractNetworkParams {
23  
24      private static final String IPV6_ROUTE_HEADER = "Internet6:";
25  
26      private static final String DEFAULT_GATEWAY = "default";
27  
28      /** macOS prints the interface in the fourth column: Destination Gateway Flags Netif Expire. */
29      private static final int IF_NAME_INDEX = 3;
30  
31      /** Default constructor. */
32      protected MacNetworkParams() {
33      }
34  
35      @Override
36      public String getIpv4DefaultGateway() {
37          return searchGateway(ExecutingCommand.runNative("route -n get default"));
38      }
39  
40      @Override
41      public String getIpv6DefaultGateway() {
42          List<String> lines = ExecutingCommand.runNative("netstat -nr");
43          boolean v6Table = false;
44          for (String line : lines) {
45              if (v6Table && line.startsWith(DEFAULT_GATEWAY)) {
46                  String[] fields = ParseUtil.whitespaces.split(line, -1);
47                  if (fields.length > 2 && fields[2].contains("G")) {
48                      return fields[1].split("%", -1)[0];
49                  }
50              } else if (line.startsWith(IPV6_ROUTE_HEADER)) {
51                  v6Table = true;
52              }
53          }
54          return "";
55      }
56  
57      @Override
58      public List<NetworkParams.IPRoute> getRoutes() {
59          byte[] dump = queryRouteDump();
60          if (dump.length > 0) {
61              List<NetworkParams.IPRoute> routes = RouteTableDump.parse(dump, RouteTableDump.Layout.MACOS,
62                      queryInterfaceNameByIndex());
63              if (!routes.isEmpty()) {
64                  return routes;
65              }
66          }
67          return NetstatRoute.queryRoutes("netstat -rn -f inet", "netstat -rn -f inet6", IF_NAME_INDEX,
68                  queryInterfaceIndexByName());
69      }
70  
71      /**
72       * Fetches the kernel's routing table dump.
73       *
74       * @return The bytes of a {@code NET_RT_DUMP} sysctl, or an empty array if this build cannot make the call, in which
75       *         case the routing table is read by running a command instead
76       */
77      protected byte[] queryRouteDump() {
78          return new byte[0];
79      }
80  }
