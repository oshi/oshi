33  /**
34   * Abstract base shared by the macOS OSProcess implementations (JNA and FFM). Holds the field storage, the trivial
35   * accessors, the command-line/argument/environment memoization, thread enumeration, affinity mask, and the {@code
36   * pbi_status} state mapping. The native attribute reads ({@code proc_pidinfo}, the {@code KERN_PROCARGS2} sysctl, the
37   * {@code getrlimit} file limits, and the logical-processor sysctl) are provided by the per-backend subclasses.
38   */

91      protected MacOSProcess(int pid, int major, int minor, MacOperatingSystem os) {
92          super(pid);
93          this.majorVersion = major;
94          this.minorVersion = minor;
95          this.os = os;
96          // macOS historically reports an empty (not null) name until updateAttributes() populates it

144     @Override
145     public List<OSThread> getThreadDetails() {
146         long now = System.currentTimeMillis();
147         return ThreadInfo.queryTaskThreads(getProcessID()).stream().parallel().map(stat -> {
148             // For long running threads the start time calculation can overestimate
149             long start = Math.max(now - stat.getUpTime(), getStartTime());
150             return new MacOSThread(getProcessID(), stat.getThreadId(), stat.getState(), stat.getSystemTime(),
151                     stat.getUserTime(), start, now - start, stat.getPriority());
152         }).collect(Collectors.toList());
153     }

175     @Override
176     public long getAffinityMask() {
177         // macOS doesn't do affinity. Return a bitmask of the current processors.
178         int logicalProcessorCount = queryLogicalProcessorCount();
179         return logicalProcessorCount < 64 ? (1L << logicalProcessorCount) - 1 : -1L;

207     /**
208      * Maps a macOS {@code pbi_status} value to an {@link State}.
209      *
210      * @param status the {@code pbi_status} value
211      * @return the corresponding process state
212      */
213     protected static State stateFromStatus(int status) {
214         switch (status) {
215             case SSLEEP:
216                 return SLEEPING;
217             case SWAIT:
218                 return WAITING;
219             case SRUN:
220                 return RUNNING;
221             case SIDL:
222                 return NEW;
223             case SZOMB:
224                 return ZOMBIE;
225             case SSTOP:
226                 return STOPPED;
227             default:
228                 return OTHER;
229         }
230     }

232     /**
233      * Parses the buffer returned by the {@code KERN_PROCARGS2} sysctl into the process arguments and environment.
234      * <p>
235      * The buffer holds an {@code int} argument count, a null-terminated {@code exec_path} string, null padding, then
236      * exactly that many contiguous null-terminated arguments (the first of which repeats {@code exec_path}), then the
237      * null-terminated environment entries.
238      *
239      * @param procargs the raw sysctl buffer
240      * @param size     the number of valid bytes in {@code procargs}, as reported by the sysctl
241      * @return a pair of the argument list and the environment map, both empty if the buffer is unusable
242      */
243     protected static Pair<List<String>, Map<String, String>> parseProcArgs(byte[] procargs, int size) {
244         List<String> args = new ArrayList<>();
245         // API does not specify any particular order of entries, but it is reasonable to
246         // maintain whatever order the OS provided to the end user
247         Map<String, String> env = new LinkedHashMap<>();
248         int limit = Math.min(size, procargs.length);
249         if (limit > Integer.BYTES) {
250             int nargs = (int) ParseUtil.byteArrayToLong(procargs, Integer.BYTES, false);
251             // Sanity check. Every argument occupies at least its null terminator, so the bytes remaining after the
252             // count bound how many there can be. A process may legitimately have many thousands of arguments.
253             if (nargs > 0 && nargs <= limit - Integer.BYTES) {
254                 // Skip the leading int and the exec_path string, which the first argument repeats, then the padding
255                 // between exec_path and the arguments
256                 int offset = nextNull(procargs, Integer.BYTES, limit);
257                 while (offset < limit && procargs[offset] == 0) {
258                     offset++;
259                 }
260                 // The arguments are contiguous, so consume exactly nargs of them rather than treating a null as
261                 // padding: an empty argument is a legal argv entry, and skipping it would both drop it and pull an
262                 // environment entry into the argument list.
263                 for (int i = 0; i < nargs && offset < limit; i++) {
264                     int end = nextNull(procargs, offset, limit);
265                     if (end == limit) {
266                         // Unterminated: the reported size cut through this entry, so what is left is a fragment
267                         // rather than a value the process was given
268                         break;
269                     }
270                     args.add(decode(procargs, offset, end));
271                     offset = end + 1;
272                 }
273                 // The environment entries follow, separated by nulls
274                 while (offset < limit) {
275                     while (offset < limit && procargs[offset] == 0) {
276                         offset++;
277                     }
278                     if (offset >= limit) {
279                         break;
280                     }
281                     int end = nextNull(procargs, offset, limit);
282                     if (end == limit) {
283                         // Unterminated, as above: a truncated value is worse than a missing one
284                         break;
285                     }
286                     String entry = decode(procargs, offset, end);
287                     int idx = entry.indexOf('=');
288                     if (idx > 0) {
289                         env.put(entry.substring(0, idx), entry.substring(idx + 1));
290                     }
291                     offset = end;
292                 }
293             }
294         }
295         return new Pair<>(Collections.unmodifiableList(args), Collections.unmodifiableMap(env));
296     }

298     /**
299      * Decodes the bytes in {@code [from, to)} as UTF-8. The range must be delimited by scanning for the null
300      * terminator: a multi-byte character is longer in bytes than the decoded String is in chars, so advancing an offset
301      * by the char count lands mid-character and corrupts every entry that follows.
302      */
303     private static String decode(byte[] buf, int from, int to) {
304         return new String(buf, from, to - from, StandardCharsets.UTF_8);
305     }
306 
307     /**
308      * Returns the index of the next null byte at or after {@code from}, or {@code limit} if there is none.
309      */
310     private static int nextNull(byte[] buf, int from, int limit) {
311         int idx = from;
312         while (idx < limit && buf[idx] != 0) {
313             idx++;
314         }
315         return idx;
316     }
