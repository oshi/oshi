5   package oshi.software.common.os.windows;
6   
7   import java.util.Collections;
8   import java.util.LinkedHashMap;
9   import java.util.Map;
10  
11  import oshi.annotation.concurrent.ThreadSafe;
12  import oshi.driver.common.windows.perfmon.ProcessInformation.HandleCountProperty;
13  import oshi.software.common.AbstractFileSystem;
14  
15  /**
16   * Common file-descriptor logic for the Windows file system implementations. The volume enumeration
17   * ({@code getFileStores}) is backend-specific (Kernel32/WMI via JNA vs FFM) and remains in the subclasses; only the
18   * handle-count query differs by a single native call, abstracted here via {@link #queryHandles()}.
19   */
20  @ThreadSafe
21  public abstract class WindowsFileSystem extends AbstractFileSystem {
22  
23      private static final int FILE_READ_ONLY_VOLUME = 0x00080000;
24  
25      /**
26       * The {@code GetVolumeInformation} file system flags reported as mount options, in the order they are listed. The
27       * names follow the {@code winnt.h} constants: {@code FILE_CASE_PRESERVED_NAMES} is {@code casepn}, and so on.
28       */
29      private static final Map<Integer, String> OPTIONS_MAP;
30  
31      static {
32          Map<Integer, String> options = new LinkedHashMap<>();
33          options.put(0x00000002, "casepn"); // FILE_CASE_PRESERVED_NAMES
34          options.put(0x00000001, "casess"); // FILE_CASE_SENSITIVE_SEARCH
35          options.put(0x00000010, "fcomp"); // FILE_FILE_COMPRESSION
36          options.put(0x20000000, "dax"); // FILE_DAX_VOLUME
37          options.put(0x00040000, "streams"); // FILE_NAMED_STREAMS
38          options.put(0x00000008, "acls"); // FILE_PERSISTENT_ACLS
39          options.put(0x00100000, "wronce"); // FILE_SEQUENTIAL_WRITE_ONCE
40          options.put(0x00020000, "efs"); // FILE_SUPPORTS_ENCRYPTION
41          options.put(0x00010000, "oids"); // FILE_SUPPORTS_OBJECT_IDS
42          options.put(0x00000080, "reparse"); // FILE_SUPPORTS_REPARSE_POINTS
43          options.put(0x00000040, "sparse"); // FILE_SUPPORTS_SPARSE_FILES
44          options.put(0x00200000, "trans"); // FILE_SUPPORTS_TRANSACTIONS
45          options.put(0x02000000, "journaled"); // FILE_SUPPORTS_USN_JOURNAL
46          options.put(0x00000004, "unicode"); // FILE_UNICODE_ON_DISK
47          options.put(0x00008000, "vcomp"); // FILE_VOLUME_IS_COMPRESSED
48          options.put(0x00000020, "quota"); // FILE_VOLUME_QUOTAS
49          OPTIONS_MAP = Collections.unmodifiableMap(options);
50      }
51  
52      /**
53       * Formats the flags {@code GetVolumeInformation} returns as a mount options string: {@code ro} or {@code rw},
54       * followed by each set flag's name in a fixed order.
55       *
56       * @param flags the file system flags
57       * @return the options, such as {@code rw,casepn,casess,unicode}
58       */
59      protected static String volumeOptions(int flags) {
60          StringBuilder options = new StringBuilder((FILE_READ_ONLY_VOLUME & flags) == 0 ? "rw" : "ro");
61          OPTIONS_MAP.forEach((flag, name) -> {
62              if ((flag & flags) != 0) {
63                  options.append(',').append(name);
64              }
65          });
66          return options.toString();
67      }
68  
69      /**
70       * The maximum number of Windows handles. Both the 32-bit and 64-bit limits are essentially infinite for practical
71       * purposes. See
72       * <a href="https://blogs.technet.microsoft.com/markrussinovich/2009/09/29/pushing-the-limits-of-windows-handles/">
73       * Pushing the Limits of Windows: Handles</a>.
74       */
75      protected static final long MAX_WINDOWS_HANDLES;
76      static {
77          if (System.getenv("ProgramFiles(x86)") == null) {
78              MAX_WINDOWS_HANDLES = 16_777_216L - 32_768L;
79          } else {
80              MAX_WINDOWS_HANDLES = 16_777_216L - 65_536L;
81          }
82      }
83  
84      @Override
85      public long getOpenFileDescriptors() {
86          return queryHandles().getOrDefault(HandleCountProperty.HANDLECOUNT, 0L);
87      }
88  
89      @Override
90      public long getMaxFileDescriptors() {
91          return MAX_WINDOWS_HANDLES;
92      }
93  
94      @Override
95      public long getMaxFileDescriptorsPerProcess() {
96          // Windows has no separate per-process handle limit; return the system-wide maximum.
97          return MAX_WINDOWS_HANDLES;
98      }
99  
100     /**
101      * Queries the current system-wide open handle count via the subclass's native mechanism (JNA or FFM).
102      *
103      * @return a map including the {@link HandleCountProperty#HANDLECOUNT} total
104      */
105     protected abstract Map<HandleCountProperty, Long> queryHandles();
106 }
