5   package oshi.software.common.os.linux;
6   
7   import java.util.ArrayList;
8   import java.util.Collections;
9   import java.util.List;
10  import java.util.Map;
11  
12  import oshi.annotation.concurrent.ThreadSafe;
13  import oshi.software.common.AbstractNetworkParams;
14  import oshi.software.os.NetworkParams;
15  import oshi.util.FileUtil;
16  import oshi.util.ParseUtil;
17  import oshi.util.driver.linux.proc.RouteTable;
18  import oshi.util.linux.ProcPath;
19  
20  /**
21   * Linux network parameters. Provides the host name and default gateway implementations shared by all backends. The JNA
22   * and FFM subclasses override {@link #getDomainName()} with a native call; the native-free subclass inherits the
23   * Java-based resolution from {@link AbstractNetworkParams}.
24   */
25  @ThreadSafe
26  public abstract class LinuxNetworkParams extends AbstractNetworkParams {
27  
28      /**
29       * Default constructor.
30       */
31      protected LinuxNetworkParams() {
32      }
33  
34      @Override
35      public String getHostName() {
36          // The kernel.hostname sysctl is the same utsname.nodename that gethostname(2) returns, per UTS namespace, so
37          // this stays correct inside a container. Reading it avoids the name resolution the AbstractNetworkParams
38          // fallback performs, which truncates an FQDN at the first dot and reports "localhost" when the host name does
39          // not resolve.
40          String hostname = FileUtil.getStringFromFile(ProcPath.SYS_KERNEL_HOSTNAME);
41          return hostname.isEmpty() ? super.getHostName() : hostname;
42      }
43  
44      @Override
45      public String getIpv4DefaultGateway() {
46          return defaultGateway(RouteTable.queryIpv4Routes(Collections.emptyMap()));
47      }
48  
49      @Override
50      public String getIpv6DefaultGateway() {
51          return defaultGateway(RouteTable.queryIpv6Routes(Collections.emptyMap()));
52      }
53  
54      /**
55       * Selects the default gateway from a routing table: the next hop of the default route with the lowest metric, the
56       * one the kernel uses.
57       *
58       * @param routes the routes of one address family
59       * @return the gateway address text, or an empty string if no default route has a gateway
60       */
61      static String defaultGateway(List<NetworkParams.IPRoute> routes) {
62          NetworkParams.IPRoute best = null;
63          for (NetworkParams.IPRoute route : routes) {
64              if (route.getPrefixLength() == 0 && route.isGateway() && route.getGateway().length > 0
65                      && (best == null || route.getMetric() < best.getMetric())) {
66                  best = route;
67              }
68          }
69          return best == null ? "" : ParseUtil.formatIpAddress(best.getGateway());
70      }
71  
72      @Override
73      public List<NetworkParams.IPRoute> getRoutes() {
74          // /proc and /sys are kernel interfaces rather than real disk, so reading them is cheaper than the process
75          // spawn the route command costs, and is not affected by net-tools being absent.
76          Map<String, Integer> ifIndexByName = queryInterfaceIndexByName();
77          List<NetworkParams.IPRoute> routes = new ArrayList<>(RouteTable.queryIpv4Routes(ifIndexByName));
78          routes.addAll(RouteTable.queryIpv6Routes(ifIndexByName));
79          return routes;
80      }
81  }
