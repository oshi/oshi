5   package oshi.metrics;
6   
7   import java.util.Arrays;
8   import java.util.List;
9   import java.util.function.Supplier;
10  import java.util.function.ToDoubleFunction;
11  
12  import oshi.software.os.OSFileStore;
13  import oshi.util.Memoizer;
14  
15  import io.micrometer.core.instrument.Gauge;
16  import io.micrometer.core.instrument.MeterRegistry;
17  import io.micrometer.core.instrument.Tag;
18  import io.micrometer.core.instrument.Tags;
19  import io.micrometer.core.instrument.binder.MeterBinder;
20  
21  /**
22   * {@link MeterBinder} for system filesystem metrics following
23   * <a href="https://opentelemetry.io/docs/specs/semconv/system/system-metrics/#filesystem-metrics">OpenTelemetry
24   * semantic conventions</a>.
25   *
26   * <p>
27   * Registers per filesystem:
28   * <ul>
29   * <li>{@code system.filesystem.usage} — filesystem space usage by state ({@code used}, {@code free}, {@code reserved}),
30   * in bytes</li>
31   * <li>{@code system.filesystem.utilization} — fraction of filesystem space in use by state (0.0–1.0)</li>
32   * <li>{@code system.filesystem.limit} — total capacity of the filesystem, in bytes</li>
33   * </ul>
34   *
35   * <p>
36   * The three states partition the filesystem, so the {@code usage} gauges sum to {@code limit} and the
37   * {@code utilization} gauges sum to 1.0. {@code free} is the space available to the calling process
38   * ({@link OSFileStore#getUsableSpace()}), and {@code reserved} is unused space that is not available to it: the
39   * superuser reserve many UNIX filesystems hold back, or the caller's quota on Windows. Filesystems that reserve nothing
40   * at this layer, such as ZFS and APFS, report {@code reserved} as 0.
41   *
42   * <p>
43   * A filesystem's space is re-read as its gauges are sampled, once per filesystem per
44   * {@link Memoizer#defaultExpiration()} window rather than once per gauge, so that all of its gauges within a scrape
45   * report the same reading.
46   */
47  public class FileSystemMetrics implements MeterBinder {
48  
49      private static final String FS_USAGE = "system.filesystem.usage";
50      private static final String FS_UTILIZATION = "system.filesystem.utilization";
51      private static final String FS_LIMIT = "system.filesystem.limit";
52      private static final String DEVICE_KEY = "system.device";
53      private static final String MOUNTPOINT_KEY = "system.filesystem.mountpoint";
54      private static final String TYPE_KEY = "system.filesystem.type";
55      private static final String MODE_KEY = "system.filesystem.mode";
56      private static final String STATE_KEY = "system.filesystem.state";
57      private static final Tag STATE_USED = Tag.of(STATE_KEY, "used");
58      private static final Tag STATE_FREE = Tag.of(STATE_KEY, "free");
59      private static final Tag STATE_RESERVED = Tag.of(STATE_KEY, "reserved");
60  
61      private final Supplier<List<OSFileStore>> fileStoreSupplier;
62  
63      /**
64       * Creates a new {@code FileSystemMetrics} binder.
65       *
66       * <p>
67       * Note: {@link #bindTo(MeterRegistry)} calls the supplier once to capture a snapshot of filesystems. Filesystems
68       * mounted after binding will not be tracked; unmounted filesystems may leave stale meters. To refresh, create and
69       * bind a new instance.
70       *
71       * @param fileStoreSupplier supplier that returns the current list of {@link OSFileStore} instances
72       */
73      public FileSystemMetrics(Supplier<List<OSFileStore>> fileStoreSupplier) {
74          this.fileStoreSupplier = fileStoreSupplier;
75      }
76  
77      @Override
78      public void bindTo(MeterRegistry registry) {
79          for (OSFileStore fs : fileStoreSupplier.get()) {
80              String opts = fs.getOptions();
81              String mode = Arrays.asList(opts.split(",")).contains("rw") ? "rw" : "ro";
82              Tags tags = Tags.of(DEVICE_KEY, fs.getVolume(), MOUNTPOINT_KEY, fs.getMount(), TYPE_KEY, fs.getType(),
83                      MODE_KEY, mode);
84              // A bound OSFileStore is a snapshot taken when the binder was created, so it has to be refreshed as it is
85              // sampled. The seven gauges below read it through one memoized supplier, both to spare the filesystem six
86              // redundant queries per scrape and so that a single scrape reads one snapshot: the states can only be shown
87              // to partition the filesystem if they were all measured against the same reading of it.
88              Supplier<OSFileStore> refreshed = Memoizer.memoize(() -> {
89                  fs.updateAttributes();
90                  return fs;
91              }, Memoizer.defaultExpiration());
92  
93              // system.filesystem.usage — UpDownCounter (Gauge), unit "By", attr: state, device, mount, type, mode.
94              // OSFileStore guarantees usable <= free <= total, so none of these differences can be negative and the
95              // three states sum to the total.
96              registerUsage(registry, refreshed, tags, STATE_USED, f -> f.getTotalSpace() - f.getFreeSpace());
97              registerUsage(registry, refreshed, tags, STATE_FREE, OSFileStore::getUsableSpace);
98              registerUsage(registry, refreshed, tags, STATE_RESERVED, f -> f.getFreeSpace() - f.getUsableSpace());
99  
100             // system.filesystem.utilization — Gauge, unit "1", attr: state, device, mount, type, mode
101             registerUtilization(registry, refreshed, tags, STATE_USED,
102                     f -> fraction(f.getTotalSpace() - f.getFreeSpace(), f.getTotalSpace()));
103             registerUtilization(registry, refreshed, tags, STATE_FREE,
104                     f -> fraction(f.getUsableSpace(), f.getTotalSpace()));
105             registerUtilization(registry, refreshed, tags, STATE_RESERVED,
106                     f -> fraction(f.getFreeSpace() - f.getUsableSpace(), f.getTotalSpace()));
107 
108             // system.filesystem.limit — UpDownCounter (Gauge), unit "By", attr: device, mount, type, mode
109             Gauge.builder(FS_LIMIT, refreshed, s -> s.get().getTotalSpace()).tags(tags)
110                     .description("Total capacity of the filesystem").baseUnit("By").strongReference(true)
111                     .register(registry);
112         }
113     }
114 
115     private static void registerUsage(MeterRegistry registry, Supplier<OSFileStore> refreshed, Tags tags, Tag state,
116             ToDoubleFunction<OSFileStore> bytes) {
117         Gauge.builder(FS_USAGE, refreshed, s -> bytes.applyAsDouble(s.get())).tags(tags.and(state))
118                 .description("Filesystem space usage").baseUnit("By").strongReference(true).register(registry);
119     }
120 
121     private static void registerUtilization(MeterRegistry registry, Supplier<OSFileStore> refreshed, Tags tags,
122             Tag state, ToDoubleFunction<OSFileStore> ratio) {
123         Gauge.builder(FS_UTILIZATION, refreshed, s -> ratio.applyAsDouble(s.get())).tags(tags.and(state))
124                 .description("Filesystem utilization").strongReference(true).register(registry);
125     }
126 
127     private static double fraction(long value, long total) {
128         return total == 0 ? 0d : (double) value / total;
129     }
130 }
