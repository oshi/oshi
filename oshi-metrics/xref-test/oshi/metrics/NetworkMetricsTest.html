28  /**
29   * Tests that {@link NetworkMetrics} reads an interface once per scrape rather than once per meter, which the live tests
30   * in {@code OshiMetricsTest} cannot do: a real interface's counters may or may not move between samples.
31   */
32  class NetworkMetricsTest {
33  
34      private static final String NET_IO = "system.network.io";
35      private static final String NET_PACKETS = "system.network.packet.count";
36      private static final String NET_DROPPED = "system.network.packet.dropped";
37      private static final String NET_ERRORS = "system.network.errors";
38      private static final String DIRECTION = "network.io.direction";
39  
40      private MeterRegistry registry;
41  
42      /**
43       * An interface whose counters advance on every refresh, and which counts how often it was refreshed. Backed by a
44       * real {@link NetworkInterface} because that is all {@link AbstractNetworkIF} needs; only the counters below, which
45       * the fixture owns, are read by the meters under test.
46       */
47      private static class CountingNetworkIF extends AbstractNetworkIF {
48          private int refreshes;
49  
50          CountingNetworkIF(NetworkInterface netint) throws InstantiationException {
51              super(netint);
52          }
53  
54          @Override
55          public boolean updateAttributes() {
56              this.refreshes++;
57              // Advance every counter by the same amount, so any two of them read from one refresh agree
58              long n = this.refreshes;
59              this.bytesRecv = n;
60              this.bytesSent = n;
61              this.packetsRecv = n;
62              this.packetsSent = n;
63              this.inErrors = n;
64              this.outErrors = n;
65              this.inDrops = n;
66              return true;
67          }
68  
69          int refreshes() {
70              return this.refreshes;
71          }
72      }
73  
74      /** Reports no connections, so the connection-count gauges register without needing a live query. */
75      private static final class NoConnections implements InternetProtocolStats {
76          @Override
77          public TcpStats getTCPv4Stats() {
78              throw new UnsupportedOperationException("not sampled by NetworkMetrics");
79          }
80  
81          @Override
82          public TcpStats getTCPv6Stats() {
83              throw new UnsupportedOperationException("not sampled by NetworkMetrics");
84          }
85  
86          @Override
87          public UdpStats getUDPv4Stats() {
88              throw new UnsupportedOperationException("not sampled by NetworkMetrics");
89          }
90  
91          @Override
92          public UdpStats getUDPv6Stats() {
93              throw new UnsupportedOperationException("not sampled by NetworkMetrics");
94          }
95  
96          @Override
97          public List<IPConnection> getConnections() {
98              return Collections.emptyList();
99          }
100     }
101 
102     @BeforeEach
103     void setUp() {
104         registry = new SimpleMeterRegistry();
105     }
106 
107     private static CountingNetworkIF loopbackFixture() throws SocketException, InstantiationException {
108         NetworkInterface netint = NetworkInterface.getByInetAddress(InetAddress.getLoopbackAddress());
109         assumeTrue(netint != null, "No loopback interface to back the fixture; skipping");
110         return new CountingNetworkIF(netint);
111     }
112 
113     private CountingNetworkIF bind() throws SocketException, InstantiationException {
114         CountingNetworkIF net = loopbackFixture();
115         new NetworkMetrics(() -> Collections.<NetworkIF>singletonList(net), new NoConnections()).bindTo(registry);
116         return net;
117     }
118 
119     private double counter(String name, String direction, String device) {
120         FunctionCounter counter = registry.find(name).tag("system.device", device).tag(DIRECTION, direction)
121                 .functionCounter();
122         assertNotNull(counter, name + "{" + DIRECTION + "=" + direction + "} should be registered");
123         return counter.count();
124     }
125 
126     @Test
127     void allMetersOfOneInterfaceShareASingleReading() throws SocketException, InstantiationException {
128         // Sampling seven meters used to query the interface seven times, so each meter's value came from a different
129         // reading. They now share one memoized refresh per interface, which holds for 300 ms by default -- far longer
130         // than sampling seven meters in process.
131         CountingNetworkIF net = bind();
132         String device = net.getName();
133 
134         double bytesRecv = counter(NET_IO, "receive", device);
135         double bytesSent = counter(NET_IO, "transmit", device);
136         double packetsRecv = counter(NET_PACKETS, "receive", device);
137         double packetsSent = counter(NET_PACKETS, "transmit", device);
138         double drops = counter(NET_DROPPED, "receive", device);
139         double inErrors = counter(NET_ERRORS, "receive", device);
140         double outErrors = counter(NET_ERRORS, "transmit", device);
141 
142         assertEquals(1, net.refreshes(), "One scrape should read the interface once, not once per meter");
143         assertEquals(bytesRecv, bytesSent, "Bytes received and sent should come from the same reading");
144         assertEquals(packetsRecv, packetsSent, "Packets received and sent should come from the same reading");
145         assertEquals(bytesRecv, packetsRecv, "Bytes and packets should come from the same reading");
146         assertEquals(bytesRecv, drops, "Drops and bytes should come from the same reading");
147         assertEquals(inErrors, outErrors, "Errors in and out should come from the same reading");
148         assertEquals(bytesRecv, inErrors, "Errors and bytes should come from the same reading");
149     }
150 
151     @Test
152     void bindingDoesNotReadTheInterface() throws SocketException, InstantiationException {
153         // The refresh is memoized lazily, so binding an interface registers meters without querying it
154         CountingNetworkIF net = bind();
155         assertEquals(0, net.refreshes(), "Binding should not query the interface");
156     }
