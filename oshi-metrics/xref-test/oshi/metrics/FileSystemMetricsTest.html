5   package oshi.metrics;
6   
7   import static org.junit.jupiter.api.Assertions.assertEquals;
8   import static org.junit.jupiter.api.Assertions.assertNotNull;
9   import static org.junit.jupiter.api.Assertions.assertTrue;
10  
11  import java.util.Collections;
12  
13  import org.junit.jupiter.api.BeforeEach;
14  import org.junit.jupiter.api.Test;
15  
16  import oshi.software.common.AbstractOSFileStore;
17  import oshi.software.os.OSFileStore;
18  
19  import io.micrometer.core.instrument.Gauge;
20  import io.micrometer.core.instrument.MeterRegistry;
21  import io.micrometer.core.instrument.simple.SimpleMeterRegistry;
22  
23  /**
24   * Tests the state arithmetic of {@link FileSystemMetrics} against fixed values, which the live tests in
25   * {@code OshiMetricsTest} cannot do: a real filesystem's space changes between samples.
26   */
27  class FileSystemMetricsTest {
28  
29      private static final String USAGE = "system.filesystem.usage";
30      private static final String UTILIZATION = "system.filesystem.utilization";
31      private static final String LIMIT = "system.filesystem.limit";
32      private static final String STATE = "system.filesystem.state";
33      private static final String[] STATES = { "used", "free", "reserved" };
34      private static final double DELTA = 1e-9;
35  
36      private MeterRegistry registry;
37  
38      /** A file store with fixed space values, so a gauge reads the same number every time it is sampled. */
39      private static class FixedOSFileStore extends AbstractOSFileStore {
40          FixedOSFileStore(long freeSpace, long usableSpace, long totalSpace) {
41              super("fixed", "/dev/fixed", "label", "/mnt/fixed", "rw,noatime", "uuid", true, "lv", "Local Disk", "ext4",
42                      freeSpace, usableSpace, totalSpace, 100L, 500L);
43          }
44  
45          @Override
46          public boolean updateAttributes() {
47              return true;
48          }
49      }
50  
51      /** A file store that reports different space every time it is refreshed, as a busy one does. */
52      private static class VaryingOSFileStore extends AbstractOSFileStore {
53          private long free = 1000L;
54  
55          VaryingOSFileStore() {
56              super("varying", "/dev/varying", "label", "/mnt/varying", "rw", "uuid", true, "lv", "Local Disk", "ext4",
57                      1000L, 900L, 10000L, 100L, 500L);
58          }
59  
60          @Override
61          public boolean updateAttributes() {
62              this.free -= 100L;
63              updateSpace(this.free, this.free - 100L, 10000L);
64              return true;
65          }
66      }
67  
68      @BeforeEach
69      void setUp() {
70          registry = new SimpleMeterRegistry();
71      }
72  
73      private void bind(long freeSpace, long usableSpace, long totalSpace) {
74          OSFileStore fs = new FixedOSFileStore(freeSpace, usableSpace, totalSpace);
75          new FileSystemMetrics(() -> Collections.<OSFileStore>singletonList(fs)).bindTo(registry);
76      }
77  
78      private double usage(String state) {
79          Gauge gauge = registry.find(USAGE).tag(STATE, state).gauge();
80          assertNotNull(gauge, USAGE + "{state=" + state + "} should be registered");
81          return gauge.value();
82      }
83  
84      private double utilization(String state) {
85          Gauge gauge = registry.find(UTILIZATION).tag(STATE, state).gauge();
86          assertNotNull(gauge, UTILIZATION + "{state=" + state + "} should be registered");
87          return gauge.value();
88      }
89  
90      private double limit() {
91          Gauge gauge = registry.find(LIMIT).gauge();
92          assertNotNull(gauge, LIMIT + " should be registered");
93          return gauge.value();
94      }
95  
96      @Test
97      void reservedSpaceIsReportedSeparatelyFromUsedAndFree() {
98          // 1000 total, 200 unused, only 150 of it available to the caller: the other 50 is the superuser reserve
99          bind(200L, 150L, 1000L);
100         assertEquals(800d, usage("used"), "used should exclude the reserve");
101         assertEquals(150d, usage("free"), "free should be the space available to the caller");
102         assertEquals(50d, usage("reserved"), "reserved should be the unused space unavailable to the caller");
103     }
104 
105     @Test
106     void usageStatesSumToLimit() {
107         // OpenTelemetry's filesystem conventions require summing usage across states to equal the limit
108         bind(200L, 150L, 1000L);
109         assertEquals(limit(), usage("used") + usage("free") + usage("reserved"),
110                 "usage states should sum to the filesystem limit");
111     }
112 
113     @Test
114     void utilizationStatesSumToOne() {
115         bind(200L, 150L, 1000L);
116         assertEquals(0.8d, utilization("used"), DELTA);
117         assertEquals(0.15d, utilization("free"), DELTA);
118         assertEquals(0.05d, utilization("reserved"), DELTA);
119         assertEquals(1d, utilization("used") + utilization("free") + utilization("reserved"), DELTA,
120                 "utilization states should sum to 1.0");
121     }
122 
123     @Test
124     void filesystemWithoutAReserveReportsZeroReserved() {
125         // ZFS and APFS reserve nothing at this layer, so statvfs reports f_bfree == f_bavail
126         bind(200L, 200L, 1000L);
127         assertEquals(800d, usage("used"));
128         assertEquals(200d, usage("free"));
129         assertEquals(0d, usage("reserved"));
130     }
131 
132     @Test
133     void emptyFilesystemReportsZeroUtilization() {
134         // A pseudo filesystem with no capacity would otherwise divide by zero
135         bind(0L, 0L, 0L);
136         assertEquals(0d, limit());
137         assertEquals(0d, utilization("used"));
138         assertEquals(0d, utilization("free"));
139         assertEquals(0d, utilization("reserved"));
140     }
141 
142     @Test
143     void allGaugesOfOneFilesystemShareASingleReading() {
144         // A filesystem whose space changes on every refresh would break the partition rule if each gauge refreshed
145         // separately: the states would be measured against four different readings. They share one memoized refresh
146         // per filesystem, which holds for 300 ms by default -- far longer than sampling four gauges in process.
147         OSFileStore fs = new VaryingOSFileStore();
148         new FileSystemMetrics(() -> Collections.singletonList(fs)).bindTo(registry);
149         assertEquals(limit(), usage("used") + usage("free") + usage("reserved"),
150                 "usage states should sum to the limit even while the filesystem is changing");
151     }
152 
153     @Test
154     void inconsistentSpaceValuesCannotProduceNegativeStates() {
155         // AbstractOSFileStore clamps usable <= free <= total, so no state can go negative even when the operating
156         // system reports the three values inconsistently
157         bind(6000L, 5500L, 5000L);
158         for (String state : STATES) {
159             assertTrue(usage(state) >= 0d, USAGE + "{state=" + state + "} should not be negative");
160             assertTrue(utilization(state) >= 0d, UTILIZATION + "{state=" + state + "} should not be negative");
161         }
162         assertEquals(limit(), usage("used") + usage("free") + usage("reserved"));
163     }
164 }
